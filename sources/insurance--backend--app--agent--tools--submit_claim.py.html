32async def prepare_claim_submission(
33    policy_number: str,
34    claim_type: str,
35    amount: float,
36    description: str,
37) -> dict[str, Any]:
38    """Prepare a new OmniCare insurance claim submission for confirmation.
39
40    Use this tool when the user has confirmed they want to file a new claim
41    after all required fields have been collected. This tool validates the
42    inputs and creates a pending record in the database, returning a
43    confirmation_token that the frontend must present to the /confirm endpoint.
44
45    The agent should instruct the user to confirm the submission in the UI
46    after receiving the confirmation_token.
47
48    Args:
49        policy_number: The policyholder's policy number (e.g., "POL-1092").
50        claim_type: Category of the claim (e.g., "Water Damage", "Personal Property").
51        amount: Claimed amount in US dollars. Must be greater than 0.
52        description: Factual description of the incident (minimum 10 characters).
53
54    Returns:
55        dict: Preparation result. On success, includes success=True,
56        confirmation_token, expires_at, and a message instructing the user
57        to confirm. On failure, includes success=False and an error message.
58    """
59    try:
60        user_uuid = current_user_id.get()
61    except LookupError:
62        logger.error("current_user_id not found in context during claim preparation.")
63        return {"success": False, "error": "Unauthorized submission."}
64
65    try:
66        validated = ClaimSubmission(
67            policy_number=policy_number,
68            claim_type=claim_type,
69            amount=amount,
70            description=description,
71        )
72    except ValidationError as exc:
73        errors = [
74            f"{err['loc'][-1] if err['loc'] else 'field'}: {err['msg']}" for err in exc.errors()
75        ]
76        logger.warning("Claim preparation validation failed: %s", errors)
77        return {"success": False, "validation_errors": errors}
78
79    confirmation_token = str(uuid.uuid4())
80    expires_at = datetime.now(UTC) + timedelta(minutes=15)
81
82    claim_data = {
83        "policy_number": validated.policy_number,
84        "claim_type": validated.claim_type,
85        "amount": str(validated.amount),
86        "description": validated.description,
87    }
88
89    try:
90        async with async_session_factory() as session:
91            submission = ClaimSubmissionModel(
92                user_id=user_uuid,
93                confirmation_token=confirmation_token,
94                claim_data=claim_data,
95                status="pending",
96                expires_at=expires_at,
97            )
98            session.add(submission)
99            await session.commit()
100    except Exception as exc:
101        logger.exception("Failed to persist pending claim submission: %s", exc)
102        return {
103            "success": False,
104            "error": "Your claim could not be prepared. Please try again or contact support.",
105        }
106
107    logger.info(
108        "Prepared claim submission for user '%s' with token '%s'.",
109        user_uuid,
110        confirmation_token,
111    )
112
113    return {
114        "success": True,
115        "confirmation_token": confirmation_token,
116        "expires_at": expires_at.isoformat(),
117        "status": "pending",
118        "message": (
119            "Your claim has been prepared. Please confirm the submission in the UI "
120            "to complete your claim."
121        ),
122    }

125async def submit_claim_internal(  # noqa: PLR0913, PLR0917
126    policy_number: str,
127    claim_type: str,
128    amount: float,
129    description: str,
130    user_uuid: uuid.UUID,
131    session: AsyncSession | None = None,
132) -> dict[str, Any]:
133    """Submit a new OmniCare insurance claim on behalf of the authenticated user.
134
135    This is the internal function that performs the actual database write.
136    It is called by the /confirm endpoint after the user has confirmed the
137    submission via the frontend.
138
139    Args:
140        policy_number: The policyholder's policy number.
141        claim_type: Category of the claim.
142        amount: Claimed amount in US dollars.
143        description: Factual description of the incident.
144        user_uuid: UUID of the authenticated user.
145        session: Optional existing database session. When provided, the claim
146            and embedding job are written to the same transaction and the
147            caller is responsible for committing.
148
149            This function never commits a caller-supplied session. Committing here
150            would release the ``SELECT ... FOR UPDATE`` row lock that
151            ``confirm_claim_submission`` holds on the pending submission, letting a
152            second concurrent confirmation read the same row as ``pending`` and insert
153            a duplicate claim. It only flushes, so the caller can still mark the
154            submission ``completed`` and commit both writes atomically.
155
156    Returns:
157        dict: Submission confirmation or error payload.
158    """
159    try:
160        validated = ClaimSubmission(
161            policy_number=policy_number,
162            claim_type=claim_type,
163            amount=amount,
164            description=description,
165        )
166    except ValidationError as exc:
167        errors = [
168            f"{err['loc'][-1] if err['loc'] else 'field'}: {err['msg']}" for err in exc.errors()
169        ]
170        logger.warning("Claim submission validation failed: %s", errors)
171        return {"success": False, "validation_errors": errors}
172
173    confirmation_id = f"CLM-{uuid.uuid4().hex[:8].upper()}"
174
175    async def _submit(session: AsyncSession) -> dict[str, Any]:
176        new_claim = Claim(
177            claim_id=confirmation_id,
178            policy_number=validated.policy_number,
179            claim_type=validated.claim_type,
180            status=ClaimStatus.SUBMITTED.value,
181            amount=validated.amount,
182            description=validated.description,
183            owner_id=user_uuid,
184        )
185        session.add(new_claim)
186        await session.flush()
187
188        job = EmbeddingJob(
189            claim_uuid=new_claim.id,
190            claim_id=new_claim.claim_id,
191            owner_id=new_claim.owner_id,
192            claim_type=new_claim.claim_type,
193            description=new_claim.description,
194            policy_number=new_claim.policy_number,
195            claim_status=new_claim.status,
196            status="pending",
197            status_detail="pending",
198        )
199        session.add(job)
200        await session.flush()
201
202        logger.info(
203            "Claim '%s' submitted for policy '%s'.",
204            confirmation_id,
205            validated.policy_number,
206        )
207
208        return {
209            "success": True,
210            "confirmation_id": confirmation_id,
211            "status": ClaimStatus.SUBMITTED.value,
212            "policy_number": validated.policy_number,
213            "claim_type": validated.claim_type,
214            "amount": validated.amount,
215            "description": validated.description,
216            "citation": (
217                f"Claim {confirmation_id} recorded in the OmniCare claims system "
218                f"under policy {validated.policy_number}."
219            ),
220            "message": (
221                f"Your claim {confirmation_id} has been successfully submitted "
222                "and is now being processed. Keep this ID for your records."
223            ),
224        }
225
226    try:
227        if session is None:
228            async with async_session_factory() as owned_session:
229                result = await _submit(owned_session)
230                await owned_session.commit()
231                return result
232        else:
233            return await _submit(session)
234    except Exception as exc:
235        logger.exception("Failed to persist claim '%s': %s", confirmation_id, exc)
236        return {
237            "success": False,
238            "error": "Your claim could not be saved. Please try again or contact support.",
239        }
