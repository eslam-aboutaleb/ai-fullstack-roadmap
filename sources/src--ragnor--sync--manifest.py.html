26@dataclass
27class ManifestRecord:
28    """Tracks document identity, fingerprints, generation, and status.
29
30    Attributes:
31        scope: Scoped namespace, collection, or source identifier.
32        document_id: Document identity within the scope.
33        observed_revision: Opaque revision hint (e.g. ETag, mtime, or version string).
34            Treated strictly as a revision hint, not a universal content hash.
35        content_hash: Cryptographic hash of normalized document content.
36        extractor_fingerprint: Identity/version/options of the extractor.
37        chunker_fingerprint: Configuration/version of the chunker.
38        embedding_fingerprint: Model name/dimension of the embedding function.
39        active_generation: Generation currently active and visible to readers.
40        staged_generation: Generation currently staged and awaiting activation.
41        status: Lifecycle status of the document.
42        last_successful_observation: Epoch timestamp of last successful observation.
43        chunk_ids: IDs of vector chunks active for this document.
44        staged_chunk_ids: IDs of vector chunks staged for this document.
45        metadata: Document metadata preserved through normalization.
46        provenance: Original source provenance.
47        fence_token: Monotonically increasing version used for compare-and-swap.
48    """
49
50    scope: str
51    document_id: str
52    observed_revision: str | None = None
53    content_hash: str = ""
54    extractor_fingerprint: str = ""
55    chunker_fingerprint: str = ""
56    embedding_fingerprint: str = ""
57    active_generation: int = 0
58    staged_generation: int = 0
59    status: ManifestStatus | str = ManifestStatus.ACTIVE
60    last_successful_observation: float | None = None
61    chunk_ids: list[str] = field(default_factory=list)
62    staged_chunk_ids: list[str] = field(default_factory=list)
63    metadata: dict[str, Any] = field(default_factory=dict)
64    provenance: dict[str, Any] = field(default_factory=dict)
65    fence_token: int = 0

106class InMemoryManifestStore:
107    """Thread-safe and async-safe in-memory ManifestStore implementation."""
108
109    def __init__(self) -> None:
110        self._records: dict[tuple[str, str], ManifestRecord] = {}
111        self._lock = asyncio.Lock()
112
113    async def get(self, scope: str, document_id: str) -> ManifestRecord | None:
114        async with self._lock:
115            rec = self._records.get((scope, document_id))
116            if rec is None:
117                return None
118            return deepcopy(rec)
119
120    async def list(
121        self, scope: str, status: ManifestStatus | str | None = None
122    ) -> list[ManifestRecord]:
123        async with self._lock:
124            records = [
125                deepcopy(rec)
126                for (s, _), rec in self._records.items()
127                if s == scope and (status is None or rec.status == status)
128            ]
129            return sorted(records, key=lambda r: r.document_id)
130
131    async def put(self, record: ManifestRecord) -> None:
132        async with self._lock:
133            record.fence_token += 1
134            self._records[(record.scope, record.document_id)] = deepcopy(record)
135
136    async def compare_and_swap(self, record: ManifestRecord, expected_fence: int) -> bool:
137        async with self._lock:
138            key = (record.scope, record.document_id)
139            existing = self._records.get(key)
140            current_fence = existing.fence_token if existing is not None else 0
141            if current_fence != expected_fence:
142                return False
143            record.fence_token = expected_fence + 1
144            self._records[key] = deepcopy(record)
145            return True
146
147    async def compare_and_swap_many(self, updates: list[tuple[ManifestRecord, int]]) -> bool:
148        async with self._lock:
149            keys = [(record.scope, record.document_id) for record, _ in updates]
150            if len(keys) != len(set(keys)):
151                raise ValueError("Duplicate document in manifest commit")
152            for record, expected in updates:
153                existing = self._records.get((record.scope, record.document_id))
154                if (existing.fence_token if existing is not None else 0) != expected:
155                    return False
156            for record, expected in updates:
157                record.fence_token = expected + 1
158                self._records[(record.scope, record.document_id)] = deepcopy(record)
159            return True
160
161    async def delete(self, scope: str, document_id: str) -> bool:
162        async with self._lock:
163            key = (scope, document_id)
164            if key in self._records:
165                del self._records[key]
166                return True
167            return False
