1"""
2In-app notification feed endpoints for the OmniCare backend.
3
4``GET /notifications`` returns the caller's delivered notifications, newest first,
5and ``POST /notifications/{id}/read`` marks one of them read. Both are scoped to the
6caller's own id, so another user's notification is answered as not found rather than
7forbidden — the response must not confirm that the id exists.
8
9The feed is a read model, not the queue: ``notification_jobs`` holds what is waiting
10to be delivered, and the drain worker writes a row into ``notifications`` when it
11delivers. Nothing here can enqueue, retry, or reorder a delivery, and no endpoint on
12the request path performs one, because a claimant waiting on a status change must
13not also wait on a delivery.
14
15The request and response models live here rather than in ``app.schemas.models``
16because they belong to these two tables and nothing else: the shared module is the
17frozen contract surface for chat, claims, and auth.
18"""

41class NotificationItem(BaseModel):
42    """One delivered notification, as the recipient's feed shows it.
43
44    Attributes:
45        id: Primary key of the delivered notification.
46        job_type: Which claim event produced it, so a client can key an icon or a
47            grouping on it without parsing the text.
48        channel: Which provider delivered it. Only ``in_app`` today.
49        title: Rendered subject line.
50        body: Rendered message text.
51        claim_id: The claim this is about, when the notification has one. Present so
52            a "view claim" action can deep link without a second request.
53        read_at: When the caller acknowledged it, or NULL while it is still unread.
54        created_at: When it was delivered.
55    """
56
57    id: uuid.UUID = Field(..., description="Primary key of the delivered notification")
58    job_type: str = Field(..., description="Claim event that produced the notification")
59    channel: str = Field(..., description="Delivery channel that produced the row")
60    title: str = Field(..., description="Rendered subject line")
61    body: str = Field(..., description="Rendered message text")
62    claim_id: str | None = Field(default=None, description="Claim the notification is about")
63    read_at: datetime | None = Field(default=None, description="When it was read, if it was")
64    created_at: datetime = Field(..., description="When it was delivered")

106async def list_notifications(  # noqa: PLR0913
107    unread_only: bool = Query(
108        default=False,
109        description="Return only notifications the caller has not acknowledged yet.",
110    ),
111    limit: int = Query(
112        default=20,
113        ge=1,
114        le=50,
115        description="Maximum number of notifications to return.",
116    ),
117    offset: int = Query(
118        default=0,
119        ge=0,
120        description="Number of newer notifications to skip.",
121    ),
122    current_user_id: str = Depends(get_current_user),
123    session: AsyncSession = Depends(get_db),
124) -> NotificationListResponse:
125    """List the authenticated user's delivered notifications, newest first.
126
127    The owner predicate is the first thing in the WHERE clause, before the unread
128    filter and before the pagination window, so a caller can never trim somebody
129    else's feed into their own response by choosing a limit.
130
131    Args:
132        unread_only: Whether to return only unacknowledged notifications.
133        limit: Page size, defaulting to 20 and capped at 50.
134        offset: Number of newest notifications to skip.
135        current_user_id: The authenticated user's UUID, injected by
136            ``get_current_user``.
137        session: Async database session.
138
139    Returns:
140        NotificationListResponse: One page of the caller's notifications plus the
141        pagination values that produced it.
142
143    Raises:
144        HTTPException: 401 if the authenticated user id is not a valid UUID.
145    """
146    try:
147        user_uuid = uuid.UUID(current_user_id)
148    except ValueError:
149        raise HTTPException(
150            status_code=status.HTTP_401_UNAUTHORIZED,
151            detail="Invalid user",
152        ) from None
153
154    stmt = select(Notification).where(Notification.user_id == user_uuid)
155    if unread_only:
156        stmt = stmt.where(Notification.read_at.is_(None))
157    # id breaks ties so a page ordered on a same-microsecond delivery is still
158    # deterministic across repeated requests.
159    stmt = stmt.order_by(desc(Notification.created_at), desc(Notification.id))
160    stmt = stmt.limit(limit).offset(offset)
161
162    notifications = (await session.execute(stmt)).scalars().all()
163
164    return NotificationListResponse(
165        notifications=[_notification_item(row) for row in notifications],
166        count=len(notifications),
167        limit=limit,
168        offset=offset,
169    )

176    description=(

188            "description": "Not authenticated.",
189            "model": ErrorResponse,

191        status.HTTP_404_NOT_FOUND: {
192            "description": "No notification with that id belongs to the caller.",
193            "model": ErrorResponse,
194        },
195        status.HTTP_422_UNPROCESSABLE_ENTITY: {
196            "description": "The notification id is not a valid UUID.",
197            "model": ErrorResponse,
198        },
199    },
200)
201async def mark_notification_read(
202    notification_id: uuid.UUID,
203    current_user_id: str = Depends(get_current_user),
204    session: AsyncSession = Depends(get_db),
205) -> NotificationItem:
206    """Mark one of the caller's notifications as read.
207
208    Args:
209        notification_id: The delivered notification to acknowledge.
210        current_user_id: The authenticated user's UUID, injected by
211            ``get_current_user``.
212        session: Async database session.
213
214    Returns:
215        NotificationItem: The notification with its new ``read_at``.
216
217    Raises:
218        HTTPException: 401 if the authenticated user id is not a valid UUID, 404 if
219            the notification does not exist or belongs to another user.
220    """
221    try:
222        user_uuid = uuid.UUID(current_user_id)
223    except ValueError:
224        raise HTTPException(
225            status_code=status.HTTP_401_UNAUTHORIZED,
226            detail="Invalid user",
227        ) from None
228
229    # SECURITY: Both predicates live in the WHERE clause so a notification id
230    # belonging to another user is indistinguishable from one that was never issued.
231    notification = (
232        await session.execute(
233            select(Notification).where(
234                Notification.id == notification_id,
235                Notification.user_id == user_uuid,
236            )
237        )
238    ).scalar_one_or_none()
239
240    if notification is None:
241        logger.warning(
242            "Notification '%s' read requested by non-owner '%s'; reported as not found.",
243            notification_id,
244            current_user_id,
245        )
246        raise HTTPException(
247            status_code=status.HTTP_404_NOT_FOUND,
248            detail="Notification not found",
249        )
250
251    # First acknowledgement wins, so a double tap cannot push the timestamp forward
252    # and make an older notification look newer than one that arrived after it.
253    if notification.read_at is None:
254        notification.read_at = datetime.now(UTC)
255        await session.commit()
256
257    return _notification_item(notification)

260def _notification_item(row: Notification) -> NotificationItem:
261    """Project a delivered notification onto the API's response model.
262
263    Args:
264        row: The delivered notification to project.
265
266    Returns:
267        NotificationItem: The notification as the feed shows it.
268    """
269    return NotificationItem(
270        id=row.id,
271        job_type=row.job_type,
272        channel=row.channel,
273        title=row.title,
274        body=row.body,
275        claim_id=row.claim_id,
276        read_at=row.read_at,
277        created_at=row.created_at,
278    )
