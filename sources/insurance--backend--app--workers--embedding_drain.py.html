1"""
2Outbox drainer for the ``embedding_jobs`` table.
3
4Thin app-side entry point: wires the
5OmniCare job store and claim processor into ragnor's
6generic drainer. The drain loop itself — per-pass stale
7reclaim, ``FOR UPDATE SKIP LOCKED`` claiming, failure
8retry on the next tick, graceful SIGINT/SIGTERM — lives
9in :mod:`ragnor.jobs.drainer`.
10
11Run it directly:
12
13    python -m app.workers.embedding_drain
14
15Configuration (all optional, validated by ``app.core.config.Settings``):
16
17    EMBEDDING_DRAIN_INTERVAL_SECONDS  seconds to sleep when the queue is empty
18    EMBEDDING_DRAIN_BATCH_SIZE       jobs claimed per pass
19
20Multiple drainers can run concurrently: ``SKIP LOCKED`` keeps two workers from
21processing the same job, so this scales horizontally without coordination.
22"""
23
24from __future__ import annotations
25
26import sys
27
28from app.core.config import settings
29from app.domain.claims.outbox import ClaimJobProcessor, SqlAlchemyJobStore
30from ragnor.jobs.drainer import main as _ragnor_main
31
32
33def main() -> int:
34    """Console entry point: drain the embedding outbox until signaled."""
35    store = SqlAlchemyJobStore()
36    processor = ClaimJobProcessor()
37    return _ragnor_main(
38        store,
39        processor,
40        interval=settings.embedding_drain_interval_seconds,
41        batch_size=settings.embedding_drain_batch_size,
42        worker_id=settings.worker_id,
43    )
44
45
46if __name__ == "__main__":
47    sys.exit(main())
