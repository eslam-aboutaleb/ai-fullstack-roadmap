1"""Vector store abstraction for ragnor.
2
3Defines the interface for hybrid retrieval (vector + keyword
4search) operations, allowing different backends (pgvector,
5in-memory, etc.) to be swapped via the store registry without
6changing the retrieval or ingestion code.
7
8Storage is split from retrieval strategy. Three structural
9capability Protocols describe what a backend can do:
10
11* :class:`VectorIndex` -- nearest-neighbor search over
12  embeddings;
13* :class:`LexicalIndex` -- keyword search over document
14  text;
15* :class:`DocumentStore` -- document lifecycle (upsert,
16  count, delete, get).
17
18A host adapts an existing search engine to a capability
19Protocol by implementing its methods -- no ragnor
20inheritance required. :class:`HybridVectorStore` composes
21the two search capabilities into the legacy
22:class:`VectorStore` hybrid contract (Reciprocal Rank
23Fusion), and :class:`LegacyHybridStoreAdapter` wraps a
24pre-refactor backend that only implements the fused
25contract.
26"""

51@runtime_checkable
52class VectorIndex(Protocol):
53    """Structural capability: similarity search over embeddings.
54
55    Implementations search a vector index for the nearest
56    neighbors of ``embedding`` and return at most
57    ``n_results`` result dicts ordered by similarity (best
58    first). The method requires a valid embedding -- it does
59    not auto-embed the query -- and it always raises on
60    operational failure (there is no ``strict`` parameter;
61    the opt-in swallow contract lives in the legacy
62    ``hybrid_search`` and in the retrieval strategies).
63
64    Result dict shape: ``id``, ``document``, ``metadata``,
65    ``distance`` (the similarity distance), and
66    ``keyword_score`` (``None`` -- this stage found nothing
67    by keyword). ``**filters`` are equality filters matched
68    against document fields; keys must be valid SQL
69    identifiers. A supplied ``session`` is caller-owned;
70    without one, each backend follows its documented
71    lifecycle.
72    """

88@runtime_checkable
89class LexicalIndex(Protocol):
90    """Structural capability: keyword search over document text.
91
92    Implementations search their lexical index for documents
93    matching ``query`` and return at most ``n_results``
94    result dicts ordered by keyword relevance (best first).
95    There is no ``strict`` parameter: operational failures
96    always propagate (the opt-in swallow contract lives in
97    the legacy ``hybrid_search`` and in the retrieval
98    strategies).
99
100    Result dict shape: ``id``, ``document``, ``metadata``,
101    ``distance`` (``None`` -- this stage found nothing by
102    vector similarity), and ``keyword_score`` (the keyword
103    relevance score). ``**filters`` are equality filters
104    matched against document fields; keys must be valid SQL
105    identifiers. A supplied ``session`` is caller-owned;
106    without one, each backend follows its documented
107    lifecycle.
108
109    Performance notice: implementing this Protocol does not
110    guarantee production-scale keyword search. pgvector is
111    the only built-in native lexical index (PostgreSQL
112    full-text search); the other built-in backends expose
113    this Protocol through an in-process token-overlap scan
114    intended for local development, small corpora, and
115    testing. Hosts at scale combine a pgvector
116    :class:`VectorIndex` with an external lexical engine
117    adapted to this Protocol.
118    """

133@runtime_checkable
134class DocumentStore(Protocol):
135    """Structural capability: document lifecycle operations.
136
137    Implementations persist documents (``upsert``), count
138    them (``count``), delete them (``delete``), and fetch
139    them directly (``get``). ``delete`` and ``get`` have no
140    ``strict`` parameter: failures always propagate, and
141    ``delete`` requires at least one selector (``ids`` or
142    ``filters``) and rejects an empty ``ids`` list -- an
143    empty id list never means "delete everything". When both
144    selectors are supplied, both restrictions apply
145    (conjunction). A supplied ``session`` is caller-owned;
146    without one, each backend follows its documented
147    lifecycle. ``delete`` returns the number of deleted
148    documents, with consistent meaning across backends.
149    ``get`` returns ``id``/``document``/``metadata`` dicts
150    (score keys are meaningless for direct fetches); with no
151    selector it returns every document.
152    """

196class VectorStore(ABC):
197    """Abstract interface for vector store operations.
198
199    The legacy composite contract: hybrid search (dense +
200    keyword, fused with Reciprocal Rank Fusion), counting,
201    and upsert. New backends should implement the capability
202    Protocols (:class:`VectorIndex`, :class:`LexicalIndex`,
203    :class:`DocumentStore`) and subclass
204    :class:`HybridVectorStore`; this ABC remains the
205    registry-facing name the host code and the docs depend
206    on.
207
208    Session ownership rule: when the caller supplies a
209    ``session`` to a method, the caller owns the transaction
210    and is responsible for committing or rolling it back.
211    When no session is supplied, the store opens its own
212    session and commits where the operation is transactional.
213    """
214

217        raise NotImplementedError("Store does not support embedding fetch for metadata updates")

304class HybridVectorStore(VectorStore):
305    """Composite base: RRF-fused hybrid search over capability stages.
306
307    Built-in in-process backends subclass this base and
308    implement the :class:`VectorIndex` and
309    :class:`LexicalIndex` capability methods
310    (:meth:`vector_search` and :meth:`keyword_search`); this
311    base provides the legacy :meth:`hybrid_search` by running
312    both stages -- each independently ranked and truncated to
313    ``n_results`` by the capability method -- and fusing the
314    two truncated result lists with Reciprocal Rank Fusion
315    (:func:`ragnor.stores._scoring.fuse_hybrid`).
316
317    The constructor accepts the uniform registry kwargs
318    (``table_name``, ``id_field``, ``embedding_dim``,
319    ``session_factory``, ``settings``, ``embedding_fn``,
320    ``metric``) so subclasses stay drop-in compatible with
321    :func:`ragnor.stores.get_vector_store`.
322
323    ``metric`` selects the vector stage's distance metric
324    (``"l2"`` by default, ``"cosine"`` opt-in); see
325    :mod:`ragnor.metrics`. Distance thresholds are distances
326    under that metric: a threshold is never comparable across
327    two different metrics, so a host that switches metrics must
328    re-tune it.
329    """
330
331    def __init__(  # noqa: PLR0913, PLR0917
332        self,
333        table_name: str = "documents",
334        id_field: str = "id",
335        embedding_dim: int | None = None,
336        session_factory: Any = None,
337        settings: Any = None,
338        embedding_fn: EmbeddingFunction | None = None,
339        metric: str | None = None,
340    ) -> None:
341        """Initialize the composite store.
342
343        Args:
344            table_name: Logical table (or collection) name.
345            id_field: Name of the primary key field.
346            embedding_dim: Expected embedding dimensionality. When
347                provided, embeddings are validated like the pgvector
348                store does.
349            session_factory: Session provider the store uses when a
350                method is called without a session (backends without
351                a database ignore it).
352            settings: Host settings (backends without a database
353                ignore them). The optional ``rag_distance_metric``
354                field selects the metric when ``metric`` is omitted.
355            embedding_fn: Async callable that embeds a list of texts,
356                used to auto-embed queries when ``hybrid_search`` is
357                called with an empty embedding.
358            metric: Distance metric for the vector stage (``"l2"``
359                by default, ``"cosine"`` opt-in; resolved
360                case-insensitively).
361
362        Raises:
363            ValueError: If ``table_name`` or ``id_field`` contain
364                unsafe characters, or ``metric`` is not a supported
365                distance metric.
366        """
367        validate_identifier(table_name, "table_name")
368        validate_identifier(id_field, "id_field")
369        self.table_name = table_name
370        self.id_field = id_field
371        self.embedding_dim = embedding_dim
372        self._session_factory = session_factory
373        self._settings = settings
374        self._embedding_fn = embedding_fn
375        self.metric: DistanceMetric = resolve_distance_metric(metric, settings)

404    async def hybrid_search(  # noqa: PLR0913, PLR0917
405        self,
406        query: str,
407        embedding: list[float],
408        n_results: int,
409        threshold: float,
410        text_field: str = "text",
411        metadata_fields: list[str] | None = None,
412        session: Any = None,
413        strict: bool = False,
414        **filters: Any,
415    ) -> list[dict[str, Any]]:
416        """Hybrid search: vector stage plus keyword stage, RRF-fused.
417
418        Validates identifiers and ``n_results`` and auto-embeds an
419        empty query embedding via the injected ``embedding_fn``
420        (raising ``ValueError`` when none is configured) before the
421        search runs; those ``ValueError`` s propagate. The two
422        capability stages then run and their truncated results are
423        fused with Reciprocal Rank Fusion. A stage failure is
424        logged and swallowed into an empty list unless
425        ``strict=True``, which raises
426        :class:`ragnor.types.RetrievalError`.
427
428        Args:
429            query: Natural language search query.
430            embedding: Query embedding vector. If empty, the query is
431                embedded automatically using the injected
432                ``embedding_fn``.
433            n_results: Maximum number of results to return.
434            threshold: Maximum distance for the vector search part.
435            text_field: Name of the field containing the document text.
436            metadata_fields: Fields to include in the result metadata.
437            session: Optional external session. When supplied, the
438                caller owns the transaction (see the class docstring).
439            strict: When ``True``, a failed search raises
440                :class:`ragnor.types.RetrievalError`.
441            **filters: Equality filters matched against document
442                fields (metadata and extra fields).
443
444        Returns:
445            list[dict]: Result dicts sorted by RRF score, with the
446            same keys as the other backends' hybrid search.
447
448        Raises:
449            ValueError: If identifiers are unsafe, ``n_results`` is
450                not positive, or the embedding is empty and no
451                ``embedding_fn`` is configured.
452            ragnor.types.RetrievalError: If a search stage fails and
453                ``strict=True``.
454        """
455        validate_search_identifiers(text_field, filters, metadata_fields)
456        metadata_fields = metadata_fields or []
457
458        if n_results < 1:
459            raise ValueError(f"n_results must be a positive integer; got {n_results}")
460
461        if not embedding:
462            if self._embedding_fn is None:
463                raise ValueError(
464                    "embedding_fn is required to auto-embed the query; "
465                    "pass an embedding or construct the store with "
466                    "embedding_fn"
467                )
468            embedding_list = await self._embedding_fn([query])
469            embedding = embedding_list[0]
470
471        if self.embedding_dim is not None:
472            validate_embedding(embedding, self.embedding_dim, label="query embedding")
473
474        try:
475            vector_results = await self.vector_search(
476                embedding=embedding,
477                n_results=n_results,
478                threshold=threshold,
479                text_field=text_field,
480                metadata_fields=metadata_fields,
481                session=session,
482                **filters,
483            )
484            keyword_results = await self.keyword_search(
485                query=query,
486                n_results=n_results,
487                text_field=text_field,
488                metadata_fields=metadata_fields,
489                session=session,
490                **filters,
491            )
492            return fuse_hybrid(vector_results, keyword_results, n_results)
493        except Exception as exc:
494            logger.error("Hybrid search failed on %s: %s", self.table_name, exc)
495            if strict:
496                raise RetrievalError(f"Hybrid search failed on {self.table_name}: {exc}") from exc
497            return []

500class LegacyHybridStoreAdapter(VectorStore):
501    """Wrap a pre-refactor custom backend for the default retrieval path.
502
503    A legacy backend implements the fused :class:`VectorStore`
504    contract (``hybrid_search``/``count``/``upsert``) but not
505    the capability Protocols. A fused hybrid result cannot be
506    decomposed into independent vector and keyword stages, so
507    this adapter deliberately implements no capability methods:
508    it must not pretend to satisfy :class:`VectorIndex`,
509    :class:`LexicalIndex`, or :class:`DocumentStore`
510    (``isinstance(adapter, VectorIndex)`` is ``False``).
511    Legacy backends keep working on the default retrieval path
512    with zero changes -- wrap them in this adapter and register
513    the adapter under the backend's provider name.
514    """
515
516    def __init__(self, backend: VectorStore) -> None:
517        """Wrap a legacy backend.
518
519        Args:
520            backend: A pre-refactor backend implementing the
521                :class:`VectorStore` ABC.
522        """
523        self._backend = backend
524
525    @property
526    def backend(self) -> VectorStore:
527        """The wrapped legacy backend."""
528        return self._backend
529
