1"""Optional reranker integration implementing the existing Reranker seam.
2
3Provides :class:`LiteLLMReranker`, an implementation of the
4:class:`~ragnor.retrieval.strategies.Reranker` protocol that uses
5``litellm.rerank`` to re-score candidate documents with cross-encoders
6or commercial reranking APIs (e.g. Cohere, Voyage, Jina, etc.).
7
8This module ships behind the ``rerank`` extra and is imported lazily
9so ``ragnor.retrieval`` remains completely dependency-free.
10"""
11
12from __future__ import annotations
13
14import asyncio
15from typing import Any
16
17
18class LiteLLMReranker:
19    """Reranker implementation backed by LiteLLM.
20
21    Implements the :class:`~ragnor.retrieval.strategies.Reranker`
22    protocol. Requires the ``rerank`` extra (``pip install 'ragnor[rerank]'``).
23
24    Attributes:
25        model: Model identifier passed to ``litellm.rerank`` (e.g.
26            ``"cohere/rerank-v3.5"``).
27        api_key: Optional provider API key.
28        api_base: Optional provider API base URL.
29        top_n: Optional limit on the number of reranked results returned.
30    """
31
32    def __init__(
33        self,
34        model: str = "cohere/rerank-v3.5",
35        *,
36        api_key: str | None = None,
37        api_base: str | None = None,
38        top_n: int | None = None,
39    ) -> None:
40        self.model = model
41        self.api_key = api_key
42        self.api_base = api_base
43        self.top_n = top_n
44
45    def _rerank_sync(
46        self,
47        query: str,
48        documents: list[str],
49    ) -> list[dict[str, Any]]:
50        """Synchronous call to litellm.rerank running in worker thread."""
51        try:
52            import litellm  # noqa: PLC0415
53        except ImportError as exc:
54            raise ImportError(
55                "The LiteLLM reranker requires the 'rerank' extra: pip install 'ragnor[rerank]'"
56            ) from exc
57
58        kwargs: dict[str, Any] = {"model": self.model, "query": query, "documents": documents}
59        if self.api_key:
60            kwargs["api_key"] = self.api_key
61        if self.api_base:
62            kwargs["api_base"] = self.api_base
63        if self.top_n is not None:
64            kwargs["top_n"] = self.top_n
65
66        response = litellm.rerank(**kwargs)
67        # Standard litellm rerank response format: response.results or list of dicts
68        raw_results = getattr(response, "results", response)
69        items: list[dict[str, Any]] = []
70        for item in raw_results:
71            idx = item.get("index") if isinstance(item, dict) else getattr(item, "index", 0)
72            score = (
73                item.get("relevance_score")
74                if isinstance(item, dict)
75                else getattr(item, "relevance_score", 0.0)
76            )
77            items.append({"index": idx, "score": float(score)})
78        return items
79
80    async def rerank(
81        self,
82        query: str,
83        results: list[dict[str, Any]],
84    ) -> list[dict[str, Any]]:
85        """Rerank ``results`` for ``query``, sorted by relevance score descending.
86
87        Args:
88            query: Natural language search query.
89            results: List of candidate result dicts carrying at least
90                a ``"document"`` key.
91
92        Returns:
93            Re-ordered list of result dicts, each carrying a
94            ``"_reranker_score"`` key with the raw score.
95        """
96        if not results:
97            return []
98
99        doc_texts = [str(r.get("document", "")) for r in results]
100        reranked_items = await asyncio.to_thread(self._rerank_sync, query, doc_texts)
101
102        output: list[dict[str, Any]] = []
103        for item in reranked_items:
104            idx = item["index"]
105            score = item["score"]
106            if 0 <= idx < len(results):
107                updated = dict(results[idx])
108                updated["_reranker_score"] = score
109                output.append(updated)
110
111        return output
