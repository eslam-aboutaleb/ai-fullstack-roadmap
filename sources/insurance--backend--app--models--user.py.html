1"""
2SQLAlchemy ORM model for the ``users`` table.
3
4Represents an authenticated OmniCare user. Password storage uses Argon2id
5hashes, which are intentionally stored in a separate column from the username
6to avoid accidental leakage in query logs or admin interfaces.
7
8Every user carries a role, because an API that serves both policyholders and
9insurers is only as safe as its weakest authorization check. The default is
10``policyholder``: the tier that existed before the column was added, so an
11existing row keeps the access it already had and no endpoint changes behaviour
12for the accounts it already had.
13"""
14
15import uuid
16from datetime import datetime
17from enum import Enum
18
19from sqlalchemy import DateTime, String, Uuid, func
20from sqlalchemy.orm import Mapped, mapped_column
21
22from app.models.base import Base
23
24
25class UserRole(str, Enum):
26    """The authorization tiers an account can hold.
27
28    Attributes:
29        POLICYHOLDER: An end user. May act on their own claims, conversations,
30            and ratings, and on nothing else.
31        ADMIN: An operator. May read the cross-owner queue, the audit trail, and
32            the embedding outbox, and may run the evaluation harness.
33        CLAIMS_HANDLER: Reserved for the handler tier that decides a claim's
34            fate. No endpoint reads it yet -- the lifecycle route still refuses
35            administrative transitions to everyone -- so it exists to keep the
36            column's vocabulary complete rather than to grant anything today.
37    """
38
39    POLICYHOLDER = "policyholder"
40    ADMIN = "admin"
41    CLAIMS_HANDLER = "claims_handler"
42
43
44class User(Base):
45    """Database model for an OmniCare user account.
46
47    Attributes:
48        id: Primary key UUID.
49        username: Unique, indexed username for authentication.
50        password_hash: Argon2id hash of the user's password. Never store
51            plaintext passwords.
52        role: Authorization tier, one of ``UserRole``. Defaults to
53            ``policyholder`` at both the ORM and database level, so a row
54            written before the column existed and a row inserted by raw SQL
55            both land in the tier that changes nothing.
56        created_at: Timestamp of account creation, set automatically by the
57            database server.
58    """
59
60    __tablename__ = "users"
61
62    id: Mapped[uuid.UUID] = mapped_column(Uuid(as_uuid=True), primary_key=True, default=uuid.uuid4)
63    username: Mapped[str] = mapped_column(String(200), unique=True, index=True, nullable=False)
64    password_hash: Mapped[str] = mapped_column(String(300), nullable=False)
65    role: Mapped[str] = mapped_column(
66        String(32),
67        nullable=False,
68        default=UserRole.POLICYHOLDER.value,
69        server_default=UserRole.POLICYHOLDER.value,
70    )
71    created_at: Mapped[datetime] = mapped_column(
72        DateTime(timezone=True), nullable=False, server_default=func.now()
73    )
74
75
76__all__ = ["User", "UserRole"]
