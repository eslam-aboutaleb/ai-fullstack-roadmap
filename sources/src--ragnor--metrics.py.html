1"""Distance metrics for the vector stores.
2
3A store's distance metric decides how the vector stage scores
4documents: ``l2`` (Euclidean distance) is ragnor's historical
5default, and ``cosine`` (``1 - cosine_similarity``) is an opt-in
6for hosts whose indexes and thresholds are cosine end-to-end.
7
8Thresholds are metric-relative: a threshold tuned for one metric
9means something different under another, so a host that switches
10metrics must re-tune ``rag_distance_threshold``.
11
12``rag_distance_metric`` is an OPTIONAL host settings field, read
13through ``getattr`` exactly like the other optional host fields
14(``embedding_provider``, ``llm_model``). ragnor never adds it to
15the :class:`ragnor.config.RagSettings` protocol, so a host settings
16object keeps satisfying the protocol whether or not it declares
17the field.
18"""
19
20from __future__ import annotations

22import math
23from collections.abc import Sequence
24from typing import Any, Literal

26DistanceMetric = Literal["l2", "cosine"]
27"""Distance metric a store ranks by (``l2`` or ``cosine``)."""
28
29DEFAULT_DISTANCE_METRIC: DistanceMetric = "l2"
30"""Metric used when neither an argument nor the host settings pick one."""
31
32SUPPORTED_DISTANCE_METRICS: tuple[DistanceMetric, ...] = ("l2", "cosine")
33"""Every metric the stores accept."""

36def resolve_distance_metric(metric: str | None = None, settings: Any = None) -> DistanceMetric:
37    """Resolve the distance metric from an argument or host settings.
38
39    Resolution order: the explicit ``metric`` argument, then the
40    optional ``rag_distance_metric`` host settings field, then
41    :data:`DEFAULT_DISTANCE_METRIC`. Values are normalized
42    case-insensitively (mirroring ragnor's provider registries) and
43    surrounding whitespace is ignored.
44
45    Args:
46        metric: Requested metric, or ``None`` to fall back to the
47            host settings.
48        settings: Host settings object. The optional
49            ``rag_distance_metric`` field is read with ``getattr``,
50            so a settings object that does not declare it keeps
51            working; ``None`` is accepted.
52
53    Returns:
54        The resolved :class:`DistanceMetric`.
55
56    Raises:
57        ValueError: If the resolved value is not a supported
58            metric. The message lists the supported names.
59    """
60    if metric is None and settings is not None:
61        metric = getattr(settings, "rag_distance_metric", None)
62    if metric is None:
63        return DEFAULT_DISTANCE_METRIC
64    normalized = str(metric).strip().lower()
65    for supported in SUPPORTED_DISTANCE_METRICS:
66        if normalized == supported:
67            return supported
68    supported_names = ", ".join(SUPPORTED_DISTANCE_METRICS)
69    raise ValueError(
70        f"Unknown distance metric {metric!r}; supported metrics are: {supported_names}"
71    )

74def cosine_similarity(left: Sequence[float], right: Sequence[float]) -> float:
75    """Return the cosine similarity of two vectors.
76
77    The similarity is the dot product divided by the product of
78    the vector norms, in ``[-1.0, 1.0]``. A zero-norm vector has
79    no direction, so the similarity is defined as ``0.0``.
80
81    Args:
82        left: First vector.
83        right: Second vector.
84
85    Returns:
86        The cosine similarity of the two vectors.
87
88    Raises:
89        ValueError: If the vectors have different lengths.
90    """
91    if len(left) != len(right):
92        raise ValueError(f"vectors must have the same length; got {len(left)} and {len(right)}")
93    left_norm = math.sqrt(sum(value * value for value in left))
94    right_norm = math.sqrt(sum(value * value for value in right))
95    if left_norm == 0.0 or right_norm == 0.0:
96        return 0.0
97    return sum(a * b for a, b in zip(left, right, strict=False)) / (left_norm * right_norm)
98

100def cosine_distance(left: Sequence[float], right: Sequence[float]) -> float:
101    """Return the cosine distance between two vectors.
102
103    The distance is ``1.0 - cosine_similarity(left, right)``, in
104    ``[0.0, 2.0]``: identical vectors score ``0.0``, orthogonal
105    vectors score ``1.0``, and opposite vectors score ``2.0``.
106
107    Args:
108        left: First vector.
109        right: Second vector.
110
111    Returns:
112        The cosine distance between the two vectors.
113
114    Raises:
115        ValueError: If the vectors have different lengths.
116    """
117    return 1.0 - cosine_similarity(left, right)

120def normalize(vector: Sequence[float]) -> list[float]:
121    """Return the unit-length form of ``vector``.
122
123    A zero vector has no direction to normalize to, so it is
124    returned unchanged.
125
126    Args:
127        vector: Vector to scale to unit length.
128
129    Returns:
130        A new list holding the unit vector (``[0.0, ...]`` for a
131        zero vector). The input is not modified.
132    """
133    norm = math.sqrt(sum(value * value for value in vector))
134    if norm == 0.0:
135        return list(vector)
136    return [value / norm for value in vector]
