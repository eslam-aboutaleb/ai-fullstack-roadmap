1"""Generic provider registry.
2
3Providers (embedders, chunkers, vector stores) register themselves
4under a name and are looked up case-insensitively. Host applications
5instantiate a registry per process; this module creates no
6module-level singletons.
7"""
8
9from __future__ import annotations
10
11from collections.abc import Callable
12from typing import Generic, TypeVar
13
14T = TypeVar("T")
15
16
17class Registry(Generic[T]):
18    """Case-insensitive registry mapping provider names to classes."""
19
20    def __init__(self) -> None:
21        self._providers: dict[str, type[T]] = {}
22        self._default_name: str | None = None
23
24    def register(self, name: str) -> Callable[[type[T]], type[T]]:
25        """Return a decorator that registers a provider class under ``name``.
26
27        Args:
28            name: Provider name. Lookup is case-insensitive.
29
30        Returns:
31            A decorator that registers the class and returns it unchanged.
32        """
33
34        def decorator(cls: type[T]) -> type[T]:
35            self._providers[name.lower()] = cls
36            return cls
37
38        return decorator
39
40    def get(self, name: str) -> type[T]:
41        """Return the class registered under ``name`` (case-insensitive).
42
43        Raises:
44            KeyError: If no provider is registered under ``name``. The
45                message lists the registered names.
46        """
47        try:
48            return self._providers[name.lower()]
49        except KeyError:
50            raise KeyError(
51                f"Unknown provider {name!r}. Registered: {', '.join(self.list()) or '(none)'}"
52            ) from None
53
54    def list(self) -> list[str]:
55        """Return the registered provider names in registration order."""
56        return list(self._providers.keys())
57
58    def default(self, name: str) -> None:
59        """Set the default provider name.
60
61        Args:
62            name: Provider name to use when no explicit provider is given.
63
64        Raises:
65            KeyError: If no provider is registered under ``name``.
66        """
67        if name.lower() not in self._providers:
68            raise KeyError(
69                f"Unknown provider {name!r}. Registered: {', '.join(self.list()) or '(none)'}"
70            )
71        self._default_name = name.lower()
72
73    @property
74    def default_name(self) -> str | None:
75        """The registered default provider name, or ``None`` if unset."""
76        return self._default_name
