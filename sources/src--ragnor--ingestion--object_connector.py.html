36def _error(exc: Exception, path: str) -> tuple[SourceStatus, dict[str, Any]]:
37    """Expose only safe structured error fields, never SDK exception strings."""
38    response = getattr(exc, "response", {})
39    response = response if isinstance(response, dict) else {}
40    code = str(response.get("Error", {}).get("Code", ""))
41    status_code = response.get("ResponseMetadata", {}).get("HTTPStatusCode")
42    if code in {"NoSuchKey", "NoSuchBucket", "NotFound", "404"} or status_code == 404:
43        status = SourceStatus.NOT_FOUND
44    elif code in {"AccessDenied", "InvalidAccessKeyId", "SignatureDoesNotMatch"} or status_code in {
45        401,
46        403,
47    }:
48        status = SourceStatus.INACCESSIBLE
49    else:
50        status = SourceStatus.TRANSIENT
51    return status, {"path": path, "reason": status.value}

54class ObjectStorageConnector:
55    """Bounded S3-compatible snapshot connector.
56
57    Configure ``max_pages``/``max_items`` to cap a run. Hitting a cap before
58    exhaustion, a malformed token, a changed object, or any read failure makes
59    the result non-authoritative. No retries happen here: SDK retry/backoff and
60    rate limiting belong to the injected client. Reconciliation restarts at the
61    first page; a listing is not a point-in-time snapshot under concurrent writes.
62    """
63
64    def __init__(self, bucket: str, **options: Any) -> None:
65        self.bucket = bucket
66        self.prefix = str(options.get("prefix", options.get("root", "")))
67        self.client = options.get("client")
68        self._credentials = options.get("credentials")
69        self._client_factory = options.get("client_factory")
70        self._endpoint_url = options.get("endpoint_url")
71        self._region = options.get("region")
72        self.max_file_size = int(options.get("max_file_size", options.get("max_bytes", 10485760)))
73        self.page_size = int(options.get("page_size", 100))
74        self.max_pages = options.get("max_pages")
75        self.max_items = options.get("max_items")
76        extensions: Sequence[str] = options.get("extensions", options.get("allowed_extensions", []))
77        self.extensions = {str(extension).lower().lstrip(".") for extension in extensions}
78        self.extractor = options.get("extractor")
79        self._last_result: LoadResult | None = None
80        if self.max_file_size <= 0 or not 0 < self.page_size <= 1000:
81            raise ValueError("max_file_size must be positive and page_size must be 1..1000")
82        for name in ("max_pages", "max_items"):
83            value = getattr(self, name)
84            if value is not None and (not isinstance(value, int) or value <= 0):
85                raise ValueError(f"{name} must be a positive integer")

101    def _client(self) -> ObjectStorageClient:
102        if self.client is None:
103            if self._client_factory is not None:
104                self.client = self._client_factory(self._credentials)
105            else:
106                try:
107                    import boto3  # noqa: PLC0415
108                except ImportError:
109                    raise ImportError(
110                        "Install ragnor[object-storage] or inject an ObjectStorageClient"
111                    ) from None
112                self.client = boto3.client(
113                    "s3", endpoint_url=self._endpoint_url, region_name=self._region
114                )
115        return self.client

117    async def discover(self, *, cursor: Any = None, limit: int | None = None) -> DiscoveryPage:
118        size = self.page_size if limit is None else limit
119        if not isinstance(size, int) or not 0 < size <= 1000:
120            raise ValueError("limit must be 1..1000")
121        if cursor is not None and (not isinstance(cursor, str) or not cursor):
122            raise ValueError("cursor must be a nonempty opaque listing token")
123        kwargs = {"Bucket": self.bucket, "Prefix": self.prefix, "MaxKeys": size}
124        if cursor is not None:
125            kwargs["ContinuationToken"] = cursor
126        try:
127            response = await asyncio.to_thread(self._client().list_objects_v2, **kwargs)
128        except ImportError:
129            raise
130        except Exception as exc:
131            status, error = _error(exc, self.bucket)
132            return DiscoveryPage([], status=status, errors=[error], complete=False)
133        if not isinstance(response, dict) or not isinstance(response.get("Contents", []), list):
134            return DiscoveryPage(
135                [],
136                status=SourceStatus.TRANSIENT,
137                errors=[{"reason": "malformed_listing"}],
138                complete=False,
139            )
140        has_more = bool(response.get("IsTruncated", False))
141        token = response.get("NextContinuationToken")
142        errors = []
143        if has_more and (not isinstance(token, str) or not token or token == cursor):
144            errors.append({"reason": "invalid_continuation"})
145        items = []
146        for entry in response.get("Contents", []):
147            if not isinstance(entry, dict) or not isinstance(entry.get("Key"), str):
148                errors.append({"reason": "malformed_item"})
149                continue
150            key = entry["Key"]
151            if not key.startswith(self.prefix):
152                errors.append({"reason": "key_outside_prefix"})
153                continue
154            if key.endswith("/"):
155                continue
156            items.append(
157                SourceItem(
158                    key=key,
159                    relative_path=key[len(self.prefix) :].lstrip("/"),
160                    size=entry.get("Size"),
161                    etag=entry.get("ETag"),
162                    last_modified=entry.get("LastModified"),
163                )
164            )
165        return DiscoveryPage(
166            items,
167            cursor=cursor,
168            next_cursor=token if has_more else None,
169            has_more=has_more,
170            complete=not has_more and not errors,
171            errors=errors,
172            status=SourceStatus.TRANSIENT if errors else SourceStatus.OK,
173        )
174
175    def _read(self, item: SourceItem) -> tuple[bytes | None, dict[str, Any]]:
176        kwargs = {"Bucket": self.bucket, "Key": item.key}
177        if item.get("etag"):
178            kwargs["IfMatch"] = item.etag
179        response = self._client().get_object(**kwargs)
180        body = response["Body"]
181        try:
182            size = response.get("ContentLength", 0)
183            if size > self.max_file_size:
184                return None, response
185            pieces = []
186            total = 0
187            while total <= self.max_file_size:
188                piece = body.read(min(65536, self.max_file_size + 1 - total))
189                if not piece:
190                    break
191                total += len(piece)
192                if total > self.max_file_size:
193                    return None, response
194                pieces.append(piece)
195            data = b"".join(pieces)
196            if size and len(data) != size:
197                raise OSError("incomplete object read")
198            return data, response
199        finally:
200            body.close()

202    async def load_page(self, *, cursor: Any = None, limit: int | None = None) -> LoadResult:
203        page = await self.discover(cursor=cursor, limit=limit)
204        result = LoadResult(
205            [],
206            status=page.status,
207            errors=page.errors,
208            next_cursor=page.next_cursor,
209            has_more=page.has_more,
210            complete=page.complete,
211            total_items=len(page),
212        )
213        for item in page:
214            relative = item.relative_path
215            suffix = relative.rsplit(".", 1)[-1].lower() if "." in relative else ""
216            if self.extensions and suffix not in self.extensions:
217                result.skipped.append(SkipRecord(relative, "unsupported_extension"))
218                continue
219            if item.get("size") is not None and item.size > self.max_file_size:
220                result.skipped.append(SkipRecord(relative, "oversized"))
221                continue
222            try:
223                data, response = await asyncio.to_thread(self._read, item)
224            except Exception as exc:
225                status, error = _error(exc, relative)
226                result.errors.append(error)
227                result.status = status
228                result.complete = False
229                continue
230            if data is None:
231                result.skipped.append(SkipRecord(relative, "oversized"))
232                continue
233            extracted = extract_documents(
234                data,
235                source_name=relative,
236                external_id=item.key,
237                metadata={
238                    "media_type": response.get("ContentType", ""),
239                    "size": len(data),
240                    "last_modified": response.get("LastModified") or item.get("last_modified"),
241                },
242                provenance={"bucket": self.bucket, "key": item.key, "etag": item.get("etag")},
243                extractor=self.extractor,
244            )
245            result.extend(extracted)
246            result.skipped.extend(extracted.skipped)
247            result.errors.extend(extracted.errors)
248            if extracted.errors:
249                result.complete = False
250                result.status = extracted.status
251        if not result and not result.errors and not page.has_more:
252            result.status = SourceStatus.EMPTY
253        self._last_result = result
254        return result

256    async def load_all(
257        self, *, max_pages: int | None = None, page_size: int | None = None
258    ) -> LoadResult:
259        count = 0
260
261        async def bounded_page(*, cursor=None, limit=None):
262            nonlocal count
263            remaining = None if self.max_items is None else self.max_items - count
264            size = self.page_size if limit is None else limit
265            if remaining is not None:
266                size = min(size, remaining)
267            page = await self.load_page(cursor=cursor, limit=size)
268            count += len(page)
269            if remaining is not None and len(page) > remaining:
270                del page[remaining:]
271                page.errors.append({"reason": "max_items"})
272                page.complete = False
273                page.status = SourceStatus.INCOMPLETE
274            elif self.max_items is not None and count >= self.max_items and page.has_more:
275                page.errors.append({"reason": "max_items"})
276                page.complete = False
277                page.status = SourceStatus.INCOMPLETE
278            return page
279
280        result = await collect_pages(
281            SimpleNamespace(load_page=bounded_page),
282            max_pages=self.max_pages if max_pages is None else max_pages,
283            page_size=self.page_size if page_size is None else page_size,
284        )
285        self._last_result = result
286        return result

296    async def load_incremental(self, *, checkpoint: Any = None) -> LoadResult:
297        """Load objects modified after checkpoint."""
298        since = parse_checkpoint_since(checkpoint)
299
300        full = await self.load_all()
301        if not full.complete or full.errors:
302            full.incremental = True
303            return full
304
305        latest = since or 0.0
306        docs = []
307        skipped = list(full.skipped)
308        for doc in full:
309            mtime = doc.metadata.get("last_modified")
310            mtime_val = 0.0
311            if isinstance(mtime, (int, float)):
312                mtime_val = float(mtime)
313            elif hasattr(mtime, "timestamp"):
314                mtime_val = mtime.timestamp()
315            elif isinstance(mtime, str):
316                from datetime import datetime  # noqa: PLC0415
317
318                try:
319                    mtime_val = datetime.fromisoformat(mtime).timestamp()
320                except Exception:
321                    mtime_val = 0.0
322            latest = max(latest, mtime_val)
323            if since is not None and mtime_val <= since:
324                skipped.append(SkipRecord(doc.source_name, "unmodified"))
325            else:
326                docs.append(doc)
327
328        new_cp = (
329            str(latest) if latest > 0 else (str(checkpoint) if checkpoint is not None else None)
330        )
331        status = SourceStatus.OK if docs else SourceStatus.EMPTY
332        return LoadResult(
333            docs,
334            status=status,
335            skipped=skipped,
336            errors=full.errors,
337            complete=full.complete,
338            incremental=True,
339            checkpoint=new_cp,
340            metadata={**full.metadata, "incremental": True},
341        )
