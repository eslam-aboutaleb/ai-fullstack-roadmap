1"""Session provider protocol and standalone session factory.
2
3ragnor talks to databases through the :class:`SessionProvider`
4protocol: any callable that returns an async session context
5manager works structurally. The host application keeps its own
6engine and session factory (including its NullPool test
7behavior); :func:`create_session_factory` is the standalone
8helper for projects that use ragnor without the host
9application. It delegates to the default
10:class:`ragnor.db.sql_factory.SQLAlchemyFactory`.
11"""
12
13from __future__ import annotations
14
15from typing import TYPE_CHECKING, Protocol, runtime_checkable
16
17if TYPE_CHECKING:
18    from sqlalchemy.ext.asyncio import AsyncSession
19
20
21@runtime_checkable
22class SessionProvider(Protocol):
23    """Callable that yields an async session as a context manager."""
24
25    def __call__(self) -> AsyncSession: ...
26
27
28def create_session_factory(database_url: str) -> SessionProvider:
29    """Build a session factory bound to a new async engine.
30
31    Args:
32        database_url: SQLAlchemy async database URL (for example
33            ``postgresql+asyncpg://user:pass@host/db``).
34
35    Returns:
36        A session factory. Each call yields an independent
37        ``AsyncSession`` with ``expire_on_commit=False``. The caller
38        owns the engine's lifecycle; ragnor creates no module-level
39        singletons.
40    """
41    from ragnor.db.sql_factory import SQLAlchemyFactory  # noqa: PLC0415
42
43    return SQLAlchemyFactory().create_session_factory(database_url)
