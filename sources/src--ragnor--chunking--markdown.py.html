1"""Markdown section-aware chunking.
2
3Splits a Markdown document on ``##`` section headers to
4preserve semantic boundaries, then delegates each
5reconstructed section to an inner chunker (typically
6:class:`ragnor.chunking.sliding_window.SlidingWindowChunker`)
7to produce overlapping sub-chunks.
8
9The file-reading logic lives in the
10:class:`ragnor.ingestion.sources.FileDocumentSource`;
11this chunker operates on text and takes the source name
12as a constructor argument.
13"""
14
15from __future__ import annotations
16
17import re
18from typing import Any
19
20from ragnor.chunking.base import Chunker
21from ragnor.types import Chunk
22
23
24class MarkdownSectionChunker:
25    """Chunker that splits Markdown on ``##`` section headers.
26
27    Each section is reconstructed with its H2 header
28    (``f"## {title}\\n\\n{body}"``) so the section title is
29    part of the chunk text, then passed to the inner chunker.
30    """
31
32    def __init__(self, inner: Chunker, source_name: str = "document") -> None:
33        """
34        Args:
35            inner: Chunker applied to each reconstructed section.
36            source_name: Source identifier recorded in chunk metadata
37                (the host application derives it from the document
38                path; ``FileDocumentSource`` supplies it).
39        """
40        self._inner = inner
41        self._source_name = source_name
42
43    def chunk(self, text: str) -> list[Chunk]:
44        """Parse a Markdown document into structured, overlapping chunks.
45
46        Args:
47            text: The Markdown document text.
48
49        Returns:
50            Chunks whose metadata carries the frozen keys
51            ``chunk_id``, ``section``, ``source``,
52            ``chunk_index``, and ``sub_chunk_index``.
53        """
54        # Split on Markdown H2 headers to preserve section boundaries. The
55        # regex captures the header title in group 1, allowing us to
56        # reconstruct the document as alternating title/body pairs.
57        sections = re.split(r"(?m)^##\s+(.*)$", text)
58
59        preamble = sections[0].strip()
60        parsed_sections: list[dict[str, str]] = []
61        if preamble:
62            title_match = re.search(r"^#\s+(.+)$", preamble, re.MULTILINE)
63            preamble_title = title_match.group(1).strip() if title_match else "Introduction"
64            parsed_sections.append({"title": preamble_title, "content": preamble})
65
66        for i in range(1, len(sections), 2):
67            title = sections[i].strip()
68            body = sections[i + 1].strip() if i + 1 < len(sections) else ""
69            parsed_sections.append({"title": title, "content": body})
70
71        chunks: list[Chunk] = []
72        global_chunk_idx = 0
73
74        for section in parsed_sections:
75            section_title = section["title"]
76            section_text = section["content"]
77
78            # Reconstruct the full section text with its H2 header so that
79            # the section title is included in the chunk content for better
80            # retrieval context.
81            full_text = f"## {section_title}\n\n{section_text}"
82            sub_chunks = self._inner.chunk(full_text)
83
84            for sub_idx, sub_chunk in enumerate(sub_chunks):
85                chunks.append(
86                    Chunk(
87                        id=sub_chunk.id,
88                        text=sub_chunk.text,
89                        metadata={
90                            "chunk_id": f"chunk_{global_chunk_idx}",
91                            "section": section_title,
92                            "source": self._source_name,
93                            "chunk_index": global_chunk_idx,
94                            "sub_chunk_index": sub_idx,
95                        },
96                    )
97                )
98                global_chunk_idx += 1
99
100        return chunks
101
102    @property
103    def version(self) -> str:
104        """Version tag for ingestion snapshot comparison."""
105        return "section-aware-v1"
106
107    @property
108    def config(self) -> dict[str, Any]:
109        """Inner chunker tunables for ingestion snapshots."""
110        return dict(self._inner.config)
