1"""
2Rate limiting configuration for the OmniCare backend.
3
4This module instantiates a SlowAPI ``Limiter`` that is applied globally to
5protect the backend from abuse and ensure fair resource usage across clients.
6
7Rate limit strategy:
8  - Key function: Extracts the user ID from the JWT token. If no valid user ID
9    can be extracted (unauthenticated), it falls back to the client IP address.
10  - Current limit: 20 requests/minute for chat endpoints, 10/minute for
11    session reset. These limits are conservative enough for normal human
12    interaction while blocking automated abuse.
13  - Storage: in-process counters by default, which is exact for a single
14    replica. A Redis-backed storage registered against the ``redis`` scheme
15    makes the counters global, so the 20/minute chat limit is 20/minute for the
16    deployment rather than for each instance.
17"""

45class RedisRateLimitStorage(Storage):
46    """Fixed-window rate-limit counters kept in Redis.
47
48    Registering against the ``redis`` scheme is what makes the ordinary
49    ``Limiter(storage_uri=...)`` wiring resolve here: ``limits`` picks a storage
50    class from the URI scheme, and the class in the package needs the ``redis``
51    client library, which this deployment deliberately does not carry. The
52    commands a fixed window actually needs -- INCRBY, EXPIRE, GET, PTTL, DEL,
53    PING and a SCAN loop for reset -- are spoken directly instead.
54
55    Only the fixed-window strategy is supported. SlowAPI selects it by default
56    and a windowed counter is the whole of what a per-minute quota needs, so
57    the moving-window and sliding-window interfaces stay unimplemented.
58
59    Args:
60        uri: The Redis URI SlowAPI was given.
61        wrap_exceptions: Whether to wrap storage errors in StorageError.
62
63    Returns:
64        None
65    """
66
67    STORAGE_SCHEME = ["redis", "rediss"]
68
69    # A counter is a few bytes, so a generous SCAN batch keeps the whole reset to
70    # one round trip for the key counts this service produces.
71    _SCAN_COUNT = 200
72    # Bounded so a pathological keyspace degrades to a failed reset rather than
73    # an unbounded loop.
74    _MAX_SCAN_PASSES = 1_000
75
76    def __init__(
77        self,
78        uri: str | None = None,
79        wrap_exceptions: bool = False,
80        **_: str | bool | float,
81    ) -> None:
82        parsed = urlsplit(uri or "redis://localhost:6379/0")
83        self._uri = uri or "redis://localhost:6379/0"
84        self._host = parsed.hostname or "localhost"
85        self._port = parsed.port or 6379
86        self._db = int((parsed.path or "/0").lstrip("/") or 0)
87        self._password = parsed.password or None
88        self._client: RedisClient | None = None
89        super().__init__(uri, wrap_exceptions=wrap_exceptions)
90
91    def _connection(self) -> RedisClient:
92        """Return the client, building it on first use so import stays cheap."""
93        if self._client is None:
94            self._client = RedisClient(
95                self._host,
96                self._port,
97                db=self._db,
98                password=self._password,
99                connect_timeout=settings.redis_connect_timeout_seconds,
100                socket_timeout=settings.redis_socket_timeout_seconds,
101            )
102        return self._client
103
104    def _namespaced(self, key: str) -> str:
105        return f"{_KEY_PREFIX}{key}"
106
107    @property
108    def base_exceptions(self) -> tuple[type[BaseException], ...]:
109        """The failures that mean Redis, not the caller, is the problem."""
110        return (RedisUnavailable, RedisProtocolError)
111
112    def incr(self, key: str, expiry: int, amount: int = 1) -> int:
113        """Count one hit in the current window and start the window if new.
114
115        Args:
116            key: The rate-limit key SlowAPI derived for this request.
117            expiry: Window length in seconds.
118            amount: The cost of the hit; the fixed-window strategy leaves it at 1.
119
120        Returns:
121            int: The number of hits recorded in the current window, including
122            this one.
123        """
124        client = self._connection()
125        prefixed = self._namespaced(key)
126        current = int(client.execute("INCRBY", prefixed, int(amount)))
127        if current == int(amount):
128            # The window starts with the first hit that lands in it.
129            client.execute("EXPIRE", prefixed, int(expiry))
130        return current
131
132    def get(self, key: str) -> int:
133        """Return the hits recorded in the current window.
134
135        Args:
136            key: The rate-limit key.
137
138        Returns:
139            int: The counter, or 0 when no window is open.
140        """
141        reply = self._connection().get(self._namespaced(key))
142        return 0 if reply is None else int(reply)
143
144    def get_expiry(self, key: str) -> float:
145        """Return when the current window for a key ends.
146
147        Args:
148            key: The rate-limit key.
149
150        Returns:
151            float: A POSIX timestamp. A key with no live window reports the
152            current time, matching the in-memory storage.
153        """
154        ttl_ms = int(self._connection().execute("PTTL", self._namespaced(key)))
155        if ttl_ms < 0:
156            return time.time()
157        return time.time() + ttl_ms / 1000.0
158
159    def clear(self, key: str) -> None:
160        """Close the window for a key.
161
162        Args:
163            key: The rate-limit key.
164
165        Returns:
166            None
167        """
168        self._connection().delete(self._namespaced(key))
169
170    def check(self) -> bool:
171        """Report whether Redis answers, which lets SlowAPI recover after an outage.
172
173        Returns:
174            bool: True when Redis responded to PING.
175        """
176        return self._connection().ping()
177
178    def reset(self) -> int:
179        """Clear every counter this service owns and report how many there were.
180
181        Only this module's namespace is scanned, so a shared Redis database is
182        never emptied by a reset.
183
184        Returns:
185            int: The number of counters removed.
186        """
187        client = self._connection()
188        keys = self._scan_namespace_keys()
189        client.delete(*keys)
190        return len(keys)
191
192    def _scan_namespace_keys(self) -> list[str]:
193        """Collect this service's keys with SCAN, which never blocks Redis."""
194        client = self._connection()
195        pattern = f"{_KEY_PREFIX}*"
196        cursor = "0"
197        keys: dict[str, None] = {}
198        for _ in range(self._MAX_SCAN_PASSES):
199            reply = client.execute(
200                "SCAN",
201                cursor,
202                "MATCH",
203                pattern,
204                "COUNT",
205                str(self._SCAN_COUNT),
206            )
207            next_cursor, batch = reply[0], reply[1]
208            cursor = next_cursor.decode() if isinstance(next_cursor, bytes) else str(next_cursor)
209            for key in batch or []:
210                keys[key.decode() if isinstance(key, bytes) else str(key)] = None
211            if cursor == "0":
212                break
213        return list(keys)

216def _user_or_ip_key_func(request):
217    """Extract the client IP address or user ID from the incoming request.
218
219    SlowAPI calls this function for every request to determine which rate
220    limit bucket to apply. We prioritize user ID from the token for authenticated
221    users to implement production-level rate limiting, and fall back to IP.
222
223    Args:
224        request: The ASGI request object.
225
226    Returns:
227        str: The rate limit key, e.g., 'user:UUID' or 'ip:127.0.0.1'.
228    """
229    token = None
230    # Check Authorization header first
231    auth_header = request.headers.get("Authorization")
232    if auth_header and auth_header.startswith("Bearer "):
233        token = auth_header.split(" ")[1]
234
235    # Check cookie fallback
236    if not token:
237        token = request.cookies.get("omnicare_access_token")
238
239    if token:
240        try:
241            from app.security.auth import _decode_token  # noqa: PLC0415
242
243            user_id = _decode_token(token)
244            return f"user:{str(user_id)}"
245        except Exception:  # noqa: S110
246            pass
247
248    return f"ip:{get_remote_address(request)}"

251def record_rate_limit_rejection(route: str) -> None:
252    """Report one rejected request to the telemetry store.
253
254    Rejections are the visible symptom of either abuse or a limit set too tight,
255    so they belong beside the cost and latency data an operator reads when the
256    service looks unhealthy. Kept as a module-level hook so the caller -- the
257    application's exception handler -- stays unaware of how telemetry is stored.
258
259    Args:
260        route: Request path that was refused.
261    """
262    get_metrics().record_limiter_rejection(
263        LimiterRejection(
264            timestamp=time.time(),
265            route=route,
266            request_id=get_request_id() or "-",
267        )
268    )

271def _build_limiter() -> Limiter:
272    """Build the SlowAPI limiter selected by the settings.
273
274    Redis-backed counters are only used when the deployment asks for them *and*
275    a Redis URI is configured; otherwise, and if the URI cannot be turned into a
276    storage, the limiter keeps its in-process counters. SlowAPI is also told to
277    swallow storage errors in the Redis case: a Redis outage should cost the
278    service its per-minute quota, not the availability of the chat endpoints
279    that quota exists to protect.
280
281    Returns:
282        Limiter: A limiter with global counters when Redis is configured, and
283        per-process counters otherwise.
284    """
285    if settings.rate_limit_storage == "redis" and settings.redis_enabled:
286        try:
287            return Limiter(
288                key_func=_user_or_ip_key_func,
289                storage_uri=settings.redis_url,
290                swallow_errors=True,
291            )
292        except Exception:
293            logger.warning(
294                "Redis rate-limit storage could not be initialised from REDIS_URL "
295                "'%s'; falling back to per-process counters.",
296                settings.redis_url,
297                exc_info=True,
298            )
299    return Limiter(key_func=_user_or_ip_key_func)

302# Module-level limiter instance. Import this and apply ``@limiter.limit()``
303# decorators to FastAPI route handlers to enforce rate limits.
304limiter = _build_limiter()
