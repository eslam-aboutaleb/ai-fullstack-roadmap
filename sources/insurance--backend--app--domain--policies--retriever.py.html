30class PolicyRetriever:
31    """Hybrid retrieval over the OmniCare policy chunk table.
32
33    Wraps ragnor's :class:`ragnor.retrieval.HybridRetriever`
34    with the policy chunk binding. Any retrieval failure is logged and reported as an empty
35    result set rather than raised.
36    """
37
38    def __init__(self, settings: Any = None, session_factory: Any = None) -> None:
39        """Initialize the policy retriever.
40
41        Args:
42            settings: Application settings. Defaults to
43                ``get_settings()``.
44            session_factory: Session provider the store opens
45                its transactions with. Defaults to the
46                application's ``async_session_factory``.
47        """
48        self._settings = settings or get_settings()
49        self._session_factory = session_factory or async_session_factory
50        self._retriever = HybridRetriever(
51            spec=POLICY_RETRIEVER_SPEC,
52            settings=self._settings,
53            session_factory=self._session_factory,
54        )
55
56    async def retrieve_hybrid(  # noqa: PLR0913
57        self,
58        query: str,
59        n_results: int = 5,
60        distance_threshold: float | None = None,
61    ) -> list[dict[str, Any]]:
62        """Run hybrid (vector + PostgreSQL full-text) search over policy chunks.
63
64        Args:
65            query: Natural language query.
66            n_results: Maximum number of results to return.
67            distance_threshold: Maximum L2 distance for the
68                vector search part. Defaults to
69                ``settings.rag_distance_threshold``.
70
71        Returns:
72            List of result dicts with ``document``,
73            ``metadata``, ``distance``, and ``_rrf_score`` keys.
74        """
75        if distance_threshold is None:
76            distance_threshold = self._settings.rag_distance_threshold
77        results = await self._retriever.retrieve(
78            query,
79            n_results=n_results,
80            distance_threshold=distance_threshold,
81        )
82        return as_dicts(results)
