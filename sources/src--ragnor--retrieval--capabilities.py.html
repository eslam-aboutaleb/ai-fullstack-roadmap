1"""Backend capability declarations and constraint validation.
2
3Storage backends declare their supported features and filter operators.
4When a retrieval or storage request specifies mandatory capability
5constraints that the target backend does not support, execution **fails
6closed** with :class:`CapabilityUnsupportedError` rather than silently
7ignoring constraints or returning incomplete/unfiltered results.
8"""
9
10from __future__ import annotations
11
12from collections.abc import Iterable, Mapping
13from dataclasses import dataclass, field
14from enum import StrEnum
15from typing import Any
16
17from ragnor.types import RetrievalError

20class CapabilityUnsupportedError(RetrievalError):
21    """Raised when a mandatory capability constraint is not supported by the backend.
22
23    Fail-closed behavior: unsupported security, filter, or transaction
24    guarantees reject the request immediately to prevent silent leakage
25    or incorrect results.
26    """
27
28
29class FilterOperator(StrEnum):
30    """Filter operators that a storage backend may support."""
31
32    EQ = "eq"
33    IN = "in"
34    GT = "gt"
35    GTE = "gte"
36    LT = "lt"
37    LTE = "lte"
38    NE = "ne"
39
40
41class BackendFeature(StrEnum):
42    """Functional features that a storage backend may declare."""
43
44    EQUALITY_FILTER = "equality_filter"
45    IN_FILTER = "in_filter"
46    RANGE_FILTER = "range_filter"
47    NATIVE_LEXICAL_SEARCH = "native_lexical_search"
48    APPROXIMATE_LEXICAL_SEARCH = "approximate_lexical_search"
49    DOCUMENT_FETCH = "document_fetch"
50    DOCUMENT_DELETE = "document_delete"
51    GENERATION_VISIBILITY = "generation_visibility"
52    ISOLATION_SCOPE = "isolation_scope"
53    TRANSACTION_GUARANTEE = "transaction_guarantee"
54
55
56@dataclass(frozen=True)
57class StoreCapabilities:
58    """Declared capabilities of a vector or lexical storage backend.
59
60    Attributes:
61        filter_operators: Set of supported :class:`FilterOperator` values.
62        features: Set of supported :class:`BackendFeature` values.
63    """
64
65    filter_operators: frozenset[str] = field(default_factory=frozenset)
66    features: frozenset[str] = field(default_factory=frozenset)
67
68    def supports_operator(self, operator: str | FilterOperator) -> bool:
69        """Return whether ``operator`` is supported."""
70        return str(operator) in self.filter_operators
71
72    def supports_feature(self, feature: str | BackendFeature) -> bool:
73        """Return whether ``feature`` is supported."""
74        return str(feature) in self.features
75
76    def validate_constraints(self, mandatory_constraints: Iterable[str]) -> None:
77        """Validate that all mandatory constraints are supported.
78
79        Fails closed: raises :class:`CapabilityUnsupportedError` on the
80        first unsupported constraint.
81
82        Args:
83            mandatory_constraints: Capability or feature names required
84                for safe query execution.
85
86        Raises:
87            CapabilityUnsupportedError: If any constraint is unsupported.
88        """
89        for constraint in mandatory_constraints:
90            norm = str(constraint)
91            is_supported = (
92                norm in self.features
93                or norm in self.filter_operators
94                or (
95                    norm == "transactions"
96                    and self.supports_feature(BackendFeature.TRANSACTION_GUARANTEE)
97                )
98                or (
99                    norm == "lexical"
100                    and (
101                        self.supports_feature(BackendFeature.NATIVE_LEXICAL_SEARCH)
102                        or self.supports_feature(BackendFeature.APPROXIMATE_LEXICAL_SEARCH)
103                    )
104                )
105            )
106            if not is_supported:
107                raise CapabilityUnsupportedError(
108                    f"Backend does not support required capability constraint: {constraint!r}. "
109                    "Failing closed to prevent unconstrained or leaky query execution."
110                )
111
112    def validate_filters(self, filters: Mapping[str, Any]) -> None:
113        """Validate that the operators implied by ``filters`` are supported.
114
115        If a filter value is a list/tuple/set, requires :attr:`FilterOperator.IN`.
116        Otherwise requires :attr:`FilterOperator.EQ`.
117
118        Raises:
119            CapabilityUnsupportedError: If a required operator is unsupported.
120        """
121        for key, value in filters.items():
122            if isinstance(value, (list, tuple, set, frozenset)):
123                if not self.supports_operator(FilterOperator.IN):
124                    raise CapabilityUnsupportedError(
125                        f"Backend does not support IN filters for field {key!r}. "
126                        "Failing closed to prevent dropping mandatory filter."
127                    )
128            elif not self.supports_operator(FilterOperator.EQ):
129                raise CapabilityUnsupportedError(
130                    f"Backend does not support equality filters for field {key!r}."
131                )

168def get_backend_capabilities(backend: Any) -> StoreCapabilities:
169    """Extract declared capabilities from ``backend`` or infer defaults.
170
171    Args:
172        backend: A vector index, lexical index, document store, or vector store.
173
174    Returns:
175        The declared or inferred :class:`StoreCapabilities`.
176    """
177    if hasattr(backend, "capabilities") and isinstance(backend.capabilities, StoreCapabilities):
178        return backend.capabilities
179
180    # Infer capabilities from type or attributes
181    type_name = type(backend).__name__.lower()
182    if "memory" in type_name:
183        return DEFAULT_MEMORY_CAPABILITIES
184    if "pgvector" in type_name:
185        return DEFAULT_PGVECTOR_CAPABILITIES
186
187    # Default fallback: minimal equality filter & fetch/delete if methods exist
188    features: set[str] = {BackendFeature.EQUALITY_FILTER}
189    if hasattr(backend, "get"):
190        features.add(BackendFeature.DOCUMENT_FETCH)
191    if hasattr(backend, "delete"):
192        features.add(BackendFeature.DOCUMENT_DELETE)
193    if hasattr(backend, "keyword_search"):
194        features.add(BackendFeature.APPROXIMATE_LEXICAL_SEARCH)
195
196    return StoreCapabilities(
197        filter_operators=frozenset({FilterOperator.EQ}),
198        features=frozenset(features),
199    )
