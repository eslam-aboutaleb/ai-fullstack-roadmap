68@limiter.limit("5/minute")
69async def evaluate_rag(
70    payload: RagEvalRequest,
71    request: Request,
72    current_user_id: str = Depends(get_current_user),
73    _admin: str = Depends(require_role("admin")),
74) -> RagEvalResponse:
75    """Run RAG evaluation pipeline and return structured metrics."""
76    try:
77        category_enum = None
78        if payload.category:
79            try:
80                category_enum = QuestionCategory(payload.category.lower())
81            except ValueError:
82                valid = [c.value for c in QuestionCategory]
83                raise HTTPException(
84                    status_code=status.HTTP_422_UNPROCESSABLE_ENTITY,
85                    detail=f"Invalid category '{payload.category}'. Valid: {valid}",
86                ) from None
87
88        difficulty_enum = None
89        if payload.difficulty:
90            try:
91                difficulty_enum = Difficulty(payload.difficulty.lower())
92            except ValueError:
93                valid = [d.value for d in Difficulty]
94                raise HTTPException(
95                    status_code=status.HTTP_422_UNPROCESSABLE_ENTITY,
96                    detail=f"Invalid difficulty '{payload.difficulty}'. Valid: {valid}",
97                ) from None
98
99        config = EvalConfig(
100            mode=payload.mode,
101            judge=payload.judge,
102            n_results=payload.n_results,
103            category=category_enum,
104            difficulty=difficulty_enum,
105        )
106
107        result = await run_evaluation(config)
108
109        return RagEvalResponse(
110            timestamp=result.timestamp,
111            config=result.config,
112            retrieval_metrics=result.retrieval_metrics,
113            answer_metrics=result.answer_metrics,
114            per_sample_results=result.per_sample_results,
115            summary=result.summary,
116            duration_seconds=result.duration_seconds,
117        )
118
119    except HTTPException:
120        raise
121    except Exception as exc:
122        logger.exception("RAG evaluation failed: %s", exc)
123        raise HTTPException(
124            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
125            detail="RAG evaluation failed. See server logs for details.",
126        ) from exc
