1"""Generic ingestion pipeline.
2
3The pipeline orchestrates: load source, hash the
4source manifest, compare the ingestion snapshot,
5chunk, embed, and validate -- outside any
6transaction -- then retire the old version and
7upsert the new chunks inside one short
8transaction, serialized by an advisory lock taken
9through the SQL factory when a lock key is
10supplied.
11
12Splitting the work into a prepare phase and a
13short critical section keeps a slow remote
14embedding request from holding a transaction and
15an advisory lock open: the embedding happens
16before any session is opened for the version
17mutations, and the critical section re-checks the
18snapshot under the lock (double-checked locking)
19so a concurrent process that ingested meanwhile
20still wins.
21
22Version *storage* is application-side: the pipeline talks
23to a :class:`VersionStore` protocol, which the host
24application implements against its version tables.
25"""

103    def __init__(  # noqa: PLR0913, PLR0917
104        self,
105        store: VectorStore,
106        embedder: EmbeddingFunction,
107        chunker: Chunker,
108        session_factory: SessionProvider,

114        """Initialize the pipeline.
115
116        Args:
117            store: Vector store the chunks are upserted into.
118            embedder: Embedding function for chunk texts.
119            chunker: Chunker that splits document content.
120            session_factory: Session provider the pipeline
121                owns its transactions with.
122            settings: Settings carrying the embedding model.
123            version_id_key: Metadata key the version id is

127                stamped under (the target table's source
128                column name).
129            sql_factory: SQL factory used for advisory
130                locking and session creation; defaults to
131                the SQLAlchemy factory. Inject a custom
132                :class:`ragnor.db.sql_factory.SQLFactory`
133                to run the pipeline on another stack.
134        """
135        self._store = store
136        self._embedder = embedder
137        self._chunker = chunker
138        self._session_factory = session_factory
139        self._settings = settings
140        self._version_id_key = version_id_key
141        self._source_id_key = source_id_key
142        self._sql_factory = sql_factory
143        self._last_load_result = None

145    async def run(  # noqa: PLR0913, PLR0911
146        self,
147        source: DocumentSource,
148        version_store: VersionStore,
149        lock_key: str | None = None,
150        session: Any = None,
151        purge_on_empty: bool = False,
152    ) -> int:
153        """Ingest the source and return the stored chunk count.
154
155        The run has two phases. The *prepare* phase opens
156        no transaction and takes no lock: it loads the
157        source, hashes the source manifest, compares the
158        snapshot (an unchanged source skips without
159        embedding), chunks, embeds, and validates. The
160        *critical section* is short: it re-checks the
161        snapshot under the advisory lock (double-checked
162        locking -- a concurrent process may have ingested
163        the source while the embedding request was in
164        flight), retires the old version, creates the new
165        one, and upserts the chunks.
166
167        The pipeline owns its session -- and commits it --
168        when a ``lock_key`` is supplied: the advisory lock
169        serializes concurrent ingestion of the same source
170        across instances. A caller-supplied ``session`` is
171        never committed; the caller owns its transaction
172        (and its locking).
173
174        Args:
175            source: Document source to load from.
176            version_store: Application-side version storage.
177            lock_key: Advisory-lock key (the source path).
178                When supplied, the pipeline opens its own
179                session, locks it, and commits it.
180            session: Caller-supplied session. When supplied,
181                the pipeline runs on it without committing.
182            purge_on_empty: Treat an empty load as
183                authoritative. When ``True``, an empty
184                source retires the active version and
185                deletes its chunks from the search index,
186                so deleted content stops being searchable;
187                when ``False`` (the default), an empty
188                load is a historical no-op.
189
190        Returns:
191            The number of chunks stored, or the stored
192            chunk count when the source was unchanged.
193        """
194        if hasattr(source, "load"):
195            loaded = await source.load()
196        else:
197            loaded = source
198        if isinstance(loaded, LoadResult):
199            documents = list(loaded)
200            self._last_load_result = loaded
201            if not loaded.authoritative:
202                logger.warning(
203                    "Non-authoritative source scan; skipping ingestion without mutation."
204                )
205                return 0
206        else:
207            documents = list(loaded) if not isinstance(loaded, list) else loaded
208            self._last_load_result = None
209
210        sql_factory = None
211        if lock_key is not None:
212            # Imported lazily so importing the pipeline does
213            # not require a database driver; the default is
214            # the SQLAlchemy factory.
215            from ragnor.db.sql_factory import SQLAlchemyFactory  # noqa: PLC0415
216
217            sql_factory = self._sql_factory or SQLAlchemyFactory()
218
219        if not documents:
220            if not purge_on_empty:
221                logger.warning("No documents loaded from source; skipping ingestion.")
222                return 0
223            # An authoritative empty dataset: retire the
224            # active version so deleted content stops being
225            # searchable.
226            async with self._transacting_session(lock_key, sql_factory, session) as purge_session:
227                await self._retire_active(purge_session, version_store)
228            return 0
229
230        source_hash = compute_source_hash(documents)
231        extractor = getattr(source, "extractor", None)
232        if extractor is None and documents and documents[0].extractor:
233            first_ext = documents[0].extractor
234            if all(doc.extractor == first_ext for doc in documents):
235                extractor = first_ext
236        current_snapshot = build_snapshot(
237            source_hash,
238            self._settings,
239            self._chunker,
240            extractor=extractor,
241            embedding_profile=getattr(self._embedder, "profile", None),
242        )
243
244        # Prepare phase: no transaction, no lock. The
245        # read-only snapshot check runs on a short-lived
246        # session (or read-only on the caller's session),
247        # so an unchanged source skips without embedding.
248        if session is not None:
249            # Caller-supplied session: the caller owns the
250            # transaction, so the pipeline never commits it.
251            active = await version_store.find_active(session)
252            if snapshot_matches(current_snapshot, active.snapshot if active else None):
253                return await self._skip_unchanged(session, current_snapshot)
254        else:
255            async with self._session_factory() as check_session:
256                active = await version_store.find_active(check_session)
257                if snapshot_matches(current_snapshot, active.snapshot if active else None):
258                    return await self._skip_unchanged(check_session, current_snapshot)
259
260        chunks = self._chunk(documents)
261        if not chunks:
262            # Abort before touching the version table or the
263            # search index: a version row carrying the current
264            # snapshot would make every later run skip this
265            # source forever (the snapshot would compare equal),
266            # leaving it permanently un-ingested. Aborting here
267            # keeps the previous version active so the index
268            # still reflects it and the next run retries.
269            logger.warning("No chunks extracted from source. Ingestion aborted.")
270            return 0
271
272        # Embedding and validation run outside any transaction
273        # and outside the advisory lock: a slow remote call
274        # must not hold either open.
275        embeddings = await self._embedder([chunk.text for chunk in chunks])
276        embedding_dim = current_snapshot.embedding_dim
277        documents_to_upsert: list[dict[str, Any]] = []
278        for chunk, embedding in zip(chunks, embeddings, strict=True):
279            validate_embedding(embedding, expected_dim=embedding_dim, label="chunk embedding")
280            documents_to_upsert.append(
281                {
282                    "id": chunk.id,
283                    "text": chunk.text,
284                    "metadata": chunk.metadata,
285                    "embedding": embedding,
286                }
287            )
288
289        # Critical section: short. The snapshot is re-checked
290        # under the lock -- a concurrent process may have
291        # ingested this source while the embedding request
292        # was in flight.
293        async with self._transacting_session(lock_key, sql_factory, session) as critical_session:
294            return await self._commit_ingestion(
295                critical_session, current_snapshot, documents_to_upsert, version_store
296            )

298    @asynccontextmanager
299    async def _transacting_session(
300        self,
301        lock_key: str | None,
302        sql_factory: SQLFactory | None,
303        session: Any,
304    ) -> AsyncIterator[Any]:
305        """Yield the session a critical section transacts on.
306
307        A caller-supplied session is yielded as-is -- the
308        caller owns its transaction, so it is never
309        committed -- and the advisory lock is taken on it
310        when a lock key is supplied. Otherwise the
311        pipeline opens its own session, takes the advisory
312        lock on it when a lock key is supplied, and
313        commits it when the block succeeds.
314
315        Args:
316            lock_key: Advisory-lock key, or ``None``.
317            sql_factory: SQL factory for the advisory lock,
318                or ``None`` when no lock key was supplied.
319            session: Caller-supplied session, or ``None``.
320
321        Yields:
322            The session the critical section runs on.
323        """
324        if session is not None:
325            # Caller-supplied session: the caller owns the
326            # transaction, so the pipeline never commits it.
327            # The advisory lock is still taken on the
328            # caller's session when a lock key is supplied.
329            if sql_factory is not None:
330                async with sql_factory.advisory_lock(session, lock_key):
331                    yield session
332            else:
333                yield session
334            return
335
336        if sql_factory is not None:
337            async with (
338                self._session_factory() as owned_session,
339                sql_factory.advisory_lock(owned_session, lock_key),
340            ):
341                yield owned_session
342                await owned_session.commit()
343            return
344
345        async with self._session_factory() as owned_session:
346            yield owned_session
347            await owned_session.commit()

349    async def _retire_active(
350        self,
351        session: Any,
352        version_store: VersionStore,
353    ) -> None:
354        """Close the active version and delete its chunks (no commit).
355
356        Args:
357            session: Session to run the retirement on.
358            version_store: Application-side version storage.
359        """
360        active = await version_store.find_active(session)
361        if active is None:
362            return
363        await version_store.close_active(session)
364        await version_store.delete_chunks(session, active.version_id)

377        Returns:
378            The stored chunk count.

381        logger.info(

388    async def _commit_ingestion(  # noqa: PLR0913
389        self,
390        session: Any,
391        current_snapshot: IngestionSnapshot,
392        documents_to_upsert: list[dict[str, Any]],
393        version_store: VersionStore,
394    ) -> int:
395        """Run the version-table mutations and the upsert on ``session`` (no commit).
396
397        The snapshot is re-checked first (double-checked
398        locking): a concurrent process may have ingested
399        this source while the embedding request was in
400        flight, in which case the critical section skips
401        without touching the version table or the index.
402
403        Args:
404            session: Session to run the transaction on.
405            current_snapshot: The snapshot to record.
406            documents_to_upsert: Validated chunk documents
407                (id, text, metadata, embedding).
408            version_store: Application-side version storage.
409
410        Returns:
411            The number of chunks stored, or the stored chunk
412            count when a concurrent run ingested the source.
413        """
414        active = await version_store.find_active(session)
415        if snapshot_matches(current_snapshot, active.snapshot if active else None):
416            return await self._skip_unchanged(session, current_snapshot)
417
418        if active is not None:
419            # Close the old active version and remove its
420            # chunks from the search index. chunk_id is
421            # position-based, so the new version's chunks
422            # would violate the unique constraint on chunk_id
423            # if the retired version's chunks stayed. The
424            # index always reflects the current version;
425            # version history itself remains in the version
426            # table.
427            await version_store.close_active(session)
428            await version_store.delete_chunks(session, active.version_id)
429
430        version = await version_store.create_version(session, current_snapshot)
431
432        for document in documents_to_upsert:
433            document["metadata"][self._version_id_key] = version.version_id
434            if version.source_id is not None:
435                document["metadata"][self._source_id_key] = version.source_id
436
437        await self._store.upsert(documents_to_upsert, session=session)
438        return len(documents_to_upsert)

440    def _chunk(self, documents: list[RawDocument]) -> list[Chunk]:
441        """Chunk the loaded documents, stamping each with its source name.
442
443        The source name comes from the loaded document, not
444        from the chunker, so a multi-document source stamps
445        every chunk with the document it actually came from.
446        """
447        chunks: list[Chunk] = []
448        for index, document in enumerate(documents):
449            document_chunks = self._chunker.chunk(document.content)
450            for chunk in document_chunks:
451                if len(documents) > 1:
452                    chunk.id = f"document_{index}:{chunk.id}"
453                chunk.metadata = {**document.metadata, **chunk.metadata}
454                chunk.metadata["source"] = document.source_name
455                if document.external_id is not None:
456                    chunk.metadata["external_id"] = document.external_id
457                if document.provenance:
458                    chunk.metadata["provenance"] = dict(document.provenance)
459                if document.location is not None:
460                    chunk.metadata["location"] = document.location
461                if document.extractor is not None:
462                    chunk.metadata["extractor"] = document.extractor
463            chunks.extend(document_chunks)
464        return chunks
