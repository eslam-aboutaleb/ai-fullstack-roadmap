35class EmbeddingFactory:
36    """
37    Factory for creating embedding functions.
38
39    The factory reads the embedding configuration from the centralized
40    Pydantic settings and returns a ready-to-use embedding function
41    instance. This decouples the embedding provider choice from the
42    rest of the RAG pipeline.
43    """
44
45    @classmethod
46    def get_embedding_function(cls) -> EmbeddingFunction:
47        """Create and return an embedding function.
48
49        Delegates to the ragnor embedding registry, which selects the
50        provider from ``settings.embedding_provider`` (defaulting to
51        the LiteLLM provider) and configures it with the API key and
52        model name from the application settings.
53
54        Returns:
55            An embedding function instance ready for use with the
56            retriever and ingestion pipelines.
57        """
58        return _get_embedding_function(get_settings())

61def get_vector_store(
62    table_name: str,
63    id_field: str = "id",
64    embedding_dim: int | None = None,
65) -> VectorStore:
66    """Factory function to create a vector store instance.
67
68    Args:
69        table_name: Name of the database table.
70        id_field: Name of the ID column.
71        embedding_dim: Dimension of the embedding vectors. Defaults
72            to the dimension configured for the active embedding model.
73
74    Returns:
75        Configured VectorStore instance based on provider setting.
76    """
77    settings = get_settings()
78    return _ragnor_get_vector_store(
79        table_name=table_name,
80        id_field=id_field,
81        embedding_dim=(embedding_dim if embedding_dim is not None else get_embedding_dimension()),
82        settings=settings,
83        session_factory=async_session_factory,
84        embedding_fn=EmbeddingFactory.get_embedding_function(),
85    )
