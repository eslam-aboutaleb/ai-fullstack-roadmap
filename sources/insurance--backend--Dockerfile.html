1# syntax=docker/dockerfile:1
2# ==============================================================================
3# OmniCare Financial Backend - Production Dockerfile
4# ==============================================================================
5# Build context: the REPOSITORY ROOT (see docker-compose.yml), not ./backend.
6# The generic RAG machinery comes from the published ``ragnor``
7# package, pinned to version 0.4.1 with the pgvector extra in
8# backend/pyproject.toml.
9# ==============================================================================
10
11FROM python:3.11-slim
12
13# Set environment variables for Python runtime optimization
14ENV PYTHONUNBUFFERED=1 \
15    PYTHONDONTWRITEBYTECODE=1 \
16    UV_NO_CACHE=1 \
17    UV_PYTHON_PREFER_SYSTEM=1
18
19# Install system dependencies required to build Python packages
20RUN apt-get update && apt-get install -y --no-install-recommends \
21    build-essential \
22    curl \
23    && rm -rf /var/lib/apt/lists/*
24
25# Install UV package manager (pinned version; the image uses
26# uv exclusively for dependency management; pip is removed below).
27# The versioned installer URL pins both the installer script and
28# the uv version it installs.
29RUN curl -LsSf https://github.com/astral-sh/uv/releases/download/0.12.23/uv-installer.sh | sh
30ENV PATH="/root/.local/bin:$PATH"
31
32WORKDIR /app
33
34# Copy dependency manifest and install the backend package.
35# The pinned ragnor distribution is resolved from PyPI
36# as part of this install.
37COPY backend/pyproject.toml ./
38RUN uv pip install --system .
39# Ensure pip is entirely removed
40RUN rm -rf /usr/local/bin/pip* /usr/local/lib/python*/site-packages/pip*
41
42# Copy backend application source code
43COPY backend/app/ ./app/
44COPY backend/tests/ ./tests/
45COPY backend/alembic.ini ./
46COPY backend/alembic/ ./alembic/
47COPY .env.example ./.env.example
48
49# Create a non-root user to run the application
50RUN groupadd -r appuser && useradd -r -g appuser -d /app -s /sbin/nologin appuser \
51    && chown -R appuser:appuser /app
52
53USER appuser
54
55# Expose backend API port
56EXPOSE 8000
57
58# Start FastAPI application using Uvicorn server
59CMD ["uvicorn", "app.main:app", "--host", "0.0.0.0", "--port", "8000"]
