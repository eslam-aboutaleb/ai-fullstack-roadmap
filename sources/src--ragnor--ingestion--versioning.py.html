1"""Ingestion snapshot comparison.
2
3The snapshot captures everything that determines
4whether a stored index is still current: the source
5hash, the embedding model and dimension, and the
6chunker version and tunables. Comparing the current
7snapshot against the stored one makes ingestion
8idempotent -- an unchanged source is skipped instead
9of re-embedded and re-upserted.
10
11The source hash is a *manifest* hash: every
12document's source name, external id, and content
13hash are recorded in a canonical JSON manifest, so
14document boundaries, source names, external ids, and
15content all feed the hash. Concatenating document
16content (``"ab" + "c"`` vs ``"a" + "bc"``) is not
17enough to change it -- the manifest is.
18
19Version *storage* is application-side; this module
20defines the value type and the comparison only.
21"""
22

41class IngestionSnapshot:

43
44    Attributes:

74def snapshot_matches(
75    current: IngestionSnapshot,
76    stored: IngestionSnapshot | None,
77) -> bool:
78    """Return whether ``current`` equals the ``stored`` snapshot.
79
80    Args:
81        current: Snapshot computed from the source and the
82            current configuration.
83        stored: Snapshot recorded with the active version,
84            or ``None`` when no version exists yet.
85
86    Returns:
87        ``True`` when the stored index is still current
88        (``False`` when there is nothing stored).
89    """
90    if stored is None:
91        return False
92    return current == stored

95def compute_source_hash(documents: list[RawDocument]) -> str:
96    """Compute the manifest hash of a loaded document set.
97
98    The hash covers the whole manifest -- every
99    document's source name, external id, and content
100    hash -- serialized as canonical JSON (sorted
101    keys, no whitespace) before hashing. Document
102    boundaries, source names, external ids, and
103    content all feed the result, so a changed
104    document identity or metadata produces a
105    different hash even when the concatenated
106    content is unchanged.
107
108    Args:
109        documents: Loaded documents in source order.
110
111    Returns:
112        SHA-256 hex digest of the canonical manifest.
113    """
114    entries = []
115    for doc in documents:
116        entry = {
117            "source_name": doc.source_name,
118            "external_id": doc.external_id,
119            "content_hash": hashlib.sha256(doc.content.encode("utf-8")).hexdigest(),
120        }
121        if doc.extractor is not None:
122            entry["extractor"] = doc.extractor
123        if "extraction" in doc.metadata:
124            entry["extraction"] = safe_config(doc.metadata["extraction"])
125        if doc.metadata:
126            entry["metadata"] = safe_config(doc.metadata)
127        if doc.location is not None:
128            entry["location"] = safe_config(doc.location)
129        if doc.provenance:
130            entry["provenance"] = safe_config(doc.provenance)
131        entries.append(entry)
132    manifest = {"documents": entries}
133    canonical = json.dumps(manifest, sort_keys=True, separators=(",", ":"))
134    return hashlib.sha256(canonical.encode("utf-8")).hexdigest()

137def build_snapshot(  # noqa: PLR0913, PLR0917
138    source_hash: str,
139    settings: RagSettings,
140    chunker: Chunker,
141    extractor: Any = None,
142    extractor_id: str | None = None,
143    extractor_version: str | None = None,
144    extractor_options: dict[str, Any] | None = None,
145    embedding_profile: Any | None = None,
146) -> IngestionSnapshot:
147    """Build the current snapshot for ``source_hash``.
148
149    The embedding model and dimension come from the
150    settings (the dimension via the ragnor dimension
151    registry); the chunker version and tunables come
152    from the chunker, so a chunker upgrade changes the
153    snapshot and forces a re-ingest.
154
155    Args:
156        source_hash: SHA-256 hex digest of the source content.
157        settings: Settings carrying the embedding model.
158        chunker: Chunker that splits the document.
159        extractor: Optional extractor instance or string id.
160        extractor_id: Optional extractor identifier (overridden by
161            ``extractor`` when provided).
162        extractor_version: Optional extractor version.
163        extractor_options: Optional non-secret extractor options.
164        embedding_profile: Optional
165            :class:`~ragnor.embeddings.profile.EmbeddingProfile`.
166            When provided its :meth:`canonical_hash` is stored in the
167            snapshot so profile changes force a re-ingest.
168
169    Returns:
170        The snapshot describing this ingestion configuration.
171    """
172    config: dict[str, Any] = chunker.config
173    if extractor is not None:
174        extractor_id = (
175            extractor
176            if isinstance(extractor, str)
177            else getattr(extractor, "id", type(extractor).__name__)
178        )
179        extractor_version = getattr(extractor, "version", extractor_version)
180        extractor_options = getattr(extractor, "options", extractor_options)
181    profile_hash = ""
182    if embedding_profile is not None and hasattr(embedding_profile, "canonical_hash"):
183        profile_hash = embedding_profile.canonical_hash()
184    return IngestionSnapshot(
185        source_hash=source_hash,
186        embedding_model=(
187            embedding_profile.model_id
188            if embedding_profile is not None
189            else settings.embedding_model
190        ),
191        embedding_dim=(
192            embedding_profile.dimension
193            if embedding_profile is not None
194            else get_embedding_dimension(settings.embedding_model)
195        ),
196        chunker_version=chunker.version,
197        chunk_size=int(config.get("chunk_size", 0)),
198        overlap=int(config.get("overlap", 0)),
199        retrieval_schema_version=RETRIEVAL_SCHEMA_VERSION,
200        extractor_id=extractor_id or "none",
201        extractor_version=extractor_version or "0",
202        extractor_options=safe_config(extractor_options or {}),
203        embedding_profile_hash=profile_hash,
204    )
