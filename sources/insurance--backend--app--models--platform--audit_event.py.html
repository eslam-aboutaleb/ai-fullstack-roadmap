1"""
2SQLAlchemy ORM model for the ``audit_events`` table.
3
4The append-only record of who did what to which entity. ``claims.status`` and
5``claim_events`` explain how a claim reached its current state; this table
6answers the wider question a regulated insurer is actually asked: who
7submitted, signed in, confirmed, withdrew, and rated, and when. The row is
8written once and never updated -- an entry in an audit trail that can be
9edited is not an audit trail -- so there is no update path anywhere in the
10application.
11
12The actor reference is released (``SET NULL``) rather than enforced, matching
13``claim_events``: removing an account must not delete the history of what that
14account did, and a transition the system performed has no person to name.
15"""
16
17from __future__ import annotations
18
19import uuid
20from datetime import datetime
21
22from sqlalchemy import DateTime, ForeignKey, Index, String, func
23from sqlalchemy.dialects.postgresql import JSONB, UUID
24from sqlalchemy.orm import Mapped, mapped_column
25
26from app.models.base import Base

28
29class AuditEvent(Base):
30    """Database model for one immutable entry in the audit trail.
31
32    Attributes:
33        id: Primary key UUID.
34        actor_id: Who performed the action. Nullable so a system or anonymous
35            action (a failed sign-in attempt) can be recorded without
36            attributing it to a person, and released by the database when the
37            actor's account is removed.
38        action: The verb that was performed, e.g. ``"auth.signin"``,
39            ``"claim.status_changed"``.
40        entity_type: The kind of thing acted on, e.g. ``"user"``, ``"claim"``,
41            ``"answer_feedback"``.
42        entity_id: The identifier of the thing acted on. Human-readable where
43            the entity has one (a claim id), a UUID string where it does not.
44        before: JSONB snapshot of the fields that changed before the action,
45            or None when the action did not overwrite anything.
46        after: JSONB snapshot of the fields after the action, or None.
47        ip_address: Client IP the action was performed from, when known.
48        user_agent: Caller's user agent, when supplied.
49        correlation_id: The request's ``X-Request-ID``, so an entry can be
50            read beside its own line in the application log. Null when the
51            action ran outside a request (a startup job, a direct call).
52        created_at: Timestamp the row was written, generated by the database so
53            an entry written by a worker with a skewed clock still orders
54            correctly against its neighbours.
55    """
56
57    __tablename__ = "audit_events"
58
59    id: Mapped[uuid.UUID] = mapped_column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
60    actor_id: Mapped[uuid.UUID | None] = mapped_column(
61        UUID(as_uuid=True),
62        ForeignKey("users.id", ondelete="SET NULL"),
63        nullable=True,
64        index=True,
65    )
66    action: Mapped[str] = mapped_column(String(64), nullable=False, index=True)
67    entity_type: Mapped[str] = mapped_column(String(64), nullable=False)
68    entity_id: Mapped[str | None] = mapped_column(String(128), nullable=True)
69    before: Mapped[dict | None] = mapped_column(JSONB, nullable=True)
70    after: Mapped[dict | None] = mapped_column(JSONB, nullable=True)
71    ip_address: Mapped[str | None] = mapped_column(String(64), nullable=True)
72    user_agent: Mapped[str | None] = mapped_column(String(512), nullable=True)
73    correlation_id: Mapped[str | None] = mapped_column(String(64), nullable=True)
74    created_at: Mapped[datetime] = mapped_column(
75        DateTime(timezone=True), nullable=False, server_default=func.now()
76    )
77
78    __table_args__ = (
79        # The admin trail filters on the pair (entity type, entity id) -- "show
80        # me everything that ever happened to this claim" -- so both columns
81        # belong in one index rather than two.
82        Index("ix_audit_events_entity", "entity_type", "entity_id"),
83        # The listing is paged by newest first, so the ordering column is
84        # indexed for the same reason.
85        Index("ix_audit_events_created_at", "created_at"),
86    )
