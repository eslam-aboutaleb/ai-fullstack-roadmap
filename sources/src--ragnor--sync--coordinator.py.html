193        return hashlib.sha256(canonical.encode()).hexdigest()

221    @staticmethod
222    def _group_documents(
223        documents: Sequence[RawDocument],
224    ) -> list[tuple[RawDocument, list[RawDocument]]]:
225        groups: dict[str, list[RawDocument]] = {}
226        for doc in documents:
227            groups.setdefault(doc.external_id or doc.source_name, []).append(doc)
228        grouped = []
229        for artifacts in groups.values():
230            if len(artifacts) == 1:
231                grouped.append((artifacts[0], artifacts))
232                continue
233            locations = [json.dumps(safe_config(doc.location), sort_keys=True) for doc in artifacts]
234            if any(doc.location is None for doc in artifacts) or len(set(locations)) != len(
235                locations
236            ):
237                raise ValueError("Duplicate document artifacts require distinct locations")
238            ordered_artifacts = sorted(
239                artifacts,
240                key=lambda doc: (
241                    (0, doc.location["page"])
242                    if isinstance(doc.location, dict) and isinstance(doc.location.get("page"), int)
243                    else (1, json.dumps(safe_config(doc.location), sort_keys=True))
244                ),
245            )
246            grouped.append(
247                (
248                    replace(
249                        ordered_artifacts[0],
250                        content="\n\n".join(doc.content for doc in ordered_artifacts),
251                    ),
252                    ordered_artifacts,
253                )
254            )
255        return grouped

257    async def plan(
258        self, scope: str, documents: Sequence[RawDocument], is_authoritative: bool = True
259    ) -> SyncPlan:
260        plan = SyncPlan()
261        seen_ids: set[str] = set()
262        for doc, artifacts in self._group_documents(documents):
263            doc_id = doc.external_id or doc.source_name
264            seen_ids.add(doc_id)
265            chash = (
266                self._compute_hash(doc.content)
267                if len(artifacts) == 1
268                else self._fingerprint(
269                    [[artifact.location, artifact.content] for artifact in artifacts]
270                )
271            )
272            rev = (
273                doc.metadata.get("revision")
274                or doc.metadata.get("etag")
275                or doc.metadata.get("version")
276            )
277            existing = await self._manifest_store.get(scope, doc_id)
278            if existing is None or existing.status == ManifestStatus.TOMBSTONED:
279                action = "added"
280            elif (
281                existing.status != ManifestStatus.ACTIVE
282                or existing.chunker_fingerprint != self._chunker_fingerprint()
283                or existing.embedding_fingerprint != self._embedding_fingerprint()
284                or existing.extractor_fingerprint != self._extractor_fingerprint(doc)
285                or existing.content_hash != chash
286            ):
287                action = "changed"
288            elif existing.metadata != doc.metadata or existing.provenance != doc.provenance:
289                action = "metadata_only"
290            else:
291                action = "unchanged"
292            item = SyncPlanItem(
293                doc_id, action, doc, chash, rev, artifacts, existing.fence_token if existing else 0
294            )
295            getattr(plan, action).append(item)
296        if is_authoritative:
297            for rec in await self._manifest_store.list(scope):
298                if rec.status != ManifestStatus.TOMBSTONED and rec.document_id not in seen_ids:
299                    plan.removed.append(
300                        SyncPlanItem(rec.document_id, "removed", expected_fence=rec.fence_token)
301                    )
302        return plan

319    async def _heartbeat(self, token: FenceToken) -> None:
320        while True:
321            await asyncio.sleep(self._lease_ttl_seconds / 3)
322            try:
323                renewed = await self._state_store.renew_lease(token, self._lease_ttl_seconds)
324            except Exception:
325                logger.warning("Lease renewal failed; activation still requires a valid fence")
326                return
327            if renewed is None:
328                return

330    async def _prepare_item(
331        self,
332        scope: str,
333        item: SyncPlanItem,
334        generation: int,
335        token: FenceToken,
336        existing_vectors: list[list[float]] | None = None,
337    ) -> tuple[ManifestRecord, list[dict[str, Any]]]:
338        doc = item.raw_document
339        assert doc is not None
340        chunks = []
341        for artifact in item.artifacts or [doc]:
342            for chunk in self._chunker.chunk(artifact.content):
343                metadata = {
344                    **artifact.metadata,
345                    **chunk.metadata,
346                }
347                if artifact.provenance:
348                    metadata["provenance"] = dict(artifact.provenance)
349                if artifact.location is not None:
350                    metadata["location"] = artifact.location
351                    if isinstance(artifact.location, dict):
352                        metadata.update(artifact.location)
353                # Location and source metadata cannot replace coordinator-owned fields.
354                metadata.update(
355                    scope=scope,
356                    doc_id=item.document_id,
357                    generation=generation,
358                    active=False,
359                    source=artifact.source_name,
360                )
361                if artifact.external_id is not None:
362                    metadata["external_id"] = artifact.external_id
363                chunks.append({"text": chunk.text, "metadata": metadata})
364        if not chunks:
365            raise SyncError("Document produced no chunks; retaining the previous active version")
366        profile = getattr(self._embedder, "profile", None)
367        dim = (
368            profile.dimension
369            if profile is not None
370            else get_embedding_dimension(self._settings.embedding_model)
371        )
372        embeddings = list(existing_vectors or [])
373        if existing_vectors is None:
374            fn = self._embedder
375            for offset in range(0, len(chunks), self._batch_size):
376                batch = chunks[offset : offset + self._batch_size]
377                vectors = await fn([chunk["text"] for chunk in batch])
378                if len(vectors) != len(batch):
379                    raise ValueError("Embedding response count does not match chunk count")
380                embeddings.extend(vectors)
381        if len(embeddings) != len(chunks):
382            raise ValueError("Stored embedding count does not match unchanged document chunks")
383        nonce = uuid4().hex
384        for index, (chunk, vector) in enumerate(zip(chunks, embeddings, strict=True)):
385            validate_embedding(vector, expected_dim=dim)
386            chunk.update(
387                id=f"{scope}:{item.document_id}:g{generation}:e{token.epoch}:{nonce}:{index}",
388                embedding=vector,
389            )
390        record = ManifestRecord(
391            scope=scope,
392            document_id=item.document_id,
393            observed_revision=item.observed_revision,
394            content_hash=item.content_hash,
395            extractor_fingerprint=self._extractor_fingerprint(doc),
396            chunker_fingerprint=self._chunker_fingerprint(),
397            embedding_fingerprint=self._embedding_fingerprint(),
398            active_generation=generation,
399            status=ManifestStatus.ACTIVE,
400            chunk_ids=[chunk["id"] for chunk in chunks],
401            metadata=dict(doc.metadata),
402            provenance=dict(doc.provenance),
403            last_successful_observation=time.time(),
404            fence_token=item.expected_fence,
405        )
406        return record, chunks

408    async def _checkpoint(self, report: SyncReport, token: FenceToken) -> None:
409        checkpoint = {
410            "phase": "completed" if report.success else "incomplete",
411            "authoritative": report.authoritative,
412            "last_error": "sync_incomplete" if not report.success else None,
413            "added": report.added_count,
414            "changed": report.changed_count,
415            "removed": report.removed_count,
416            "observed_at": time.time(),
417        }
418        previous = await self._state_store.get_checkpoint(report.scope) or {}
419        checkpoint["last_successful_sync"] = (
420            time.time() if report.success else previous.get("last_successful_sync")
421        )
422        if not await self._state_store.save_checkpoint(report.scope, checkpoint, token):
423            raise StaleWorkerError("Cannot persist checkpoint after lease takeover")

425    async def run_sync(  # noqa: PLR0915
426        self,
427        scope: str,
428        source: Any,
429        is_authoritative: bool | None = None,
430        cursor: Any = None,
431        limit: int | None = None,
432    ) -> SyncReport:
433        if not hasattr(self._state_store, "fenced"):
434            raise FencingError("SyncStateStore must implement a fenced commit context")
435        if not hasattr(self._manifest_store, "compare_and_swap_many"):
436            raise FencingError("ManifestStore must implement atomic batch commits")
437        token = await self._state_store.acquire_lease(
438            scope, self._worker_id, self._lease_ttl_seconds
439        )
440        if token is None:
441            raise LeaseAcquisitionError(f"Could not acquire lease for scope {scope!r}")
442        heartbeat = asyncio.create_task(self._heartbeat(token))
443        report = SyncReport(scope=scope, success=False, fence_epoch=token.epoch)
444        uncommitted: set[str] = set()
445        retired: list[str] = []
446        self._emit(EventKind.SCAN_STARTED, scope, token)
447        try:
448            result = await self._scan_source(source, cursor, limit)
449            report.authoritative = result.authoritative and is_authoritative is not False
450            report.errors.extend(_error_record(error) for error in result.errors)
451            plan = await self.plan(scope, result, report.authoritative)
452            for action in ("added", "changed", "removed", "metadata_only", "unchanged"):
453                setattr(report, f"{action}_count", len(getattr(plan, action)))
454            for doc in result:
455                self._emit(
456                    EventKind.DOCUMENT_LOADED,
457                    scope,
458                    token,
459                    document_id=doc.external_id or doc.source_name,
460                )
461            for item in plan.unchanged:
462                self._emit(EventKind.DOCUMENT_SKIPPED, scope, token, document_id=item.document_id)
463            active_gen = await self._state_store.get_active_generation(scope)
464            prepared: list[tuple[SyncPlanItem, ManifestRecord, list[dict[str, Any]]]] = []
465            for item in plan.added + plan.changed + plan.metadata_only:
466                try:
467                    existing = await self._manifest_store.get(scope, item.document_id)
468                    generation = (
469                        active_gen + 1
470                        if self._activation_mode == ActivationMode.COLLECTION
471                        else ((existing.active_generation if existing else 0) + 1)
472                    )
473                    existing_vectors = None
474                    if item.action == "metadata_only":
475                        assert existing is not None
476                        by_id = await self._vector_store.get_embeddings(existing.chunk_ids)
477                        existing_vectors = [by_id[chunk_id] for chunk_id in existing.chunk_ids]
478                    record, chunks = await self._prepare_item(
479                        scope, item, generation, token, existing_vectors
480                    )
481                    if not await self._state_store.validate_token(token):
482                        raise StaleWorkerError("Lease lost during embedding")
483                    uncommitted.update(record.chunk_ids)
484                    await self._vector_store.upsert(chunks)
485                    if existing_vectors is None:
486                        report.embedded_count += len(chunks)
487                    if self._job_store is not None:
488                        await self._job_store.enqueue(
489                            {
490                                "idempotency_key": (
491                                    f"{scope}:{item.document_id}:{generation}:e{token.epoch}:activate"
492                                ),
493                                "scope": scope,
494                                "document_id": item.document_id,
495                                "generation": generation,
496                                "epoch": token.epoch,
497                                "status": "pending",
498                                "operation": "reconcile_generation",
499                                "chunk_ids": record.chunk_ids,
500                            }
501                        )
502                    prepared.append((item, record, chunks))
503                except StaleWorkerError:
504                    raise
505                except Exception as exc:
506                    report.errors.append({"doc_id": item.document_id, **_error_record(exc)})
507                    if self._activation_mode == ActivationMode.COLLECTION:
508                        raise
509            if not await self._state_store.validate_token(token):
510                raise StaleWorkerError("Lease expired before activation")
511            try:
512                async with self._state_store.fenced(token):
513                    await self._activate(scope, plan, prepared, token, report, retired, uncommitted)
514                    report.success = (
515                        not report.errors
516                        and result.complete
517                        and result.status
518                        in {
519                            "ok",
520                            "empty",
521                            "success",
522                            "empty_success",
523                        }
524                    )
525                    await self._checkpoint(report, token)
526            except ValueError as exc:
527                if not await self._state_store.validate_token(token):
528                    raise StaleWorkerError("Lease lost before fenced commit") from exc
529                raise
530            return report
531        except Exception as exc:
532            self._emit(EventKind.SYNC_FAILED, scope, token, error=exc)
533            if await self._state_store.validate_token(token):
534                await self._checkpoint(report, token)
535            raise
536        finally:
537            heartbeat.cancel()
538            with suppress(asyncio.CancelledError):
539                await heartbeat
540            await self._state_store.release_lease(token)
541            await self.garbage_collect(scope, [*retired, *uncommitted])

543    async def _activate(  # noqa: PLR0913, PLR0917
544        self,
545        scope: str,
546        plan: SyncPlan,
547        prepared: list[tuple[SyncPlanItem, ManifestRecord, list[dict[str, Any]]]],
548        token: FenceToken,
549        report: SyncReport,
550        retired: list[str],
551        uncommitted: set[str],
552    ) -> None:
553        updates: list[tuple[ManifestRecord, int]] = []
554        old_ids: list[str] = []
555        for item, record, chunks in prepared:
556            current = await self._manifest_store.get(scope, item.document_id)
557            if (current.fence_token if current else 0) != item.expected_fence:
558                raise FencingError("Document changed while preparing its generation")
559            if current:
560                old_ids.extend(current.chunk_ids)
561            # Publish vectors before changing visibility. The manifest still hides their IDs.
562            await self._vector_store.upsert(
563                [{**chunk, "metadata": {**chunk["metadata"], "active": True}} for chunk in chunks]
564            )
565            updates.append((record, item.expected_fence))
566        for item in plan.removed:
567            record = await self._manifest_store.get(scope, item.document_id)
568            if record is None or record.fence_token != item.expected_fence:
569                raise FencingError("Document changed before tombstoning")
570            old_ids.extend(record.chunk_ids)
571            updates.append(
572                (
573                    replace(
574                        record, status=ManifestStatus.TOMBSTONED, active_generation=0, chunk_ids=[]
575                    ),
576                    item.expected_fence,
577                )
578            )
579        if updates:
580            if not await self._manifest_store.compare_and_swap_many(updates):
581                raise FencingError("Manifest commit compare-and-swap failed")
582            # From here these IDs are committed and must never be garbage-collected on failure.
583            for _, record, chunks in prepared:
584                uncommitted.difference_update(record.chunk_ids)
585                report.chunk_count += len(chunks)
586                self._emit(
587                    EventKind.GENERATION_ACTIVATED,
588                    scope,
589                    token,
590                    document_id=record.document_id,
591                    generation=record.active_generation,
592                    counts={"chunks": len(chunks)},
593                )
594            retired.extend(old_ids)
595            if self._activation_mode == ActivationMode.COLLECTION:
596                generation = await self._state_store.advance_generation(
597                    scope, await self._state_store.get_active_generation(scope)
598                )
599                if generation < 0:
600                    raise FencingError("Collection generation compare-and-swap failed")
601                report.active_generation = generation
602        if self._activation_mode == ActivationMode.DOCUMENT:
603            report.active_generation = await self._state_store.get_active_generation(scope)

605    async def _scan_source(
606        self, source: Any, cursor: Any = None, limit: int | None = None
607    ) -> LoadResult:
608        if isinstance(source, LoadResult):
609            return source
610        if hasattr(source, "load_page"):
611            first = True
612
613            async def page(*, cursor=None, limit=None):
614                nonlocal first
615                actual_cursor = start_cursor if first else cursor
616                first = False
617                return await source.load_page(cursor=actual_cursor, limit=limit)
618
619            start_cursor = cursor
620            result = await collect_pages(
621                SimpleNamespace(load_page=page),
622                max_pages=getattr(source, "max_pages", None) or 10000,
623                page_size=limit,
624            )
625            if start_cursor is not None:
626                result.complete = False
627            return result
628        if hasattr(source, "load"):
629            parameters = inspect.signature(source.load).parameters
630            options = {
631                key: value
632                for key, value in {"cursor": cursor, "limit": limit}.items()
633                if key in parameters
634            }
635            result = await source.load(**options)
636            if isinstance(result, LoadResult):
637                return result
638            return LoadResult(list(result), complete=cursor is None)
639        return LoadResult(list(source), complete=cursor is None)

641    async def revoke_access(self, scope: str, document_id: str) -> bool:
642        while True:
643            record = await self._manifest_store.get(scope, document_id)
644            if record is None:
645                return False
646            tombstone = replace(
647                record, status=ManifestStatus.TOMBSTONED, active_generation=0, chunk_ids=[]
648            )
649            if await self._manifest_store.compare_and_swap(tombstone, record.fence_token):
650                await self.garbage_collect(scope, record.chunk_ids)
651                return True
652
653    async def garbage_collect(self, scope: str, retired_chunk_ids: list[str] | None = None) -> int:
654        if not retired_chunk_ids:
655            return 0
656        try:
657            return await self._vector_store.delete(ids=retired_chunk_ids)
658        except Exception:
659            logger.warning("Physical cleanup deferred for scope %s", scope)
660            return 0
