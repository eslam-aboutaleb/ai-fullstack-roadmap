1"""
2SQLAlchemy ORM models for the notification outbox and the in-app feed it fills.
3
4Two tables, on purpose. ``notification_jobs`` is the outbox: a row per delivery
5attempted in the claim transaction, carrying the retry bookkeeping a failing
6consumer needs. ``notifications`` is the read model a claimant actually opens: one
7row per *delivered* notification, denormalised into the text it renders, with a
8read receipt. Keeping them apart means the outbox can gain channels, providers, or
9attempt ceilings without the feed's shape following it, and it means a job that is
10still pending, failing, or dead-lettered is simply absent from the feed rather
11than appearing there as a husk.
12
13The outbox deliberately reuses the ``embedding_jobs`` column set rather than
14generalising that table: the drain worker, the claim guard, and the backoff policy
15are proven there, and a claim-shaped NOT NULL schema is the wrong place to start a
16notification queue.
17"""

38MAX_ATTEMPTS = 4
39"""Attempt ceiling a notification job inherits unless the caller says otherwise.
40
41Matches the outbox protocol's own default so a job that ignores the ceiling and a
42job that accepts it dead-letter on the same attempt.
43"""

45# Delivery targets. ``in_app`` is the only channel with a provider; the rest are
46# wired here so the enqueue call sites can name them today and a provider can be
47# attached to one without a schema change.
48CHANNEL_IN_APP = "in_app"
49CHANNEL_EMAIL = "email"
50CHANNEL_SMS = "sms"
51CHANNEL_WEBHOOK = "webhook"
52
53NOTIFICATION_CHANNELS: frozenset[str] = frozenset(
54    {CHANNEL_IN_APP, CHANNEL_EMAIL, CHANNEL_SMS, CHANNEL_WEBHOOK}
55)

57# The claim facts worth telling a policyholder about unprompted. A ``job_type`` is
58# a delivery instruction, not a status: the worker resolves each one to a rendered
59# message from the payload, so no model call is needed to notify anybody.
60JOB_TYPE_CLAIM_SUBMITTED = "claim.submitted"
61JOB_TYPE_CLAIM_STATUS_CHANGED = "claim.status_changed"
62JOB_TYPE_CLAIM_WITHDRAWN = "claim.withdrawn"

64NOTIFICATION_JOB_TYPES: frozenset[str] = frozenset(

66        JOB_TYPE_CLAIM_SUBMITTED,

73class NotificationJob(Base):
74    """Database model for one queued notification delivery.
75
76    Attributes:
77        id: Primary key UUID.
78        user_id: The user the notification is for. The reference is enforced, so a
79            deleted account takes its queued notifications with it rather than
80            leaving rows that name a recipient nobody can look up.
81        job_type: What happened, one of ``NOTIFICATION_JOB_TYPES``. Stored as text
82            rather than an enum because a new delivery kind must not require a
83            schema change, and the worker's template table is the authority on
84            which values it can render.
85        payload: The job's business data as JSONB, exactly what the template
86            function reads.
87        channel: Delivery target, one of ``NOTIFICATION_CHANNELS``. Defaults to the
88            in-app feed, the only channel with a provider.
89        status: Queue state (pending, processing, completed, failed, dead_letter).
90        status_detail: Human-readable detail, including the last failure reason.
91        retry_count: Number of failed attempts recorded so far.
92        next_retry_at: When the job becomes eligible again after a failure.
93        max_attempts: Attempt ceiling before the job is dead-lettered.
94        locked_at: When a worker claimed the job, so a dead worker's lock can be
95            reclaimed.
96        locked_by: Identity of the worker that claimed the job.
97        created_at: When the job was queued, in the same transaction as the claim
98            change that caused it.
99        completed_at: When delivery succeeded, or was given up on.
100    """
101
102    __tablename__ = "notification_jobs"
103
104    id: Mapped[uuid.UUID] = mapped_column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
105    user_id: Mapped[uuid.UUID] = mapped_column(
106        UUID(as_uuid=True),
107        ForeignKey("users.id", ondelete="CASCADE"),
108        nullable=False,
109        index=True,
110    )
111    job_type: Mapped[str] = mapped_column(String(64), nullable=False, index=True)
112    payload: Mapped[dict] = mapped_column(JSONB, nullable=False)
113    channel: Mapped[str] = mapped_column(String(32), nullable=False, default=CHANNEL_IN_APP)
114    # Not indexed on its own: the claim query filters on status and orders by
115    # created_at together, which is the composite index below.
116    status: Mapped[str] = mapped_column(String(32), nullable=False, default="pending")
117    status_detail: Mapped[str | None] = mapped_column(String, nullable=True)
118    retry_count: Mapped[int] = mapped_column(Integer, nullable=False, default=0)
119    next_retry_at: Mapped[datetime | None] = mapped_column(DateTime(timezone=True), nullable=True)
120    max_attempts: Mapped[int] = mapped_column(Integer, nullable=False, default=MAX_ATTEMPTS)
121    locked_at: Mapped[datetime | None] = mapped_column(DateTime(timezone=True), nullable=True)
122    locked_by: Mapped[str | None] = mapped_column(String, nullable=True)
123    created_at: Mapped[datetime] = mapped_column(
124        DateTime(timezone=True), default=lambda: datetime.now(UTC), nullable=False
125    )
126    completed_at: Mapped[datetime | None] = mapped_column(DateTime(timezone=True), nullable=True)
127
128    __table_args__ = (
129        # The claim query filters on status and orders by created_at, so both
130        # columns belong in one index rather than one each.
131        Index("ix_notification_jobs_status_created_at", status, created_at),
132    )

135class Notification(Base):
136    """Database model for one delivered notification, as the recipient reads it.
137
138    The feed is written by the drain worker, never by the request that caused the
139    notification: a claimant must not wait on a delivery to move a claim, and the
140    claim transaction must not be able to fail because a feed row would not insert.
141
142    Attributes:
143        id: Primary key UUID.
144        user_id: The user whose feed holds the notification.
145        job_id: The outbox row that delivered it. Unique, so a redelivery after a
146            reclaimed lock cannot put the same notification in a feed twice.
147        job_type: Which delivery instruction produced this, for the client to key
148            on.
149        channel: Which provider delivered it, so a future email notification and
150            the in-app copy of the same event stay distinguishable.
151        title: Rendered subject line, stored rather than recomputed so a template
152            change never rewrites what a user has already read.
153        body: Rendered message text.
154        claim_id: The claim the notification is about, when it has one. Plain text
155            rather than a reference, because the feed must stay readable for a
156            claim that has since been deleted.
157        read_at: When the user acknowledged it, or NULL while it is still unread.
158        created_at: When the notification was delivered, generated by the database
159            so rows written by a worker with a skewed clock still order correctly.
160    """
161
162    __tablename__ = "notifications"
163
164    id: Mapped[uuid.UUID] = mapped_column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
165    user_id: Mapped[uuid.UUID] = mapped_column(
166        UUID(as_uuid=True),
167        ForeignKey("users.id", ondelete="CASCADE"),
168        nullable=False,
169        index=True,
170    )
171    job_id: Mapped[uuid.UUID] = mapped_column(
172        UUID(as_uuid=True),
173        ForeignKey("notification_jobs.id", ondelete="CASCADE"),
174        nullable=False,
175    )
176    job_type: Mapped[str] = mapped_column(String(64), nullable=False)
177    channel: Mapped[str] = mapped_column(String(32), nullable=False, default=CHANNEL_IN_APP)
178    title: Mapped[str] = mapped_column(String(255), nullable=False)
179    body: Mapped[str] = mapped_column(String, nullable=False)
180    claim_id: Mapped[str | None] = mapped_column(String, nullable=True, index=True)
181    read_at: Mapped[datetime | None] = mapped_column(DateTime(timezone=True), nullable=True)
182    created_at: Mapped[datetime] = mapped_column(
183        DateTime(timezone=True), nullable=False, server_default=func.now()
184    )
185
186    __table_args__ = (
187        # One delivered row per outbox job: a worker that lost its lock mid-pass
188        # retries the job, and without this the retry would notify a second time.
189        UniqueConstraint("job_id", name="uq_notifications_job_id"),
190        # The feed read is "this user's notifications, newest first, unread
191        # filtered", so the owner and the ordering columns share one index.
192        Index("ix_notifications_user_id_created_at", user_id, created_at),
193    )
