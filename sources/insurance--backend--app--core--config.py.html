372    @field_validator("cors_origins", mode="before")
373    @classmethod
374    def assemble_cors_origins(cls, value: Any) -> list[str]:
375        """Supports comma-separated strings, JSON arrays, or existing lists."""
376        if isinstance(value, str):
377            # Check for comma-delimited string
378            value = value.strip()
379            if value.startswith("[") and value.endswith("]"):
380                try:
381                    return json.loads(value)
382                except json.JSONDecodeError:
383                    pass
384            return [item.strip() for item in value.split(",") if item.strip()]
385        if isinstance(value, list | tuple):
386            return [str(item).strip() for item in value if str(item).strip()]
387        return []

389    @field_validator("log_level")
390    @classmethod
391    def validate_log_level(cls, value: str) -> str:
392        """Validates that log level is an accepted standard logging level."""
393        valid_levels = {"DEBUG", "INFO", "WARNING", "ERROR", "CRITICAL"}
394        upper_val = value.strip().upper()
395        if upper_val not in valid_levels:
396            raise ValueError(
397                f"Invalid log_level '{value}'. Must be one of: {', '.join(sorted(valid_levels))}"
398            )
399        return upper_val

401    @field_validator("jwt_secret_key")
402    @classmethod
403    def validate_jwt_secret_key(cls, value: str, info: Any) -> str:
404        """Reject a placeholder or weak JWT secret in production.
405
406        This is an authentication-bypass gate, so it is deliberately stricter than a
407        single sentinel check. It rejects:
408
409        * every placeholder value shipped in ``.env.example``, so copying that file and
410          setting ``ENVIRONMENT=production`` cannot start the service with a publicly
411          known HMAC key;
412        * any secret shorter than 32 characters, which is too weak to sign tokens with.
413        """
414        environment = info.data.get("environment", "development")
415        if environment != "production":
416            return value
417
418        normalised = value.strip().lower()
419        placeholders = {
420            "change-me",
421            "your-secret-key-here",
422            "your_secret_key_here",
423            "changeme",
424            "secret",
425            "password",
426            "string",
427        }
428        if normalised in placeholders or "change-me" in normalised:
429            raise ValueError(
430                "JWT_SECRET_KEY is still a placeholder. Generate one with "
431                '`python -c "import secrets; print(secrets.token_urlsafe(32))"` '
432                "before deploying to production."
433            )
434        if len(value) < 32:
435            raise ValueError(
436                f"JWT_SECRET_KEY must be at least 32 characters in production; got {len(value)}."
437            )
438        return value
439
440    @field_validator("cors_origins")
441    @classmethod
442    def forbid_wildcard_origin(cls, value: list[str]) -> list[str]:
443        """Reject wildcard CORS origins when credentials are allowed."""
444        if "*" in value:
445            raise ValueError(
446                "Wildcard origin '*' is not allowed when allow_credentials=True. "
447                "Specify explicit origins instead."
448            )
449        return value
