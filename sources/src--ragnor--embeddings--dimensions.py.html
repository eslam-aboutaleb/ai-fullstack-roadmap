1"""Embedding dimension registry.
2
3Single source of truth mapping embedding model names to
4their vector dimensions, so changing the embedding model cannot
5silently desynchronize the dimension checks at insert time.
6
7The built-in table covers the models ragnor knows about;
8:func:`register_dimension` is the public extensibility seam for
9models the table does not know (host applications register their
10provider's dimension at start-up).
11"""
12
13from __future__ import annotations
14
15import logging
16
17logger = logging.getLogger(__name__)
18
19# Dimension per supported embedding provider. ``text-embedding-3-small``
20# is the only provider configured today; OpenAI's ``text-embedding-ada-002``
21# is the historical model and shares its dimension.
22_DIMENSIONS: dict[str, int] = {
23    "text-embedding-3-small": 1536,
24    "text-embedding-3-large": 3072,
25    "text-embedding-ada-002": 1536,
26}
27
28FALLBACK_DIMENSION = 1536
29
30
31def register_dimension(model: str, dimension: int) -> None:
32    """Register the vector dimension for an embedding model.
33
34    Extensibility seam replacing the private dimension table: host
35    applications and tests can register dimensions for models the
36    built-in table does not know about.
37
38    Args:
39        model: Embedding model name.
40        dimension: Vector dimension the model produces.
41    """
42    _DIMENSIONS[model] = dimension
43
44
45def get_embedding_dimension(model: str | None = None) -> int:
46    """Return the vector dimension for an embedding model.
47
48    Known models return their registered dimension. Unknown models
49    warn and fall back to :data:`FALLBACK_DIMENSION` rather than
50    raising, so a configuration-only change does not crash start-up;
51    the warning makes the wrong dimension visible.
52
53    Args:
54        model: The embedding model name. Host applications resolve
55            their configured model and pass it explicitly; a bare
56            call has no model to look up and takes the same
57            warn-and-fallback path as an unknown model.
58
59    Returns:
60        The dimension, or the fallback when the model is not in the
61        known table.
62    """
63    if model is None:
64        logger.warning(
65            "No embedding model specified; assuming dimension %d. "
66            "Pass the configured model name to get_embedding_dimension().",
67            FALLBACK_DIMENSION,
68        )
69        return FALLBACK_DIMENSION
70
71    dimension = _DIMENSIONS.get(model)
72    if dimension is not None:
73        return dimension
74
75    logger.warning(
76        "No embedding dimension registered for model '%s'; assuming %d. "
77        "Add it with ragnor.embeddings.dimensions.register_dimension() if that is wrong.",
78        model,
79        FALLBACK_DIMENSION,
80    )
81    return FALLBACK_DIMENSION
