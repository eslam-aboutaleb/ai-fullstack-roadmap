1"""Minimal plug-and-play example: a FastAPI app on ragnor.
2
3This example uses ragnor with **zero host-application code**: an
4in-memory vector store (no database), a sliding-window
5chunker, a deterministic fake embedder (used unless
6``OPENAI_API_KEY`` is set in the process environment),
7and the ingestion pipeline over a bundled document.
8
9Install and run::
10
11    pip install -e libs/ragnor fastapi uvicorn
12    python libs/ragnor/examples/minimal_app.py
13
14Then query the search endpoint::
15
16    curl "http://127.0.0.1:8000/search?query=refund"
17
18No ``DATABASE_URL`` is required: the in-memory store and
19the fake embedder need no external services.
20"""
21

75def _fake_embed(text: str) -> list[float]:
76    """Deterministic hash embedding (no API key required)."""
77    digest = hashlib.sha256(text.encode("utf-8")).digest()
78    vector: list[float] = []
79    while len(vector) < FALLBACK_DIMENSION:
80        digest = hashlib.sha256(digest).digest()
81        vector.extend(byte / 255.0 for byte in digest)
82    return vector[:FALLBACK_DIMENSION]

85class FakeEmbedder:
86    """Deterministic embedder used when no API key is configured."""
87
88    async def __call__(self, input: list[str]) -> list[list[float]]:
89        return [_fake_embed(text) for text in input]
90
91    async def embed_query(self, input: str | list[str]) -> list[float]:
92        return _fake_embed(input if isinstance(input, str) else input[0])
93
94    async def embed_documents(self, input: list[str]) -> list[list[float]]:
95        return [_fake_embed(text) for text in input]

111class MemoryVersionStore:
112    """In-memory ``VersionStore`` (the example keeps no version table)."""
113
114    def __init__(self) -> None:
115        self.active: VersionRecord | None = None
116
117    async def find_active(self, session: Any) -> VersionRecord | None:
118        return self.active
119
120    async def close_active(self, session: Any) -> None:
121        self.active = None
122
123    async def create_version(self, session: Any, snapshot: IngestionSnapshot) -> VersionRecord:
124        self.active = VersionRecord(
125            version_id=f"example-{uuid.uuid4().hex[:8]}",
126            snapshot=snapshot,
127            source_id="example-document",
128        )
129        return self.active
130
131    async def delete_chunks(self, session: Any, version_id: str) -> None:
132        """No-op: the in-memory store keeps no per-version chunk rows."""

135settings = ExampleSettings()
136store = InMemoryVectorStore()
137embedder: EmbeddingFunction = (
138    LitellmEmbeddingFunction(
139        api_key=settings.openai_api_key,
140        model_name=settings.embedding_model,
141    )
142    if settings.openai_api_key
143    else FakeEmbedder()
144)
145chunker = SlidingWindowChunker()
146version_store = MemoryVersionStore()
147pipeline = IngestionPipeline(
148    store=store,
149    embedder=embedder,
150    chunker=chunker,
151    session_factory=_null_session_factory,
152    settings=settings,
153    version_id_key="version_id",
154    source_id_key="source_id",
155)

158@asynccontextmanager
159async def _lifespan(app: FastAPI):
160    """Ingest the bundled document on startup."""
161    document = Path(__file__).with_name("sample_document.md")
162    count = await pipeline.run(FileDocumentSource(str(document)), version_store)
163    print(f"ingested {count} chunks from {document.name}")
164    yield
165
166
167app = FastAPI(title="ragnor minimal example", lifespan=_lifespan)

170@app.get("/search")
171async def search(query: str, n_results: int = 5) -> dict[str, Any]:
172    """Hybrid-search the ingested document (vector + keyword, RRF-fused)."""
173    embedding = await embedder.embed_query(query)
174    results = await store.hybrid_search(
175        query=query,
176        embedding=embedding,
177        n_results=n_results,
178        threshold=settings.rag_distance_threshold,
179        text_field="text",
180        metadata_fields=["section", "source"],
181    )
182    return {"query": query, "results": results}
