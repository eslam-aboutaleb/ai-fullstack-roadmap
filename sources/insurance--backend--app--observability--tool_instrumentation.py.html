1"""Latency and outcome telemetry for the agent's tools.
2
3The tool functions and the registry that assembles them are owned by the agent
4layer, so instrumentation is applied where a handler enters the registry
5(:meth:`ToolRegistry.register`). That single hook covers every tool the registry
6ever holds without editing a tool or the registry, and it runs before the agent
7is built, which is when the tool declarations are generated.
8
9The wrapper is deliberately transparent: it forwards every argument, returns the
10handler's value untouched, and re-raises its exceptions. Per-invocation cost is
11one ``perf_counter`` pair and one bounded append, which is why this can be left
12on in production.
13"""
14

34def _with_original_contract(
35    wrapper: Callable[..., Any],
36    handler: Callable[..., Any],
37) -> Callable[..., Any]:
38    """Pin the wrapper's signature and annotations back to the original handler.
39
40    The tracing span ``instrument_tool`` applies innermost returns a callable
41    carrying its own run-control keyword (``config``), and ``functools.wraps``
42    copies that keyword into the latency wrapper built on top of it. The agent
43    generates each tool's declaration from this wrapper's signature, so without
44    this the declaration would advertise a ``config`` argument no tool actually
45    takes. The signature is therefore pinned to the original handler's, dropping
46    whatever run-control keywords the span injected.
47
48    The annotations are stored as resolved type objects for the same reason the
49    signature is pinned: they feed the tool declaration too. The tool modules use
50    ``from __future__ import annotations``, so the strings ``functools.wraps``
51    copies are evaluated against the function's own ``__globals__`` -- the tool
52    module for the handler, but this module for the wrapper. A tool-local type
53    would therefore fail to resolve on the wrapper and break the declaration, so
54    they are resolved against the handler here instead. A resolution failure
55    leaves the raw strings in place, which is the behaviour ``functools.wraps``
56    already provides.
57
58    Args:
59        wrapper: The wrapper function to pin the contract onto.
60        handler: The original tool function, before any span was applied.
61
62    Returns:
63        Callable[..., Any]: The same wrapper, carrying the handler's signature
64        and resolved annotations.
65    """
66    wrapper.__signature__ = inspect.signature(handler)
67    try:
68        resolved = get_type_hints(handler)
69    except Exception:
70        return wrapper
71    if resolved:
72        wrapper.__annotations__ = resolved
73    return wrapper

76def instrument_tool(name: str, handler: Callable[..., Any]) -> Callable[..., Any]:
77    """Wrap ``handler`` so one tool-call event is recorded per invocation.
78
79    Args:
80        name: Tool name recorded on the event. Falls back to the handler's own
81            name when empty.
82        handler: The tool function to wrap. Both async and sync callables are
83            accepted because the registry permits either.
84
85    Returns:
86        Callable[..., Any]: A wrapper carrying the handler's name, docstring,
87        signature, and resolved annotations, so the agent's tool declaration is
88        built from the original function exactly as before.
89    """
90    tool_name = name or getattr(handler, "__name__", "tool")
91
92    original = handler
93    # The tracing span is applied innermost, before the latency wrapper, for
94    # two reasons: the span nests under the trace of the turn that invoked the
95    # tool through the run context, and the outer wrappers still resolve the
96    # original handler's annotations -- which is what the agent builds the
97    # tool declaration from. Tracing off makes this a no-op.
98    handler = trace_tool(tool_name)(handler)
99
100    def _record(started: float, success: bool, error: BaseException | None) -> None:
101        try:
102            get_metrics().record_tool_call(
103                build_tool_call(
104                    name=tool_name,
105                    latency_seconds=time.perf_counter() - started,
106                    success=success,
107                    error="" if error is None else repr(error),
108                )
109            )
110        except Exception:
111            logger.debug("Tool telemetry dropped for '%s'", tool_name, exc_info=True)
112
113    if inspect.iscoroutinefunction(handler):
114
115        @functools.wraps(handler)
116        async def async_wrapper(*args: Any, **kwargs: Any) -> Any:
117            started = time.perf_counter()
118            try:
119                result = await handler(*args, **kwargs)
120            except Exception as exc:
121                _record(started, False, exc)
122                raise
123            _record(started, True, None)
124            return result
125
126        return _with_original_contract(async_wrapper, original)
127
128    @functools.wraps(handler)
129    def sync_wrapper(*args: Any, **kwargs: Any) -> Any:
130        started = time.perf_counter()
131        try:
132            result = handler(*args, **kwargs)
133        except Exception as exc:
134            _record(started, False, exc)
135            raise
136        _record(started, True, None)
137        return result
138
139    return _with_original_contract(sync_wrapper, original)

142def install_tool_instrumentation() -> None:
143    """Instrument every handler the agent's tool registry accepts.
144
145    ``ToolRegistry.register`` is the only point where a handler enters the
146    registry, so patching it there instruments tools the agent layer owns
147    without editing them. The patch is applied to the class rather than an
148    instance because the agent module constructs the registry itself, and it is
149    installed once: re-running the application lifespan must not wrap an already
150    wrapped handler.
151    """
152    from app.agent.registry import ToolRegistry  # noqa: PLC0415
153
154    register = ToolRegistry.register
155    if getattr(register, _INSTALL_MARKER, False):
156        return
157
158    @functools.wraps(register)
159    def instrumented_register(self: ToolRegistry, spec: Any) -> None:
160        spec.handler = instrument_tool(spec.name, spec.handler)
161        register(self, spec)
162
163    setattr(instrumented_register, _INSTALL_MARKER, True)
164    ToolRegistry.register = instrumented_register  # type: ignore[method-assign]
