1"""Request correlation middleware.
2
3Implemented against the raw ASGI interface rather than as a
4``BaseHTTPMiddleware`` subclass on purpose: the streaming chat endpoint is
5sensitive to an extra task group wrapping its response body, and a plain ASGI
6middleware keeps that path byte-identical while still running the downstream
7application in the same context, which is what makes the correlation id visible
8to the endpoint and to any task it spawns.
9"""
10
11from __future__ import annotations
12
13import logging
14
15from starlette.datastructures import Headers
16from starlette.types import ASGIApp, Message, Receive, Scope, Send
17
18from app.observability.request_context import (
19    REQUEST_ID_HEADER,
20    reset_request_id,
21    resolve_request_id,
22    set_request_id,
23)
24
25logger = logging.getLogger(__name__)
26
27# ASGI header names are lowercase bytes; keeping it pre-encoded avoids rebuilding
28# the pair for every response.
29_REQUEST_ID_HEADER_BYTES = REQUEST_ID_HEADER.lower().encode("latin-1")
30
31
32class RequestIdMiddleware:
33    """Bind a correlation id to each request and echo it back on the response.
34
35    An inbound ``X-Request-ID`` is honoured when it is safe to echo, so a caller
36    that already traces requests can stitch this service's logs into its own.
37    """
38
39    def __init__(self, app: ASGIApp) -> None:
40        """Wrap ``app``.
41
42        Args:
43            app: The downstream ASGI application.
44        """
45        self.app = app
46
47    async def __call__(self, scope: Scope, receive: Receive, send: Send) -> None:
48        """Bind the request id for the duration of the request.
49
50        Args:
51            scope: The ASGI connection scope.
52            receive: ASGI receive callable.
53            send: ASGI send callable.
54        """
55        if scope["type"] != "http":
56            await self.app(scope, receive, send)
57            return
58
59        inbound = Headers(scope=scope).get(REQUEST_ID_HEADER)
60        request_id = resolve_request_id(inbound)
61        token = set_request_id(request_id)
62
63        async def send_with_request_id(message: Message) -> None:
64            if message["type"] == "http.response.start":
65                _set_request_id_header(message, request_id)
66            await send(message)
67
68        try:
69            await self.app(scope, receive, send_with_request_id)
70        finally:
71            reset_request_id(token)
72
73
74def _set_request_id_header(message: Message, request_id: str) -> None:
75    """Write the correlation id onto a response start message.
76
77    An existing value is replaced rather than appended: an outer layer may have
78    echoed a header of its own, and two ``X-Request-ID`` headers would make a
79    caller's trace ambiguous rather than better.
80
81    Args:
82        message: The ASGI ``http.response.start`` message.
83        request_id: The correlation id bound to this request.
84    """
85    encoded = (_REQUEST_ID_HEADER_BYTES, request_id.encode("latin-1"))
86    headers = [
87        header
88        for header in message.setdefault("headers", [])
89        if header[0].lower() != _REQUEST_ID_HEADER_BYTES
90    ]
91    headers.append(encoded)
92    message["headers"] = headers
