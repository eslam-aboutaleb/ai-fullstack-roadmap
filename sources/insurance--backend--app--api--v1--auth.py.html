1"""
2Authentication API routes for the OmniCare backend.
3
4Exposes three endpoints:
5  - ``POST /signup`` -- Create a new user account with Argon2 password hashing.
6  - ``POST /signin`` -- Authenticate with username and password, issue JWT.
7  - ``POST /logout`` -- Clear the session cookie.
8
9All endpoints are public (no authentication required) because they are the
10entry point for obtaining credentials. After signup or signin, the client
11receives a JWT that is used for subsequent requests.
12
13Security notes:
14  - Passwords are hashed with Argon2id in a background threadpool to avoid
15    blocking the async event loop during the CPU-intensive hashing operation.
16  - A dummy hash is returned for non-existent users during signin to prevent
17    timing-based user enumeration.
18  - Session cookies are HTTP-only and SameSite=Lax to mitigate CSRF while
19    still allowing top-level navigation from external links.
20"""

47@router.post("/signup", response_model=Token, status_code=status.HTTP_201_CREATED)
48async def signup(
49    payload: UserSignup,
50    response: Response,
51    request: Request,
52    session: AsyncSession = Depends(get_db),
53):
54    """Register a new OmniCare user account.
55
56    Validates that the username is unique, hashes the password with Argon2id,
57    and issues a JWT session cookie on successful creation.
58
59    Args:
60        payload: Validated signup payload containing username and password.
61        response: FastAPI response object used to set the session cookie.
62        request: The request being audited; supplies the client IP and user
63            agent for the audit trail entry.
64        session: The async database session.
65
66    Returns:
67        Token: The access token, token type, and user ID.
68
69    Raises:
70        HTTPException: 409 if the username already exists or a database
71        integrity error occurs.
72    """
73    existing = (
74        await session.execute(select(User).where(User.username == payload.username))
75    ).scalar_one_or_none()
76    if existing is not None:
77        raise HTTPException(status_code=409, detail="An account with this username already exists")
78
79    # Argon2 hashing is CPU-intensive; offload to a threadpool to avoid
80    # blocking the async event loop.
81    password_hash = await run_in_threadpool(hash_password, payload.password)
82    user = User(
83        id=uuid.uuid4(),
84        username=payload.username.strip(),
85        password_hash=password_hash,
86    )
87    session.add(user)
88
89    # Account creation is an audit-worthy event: an operator asked "who opened
90    # this account" has no other answer. Recorded inside the signup's own
91    # transaction, so a rejected duplicate leaves neither row behind.
92    await record_audit(
93        session,
94        "auth.signup",
95        "user",
96        str(user.id),
97        after={"username": user.username},
98        actor_id=user.id,
99        request=request,
100    )
101
102    try:
103        await session.commit()
104    except IntegrityError as exc:
105        await session.rollback()
106        raise HTTPException(
107            status_code=409,
108            detail="An account with this username already exists",
109        ) from exc
110    await session.refresh(user)
111
112    access_token = create_access_token(user.id)
113    set_session_cookie(response, access_token)
114    return {"access_token": access_token, "token_type": "bearer", "user_id": str(user.id)}

117@router.post("/signin", response_model=Token)
118async def signin(
119    payload: UserSignin,
120    response: Response,
121    request: Request,
122    session: AsyncSession = Depends(get_db),
123):
124    """Authenticate an existing user and issue a JWT session.
125
126    Uses a timing-safe password comparison to prevent user enumeration
127    attacks. On success, sets an HTTP-only session cookie.
128
129    Args:
130        payload: Validated signin payload containing username and password.
131        response: FastAPI response object used to set the session cookie.
132        request: The request being audited; supplies the client IP and user
133            agent for both the success and the failure trail entries.
134        session: The async database session.
135
136    Returns:
137        Token: The access token, token type, and user ID.
138
139    Raises:
140        HTTPException: 401 if credentials are invalid.
141    """
142    user = (
143        await session.execute(select(User).where(User.username == payload.username.strip()))
144    ).scalar_one_or_none()
145
146    # Always perform password verification, even for non-existent users, to
147    # prevent timing-based user enumeration.
148    stored_hash = user.password_hash if user is not None else DUMMY_PASSWORD_HASH
149    password_matches = await run_in_threadpool(verify_password, payload.password, stored_hash)
150
151    if user is None or not password_matches:
152        # A failed sign-in is the one event an operator needs more than a
153        # successful one: it is how a credential-stuffing run announces itself.
154        # The attempted username is recorded because it is the only identifier
155        # the attempt produced, and no response ever echoes it back. The row is
156        # committed before the 401 is raised, because a refusal that leaves
157        # nothing behind is a trail with a hole exactly where it matters.
158        await record_audit(
159            session,
160            "auth.signin_failed",
161            "user",
162            payload.username,
163            request=request,
164        )
165        await session.commit()
166        raise HTTPException(status_code=401, detail="Incorrect username or password")
167
168    access_token = create_access_token(user.id)
169    set_session_cookie(response, access_token)
170
171    await record_audit(
172        session,
173        "auth.signin",
174        "user",
175        str(user.id),
176        actor_id=user.id,
177        request=request,
178    )
179    await session.commit()
180    return {"access_token": access_token, "token_type": "bearer", "user_id": str(user.id)}

183@router.post("/logout", status_code=status.HTTP_204_NO_CONTENT)
184async def logout(
185    response: Response,
186    request: Request,
187    session: AsyncSession = Depends(get_db),
188):
189    """Invalidate the current session by clearing the session cookie.
190
191    Args:
192        response: FastAPI response object used to clear the session cookie.
193        request: The request being audited; supplies the client IP and user
194            agent for the audit trail entry.
195        session: The async database session.
196
197    Returns:
198        Response: 204 No Content on success.
199    """
200    clear_session_cookie(response)
201    response.status_code = status.HTTP_204_NO_CONTENT
202
203    # The endpoint is deliberately public and reads no credential, so a logout
204    # is recorded with no actor: what is knowable is that a sign-out happened,
205    # from where, under which correlation id. An operator correlating a session
206    # against the sign-in and the sign-in failure entries still gets the full
207    # picture of the attempt.
208    await record_audit(session, "auth.logout", "session", None, request=request)
209    await session.commit()
210    return response
