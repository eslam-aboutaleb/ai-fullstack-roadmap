1"""Encoding shared by evidence citations and source lookup tools."""

3from __future__ import annotations
4
5import re
6from urllib.parse import quote, unquote
7
8_SPAN = re.compile(r"^(.*)\[(\d+):(\d+|None)\]$")
9
10
11def source_reference(
12    source_id: str,
13    document_id: str,
14    revision: str,
15    start: int | None = None,
16    end: int | None = None,
17) -> str:
18    base = f"{quote(source_id, safe='')}@{quote(revision, safe='')}:{quote(document_id, safe='')}"
19    return f"{base}[{start}:{end}]" if start is not None else base
20
21
22def parse_source_reference(ref: str) -> tuple[str, str, str | None, int | None, int | None]:
23    start = end = None
24    match = _SPAN.fullmatch(ref)
25    if match:
26        ref, raw_start, raw_end = match.groups()
27        start = int(raw_start)
28        end = None if raw_end == "None" else int(raw_end)
29    revision = None
30    if "@" in ref and ":" in ref:
31        source, rest = ref.split("@", 1)
32        revision, document = rest.split(":", 1)
33    elif ":" in ref:
34        source, document = ref.split(":", 1)
35    else:
36        source = document = ref
37    return unquote(source), unquote(document), unquote(revision) if revision else None, start, end
