1"""
2SQLAlchemy ORM model for the ``answer_feedback`` table.
3
4Every assistant answer that a user rates produces one row here. Until this
5table existed the eval harness scored only a static curated dataset, so a real
6answer that users routinely found wrong produced no signal at all; a row per
7(message, user) pair is the smallest unit that changes that.
8
9The rating is an enum rather than free text, and the pair (message, user) is
10unique, because the two halves of a feedback loop only close if the values that
11reach the column are comparable across users and a user can revise their mind
12without stacking a second row beside the first.
13"""
14

30class FeedbackRating(str, Enum):

33    Attributes:

36    """

39    DOWN = "down"

42class FeedbackCategory(str, Enum):
43    """Why a user rated an answer unhelpful.
44
45    Attributes:
46        WRONG_ANSWER: The answer contradicts the policy or the facts.
47        OUTDATED_POLICY: The answer reflects a superseded policy version.
48        UNHELPFUL_TONE: The answer was correct but unusable in its wording.
49        MISSING_INFO: The answer left out the part the user needed.

60class AnswerFeedback(Base):
61    """Database model for one user's rating of one assistant answer.
62
63    Attributes:
64        id: Primary key UUID.
65        message_id: Foreign key to the rated conversation message. The
66            constraint is real rather than an indexed column, so a rating can
67            never outlive the answer it describes.
68        user_id: Foreign key to the user who rated the answer, released with the
69            message. A rating no user can claim is not scoped to anyone, so it
70            has no reader left to protect and no meaning to keep.
71        rating: One of ``FeedbackRating``. Never free text, so rows from
72            different users can be counted against each other.
73        comment: Optional free-text reason the user typed alongside a rating.
74        category: Optional structured reason the user picked, one of
75            ``FeedbackCategory``.
76        created_at: Timestamp the row was written, generated by the database so
77            a row written by a worker with a skewed clock still orders
78            correctly against its neighbours.
79    """
80
81    __tablename__ = "answer_feedback"
82
83    id: Mapped[uuid.UUID] = mapped_column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
84    message_id: Mapped[uuid.UUID] = mapped_column(
85        UUID(as_uuid=True),
86        ForeignKey("conversation_messages.id", ondelete="CASCADE"),
87        nullable=False,
88        index=True,
89    )
90    user_id: Mapped[uuid.UUID] = mapped_column(
91        UUID(as_uuid=True),
92        ForeignKey("users.id", ondelete="CASCADE"),
93        nullable=False,
94        index=True,
95    )
96    rating: Mapped[str] = mapped_column(String(16), nullable=False)
97    comment: Mapped[str | None] = mapped_column(String(1000), nullable=True)
98    category: Mapped[str | None] = mapped_column(String(32), nullable=True)
99    created_at: Mapped[datetime] = mapped_column(
100        DateTime(timezone=True), nullable=False, server_default=func.now()
101    )
102
103    __table_args__ = (
104        # A user who changes their mind revises this row instead of inserting a
105        # second one, so the pair is the row's identity and every upsert keys on
106        # it. Without it two taps on the thumbs button would leave two rows and
107        # the counts would be a lie about how many users felt what.
108        UniqueConstraint("message_id", "user_id", name="uq_answer_feedback_message_user"),
109    )

112async def feedback_rows_for_messages(
113    session: AsyncSession,
114    message_ids: Iterable[uuid.UUID],
115) -> dict[uuid.UUID, list[AnswerFeedback]]:
116    """Read every rating attached to a set of messages, grouped by message.
117
118    This is the seam a future eval report joins through: a "down"-rated real
119    answer is the highest-value eval case a product can get, and the report
120    adds one by reading the rated answer's text back from
121    ``conversation_messages`` next to these rows. Nothing calls it yet -- the
122    join is deliberately not built here -- but it is the whole of the query
123    that join needs, kept with the table so the two cannot drift apart.
124
125    Args:
126        session: The session performing the read.
127        message_ids: The messages whose ratings are wanted.
128
129    Returns:
130        dict[uuid.UUID, list[AnswerFeedback]]: One entry per message that has
131        at least one rating, in insertion order. Messages with no rating are
132        absent rather than mapped to an empty list.
133    """
134    wanted = list(message_ids)
135    if not wanted:
136        return {}
137
138    rows = (
139        (
140            await session.execute(
141                select(AnswerFeedback)
142                .where(AnswerFeedback.message_id.in_(wanted))
143                .order_by(AnswerFeedback.created_at, AnswerFeedback.id)
144            )
145        )
146        .scalars()
147        .all()
148    )
149
150    grouped: dict[uuid.UUID, list[AnswerFeedback]] = {}
151    for row in rows:
152        grouped.setdefault(row.message_id, []).append(row)
153    return grouped
