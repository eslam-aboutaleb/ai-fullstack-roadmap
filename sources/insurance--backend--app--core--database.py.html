1"""
2Database session factory and engine configuration for the OmniCare backend.
3
4This module centralizes SQLAlchemy async engine setup and provides the
5``get_db`` FastAPI dependency used by every endpoint that requires a database
6session. It also exposes ``create_all_tables`` for startup-time schema
7initialization (used in development and Docker build contexts).
8
9Design decisions:
10  - ``expire_on_commit=False`` prevents SQLAlchemy from expiring ORM instances
11    after each commit, which avoids extra SELECT queries in read-heavy flows.
12  - ``create_all_tables`` imports models inside the function body rather than
13    at module scope. This ensures the declarative base metadata is fully
14    populated before ``create_all`` runs, while also avoiding circular import
15    issues during module initialization.
16"""
17
18import os
19from collections.abc import AsyncGenerator
20
21from sqlalchemy.ext.asyncio import AsyncSession, async_sessionmaker, create_async_engine
22from sqlalchemy.pool import NullPool
23
24from app.core.config import settings
25
26# Database URL resolved from centralized Pydantic settings.
27DATABASE_URL = settings.database_url
28
29# The async engine is a process-wide singleton. ``echo=False`` suppresses
30# SQL logging; enable it temporarily when debugging query performance.
31#
32# ``OMNICARE_TEST_NULLPOOL`` selects ``NullPool``, which opens a fresh connection
33# for every checkout and closes it on return. The test suite runs each request on a
34# different event loop (the ``TestClient`` portal loop, then the fixture reset's own
35# ``asyncio.run`` loop). A pooling engine can hand a live connection to a loop other
36# than the one that created it, which asyncpg rejects with "got Future attached to a
37# different loop" and which surfaces as an intermittent 401 or a missing row rather
38# than as an obvious error. NullPool makes that impossible.
39#
40# Honoured only when ENVIRONMENT=test. NullPool is correct for tests and pathological
41# under load, so a stray value in a deployment environment must not select it.
42_NULL_POOL = os.environ.get("OMNICARE_TEST_NULLPOOL") == "1" and settings.environment == "test"
43
44engine = create_async_engine(
45    DATABASE_URL,
46    echo=False,
47    poolclass=NullPool if _NULL_POOL else None,
48)
49
50# Session factory bound to the engine. Each ``async_session_factory()`` call
51# produces an independent ``AsyncSession`` that must be closed after use.
52async_session_factory = async_sessionmaker(
53    engine,
54    class_=AsyncSession,
55    expire_on_commit=False,
56)

59async def get_db() -> AsyncGenerator[AsyncSession, None]:
60    """FastAPI dependency that yields an async database session.
61
62    The session is automatically closed after the request completes, even if
63    the endpoint raises an exception. This is the canonical way to inject a
64    database session into route handlers.
65
66    Yields:
67        AsyncSession: A scoped SQLAlchemy async session for the duration of
68        the request.
69    """
70    async with async_session_factory() as session:
71        try:
72            yield session
73        finally:
74            await session.close()

77async def create_all_tables():
78    """Create all tables defined by SQLAlchemy ORM models.
79
80    Importing ``app.models`` inside the function body imports every model
81    module, which is what guarantees the declarative base metadata is fully
82    populated before ``create_all`` runs. Without that ordering, ``create_all``
83    would silently create no tables because the model subclasses would not yet
84    be registered.
85
86    Intended for development and CI setup. In production, use Alembic
87    migrations instead.
88    """
89    import app.models  # noqa: F401, PLC0415
90    from app.models.base import Base  # noqa: PLC0415
91
92    async with engine.begin() as conn:
93        await conn.run_sync(Base.metadata.create_all)
