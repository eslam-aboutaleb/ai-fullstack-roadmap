3The :class:`EmbeddingFunction` protocol is the boundary every
4embedding provider implements. Host applications depend on the
5protocol, not on a concrete provider, so providers can be swapped
6through the registry in :mod:`ragnor.embeddings.registry`.
7"""
8
9from __future__ import annotations

11from typing import Protocol, runtime_checkable
12
13
14@runtime_checkable
15class EmbeddingFunction(Protocol):
16    """Callable embedding function.
17
18    Implementations embed batches of texts and expose the
19    ``embed_query`` / ``embed_documents`` convenience surface used by
20    the retrieval and ingestion pipelines.
21    """
22
23    async def __call__(self, input: list[str]) -> list[list[float]]: ...
24
25    async def embed_query(self, input: str | list[str]) -> list[float]: ...
26
27    async def embed_documents(self, input: list[str]) -> list[list[float]]: ...
