1"""
2Minimal Redis clients used by the backend's two shared stores.
3
4The third-party ``redis`` package is deliberately not used: the rate-limit
5storage and the idempotency cache must not add a dependency the deployment
6image may not carry, and the commands they need -- PING, GET, SET with EX/NX,
7EXISTS, DEL, INCRBY, EXPIRE, PTTL and a SCAN loop for reset -- are a small,
8stable subset of the protocol. Speaking RESP directly is therefore cheaper
9than introducing a package.
10
11Two clients live here on purpose. ``RedisClient`` is blocking: SlowAPI's
12storage interface is synchronous and runs inside the request path, so that
13path needs a plain socket. ``AsyncRedisClient`` speaks the same protocol over
14asyncio streams for the idempotency backend, which is wired during startup.
15Both delegate decoding to one ``RespDecoder`` implementation so the wire
16protocol lives in exactly one place and is tested once.
17
18Both clients connect on first use and reconnect once when a connection is
19found dead, so a Redis restart costs one failed command rather than a cached
20error for the lifetime of the process.
21"""
22

50def encode_resp_command(args: Sequence[str | bytes]) -> bytes:
51    """Encode a command as a RESP array of bulk strings.
52
53    Args:
54        args: Command name followed by its arguments, in order. An integer
55            argument (a TTL, an increment) is rendered as its decimal digits,
56            because ``bytes(1)`` would silently send a NUL byte instead.
57
58    Returns:
59        bytes: The ``*<n>`` array with one ``$<len>`` bulk string per argument.
60    """
61    chunks = [b"*%d\r\n" % len(args)]
62    for arg in args:
63        if isinstance(arg, str):
64            payload = arg.encode()
65        elif isinstance(arg, int):
66            payload = str(arg).encode()
67        else:
68            payload = bytes(arg)
69        chunks.append(b"$%d\r\n%s\r\n" % (len(payload), payload))
70    return b"".join(chunks)

73class RespDecoder:
74    """Incremental decoder for the RESP2 replies this service asks for.
75
76    One shared implementation serves the blocking and the asyncio clients: both
77    feed it bytes as they arrive and pull complete values out, so the wire
78    protocol is implemented once and tested once.
79
80    Only the reply types the commands in use can produce are supported: simple
81    strings, integers, bulk strings, arrays, the null bulk string and error
82    replies.
83    """
84
85    def __init__(self) -> None:
86        self._buffer = bytearray()
87
88    def feed(self, chunk: bytes) -> None:
89        """Append bytes received from the socket to the buffer."""
90        self._buffer.extend(chunk)
91
92    def take(self) -> Any:
93        """Return the next complete reply.
94
95        Returns:
96            The decoded value: ``str`` for simple strings, ``int`` for
97            integers, ``bytes`` for bulk strings, ``list`` for arrays, and
98            ``None`` for a null reply.
99
100        Raises:
101            _IncompleteReply: If more bytes are needed, which the caller is
102                expected to handle by reading from the socket again.
103            RedisCommandError: If the reply is an error reply.
104            RedisProtocolError: If the reply is malformed.
105        """
106        try:
107            value, consumed = self._decode(0)
108        except IndexError as exc:  # pragma: no cover - guarded by _byte()
109            raise _IncompleteReply from exc
110        del self._buffer[:consumed]
111        return value
112
113    def _byte(self, position: int) -> int:
114        if position >= len(self._buffer):
115            raise _IncompleteReply
116        return self._buffer[position]
117
118    def _line(self, position: int) -> tuple[bytes, int]:
119        end = self._buffer.find(b"\r\n", position)
120        if end == -1:
121            raise _IncompleteReply
122        return bytes(self._buffer[position:end]), end + 2
123
124    def _integer(self, position: int) -> tuple[int, int]:
125        line, end = self._line(position)
126        try:
127            return int(line), end
128        except ValueError as exc:
129            raise RedisProtocolError(f"Malformed RESP integer {line!r}") from exc
130
131    def _decode(self, position: int) -> tuple[Any, int]:
132        marker = self._byte(position)
133        if marker == 0x2B:  # "+" simple string
134            line, end = self._line(position + 1)
135            return line.decode("utf-8", "replace"), end
136        if marker == 0x2D:  # "-" error reply
137            line, end = self._line(position + 1)
138            raise RedisCommandError(line.decode("utf-8", "replace"))
139        if marker == 0x3A:  # ":" integer
140            line, end = self._line(position + 1)
141            try:
142                return int(line), end
143            except ValueError as exc:
144                raise RedisProtocolError(f"Malformed RESP integer {line!r}") from exc
145        if marker == 0x24:  # "$" bulk string
146            length, end = self._integer(position + 1)
147            if length == -1:
148                return None, end
149            if len(self._buffer) < end + length + 2:
150                raise _IncompleteReply
151            return bytes(self._buffer[end : end + length]), end + length + 2
152        if marker == 0x2A:  # "*" array
153            count, end = self._integer(position + 1)
154            if count == -1:
155                return None, end
156            items: list[Any] = []
157            for _ in range(count):
158                item, end = self._decode(end)
159                items.append(item)
160            return items, end
161        raise RedisProtocolError(f"Unexpected RESP type byte {chr(marker)!r}")

164class RedisClient:
165    """Blocking Redis client used where the caller cannot await.
166
167    Rate-limiting counters are read and written by SlowAPI's synchronous storage
168    interface, which runs inside the request path, so that path needs a blocking
169    socket rather than an asyncio stream.
170
171    Attributes:
172        host: Redis host name.
173        port: Redis TCP port.
174        db: Logical database index.
175        password: Optional password sent with AUTH after connecting.
176        connect_timeout: Seconds to wait for the TCP connection.
177        socket_timeout: Seconds to wait for a reply.
178    """
179
180    def __init__(  # noqa: PLR0913 - one field per Redis connection setting
181        self,
182        host: str,
183        port: int,
184        *,
185        db: int = 0,
186        password: str | None = None,
187        connect_timeout: float = 1.0,
188        socket_timeout: float = 2.0,
189    ) -> None:
190        self.host = host
191        self.port = port
192        self.db = db
193        self.password = password
194        self.connect_timeout = connect_timeout
195        self.socket_timeout = socket_timeout
196        self._socket: Any = None
197        self._decoder = RespDecoder()
198
199    @classmethod
200    def from_url(
201        cls,
202        url: str,
203        *,
204        connect_timeout: float = 1.0,
205        socket_timeout: float = 2.0,
206    ) -> RedisClient:
207        """Build a client from a ``redis://host:port/db`` style URI.
208
209        Args:
210            url: Redis URI. A ``rediss://`` scheme is accepted and treated as
211                plain TCP because the stores are deployed inside a trusted
212                network; anything else raises.
213            connect_timeout: Seconds to wait for the TCP connection.
214            socket_timeout: Seconds to wait for a reply.
215
216        Returns:
217            RedisClient: A client that connects on first use.
218
219        Raises:
220            ValueError: If the URI is not a usable Redis address.
221        """
222        parsed = urlsplit(url)
223        if parsed.scheme not in {"redis", "rediss"} or not parsed.hostname:
224            raise ValueError(f"Not a Redis URI: {url!r}")
225        password = unquote(parsed.password) if parsed.password else None
226        db = int(unquote(parsed.path).lstrip("/") or 0)
227        return cls(
228            parsed.hostname,
229            parsed.port or 6379,
230            db=db,
231            password=password,
232            connect_timeout=connect_timeout,
233            socket_timeout=socket_timeout,
234        )

236    @property
237    def connected(self) -> bool:
238        """Whether a socket is currently open."""
239        return self._socket is not None
240
241    def close(self) -> None:
242        """Drop the socket, if any, and forget any buffered bytes."""
243        socket = self._socket
244        self._socket = None
245        self._decoder = RespDecoder()
246        if socket is not None:
247            try:
248                socket.close()
249            except OSError:  # pragma: no cover - best effort
250                logger.debug("Error while closing Redis socket", exc_info=True)
251
252    def _connect(self) -> None:
253        import socket as socket_module  # noqa: PLC0415
254
255        self._socket = socket_module.create_connection(
256            (self.host, self.port),
257            timeout=self.connect_timeout,
258        )
259        self._socket.settimeout(self.socket_timeout)
260        if self.password:
261            self._round_trip(["AUTH", self.password])
262        if self.db:
263            self._round_trip(["SELECT", str(self.db)])
264
265    def _read_reply(self) -> Any:
266        while True:
267            try:
268                return self._decoder.take()
269            except _IncompleteReply:
270                chunk = self._socket.recv(65536)
271                if not chunk:
272                    raise ConnectionResetError("Redis closed the connection") from None
273                self._decoder.feed(chunk)
274
275    def _round_trip(self, args: Sequence[str | bytes]) -> Any:
276        self._socket.sendall(encode_resp_command(args))
277        return self._read_reply()
278
279    def execute(self, *args: str | bytes) -> Any:
280        """Run one command, reconnecting once if the connection had dropped.
281
282        Args:
283            *args: Command name followed by its arguments.
284
285        Returns:
286            The decoded Redis reply.
287
288        Raises:
289            RedisUnavailable: If Redis could not be reached on both attempts.
290        """
291        try:
292            return self._execute_connected(args)
293        except (OSError, RedisProtocolError) as first_error:
294            self.close()
295            try:
296                return self._execute_connected(args)
297            except (OSError, RedisProtocolError) as second_error:
298                raise RedisUnavailable(str(second_error)) from first_error
299
300    def _execute_connected(self, args: Sequence[str | bytes]) -> Any:
301        if not self.connected:
302            self._connect()
303        return self._round_trip(args)
304
305    def ping(self) -> bool:
306        """Check that Redis answers; a connection failure is reported as False."""
307        try:
308            return self.execute("PING") in ("PONG", b"PONG")
309        except RedisUnavailable:
310            return False
311
312    def get(self, key: str) -> str | None:
313        """Return the stored string for a key, or None when it is absent."""
314        reply = self.execute("GET", key)
315        return None if reply is None else reply.decode("utf-8", "replace")
316
317    def set(self, key: str, value: str, *, ttl: int, nx: bool = False) -> bool:
318        """Store a value, optionally only when the key does not exist yet."""
319        args = ["SET", key, value, "EX", str(int(ttl))]
320        if nx:
321            args.append("NX")
322        try:
323            return self.execute(*args) in ("OK", b"OK")
324        except RedisCommandError:
325            return False
326
327    def exists(self, key: str) -> bool:
328        """Return whether a key is present."""
329        return int(self.execute("EXISTS", key)) == 1
330
331    def delete(self, *keys: str) -> None:
332        """Delete keys; missing keys are ignored."""
333        if keys:
334            self.execute("DEL", *keys)

406    @property
407    def connected(self) -> bool:
408        """Whether a connection is currently open."""

411    async def close(self) -> None:
412        """Close the connection, if any, and forget any buffered bytes."""
413        writer, self._writer, self._reader = self._writer, None, None
414        self._decoder = RespDecoder()
415        if writer is None:
416            return
417        try:
418            writer.close()
419            await writer.wait_closed()
420        except OSError:  # pragma: no cover - best effort
421            logger.debug("Error while closing Redis connection", exc_info=True)

423    async def _connect(self) -> None:
424        reader, writer = await asyncio.wait_for(
425            asyncio.open_connection(self._host, self._port),
426            timeout=self.connect_timeout,
427        )
428        self._reader, self._writer = reader, writer
429        self._decoder = RespDecoder()
430        if self._password:
431            await self._round_trip(["AUTH", self._password])
432        if self._db:
433            await self._round_trip(["SELECT", str(self._db)])
434
435    async def _read_reply(self) -> Any:
436        while True:
437            try:
438                return self._decoder.take()
439            except _IncompleteReply:
440                chunk = await self._reader.read(65536)
441                if not chunk:
442                    raise ConnectionResetError("Redis closed the connection") from None
443                self._decoder.feed(chunk)

445    async def _round_trip(self, args: Sequence[str | bytes]) -> Any:
446        self._writer.write(encode_resp_command(args))
447        await self._writer.drain()
448        return await asyncio.wait_for(self._read_reply(), timeout=self.socket_timeout)
