1"""
2Answer feedback endpoints for the OmniCare backend.
3
4Captures a rating on one assistant answer: ``POST /feedback`` stores or
5revises the caller's rating for a message, and ``DELETE /feedback/{message_id}``
6takes it back. Both are scoped to the caller, so a user can only ever rate an
7answer that was produced for them.
8
9The request and response models live here rather than in
10``app.schemas.models`` because they belong to this table and nothing else: the
11shared module is the frozen contract surface for chat, claims, and auth, and a
12feedback envelope has no reason to appear in it.
13"""

124async def submit_answer_feedback(
125    payload: AnswerFeedbackRequest,
126    request: Request,
127    current_user_id: str = Depends(get_current_user),
128    session: AsyncSession = Depends(get_db),
129) -> AnswerFeedbackResponse:
130    """Store or revise the caller's rating on one of their own answers.
131
132    Args:
133        payload: The message being rated, the rating, and any optional detail.
134        request: The request being audited; supplies the client IP and user
135            agent for the audit trail entry.
136        current_user_id: The authenticated user's UUID, injected by
137            ``get_current_user``.
138        session: Async database session.
139
140    Returns:
141        AnswerFeedbackResponse: The rating as stored after this request.
142
143    Raises:
144        HTTPException: 401 if the authenticated user id is not a valid UUID,
145            403 if the message does not belong to the caller, 422 if the
146            request body fails validation.
147    """
148    try:
149        user_uuid = uuid.UUID(current_user_id)
150    except ValueError:
151        raise HTTPException(
152            status_code=status.HTTP_401_UNAUTHORIZED,
153            detail="Invalid user",
154        ) from None
155
156    await _require_owned_message(
157        session,
158        message_id=payload.message_id,
159        user_uuid=user_uuid,
160    )
161
162    # ON CONFLICT rather than a read-then-write: the second tap on a thumbs
163    # button revises the row in one statement, so a double click cannot race a
164    # second copy into existence.
165    values = insert(AnswerFeedback).values(
166        message_id=payload.message_id,
167        user_id=user_uuid,
168        rating=payload.rating.value,
169        comment=payload.comment,
170        category=payload.category.value if payload.category else None,
171    )
172    upsert = values.on_conflict_do_update(
173        index_elements=[AnswerFeedback.message_id, AnswerFeedback.user_id],
174        set_={
175            "rating": values.excluded.rating,
176            "comment": values.excluded.comment,
177            "category": values.excluded.category,
178        },
179    )
180    await session.execute(upsert)
181
182    stored = (
183        await session.execute(
184            select(AnswerFeedback).where(
185                AnswerFeedback.message_id == payload.message_id,
186                AnswerFeedback.user_id == user_uuid,
187            )
188        )
189    ).scalar_one()
190
191    # A rating on a real answer is the product's highest-value quality signal,
192    # and it is only ever visible to admins through this trail, so the row that
193    # carries it is audited. The single commit below persists the rating and its
194    # trail entry together.
195    await record_audit(
196        session,
197        "feedback.submitted",
198        "answer_feedback",
199        str(payload.message_id),
200        after={
201            "rating": stored.rating,
202            "category": stored.category,
203            "comment": stored.comment,
204        },
205        actor_id=user_uuid,
206        request=request,
207    )
208    await session.commit()
209
210    logger.info(
211        "User '%s' rated message '%s' as '%s'.",
212        current_user_id,
213        payload.message_id,
214        stored.rating,
215    )
216
217    return AnswerFeedbackResponse(
218        id=stored.id,
219        message_id=stored.message_id,
220        rating=FeedbackRating(stored.rating),
221        comment=stored.comment,
222        category=FeedbackCategory(stored.category) if stored.category else None,
223        created_at=stored.created_at,
224    )

257async def withdraw_answer_feedback(
258    message_id: uuid.UUID,
259    current_user_id: str = Depends(get_current_user),
260    session: AsyncSession = Depends(get_db),
261) -> None:
262    """Remove the caller's rating on one of their own answers.
263
264    The delete is idempotent: a caller who never rated the message still gets a
265    204, because there is nothing left to say once the rating is gone and a
266    caller is entitled to learn nothing about ratings other than their own.
267
268    Withdrawing is recorded in the audit trail even though the delete is
269    idempotent. The trail is the only place the two states are distinguishable:
270    once the row is gone, "this user never rated it" and "this user withdrew a
271    rating" look identical, so an append-only record made before the delete is
272    what preserves the difference.
273
274    Args:
275        message_id: The rated conversation message.
276        current_user_id: The authenticated user's UUID, injected by
277            ``get_current_user``.
278        session: Async database session.
279
280    Raises:
281        HTTPException: 401 if the authenticated user id is not a valid UUID,
282            403 if the message does not belong to the caller.
283    """
284    try:
285        user_uuid = uuid.UUID(current_user_id)
286    except ValueError:
287        raise HTTPException(
288            status_code=status.HTTP_401_UNAUTHORIZED,
289            detail="Invalid user",
290        ) from None
291
292    await _require_owned_message(
293        session,
294        message_id=message_id,
295        user_uuid=user_uuid,
296    )
297
298    await record_audit(
299        session,
300        action="feedback.withdrawn",
301        entity_type="answer_feedback",
302        entity_id=str(message_id),
303        actor_id=user_uuid,
304        before={"message_id": str(message_id)},
305    )
306
307    await session.execute(
308        delete(AnswerFeedback).where(
309            AnswerFeedback.message_id == message_id,
310            AnswerFeedback.user_id == user_uuid,
311        )
312    )
313    await session.commit()
314
315    logger.info(
316        "User '%s' withdrew any rating on message '%s'.",
317        current_user_id,
318        message_id,
319    )

322async def _require_owned_message(
323    session: AsyncSession,
324    *,
325    message_id: uuid.UUID,
326    user_uuid: uuid.UUID,
327) -> ConversationMessage:
328    """Load a message only if the conversation it belongs to is the caller's.
329
330    Both predicates live in the WHERE clause and both failures raise the same
331    403, so the endpoint cannot be used to tell whether a message id exists:
332    a message belonging to another user is answered exactly like one that was
333    never issued.
334
335    Args:
336        session: The session the lookup runs in.
337        message_id: The message the caller claims to have received.
338        user_uuid: The user who must own the message's conversation.
339
340    Returns:
341        ConversationMessage: The message, which belongs to the caller.
342
343    Raises:
344        HTTPException: 403 when no message with that id belongs to the caller.
345    """
346    message = (
347        await session.execute(
348            select(ConversationMessage)
349            .join(Conversation, Conversation.id == ConversationMessage.conversation_id)
350            .where(
351                ConversationMessage.id == message_id,
352                Conversation.user_id == user_uuid,
353            )
354        )
355    ).scalar_one_or_none()
356
357    if message is None:
358        logger.warning(
359            "Feedback requested for message '%s' by non-owner '%s'; reported as forbidden.",
360            message_id,
361            user_uuid,
362        )
363        raise HTTPException(
364            status_code=status.HTTP_403_FORBIDDEN,
365            detail="This message is not available to you.",
366        )
367
368    return message
