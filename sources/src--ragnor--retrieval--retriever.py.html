1"""Generic hybrid retrieval orchestrator.
2
3The orchestrator is table-agnostic: the caller
4supplies a :class:`RetrieverSpec` naming the table
5and the metadata columns to surface, so no table
6name is hardcoded anywhere in ragnor.
7
8The retriever embeds the query, delegates the
9two-stage search (vector similarity + keyword
10search, merged with Reciprocal Rank Fusion) to the
11injected vector store, and returns
12:class:`ragnor.types.SearchResult` dataclasses.
13:func:`as_dicts` converts them back to the dict
14shape the stores return, so existing dict consumers
15keep working during the transition.
16
17A retrieval strategy (see
18:mod:`ragnor.retrieval.strategies`) can be
19injected instead: when a strategy is supplied, the
20constructor builds nothing -- no embedder, no store
21-- and retrieval delegates to the strategy, which
22owns its capability dependencies.
23"""
24

52@dataclass
53class RetrieverSpec:
54    """Description of the table a :class:`HybridRetriever` searches.
55
56    Attributes:
57        table_name: Table (or view) holding the documents.
58        id_field: Primary key column.
59        text_field: Column holding the document text.
60        metadata_fields: Columns surfaced in each result's
61            metadata dict.
62        embedding_dim: Expected embedding dimensionality. When
63            ``None``, the dimension registered for
64            ``settings.embedding_model`` is used.
65    """
66
67    table_name: str
68    id_field: str = "id"
69    text_field: str = "text"
70    metadata_fields: list[str] = field(default_factory=list)
71    embedding_dim: int | None = None
72
73    def __post_init__(self) -> None:
74        """Validate every SQL identifier in the spec."""
75        validate_identifier(self.table_name, "table_name")
76        validate_identifier(self.id_field, "id_field")
77        validate_identifier(self.text_field, "text_field")
78        for name in self.metadata_fields:
79            validate_identifier(name, "metadata_field")

82def _to_search_result(row: dict[str, Any]) -> SearchResult:
83    """Convert a store result dict to a :class:`SearchResult`."""
84    return SearchResult(
85        id=row.get("id"),
86        document=row["document"],
87        metadata=row.get("metadata") or {},
88        distance=row.get("distance"),
89        keyword_score=row.get("keyword_score"),
90        rrf_score=row.get("_rrf_score", 0.0),
91    )

94def as_dicts(results: list[SearchResult]) -> list[dict[str, Any]]:
95    """Convert :class:`SearchResult` dataclasses to dicts.
96
97    The dicts carry the same keys the vector stores return
98    (``id``, ``document``, ``metadata``, ``distance``,
99    ``keyword_score``, ``_rrf_score``), so callers written
100    against the dict-based API keep working.
101
102    Args:
103        results: Search results from :meth:`HybridRetriever.retrieve`.
104
105    Returns:
106        The equivalent list of result dicts.
107    """
108    return [
109        {
110            "id": result.id,
111            "document": result.document,
112            "metadata": result.metadata,
113            "distance": result.distance,
114            "keyword_score": result.keyword_score,
115            "_rrf_score": result.rrf_score,
116        }
117        for result in results
118    ]

121class HybridRetriever:
122    """Hybrid retrieval over an arbitrary document table.
123
124    By default the retriever resolves the embedder
125    and the hybrid store from ``settings`` and
126    ``spec``. When a retrieval strategy is injected,
127    the constructor builds nothing -- no embedder, no
128    store -- and :meth:`retrieve` delegates to the
129    strategy, which owns its capability dependencies
130    (for example a pgvector
131    :class:`ragnor.stores.base.VectorIndex` plus an
132    external lexical engine adapted to
133    :class:`ragnor.stores.base.LexicalIndex`, or a
134    keyword-only strategy with no embedding
135    dependency at all).
136    """
137
138    def __init__(  # noqa: PLR0913
139        self,
140        spec: RetrieverSpec,
141        settings: RagSettings,
142        session_factory: SessionProvider | None = None,
143        strategy: Any = None,
144        event_sink: Any | None = None,
145    ) -> None:
146        """Initialize the retriever.
147
148        Args:
149            spec: Table and column description of the index.
150            settings: Settings selecting the embedding provider
151                and carrying the default distance threshold.
152            session_factory: Session provider the vector store
153                uses when a method is called without a session.
154                When omitted, the store opens its own sessions
155                from ``settings.database_url``.
156            strategy: Optional retrieval strategy (see
157                :mod:`ragnor.retrieval.strategies`). When
158                supplied, the constructor builds nothing --
159                no embedder and no store -- and retrieval
160                delegates to the strategy. When omitted (the
161                default), the embedder and the hybrid store
162                are built from ``settings`` and ``spec``.
163            event_sink: Optional sink for telemetry events.
164        """
165        self._spec = spec
166        self._settings = settings
167        self._session_factory = session_factory
168        self._strategy = strategy
169        self._event_sink = event_sink if event_sink is not None else NullEventSink()
170        if strategy is None:
171            self._embed_fn = get_embedding_function(settings)
172            embedding_dim = spec.embedding_dim
173            if embedding_dim is None:
174                profile = getattr(self._embed_fn, "profile", None)
175                embedding_dim = (
176                    profile.dimension
177                    if profile is not None
178                    else get_embedding_dimension(settings.embedding_model)
179                )
180            self._store = get_vector_store(
181                table_name=spec.table_name,
182                id_field=spec.id_field,
183                embedding_dim=embedding_dim,
184                settings=settings,
185                session_factory=session_factory,
186                embedding_fn=self._embed_fn,
187            )
188        else:
189            self._embed_fn = None
190            self._store = None

192    async def retrieve(  # noqa: PLR0913
193        self,
194        query: str | RetrievalRequest,
195        n_results: int = 5,
196        distance_threshold: float | None = None,
197        strict: bool = False,
198        candidate_k: int | None = None,
199        **filters: Any,
200    ) -> list[SearchResult]:
201        """Retrieve documents matching ``query``.
202
203        Args:
204            query: The natural-language search query or a typed
205                :class:`~ragnor.retrieval.types.RetrievalRequest`.
206            n_results: Maximum number of candidates to fetch.
207            distance_threshold: Maximum L2 distance for the vector
208                search part. When ``None``, falls back to
209                ``settings.rag_distance_threshold``.
210            strict: When ``True``, store failures raise
211                :class:`ragnor.types.RetrievalError`.
212            candidate_k: Optional candidate count for initial stages.
213            **filters: Additional equality filters.
214
215        Returns:
216            Search results sorted by RRF score.
217        """
218        t0 = time.perf_counter()
219        if isinstance(query, RetrievalRequest):
220            req = replace(query, filters={**query.filters, **filters})
221        else:
222            req, _ = _coerce_request(
223                query,
224                n_results,
225                {
226                    **filters,
227                    "candidate_k": candidate_k,
228                    "threshold": distance_threshold,
229                    "strict": strict,
230                    "text_field": self._spec.text_field,
231                    "metadata_fields": self._spec.metadata_fields,
232                },
233            )
234        rows = await self._retrieve_rows(req)
235        n_results = req.final_k
236        candidate_k = req.get_candidate_k()
237
238        duration_ms = (time.perf_counter() - t0) * 1000.0
239        results = [_to_search_result(row) for row in rows]
240        self._event_sink.emit(
241            TelemetryEvent(
242                event_kind=EventKind.SEARCH_COMPLETED,
243                stage="retrieval",
244                duration_ms=duration_ms,
245                counts={
246                    "candidate_k": candidate_k if candidate_k is not None else n_results,
247                    "final_k": n_results,
248                    "result_count": len(results),
249                },
250            )
251        )
252        return results

254    async def _retrieve_rows(self, request: RetrievalRequest) -> list[dict[str, Any]]:
255        fields = list(request.metadata_fields)
256        if request.isolation_scope is not None:
257            fields.append(request.scope_field)
258        if request.expected_generation is not None:
259            fields.append("generation")
260        req = replace(
261            request,
262            threshold=(
263                request.threshold
264                if request.threshold is not None
265                else self._settings.rag_distance_threshold
266            ),
267            metadata_fields=list(dict.fromkeys(fields)),
268        )
269        if self._strategy is not None:
270            if hasattr(self._strategy, "retrieve_typed"):
271                response = await self._strategy.retrieve_typed(req)
272                rows = response.to_dicts()
273            else:
274                caps = get_backend_capabilities(self._strategy)
275                caps.validate_constraints(req.mandatory_capabilities)
276                caps.validate_filters(req.effective_filters())
277                rows = await self._strategy.retrieve(
278                    query=req.query,
279                    n_results=req.final_k,
280                    candidate_k=req.get_candidate_k(),
281                    threshold=req.threshold,
282                    text_field=req.text_field,
283                    metadata_fields=req.metadata_fields,
284                    session=req.session,
285                    strict=req.strict,
286                    **req.effective_filters(),
287                )
288        else:
289            caps = get_backend_capabilities(self._store)
290            caps.validate_constraints(req.mandatory_capabilities)
291            caps.validate_filters(req.effective_filters())
292            query_vector = await _query_vector(self._embed_fn, req.query)
293            rows = await self._store.hybrid_search(
294                query=req.query,
295                embedding=query_vector,
296                n_results=req.get_candidate_k(),
297                threshold=req.threshold,
298                text_field=req.text_field,
299                metadata_fields=req.metadata_fields,
300                session=req.session,
301                strict=req.strict,
302                **req.effective_filters(),
303            )
304        return _validate_scope_defense_in_depth(rows, req)[: req.final_k]
305
306    async def retrieve_typed(self, request: RetrievalRequest) -> RetrievalResponse:
307        """Execute the same constrained path while preserving reranker scores."""
308        rows = await self._retrieve_rows(request)
309        return RetrievalResponse.from_dicts(
310            rows,
311            query=request.query,
312            candidate_k=request.get_candidate_k(),
313            final_k=request.final_k,
314        )
