1"""
2Context variables for propagating request-scoped state through the OmniCare
3backend without explicit parameter threading.
4
5This module defines ``current_user_id``, a ``ContextVar`` that holds the UUID
6of the authenticated user for the duration of an async request.
7
8Why ContextVars instead of global state?
9  - FastAPI runs many requests concurrently on a single event loop. A global
10    variable would be shared across requests, causing data races where one
11    user's ID leaks into another user's agent tool execution.
12  - ``ContextVar`` is designed for exactly this problem: it provides
13    request-local storage that flows through ``async/await`` boundaries and
14    is isolated per concurrent task.
15
16Usage pattern:
17  - ``get_current_user`` (in ``app/security/auth.py``) sets ``current_user_id``
18    after successful JWT validation.
19  - Agent tools and other downstream code read ``current_user_id.get()`` to
20    scope database queries and audit log entries to the correct user.
21"""
22
23import contextvars
24import uuid
25
26# Context variable holding the authenticated user's UUID for the current
27# async request context. This is set by ``get_current_user`` and read by
28# agent tools and business logic to enforce user-scoped access control.
29current_user_id: contextvars.ContextVar[uuid.UUID] = contextvars.ContextVar("current_user_id")
