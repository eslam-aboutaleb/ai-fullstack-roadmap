23async def get_claim_status(claim_id: str, policy_number: str = "") -> dict[str, Any]:
24    """Look up the status of an existing OmniCare insurance claim by its ID.
25
26    Use this tool when a user asks about the status, progress, or details of
27    a specific claim. The claim ID is required. The policy number is optional
28    and currently unused but reserved for future enhanced filtering.
29
30    The lookup is scoped to the authenticated user via ``current_user_id``,
31    ensuring that users can only retrieve their own claim information.
32
33    Args:
34        claim_id: The unique claim identifier provided by the user.
35            Example: "CLM-8821"
36        policy_number: The policyholder's policy number. Currently unused
37            but accepted for forward compatibility.
38
39    Returns:
40        dict: Claim status details. On success, includes ``found=True``,
41        ``claim_id``, ``policy_number``, ``claim_type``, ``status``,
42        ``amount``, and ``citation``. On failure, includes ``found=False``
43        and an ``error`` message.
44    """
45    try:
46        user_uuid = current_user_id.get()
47    except LookupError:
48        logger.error("current_user_id not found in context.")
49        return {"found": False, "error": "Unauthorized claim lookup."}
50
51    # Normalize the claim ID to uppercase to handle case-insensitive input
52    # while matching the stored format.
53    normalised_id = claim_id.strip().upper()
54
55    try:
56        async with async_session_factory() as session:
57            # SECURITY: The WHERE clause includes both claim_id and owner_id.
58            # This dual filter prevents IDOR attacks where an attacker supplies
59            # a valid claim ID belonging to another user.
60            stmt = select(Claim).where(Claim.claim_id == normalised_id, Claim.owner_id == user_uuid)
61            result = await session.execute(stmt)
62            claim = result.scalar_one_or_none()
63
64            if not claim:
65                logger.warning(
66                    "IDOR attempt or claim not found: claim %s requested by user %s",
67                    normalised_id,
68                    user_uuid,
69                )
70                return {
71                    "found": False,
72                    "error": (
73                        f"No claim found with ID '{claim_id}'. "
74                        "Please double-check the claim ID and try again, "
75                        "or contact OmniCare support at 1-800-OMNICARE."
76                    ),
77                }
78
79            logger.info("Claim '%s' found securely for owner '%s'.", normalised_id, user_uuid)
80            return {
81                "found": True,
82                "claim_id": claim.claim_id,
83                "policy_number": claim.policy_number,
84                "claim_type": claim.claim_type,
85                "status": claim.status,
86                "amount": claim.amount,
87                "citation": (
88                    f"Claim {claim.claim_id} in the OmniCare claims system "
89                    f"(policy {claim.policy_number})."
90                ),
91            }
92    except Exception as exc:
93        logger.exception("Database error during claim lookup: %s", exc)
94        return {
95            "found": False,
96            "error": "Unable to retrieve claims at this time. Please contact support.",
97        }
