29class HashingEmbedder:
30    """Deterministic offline bag-of-words embedder."""
31
32    def __init__(self, dim: int = 16) -> None:
33        self.dim = dim
34
35    def embed(self, text: str) -> list[float]:
36        vec = [0.0] * self.dim
37        for token in re.findall(r"[a-z0-9]+", text.lower()):
38            digest = hashlib.sha256(token.encode("utf-8")).digest()
39            idx = int.from_bytes(digest[:2], "big") % self.dim
40            vec[idx] += 1.0
41        norm = math.sqrt(sum(x * x for x in vec))
42        if norm > 0.0:
43            vec = [x / norm for x in vec]
44        return vec
45
46    async def __call__(self, texts: list[str] | str) -> list[list[float]] | list[float]:
47        if isinstance(texts, list):
48            return [self.embed(t) for t in texts]
49        return self.embed(texts)
50
51    async def embed_query(self, query: str) -> list[float]:
52        return self.embed(query)
53
54    async def embed_documents(self, texts: list[str]) -> list[list[float]]:
55        return [self.embed(t) for t in texts]

75async def run_recipe() -> None:
76    """Run the policy knowledge recipe."""
77    print("--- 1. Initializing Offline Store & Bridge ---")
78    embedder = HashingEmbedder(dim=16)
79    vector_store = InMemoryVectorStore(embedding_dim=16, embedding_fn=embedder)
80    snapshot_store = MemorySnapshotStore()
81    source_lookup = SourceLookup(snapshot_store)
82    assembler = EvidenceAssembler(tokenizer=ApproximateTokenizer())
83
84    bridge = KnowledgeBridge(
85        retriever=vector_store,
86        source_lookup=source_lookup,
87        assembler=assembler,
88        allow_management=True,
89    )
90
91    # Ingest Policy v1
92    policy_doc_v1 = (
93        "Page 1: Travel and Expense Policy (rev-1)\n"
94        "All business travel must be booked at least 14 days in advance.\n"
95        "Meals are reimbursed up to $75 per diem with receipts.\n\n"
96        "Page 2: Equipment and Hardware Policy (rev-1)\n"
97        "Laptops are refreshed every 3 years. Remote monitors are eligible up to $300."
98    )
99
100    # Register immutable snapshot for rev-1
101    snapshot_store.put(
102        SnapshotRecord(
103            source_id="hr_portal",
104            document_id="travel_policy",
105            revision="rev-1",
106            content=policy_doc_v1,
107            isolation_scope="default",
108            metadata={"title": "Company Expense and Hardware Guidelines", "department": "HR"},
109        )
110    )
111
112    # Ingest chunks into vector store
113    chunk_1 = {
114        "id": "chunk_rev1_p1",
115        "text": (
116            "All business travel must be booked at least 14 days in advance. "
117            "Meals are reimbursed up to $75 per diem."
118        ),
119        "embedding": embedder.embed(
120            "travel expense reimbursement booked 14 days in advance meals $75"
121        ),
122        "metadata": {
123            "source_id": "hr_portal",
124            "document_id": "travel_policy",
125            "revision": "rev-1",
126            "page": 1,
127            "section": "Travel and Expense",
128            "title": "Expense Guidelines",
129            "start_offset": 0,
130            "end_offset": 160,
131        },
132    }
133    chunk_2 = {
134        "id": "chunk_rev1_p2",
135        "text": "Laptops are refreshed every 3 years. Remote monitors are eligible up to $300.",
136        "embedding": embedder.embed("hardware equipment laptop monitor refreshed 3 years $300"),
137        "metadata": {
138            "source_id": "hr_portal",
139            "document_id": "travel_policy",
140            "revision": "rev-1",
141            "page": 2,
142            "section": "Equipment and Hardware",
143            "title": "Expense Guidelines",
144            "start_offset": 161,
145            "end_offset": 280,
146        },
147    }
148    await vector_store.upsert([chunk_1, chunk_2])
149
150    print("--- 2. Querying Knowledge: Policy with Page Citations ---")
151    bundle: EvidenceBundle = await bridge.search_knowledge(
152        query="what is the meal reimbursement per diem",
153        budget_limit=2000,
154        final_k=3,
155    )
156    print(f"Bundle status: {bundle.status}")
157    print(f"Evidence items found: {len(bundle.items)}")
158    for item in bundle.items:
159        loc = f"Page {item.page}, Section: {item.section}"
160        print(f"Citation: {item.citation_label} [{loc}] (rev: {item.revision})")
161        print(f"  Excerpt: {item.text}")
162
163    assert len(bundle.items) > 0
164    assert bundle.items[0].page == 1
165    assert bundle.items[0].revision == "rev-1"
166
167    print("\n--- 3. Updating Policy to rev-2 (New Revision) ---")
168    policy_doc_v2 = (
169        "Page 1: Travel and Expense Policy (rev-2)\n"
170        "All business travel must be booked at least 21 days in advance.\n"
171        "Meals are reimbursed up to $100 per diem with receipts.\n\n"
172        "Page 2: Equipment and Hardware Policy (rev-2)\n"
173        "Laptops are refreshed every 2 years. Remote monitors are eligible up to $400."
174    )
175    snapshot_store.put(
176        SnapshotRecord(
177            source_id="hr_portal",
178            document_id="travel_policy",
179            revision="rev-2",
180            content=policy_doc_v2,
181            isolation_scope="default",
182            metadata={"title": "Company Expense Guidelines", "department": "HR"},
183        )
184    )
185
186    # Delete old chunks and add rev-2 chunks
187    await vector_store.delete(ids=["chunk_rev1_p1", "chunk_rev1_p2"])
188    chunk_1_v2 = {
189        "id": "chunk_rev2_p1",
190        "text": (
191            "All business travel must be booked at least 21 days in advance. "
192            "Meals are reimbursed up to $100 per diem."
193        ),
194        "embedding": embedder.embed(
195            "travel expense reimbursement booked 21 days in advance meals $100"
196        ),
197        "metadata": {
198            "source_id": "hr_portal",
199            "document_id": "travel_policy",
200            "revision": "rev-2",
201            "page": 1,
202            "section": "Travel and Expense",
203            "title": "Expense Guidelines",
204            "start_offset": 0,
205            "end_offset": 165,
206        },
207    }
208    await vector_store.upsert([chunk_1_v2])
209
210    print("--- 4. Verifying Citation Resolution to Stated Revision ---")
211    # Verify lookup for old citation rev-1 returns rev-1 ($75), NOT rev-2 ($100)
212    old_citation_span = await bridge.get_source(
213        source_ref="hr_portal",
214        document_id="travel_policy",
215        revision="rev-1",
216        start_offset=0,
217        end_offset=200,
218    )
219
220    print(f"Old citation lookup status: {old_citation_span.status}")
221    print(f"Resolved text: {old_citation_span.text[:80]}...")
222    assert "$75 per diem" in old_citation_span.text
223    assert old_citation_span.revision == "rev-1"
224
225    # Search for latest policy retrieves rev-2
226    new_bundle = await bridge.search_knowledge(query="what is the meal reimbursement per diem")
227    assert "$100 per diem" in new_bundle.items[0].text
228    assert new_bundle.items[0].revision == "rev-2"
229    print(
230        "Citation resolution verified: never silently substitutes newer content for old citations!"
231    )
232
233
234if __name__ == "__main__":
235    asyncio.run(run_recipe())
