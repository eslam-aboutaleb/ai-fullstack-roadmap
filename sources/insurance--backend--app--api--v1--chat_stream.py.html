1"""
2Streaming chat endpoint.
3
4POST /api/v1/chat/stream - Stream ADK SSE events for real-time chat responses.
5
6This endpoint returns a ``StreamingResponse`` that emits Server-Sent Events
7(SSE) formatted strings produced by the Google ADK agent runner. The client
8is responsible for parsing the SSE stream and rendering partial responses
9as they arrive.
10
11SSE event contract
12------------------
13
14Every frame is a single ``data:`` line holding a JSON object with a ``type``
15field, built from the stream models in ``app.schemas.models`` rather than
16interpolated strings:
17
18  - ``text_delta``        ``{"type": "text_delta", "content": "..."}``
19    One fragment of the assistant's answer.
20  - ``tool_call``         ``{"type": "tool_call", "tool": "...", "arguments": {...}}``
21    The agent invoked a tool.
22  - ``tool_result``       ``{"type": "tool_result", "tool": "...", "result": {...}}``
23    The tool returned a value.
24  - ``tool_error``        ``{"type": "tool_error", "tool": "...", "error": "..."}``
25    The tool reported its own failure.
26  - ``error``             ``{"type": "error", "message": "...", "code": "stream_failed"}``
27    The model stream failed. It is never the terminal frame.
28
29Every stream terminates with ``response_complete``, a failed one included,
30because the client relies on that frame to stop waiting and to render the
31sidebar entry:
32
33  - ``response_complete`` ``{"type": "response_complete", "sources": [...],
34    "status": "ok", "message_id": "...", "user_message_id": "..."}``
35
36``status`` is ``"ok"`` for a turn the agent finished and ``"error"`` for one
37that failed, which is how a client tells a truncated answer from a complete
38one: the ``error`` frame says why the stream stopped and the
39``response_complete`` that always follows says whether it stopped.
40``sources`` lists the citations collected while the turn ran and is empty on
41the error path.
42
43``message_id`` and ``user_message_id`` are the backend-assigned ids of the
44persisted assistant answer and user message. The turn is persisted *before*
45this frame is emitted, so the terminal frame can carry the ids the feedback
46endpoint will authorise against: the client adopts ``message_id`` for the answer
47it just streamed instead of rating a locally-generated id that matches no row.
48Both are ``null`` when the turn could not be persisted.
49
50The two travel together by construction: an ``error`` frame is always
51followed by ``response_complete`` carrying ``"status": "error"``, and
52``"status": "ok"`` is sent exactly when no ``error`` frame was sent.
53"""

79def _reports_failure(frame: str) -> bool:
80    """Report whether an already-mapped frame is the ``error`` event.
81
82    ``_transform_adk_event`` is a pure mapping, so a payload it rejects is
83    reported to the client through the ``error`` frame it returns rather than
84    by raising. Reading the frame back is what keeps the terminal ``status``
85    honest about why the turn stopped.
86
87    Args:
88        frame: The SSE frame produced by ``_transform_adk_event``.
89
90    Returns:
91        bool: True when the frame carries the ``error`` discriminator.
92    """
93    try:
94        payload = json.loads(frame.removeprefix("data: ").strip())
95    except json.JSONDecodeError:
96        return False
97    return isinstance(payload, dict) and payload.get("type") == "error"

100def _transform_adk_event(event_json: str, sources: list[Any] | None = None) -> str | None:
101    """Transform raw ADK event into frontend-friendly SSE event.
102
103    Converts ADK streaming events into a stable frontend contract so the
104    client does not need to understand ADK internals. Tool activity is
105    already emitted as typed events by ``run_agent_stream``; this only maps
106    the ADK text and final-response envelopes.
107
108    Every branch returns a frame whose payload is JSON: events the contract
109    maps are re-serialized from a model, and events it only forwards are
110    round-tripped through ``json`` so the client never receives a payload
111    ``JSON.parse`` would reject.
112
113    Args:
114        event_json: Raw JSON string of an ADK event, possibly prefixed
115            with ``data: `` from the SSE formatter in ``run_agent_stream``.
116        sources: Accumulated source citations collected during streaming,
117            or ``None`` if not yet available.
118
119    Returns:
120        str | None: A formatted SSE ``data:`` line, or ``None`` if the
121            event should be dropped.
122    """
123    raw = event_json.strip()
124    if raw.startswith("data: "):
125        raw = raw[6:].strip()
126
127    try:
128        event = json.loads(raw)
129    except (json.JSONDecodeError, TypeError):
130        # Forwarding a payload we cannot parse would hand the client a parse
131        # error it cannot act on. Emitting a typed error frame instead keeps
132        # the failure visible while the stream still terminates normally.
133        return sse_frame(StreamError(message="Malformed event received", code="stream_failed"))
134
135    if not isinstance(event, dict):
136        return sse_frame(StreamError(message="Malformed event received", code="stream_failed"))
137
138    event_type = event.get("type", "")
139
140    # ``is_final_response`` is a method on ADK's Event, so it is not expected in
141    # the serialized payload ``run_agent_stream`` forwards -- but it is cheap to
142    # honour and the tests pin the behaviour, so a payload that does carry it
143    # still produces exactly one terminal frame here. Should ADK ever start
144    # serializing it, this and the ``finally`` block would both emit
145    # ``response_complete``; the guard against that belongs wherever the frames
146    # are counted, not in a branch that silently changes shape.
147    if event.get("is_final_response"):
148        return sse_frame(
149            StreamResponseComplete(sources=sources or [], status=_STREAM_OK),
150        )
151
152    # ADK reports text inside ``content.parts`` and may omit ``content``
153    # entirely. Anything that is not an object there has no parts to read, so
154    # the event is forwarded instead of being turned into a stream failure.
155    content = event.get("content")
156    parts = content.get("parts", []) if isinstance(content, dict) else []
157    if event_type == "text_delta" or parts:
158        text_deltas = [p.get("text", "") for p in parts if isinstance(p, dict) and p.get("text")]
159        if text_deltas:
160            return sse_frame(StreamTextDelta(content="".join(text_deltas)))
161
162    return f"data: {json.dumps(event)}\n\n"

184@limiter.limit("20/minute")
185async def chat_stream(
186    payload: ChatRequest,
187    request: Request,
188    current_user_id: str = Depends(get_current_user),
189) -> StreamingResponse:
190    """Stream a chat interaction response as Server-Sent Events.
191
192    Validates the request payload, then delegates to the ADK agent runner
193    in streaming mode. Each ADK event is mapped onto a typed event envelope
194    and yielded as an SSE ``data:`` chunk. After the stream completes, the
195    conversation turn is persisted to the database before the stream closes.
196
197    Args:
198        payload: Validated chat request containing the user message.
199        request: The incoming FastAPI request (used for rate limiting).
200        current_user_id: The authenticated user's UUID (injected by dependency).
201
202    Returns:
203        StreamingResponse: An SSE stream with media type
204            ``text/event-stream``.
205
206    Raises:
207        HTTPException: 422 if the message is empty or whitespace-only.
208        HTTPException: 500 if the streaming generator raises an unexpected error.
209    """
210    message = payload.message
211
212    if not message.strip():
213        raise HTTPException(status_code=422, detail="Message cannot be empty")
214
215    async def event_generator():
216        stream_result = _StreamResult()
217        stream_failed = False
218        try:
219            async for chunk in run_agent_stream(
220                user_id=current_user_id,
221                message=message,
222                result=stream_result,
223            ):
224                if await request.is_disconnected():
225                    logger.info("Client disconnected during streaming")
226                    return
227
228                transformed = _transform_adk_event(chunk, stream_result.sources)
229                if transformed is not None:
230                    # A mapping failure surfaces as an ``error`` frame, so the
231                    # turn is flagged failed too: the terminal ``status`` must
232                    # never call a broken stream complete.
233                    if _reports_failure(transformed):
234                        stream_failed = True
235                    yield transformed
236        except ValueError as err:
237            # Suppress OpenTelemetry context token mismatch that can occur
238            # when the ADK generator is closed across async contexts: it is a
239            # teardown artifact rather than a model failure, so it is not
240            # logged as an exception. The turn still stopped short, so the
241            # client is told that as well.
242            if "was created in a different Context" in str(err):
243                logger.debug("Suppressed OpenTelemetry context mismatch during streaming: %s", err)
244            else:
245                logger.exception("ValueError during streaming: %s", err)
246            yield sse_frame(StreamError(message="Streaming failed", code="stream_failed"))
247            stream_failed = True
248        except Exception as err:
249            logger.exception("Error during streaming: %s", err)
250            yield sse_frame(StreamError(message="Streaming failed", code="stream_failed"))
251            stream_failed = True
252        finally:
253            # Persist the turn first, then emit response_complete, so the
254            # terminal frame can carry the backend-assigned message ids. The
255            # id a client rates must match the row the database wrote -- a
256            # client-generated id matches nothing and the feedback endpoint
257            # refuses it with a 403 -- so the ids have to exist before the
258            # frame that reports them. The turn is awaited here (rather than
259            # fired off with asyncio.create_task) so the stream only closes once
260            # the turn is durably saved; a detached task could otherwise commit
261            # after the response finished and race with the next request or a
262            # test's database reset. Every failure path still reaches this block
263            # (a failed stream included) and _persist_streamed_turn swallows its
264            # own errors, so the terminal frame is always emitted and reports
265            # null ids rather than raising.
266            saved_ids = SavedMessageIds()
267            if stream_result.session_id:
268                saved_ids = await _persist_streamed_turn(
269                    user_id=current_user_id,
270                    session_id=stream_result.session_id,
271                    message=message,
272                    result=stream_result,
273                )
274
275            yield sse_frame(
276                StreamResponseComplete(
277                    sources=stream_result.sources,
278                    status=_STREAM_FAILED if stream_failed else _STREAM_OK,
279                    message_id=saved_ids.assistant_message_id,
280                    user_message_id=saved_ids.user_message_id,
281                )
282            )
283
284    return StreamingResponse(event_generator(), media_type="text/event-stream")

287async def _persist_streamed_turn(
288    user_id: str,
289    session_id: str,
290    message: str,
291    result: _StreamResult,
292) -> SavedMessageIds:
293    """Persist a streamed conversation turn and report its message ids.
294
295    This function runs when the SSE stream finishes, before the terminal
296    frame is emitted. It catches and logs any errors without affecting the
297    streaming response, so a persistence failure surfaces as an empty
298    :class:`SavedMessageIds` (both ids ``None``) rather than a raised error.
299
300    Args:
301        user_id: The authenticated user's ID.
302        session_id: The ADK session ID.
303        message: The user's original message.
304        result: The populated _StreamResult from run_agent_stream.
305
306    Returns:
307        SavedMessageIds: The ids the backend assigned to the persisted user and
308            assistant messages, or an empty instance when the save failed.
309    """
310    try:
311        response_text = "\n".join(result.final_text_parts) if result.final_text_parts else ""
312        saved = await save_conversation_turn(
313            user_id=user_id,
314            session_id=session_id,
315            message=message,
316            response_text=response_text,
317            sources=result.sources,
318            tool_calls=result.tool_calls,
319        )
320    except Exception as exc:
321        logger.error("Failed to persist streamed conversation turn: %s", exc, exc_info=True)
322        return SavedMessageIds()
323
324    # save_conversation_turn returns SavedMessageIds in production; coerce
325    # anything else (for example a test double) to the empty instance so the
326    # terminal frame can read .assistant_message_id / .user_message_id safely.
327    return saved if isinstance(saved, SavedMessageIds) else SavedMessageIds()
