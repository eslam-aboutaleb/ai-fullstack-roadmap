1"""Qdrant implementation of the VectorStore interface.
2
3Provides hybrid search (vector similarity + token-overlap
4keyword matching with Reciprocal Rank Fusion), document
5counting, lifecycle operations, and upsert operations
6against a Qdrant collection.
7
8The store subclasses :class:`ragnor.stores.base.HybridVectorStore`:
9the vector stage lives in :meth:`vector_search` (Qdrant's
10native nearest-neighbor search) and the keyword stage in
11:meth:`keyword_search` (an in-process token-overlap scan over
12the collection's payloads, because Qdrant has no built-in
13full-text rank). The inherited composite ``hybrid_search``
14runs both stages and fuses them with RRF, so the result
15shape matches the other backends.
16
17Keyword-search performance notice: the keyword stage is an
18in-process fallback that scrolls every document in the
19collection and scores token overlap. It is intended for
20local development, small corpora, and testing -- not for
21production-scale keyword retrieval. Hosts at scale combine a
22pgvector :class:`VectorIndex` with an external lexical
23engine adapted to :class:`ragnor.stores.base.LexicalIndex`.
24
25Point ids: Qdrant identifies points by unsigned integers or
26UUIDs, so string document ids are mapped deterministically to
27a UUID (``uuid5``) and the original id is kept in the payload
28under the store's ``id_field``.
29
30Configuration (no host imports): the store receives everything
31through constructor arguments. The Qdrant client is created
32lazily from ``url``/``api_key`` (or ``settings.database_url``)
33on first use, or injected directly via ``client`` for hosts
34and tests that manage their own client.
35
36Distance metric: the store accepts the uniform ``metric``
37constructor kwarg so the registry can build every backend the
38same way, but the cosine metric is NOT supported here -- only
39the ``l2`` default is. Constructing the store with the cosine
40metric raises ``ValueError``; cosine ranking is available from
41the pgvector, memory, and faiss backends.
42"""

76def _point_id(doc_id: Any) -> Any:
77    """Map a document id to a Qdrant point id.
78
79    Qdrant point ids are unsigned integers or UUIDs. Integer
80    and UUID ids pass through; anything else (typically a
81    string id) is mapped deterministically to a UUID derived
82    from its string form. The original id is stored in the
83    payload under the store's ``id_field``.
84    """
85    if isinstance(doc_id, (int, uuid.UUID)):
86        return doc_id
87    return uuid.uuid5(_ID_NAMESPACE, str(doc_id))

90def _equality_filter(filters: dict[str, Any]) -> Any | None:
91    """Build a Qdrant ``Filter`` from equality filters, or ``None``."""
92    if not filters:
93        return None
94    from qdrant_client import FieldCondition, Filter, MatchValue  # noqa: PLC0415
95
96    return Filter(
97        must=[
98            FieldCondition(key=key, match=MatchValue(value=value)) for key, value in filters.items()
99        ]
100    )

103def _selector_filter(
104    ids: list[Any] | None,
105    filters: dict[str, Any] | None,
106    id_field: str,
107) -> Any | None:
108    """Build a Qdrant ``Filter`` selecting by ids and/or equality filters.
109
110    The id restriction is a ``MatchAny`` on the payload's
111    ``id_field``; equality filters become ``MatchValue``
112    conditions. Both restrictions apply (conjunction) when
113    both are supplied. Returns ``None`` when neither is
114    supplied (matching every point).
115    """
116    from qdrant_client import FieldCondition, Filter, MatchAny, MatchValue  # noqa: PLC0415
117
118    must: list[Any] = []
119    if ids is not None:
120        must.append(FieldCondition(key=id_field, match=MatchAny(any=list(ids))))
121    for key, value in (filters or {}).items():
122        must.append(FieldCondition(key=key, match=MatchValue(value=value)))
123    return Filter(must=must) if must else None

135    def __init__(  # noqa: PLR0913, PLR0917
136        self,
137        table_name: str,
138        id_field: str = "id",
139        embedding_dim: int | None = None,
140        url: str | None = None,
141        api_key: str | None = None,
142        settings: RagSettings | None = None,
143        embedding_fn: EmbeddingFunction | None = None,
144        client: Any | None = None,
145        session_factory: Any | None = None,
146        metric: str | None = None,
147    ):
148        """Initialize the Qdrant store.
149
150        Args:
151            table_name: Name of the Qdrant collection.
152            id_field: Payload field holding the document id.
153            embedding_dim: Dimensionality of the embedding
154                vectors. Required; pass the dimension configured
155                for the active embedding model.
156            url: Qdrant server URL (e.g. ``http://localhost:6333``).
157                Falls back to ``settings.database_url``.
158            api_key: API key for authenticated Qdrant servers.
159            settings: Host settings; ``database_url`` is used as
160                the Qdrant URL when ``url`` is omitted, and the
161                optional ``rag_distance_metric`` field selects the
162                metric when ``metric`` is omitted.
163            embedding_fn: Async callable that embeds a list of
164                texts, used to auto-embed queries.
165            client: Pre-built Qdrant client (an
166                ``AsyncQdrantClient``). When omitted, one is
167                created lazily from ``url``/``api_key``.
168            session_factory: Accepted for constructor
169                compatibility with the other stores; ignored
170                (Qdrant manages its own connections).
171            metric: Distance metric for the vector stage. Only
172                ``l2`` (the default) is supported by this
173                backend.
174
175        Raises:
176            ValueError: If ``table_name`` or ``id_field`` contain
177                unsafe characters, if ``embedding_dim`` is omitted,
178                if ``metric`` is not a supported distance metric,
179                or if the cosine metric is requested (unsupported
180                by this backend).
181        """
182        super().__init__(
183            table_name=table_name,
184            id_field=id_field,
185            embedding_dim=embedding_dim,
186            session_factory=session_factory,
187            settings=settings,
188            embedding_fn=embedding_fn,
189            metric=metric,
190        )
191        if self.metric == "cosine":
192            raise ValueError(
193                "the 'cosine' distance metric is unsupported by the Qdrant "
194                "backend; use l2 (the default), or switch to the pgvector, "
195                "memory, or faiss backend for cosine ranking"
196            )
197        if embedding_dim is None:
198            raise ValueError(
199                "embedding_dim is required; pass the dimension configured "
200                "for the active embedding model"
201            )
202        self._url = url
203        self._api_key = api_key
204        self._client = client
205        self._collection_ready = False

207    def _resolve_client(self) -> Any:
208        """Return the Qdrant client, creating it on first use.
209
210        Raises:
211            RuntimeError: If no client was injected and no URL
212                is available from ``url`` or ``settings``.
213        """
214        if self._client is None:
215            from qdrant_client import AsyncQdrantClient  # noqa: PLC0415
216
217            url = self._url
218            if url is None and self._settings is not None:
219                url = self._settings.database_url
220            if not url:
221                raise RuntimeError(
222                    "QdrantStore needs a url, a client, or settings with "
223                    "a database_url pointing at the Qdrant server"
224                )
225            self._client = AsyncQdrantClient(url=url, api_key=self._api_key)
226        return self._client

228    async def _ensure_collection(self) -> None:
229        """Create the collection on first use if it is missing."""
230        if self._collection_ready:
231            return
232        client = self._resolve_client()
233        if not await client.collection_exists(self.table_name):
234            from qdrant_client import Distance, VectorParams  # noqa: PLC0415
235
236            await client.create_collection(
237                collection_name=self.table_name,
238                vectors_config=VectorParams(size=self.embedding_dim, distance=Distance.EUCLID),
239            )
240        self._collection_ready = True

242    async def _scroll_all_points(self, client: Any, query_filter: Any) -> list[Any]:
243        all_points: list[Any] = []
244        offset: Any = None
245        while True:
246            points, offset = await client.scroll(
247                collection_name=self.table_name,
248                scroll_filter=query_filter,
249                limit=_SCROLL_BATCH,
250                offset=offset,
251                with_payload=True,
252            )
253            all_points.extend(points)
254            if offset is None:
255                break
256        return all_points

258    async def _keyword_stage(
259        self,
260        client: Any,
261        query: str,
262        text_field: str,
263        query_filter: Any,
264    ) -> tuple[dict[str, float], dict[str, dict[str, Any]]]:
265        """Score token overlap over every document in the collection.
266
267        Qdrant has no native full-text rank, so the keyword
268        stage paginates through the collection's payloads and
269        scores token overlap in process. Returns the keyword
270        scores and the payloads of every scanned document.
271        """
272        keyword_scores: dict[str, float] = {}
273        payloads: dict[str, dict[str, Any]] = {}
274        query_tokens = set(tokenize(query))
275        points = await self._scroll_all_points(client, query_filter)
276        for point in points:
277            payload = dict(point.payload or {})
278            doc_id = str(payload.get(self.id_field, point.id))
279            payloads.setdefault(doc_id, payload)
280            score = token_overlap(query_tokens, tokenize(str(payload.get(text_field, ""))))
281            if score > 0.0:
282                keyword_scores[doc_id] = score
283        return keyword_scores, payloads

285    async def vector_search(  # noqa: PLR0913, PLR0917
286        self,
287        embedding: list[float],
288        n_results: int,
289        threshold: float,
290        text_field: str = "text",
291        metadata_fields: list[str] | None = None,
292        session: Any = None,
293        **filters: Any,
294    ) -> list[dict[str, Any]]:
295        """Native nearest-neighbor search over the collection.
296
297        The ``n_results`` nearest points are fetched with their
298        vectors so the L2 distance threshold is applied in
299        process, matching the pgvector backend's semantics.
300
301        Args:
302            embedding: Query embedding vector (required; the
303                method does not auto-embed).
304            n_results: Maximum number of results to return.
305            threshold: Maximum L2 distance.
306            text_field: Payload field containing the document text.
307            metadata_fields: Payload fields to include in the
308                result metadata dict.
309            session: Accepted for interface compatibility; ignored
310                (Qdrant has no SQL session concept).
311            **filters: Equality filters applied to the search
312                (e.g., ``tenant_id="..."``).
313
314        Returns:
315            list[dict]: Result dicts ordered by distance ascending,
316            each carrying ``id``, ``document``, ``metadata``,
317            ``distance``, and ``keyword_score`` (``None``).
318
319        Raises:
320            ValueError: If identifiers are unsafe or the embedding
321                has the wrong dimension.
322            Exception: Re-raises Qdrant errors.
323        """
324        validate_identifier(text_field, "text_field")
325        for key in filters:
326            validate_identifier(key, "filter key")
327        metadata_fields = metadata_fields or []
328        for field in metadata_fields:
329            validate_identifier(field, "metadata_field")
330        validate_embedding(embedding, self.embedding_dim, label="query embedding")
331
332        await self._ensure_collection()
333        client = self._resolve_client()
334        query_filter = _equality_filter(filters)
335
336        response = await client.query_points(
337            collection_name=self.table_name,
338            query=embedding,
339            limit=n_results,
340            with_payload=True,
341            with_vectors=True,
342            query_filter=query_filter,
343        )
344        vector_ranked: list[str] = []
345        distances: dict[str, float] = {}
346        payloads: dict[str, dict[str, Any]] = {}
347        for point in response.points:
348            point_vector = list(point.vector or [])
349            distance = l2_distance(embedding, point_vector)
350            if distance >= threshold:
351                continue
352            doc_id = str(point.payload.get(self.id_field, point.id))
353            vector_ranked.append(doc_id)
354            distances[doc_id] = distance
355            payloads[doc_id] = dict(point.payload or {})
356
357        results = []
358        for doc_id in vector_ranked:
359            payload = payloads[doc_id]
360            results.append(
361                format_search_hit(
362                    doc_id=doc_id,
363                    document=payload.get(text_field),
364                    metadata=project_metadata(payload, metadata_fields),
365                    distance=distances[doc_id],
366                )
367            )
368        return results

370    async def keyword_search(  # noqa: PLR0913, PLR0917
371        self,
372        query: str,
373        n_results: int,
374        text_field: str = "text",
375        metadata_fields: list[str] | None = None,
376        session: Any = None,
377        **filters: Any,
378    ) -> list[dict[str, Any]]:
379        """Token-overlap keyword search over the collection's payloads.
380
381        Qdrant has no native full-text rank, so the keyword
382        stage scrolls every document in the collection and
383        scores token overlap in process -- an in-process
384        fallback intended for local development, small
385        corpora, and testing, not production-scale keyword
386        retrieval.
387
388        Args:
389            query: Natural language search query.
390            n_results: Maximum number of results to return.
391            text_field: Payload field containing the document text.
392            metadata_fields: Payload fields to include in the
393                result metadata dict.
394            session: Accepted for interface compatibility; ignored
395                (Qdrant has no SQL session concept).
396            **filters: Equality filters applied to the search
397                (e.g., ``tenant_id="..."``).
398
399        Returns:
400            list[dict]: Result dicts ordered by keyword score
401            descending (tie-broken by id), each carrying ``id``,
402            ``document``, ``metadata``, ``distance`` (``None``),
403            and ``keyword_score``.
404
405        Raises:
406            ValueError: If identifiers are unsafe.
407            Exception: Re-raises Qdrant errors.
408        """
409        validate_identifier(text_field, "text_field")
410        for key in filters:
411            validate_identifier(key, "filter key")
412        metadata_fields = metadata_fields or []
413        for field in metadata_fields:
414            validate_identifier(field, "metadata_field")
415
416        await self._ensure_collection()
417        client = self._resolve_client()
418        query_filter = _equality_filter(filters)
419
420        keyword_scores, payloads = await self._keyword_stage(
421            client, query, text_field, query_filter
422        )
423        keyword_ranked = sorted(
424            keyword_scores, key=lambda doc_id: (-keyword_scores[doc_id], doc_id)
425        )[:n_results]
426
427        results = []
428        for doc_id in keyword_ranked:
429            payload = payloads.get(doc_id, {})
430            results.append(
431                format_search_hit(
432                    doc_id=doc_id,
433                    document=payload.get(text_field),
434                    metadata=project_metadata(payload, metadata_fields),
435                    keyword_score=keyword_scores[doc_id],
436                )
437            )
438        return results

444        **filters: Any,

448        Args:

491    async def delete(
492        self,
493        ids: list[Any] | None = None,
494        filters: dict[str, Any] | None = None,
495        session: Any = None,
496    ) -> int:
497        """Delete points selected by ``ids`` and/or ``filters``.
498
499        At least one selector must be supplied; an empty ``ids``
500        list is rejected. When both are supplied, both
501        restrictions apply (conjunction). Qdrant's ``delete``
502        does not report how many points it removed, so the
503        deletion count is the collection count before the
504        delete minus the count after it.
505
506        Args:
507            ids: Document ids to delete (mapped to Qdrant point
508                ids with the store's ``_point_id`` mapping).
509            filters: Equality filters (e.g., ``tenant_id="..."``).
510            session: Accepted for interface compatibility; ignored
511                (Qdrant manages its own connections).
512
513        Returns:
514            int: Number of deleted documents.
515
516        Raises:
517            ValueError: If no selector is supplied, ``ids`` is
518                empty, or a filter key is unsafe.
519            Exception: Re-raises Qdrant errors.
520        """
521        validate_delete_selectors(ids, filters)
522
523        await self._ensure_collection()
524        client = self._resolve_client()
525
526        if ids is not None and not filters:
527            # A point-ids selector carries the id restriction alone, so
528            # the deletion count is the whole-collection count delta.
529            before_response = await client.count(collection_name=self.table_name)
530            before = int(before_response.count)
531            from qdrant_client import PointIdsList  # noqa: PLC0415
532
533            selector: Any = PointIdsList(points=[_point_id(doc_id) for doc_id in ids])
534            await client.delete(collection_name=self.table_name, points_selector=selector)
535            after_response = await client.count(collection_name=self.table_name)
536            after = int(after_response.count)
537        else:
538            # A filter selector carries the id restriction (a MatchAny on
539            # the payload's id_field) and/or the equality filters, so both
540            # restrictions apply when both are supplied.
541            query_filter = _selector_filter(ids, filters, self.id_field)
542            before_response = await client.count(
543                collection_name=self.table_name, count_filter=query_filter
544            )
545            before = int(before_response.count)
546            from qdrant_client import FilterSelector  # noqa: PLC0415
547
548            selector = FilterSelector(filter=query_filter)
549            await client.delete(collection_name=self.table_name, points_selector=selector)
550            after_response = await client.count(
551                collection_name=self.table_name, count_filter=query_filter
552            )
553            after = int(after_response.count)
554        return before - after

556    async def get(  # noqa: PLR0913, PLR0917
557        self,
558        ids: list[Any] | None = None,
559        filters: dict[str, Any] | None = None,
560        session: Any = None,
561        text_field: str = "text",
562        metadata_fields: list[str] | None = None,
563    ) -> list[dict[str, Any]]:
564        """Fetch documents selected by ``ids`` and/or ``filters``.
565
566        The selection scrolls the collection with a ``MatchAny``
567        filter on the payload's ``id_field`` (for ``ids``) and/or
568        equality filters. With no selector, every document is
569        returned.
570
571        Args:
572            ids: Document ids to fetch.
573            filters: Equality filters (e.g., ``tenant_id="..."``).
574            session: Accepted for interface compatibility; ignored
575                (Qdrant has no SQL session concept).
576            text_field: Payload field containing the document text.
577            metadata_fields: Payload fields to include in the
578                result metadata dict.
579
580        Returns:
581            list[dict]: ``id``/``document``/``metadata`` dicts
582            (score keys are meaningless for direct fetches).
583
584        Raises:
585            ValueError: If any identifier is unsafe.
586            Exception: Re-raises Qdrant errors.
587        """
588        validate_identifier(text_field, "text_field")
589        metadata_fields = metadata_fields or []
590        for field in metadata_fields:
591            validate_identifier(field, "metadata_field")
592        if filters is not None:
593            for key in filters:
594                validate_identifier(key, "filter key")
595
596        await self._ensure_collection()
597        client = self._resolve_client()
598        query_filter = _selector_filter(ids, filters, self.id_field)
599
600        points = await self._scroll_all_points(client, query_filter)
601        results: list[dict[str, Any]] = []
602        for point in points:
603            payload = dict(point.payload or {})
604            doc_id = str(payload.get(self.id_field, point.id))
605            results.append(
606                {
607                    "id": doc_id,
608                    "document": payload.get(text_field),
609                    "metadata": project_metadata(payload, metadata_fields),
610                }
611            )
612        return results

614    async def upsert(  # noqa: PLR0913, PLR0917
615        self,
616        documents: list[dict[str, Any]],
617        session: Any = None,
618        text_field: str = "text",
619        embedding_field: str = "embedding",
620        id_field: str | None = None,
621        extra_fields: dict[str, Any] | None = None,
622    ) -> None:
623        """Upsert documents into the collection.
624
625        Args:
626            documents: List of document dicts with keys: id, text,
627                metadata, embedding. Metadata keys are included as
628                additional payload fields.
629            session: Accepted for interface compatibility; ignored.
630            text_field: Payload field for the document text.
631                Defaults to "text".
632            embedding_field: Document dict key holding the
633                embedding vector. Defaults to "embedding".
634            id_field: Payload field for the document id. Defaults
635                to the store's configured ``id_field``.
636            extra_fields: Additional static fields to include in
637                every payload (e.g., ``{"tenant_id": "..."}``).
638
639        Raises:
640            ValueError: If any field name contains unsafe
641                characters.
642            Exception: Re-raises Qdrant errors after logging.
643        """
644        id_field = validate_upsert_documents(
645            documents,
646            text_field,
647            embedding_field,
648            id_field,
649            self.id_field,
650            extra_fields,
651            self.embedding_dim,
652        )
653        if not documents:
654            return
655
656        await self._ensure_collection()
657        client = self._resolve_client()
658
659        from qdrant_client import PointStruct  # noqa: PLC0415
660
661        points = []
662        for doc in documents:
663            payload: dict[str, Any] = {
664                id_field: doc.get("id"),
665                text_field: doc[text_field],
666            }
667            payload.update(doc.get("metadata", {}))
668            if extra_fields:
669                payload.update(extra_fields)
670            points.append(
671                PointStruct(
672                    id=_point_id(doc.get("id")),
673                    vector=list(doc[embedding_field]),
