23class RateLimiter:
24    """Sliding-window rate limiter with concurrency tracking.
25
26    Thread-safe and async-friendly in-memory limiter.
27    """
28
29    def __init__(
30        self,
31        requests_per_minute: int = 120,
32        max_concurrent: int = 20,
33    ) -> None:
34        self.requests_per_minute = requests_per_minute
35        self.max_concurrent = max_concurrent
36        self._history: dict[str, deque[float]] = defaultdict(deque)
37        self._active_counts: dict[str, int] = defaultdict(int)
38        self._lock = asyncio.Lock()
39
40    async def acquire(self, client_id: str = "global") -> None:
41        """Acquire a request slot or raise RateLimitExceededError."""
42        async with self._lock:
43            now = time.monotonic()
44            window_start = now - 60.0
45
46            # Prune timestamps older than 60s
47            queue = self._history[client_id]
48            while queue and queue[0] < window_start:
49                queue.popleft()
50
51            # Check rate limit
52            if len(queue) >= self.requests_per_minute:
53                raise RateLimitExceededError(
54                    f"Rate limit exceeded: {len(queue)} requests in last 60s "
55                    f"(maximum {self.requests_per_minute})"
56                )
57
58            # Check concurrency limit
59            active = self._active_counts[client_id]
60            if active >= self.max_concurrent:
61                raise RateLimitExceededError(
62                    f"Concurrency limit exceeded: {active} active requests "
63                    f"(maximum {self.max_concurrent})"
64                )
65
66            # Record request
67            queue.append(now)
68            self._active_counts[client_id] += 1
69
70    async def release(self, client_id: str = "global") -> None:
71        """Release an active concurrent request slot."""
72        async with self._lock:
73            if self._active_counts[client_id] > 0:
74                self._active_counts[client_id] -= 1

77class ScopeAuthorizer:
78    """Authorizes client requests against collections and isolation scopes."""
79
80    def __init__(
81        self,
82        scope_mapping: (
83            Mapping[str, Sequence[str] | set[str]]
84            | Callable[[str | None, str, str | None], bool]
85            | None
86        ) = None,
87        default_allow: bool = True,
88    ) -> None:
89        """Initialize ScopeAuthorizer.
90
91        Args:
92            scope_mapping: Mapping of client/token IDs to allowed scopes or a custom callback.
93            default_allow: Whether to allow access when no explicit scope map is configured.
94        """
95        self._mapping = scope_mapping
96        self._default_allow = default_allow
97
98    def is_authorized(
99        self,
100        client_id: str | None,
101        collection: str,
102        scope: str | None,
103    ) -> bool:
104        """Check whether client is authorized for the given collection and scope."""
105        if callable(self._mapping):
106            return self._mapping(client_id, collection, scope)
107
108        if self._mapping is None:
109            return self._default_allow
110
111        if client_id is None:
112            return False
113
114        allowed = self._mapping.get(client_id)
115        if allowed is None:
116            return False
117
118        if "*" in allowed:
119            return True
120
121        target_key = f"{collection}:{scope}" if scope else collection
122        return target_key in allowed or scope in allowed or collection in allowed
123
124    def enforce(self, client_id: str | None, collection: str, scope: str | None) -> None:
125        """Raise ScopeAuthorizationError if access is not permitted."""
126        if not self.is_authorized(client_id, collection, scope):
127            raise ScopeAuthorizationError(
128                f"Client {client_id!r} is not authorized for collection {collection!r} "
129                f"under scope {scope!r}"
130            )

133@dataclass
134class RemoteTransportConfig:
135    """Configuration for remote MCP transport deployment.
136
137    Attributes:
138        auth_token_validator: Callable validating host-provided auth token / API key.
139        rate_limiter: Configured request rate limiter.
140        scope_authorizer: Scope mapping and authorization engine.
141        host: Host IP or hostname to bind to.
142        port: Port number for HTTP/SSE/Streamable HTTP transport.
143        endpoint_path: Root endpoint path for the MCP server.
144        cors_allowed_origins: Permitted CORS origins for browser/client connections.
145        request_timeout_seconds: Bounded timeout per tool invocation.
146        metadata: Deployment environment metadata.
147    """
148
149    auth_token_validator: Callable[[str], bool] | None = None
150    rate_limiter: RateLimiter = field(default_factory=RateLimiter)
151    scope_authorizer: ScopeAuthorizer = field(default_factory=ScopeAuthorizer)
152    host: str = "127.0.0.1"
153    port: int = 8000
154    endpoint_path: str = "/mcp"
155    cors_allowed_origins: list[str] = field(default_factory=lambda: ["*"])
156    request_timeout_seconds: float = 30.0
157    metadata: dict[str, Any] = field(default_factory=dict)
158
159    def validate_token(self, token: str | None) -> bool:
160        """Validate auth token against configured validator."""
161        if self.auth_token_validator is None:
162            return True
163        if not token:
164            return False
165        return self.auth_token_validator(token)
