77# ── Shared session storage ────────────────────────────────────────────────────
78#
79# In-memory sessions live and die with the process, which is why they are not the
80# default for a deployment: a restart forgets the conversation in flight, and a
81# second replica cannot continue it. The database-backed service below keeps the
82# same ADK contract (``BaseSessionService``) so the runner, ``_ensure_session``
83# and the SSE stream are unchanged by the swap -- the choice is made when the
84# service is built, from ``SESSION_STORAGE``.
85
86_session_metadata = MetaData()

89# key because that is the tuple ADK addresses a session by, so a session id may
90# be reused across applications without colliding.

119class DatabaseSessionService(BaseSessionService):
120    """ADK session service that keeps sessions in the application's PostgreSQL.
121
122    ADK ships a database session service of its own, but it drives a
123    *synchronous* SQLAlchemy engine and this deployment's driver is async-only,
124    so it cannot be pointed at the application database. This implementation
125    keeps the same contract against the application's async engine and creates
126    its two tables, ``agent_sessions`` and ``agent_session_events``, on startup.
127    Both are owned here rather than by Alembic, exactly as ADK's own service
128    creates its tables itself, so the two can never be out of step.
129
130    Only the fixed contract ``Runner`` needs is implemented: create, get, list,
131    delete, and append. Two ADK behaviours are simplified on purpose:
132
133    * One state blob per session instead of separate app-, user- and
134      session-scoped rows: the assistant keeps no state that is meant to outlive
135      its session, so splitting it would only add write traffic. Keys with the
136      ``temp:`` prefix are still dropped, because ADK marks those as
137      invocation-scoped.
138    * An append whose in-memory session has fallen behind a concurrent replica
139      is logged rather than refused. ADK's database service raises in that
140      situation, which in a multi-replica deployment turns one interleaved turn
141      into a failed request for the user, while the append itself is still
142      correct.
143
144    Args:
145        engine: The application's async engine.
146
147    Returns:
148        None
149    """
150
151    # Only the tail of the history is replayed: a long conversation must not be
152    # pushed at the model whole, and a request never needs more than the recent
153    # events to answer it.
154    _MAX_REPLAYED_EVENTS = 200

167    async def create_session(
168        self,
169        *,
170        app_name: str,
171        user_id: str,
172        state: dict[str, Any] | None = None,
173        session_id: str | None = None,
174    ) -> Session:
175        """Create a session and return it.
176
177        A session id that already exists returns the stored session instead of
178        failing, so two replicas resolving a user's active session at the same
179        moment both converge on the same row.
180
181        Args:
182            app_name: Application the session belongs to.
183            user_id: Owning user.
184            state: Optional initial session state.
185            session_id: Optional caller-supplied id; generated when absent.
186
187        Returns:
188            Session: The created session.
189        """
190        session_id = session_id or f"session_{uuid.uuid4().hex[:12]}"
191        initial_state = {
192            key: value
193            for key, value in (state or {}).items()
194            if not key.startswith(State.TEMP_PREFIX)
195        }
196        try:
197            await self._insert_session(app_name, user_id, session_id, initial_state)
198        except IntegrityError:
199            # Somebody else created this row first; adopt theirs.
200            logger.debug("Session '%s' already exists; adopting the stored session.", session_id)
201            existing = await self.get_session(
202                app_name=app_name, user_id=user_id, session_id=session_id
203            )
204            if existing is not None:
205                return existing
206        return Session(
207            app_name=app_name,
208            user_id=user_id,
209            id=session_id,
210            state=initial_state,
211            last_update_time=time.time(),
212        )

227    async def get_session(
228        self,
229        *,
230        app_name: str,
231        user_id: str,
232        session_id: str,
233        config: Any = None,
234    ) -> Session | None:
235        """Load a session with its events, newest last.
236
237        The history is capped at ``_MAX_REPLAYED_EVENTS`` unless the caller
238        narrows it further: a turn needs the recent turns, not every event the
239        session ever recorded, and an unbounded read would push a whole
240        conversation -- tool payloads included -- at the model.
241
242        Args:
243            app_name: Application the session belongs to.
244            user_id: Owning user.
245            session_id: The session to load.
246            config: Optional ADK ``GetRequestConfig``; ``num_recent_events``
247                overrides how much history is replayed.
248
249        Returns:
250            Session | None: The session, or None when the id is unknown.
251        """
252        limit = getattr(config, "num_recent_events", None) or self._MAX_REPLAYED_EVENTS
253        async with self._engine.connect() as connection:
254            row = (
255                (
256                    await connection.execute(
257                        select(_agent_sessions).where(
258                            _agent_sessions.c.app_name == app_name,
259                            _agent_sessions.c.user_id == user_id,
260                            _agent_sessions.c.id == session_id,
261                        )
262                    )
263                )
264                .mappings()
265                .first()
266            )
267            if row is None:
268                return None
269            event_query = (
270                select(_agent_session_events.c.event)
271                .where(
272                    _agent_session_events.c.app_name == app_name,
273                    _agent_session_events.c.user_id == user_id,
274                    _agent_session_events.c.session_id == session_id,
275                )
276                .order_by(_agent_session_events.c.seq.desc())
277                .limit(limit)
278            )
279            rows = (await connection.execute(event_query)).scalars().all()
280            updated_at = row["updated_at"]
281
282        events = [Event.model_validate(payload) for payload in reversed(rows)]
283        return Session(
284            app_name=app_name,
285            user_id=user_id,
286            id=session_id,
287            state=dict(row["state"] or {}),
288            events=events,
289            last_update_time=_timestamp(updated_at),
290        )

360    async def append_event(self, session: Session, event: Event) -> Event:
361        """Persist one event and keep the in-memory session in step.
362
363        Args:
364            session: The session the event belongs to. Its ``last_update_time``
365                is updated to the value the database recorded.
366            event: The event to store. Partial (streaming) events are skipped,
367                matching ADK's own database service, because only completed
368                events belong in a history.
369
370        Returns:
371            Event: The event that was stored.
372        """
373        if event.partial:
374            return await super().append_event(session=session, event=event)
375
376        persisted_state = dict(session.state)
377        if event.actions and event.actions.state_delta:
378            # ``temp:`` keys are scoped to a single invocation and must not
379            # survive it.
380            persisted_state.update(
381                {
382                    key: value
383                    for key, value in event.actions.state_delta.items()
384                    if not key.startswith(State.TEMP_PREFIX)
385                }
386            )
387
388        payload = event.model_dump(mode="json", **_SESSION_MODEL_CONFIG)
389        async with self._engine.begin() as connection:
390            stored_at = await self._store_event(connection, session, event, payload)
391        if stored_at is not None:
392            session.last_update_time = stored_at
393        await super().append_event(session=session, event=event)
394        return event

396    async def _store_event(
397        self,
398        connection: Any,
399        session: Session,
400        event: Event,
401        payload: dict[str, Any],
402    ) -> float | None:
403        """Write the event row and refresh the session's update time.
404
405        Args:
406            connection: The open transaction to write in.
407            session: Owner of the event.
408            event: The event being stored.
409            payload: The event serialized as JSON.
410
411        Returns:
412            float | None: The timestamp the database recorded for the session,
413            or None if the session row vanished.
414        """
415        state_delta = bool(event.actions and event.actions.state_delta)
416        persisted_state = dict(session.state)
417        if state_delta:
418            persisted_state.update(
419                {
420                    key: value
421                    for key, value in event.actions.state_delta.items()
422                    if not key.startswith(State.TEMP_PREFIX)
423                }
424            )
425        await connection.execute(
426            insert(_agent_session_events).values(
427                app_name=session.app_name,
428                user_id=session.user_id,
429                session_id=session.id,
430                invocation_id=event.invocation_id,
431                author=event.author,
432                timestamp=event.timestamp,
433                event=payload,
434            )
435        )
436        session_filter = (
437            _agent_sessions.c.app_name == session.app_name,
438            _agent_sessions.c.user_id == session.user_id,
439            _agent_sessions.c.id == session.id,
440        )
441        updates: dict[str, Any] = {"updated_at": func.now()}
442        if state_delta:
443            updates["state"] = persisted_state
444        await connection.execute(update(_agent_sessions).where(*session_filter).values(**updates))
445        stored = (
446            await connection.execute(select(_agent_sessions.c.updated_at).where(*session_filter))
447        ).first()
448        if stored is None or stored[0] is None:  # pragma: no cover - defensive
449            logger.warning("Session '%s' vanished while an event was being stored.", session.id)
450            return None
451        return _timestamp(stored[0])

533def get_runner() -> Runner:
534    """Return the shared ADK Runner, building the agent on first call.
535
536    The session service is chosen from ``SESSION_STORAGE`` the first time the
537    runner (or ``_ensure_session``) needs one, and a shared store is the only
538    thing that lets a second replica continue a conversation the first one
539    started.
540
541    Returns:
542        Runner: The ADK runner bound to the current agent and session service.
543    """
544    global omnicare_agent, session_service, runner  # noqa: PLW0603
545    if session_service is None:
546        session_service = _build_session_service()
547    if runner is None:
548        omnicare_agent = create_agent(build_default_registry(), settings)
549        runner = Runner(
550            agent=omnicare_agent,
551            app_name=APP_NAME,
552            session_service=session_service,
553        )
554    return runner

579async def warm_session_service() -> None:
580    """Build the shared session service ahead of the first request.
581
582    Called from the application lifespan when ``SESSION_STORAGE=database``.
583    Creating the tables at startup makes the first request against an empty
584    database succeed, instead of the runner discovering a missing table at the
585    worst possible moment. If the database cannot be reached the process keeps
586    its in-memory service, because a deployment that cannot talk to its own
587    database has bigger problems than conversation continuity.
588    """
589    global session_service  # noqa: PLW0603
590    if settings.session_storage != "database":
591        return
592    try:
593        service = _build_session_service()
594        await service.create_tables()
595    except Exception as e:
596        logger.warning(
597            "Database session storage unavailable (%s); this process keeps in-memory "
598            "sessions, which are lost on restart.",
599            e,
600        )
601        return
602    session_service = service
603    logger.info("ADK session tables are ready.")

630    """Return the most recently updated stored session for a user, or None."""

644async def _ensure_session(user_id: str) -> str:
645    """Get or create the active session for a user.
646
647    With an in-process store the active session is the one this process created,
648    kept in ``_user_sessions`` and evicted FIFO once ``_MAX_SESSIONS`` users are
649    held. With a shared store the per-process mapping cannot be the truth: a
650    request may land on any replica, so the session is read from the database on
651    every call. That is what makes conversation continuity work across replicas,
652    and what makes a reset visible immediately everywhere.
653
654    Args:
655        user_id: The authenticated user identifier.
656
657    Returns:
658        str: The ADK session id to use for this request.
659    """
660    if _session_store_is_shared():
661        stored_id = await _latest_shared_session(user_id)
662        if stored_id is not None:
663            _user_sessions[user_id] = stored_id
664            return stored_id
665
666    if user_id not in _user_sessions:
667        # Evict oldest entry when at capacity (FIFO -- OrderedDict preserves
668        # insertion order, so the first key is always the oldest entry).
669        if len(_user_sessions) >= _MAX_SESSIONS:
670            oldest_user, evicted_session_id = _user_sessions.popitem(last=False)
671            logger.info(
672                "Session capacity reached (%d). Evicted session '%s' for user '%s'.",
673                _MAX_SESSIONS,
674                evicted_session_id,
675                oldest_user,
676            )
677            # Clean up the session object from the session service to prevent
678            # unbounded memory growth over long-running server uptime. The
679            # in-memory service holds sessions for the whole process lifetime,
680            # so deleting is what actually frees them. A shared store keeps its
681            # own retention instead: dropping the row would throw away a
682            # conversation another replica may still be reading.
683            if not _session_store_is_shared():
684                try:
685                    await _session_service().delete_session(
686                        app_name=APP_NAME,
687                        user_id=oldest_user,
688                        session_id=evicted_session_id,
689                    )
690                    logger.debug(
691                        "Cleaned up evicted session '%s' for user '%s'.",
692                        evicted_session_id,
693                        oldest_user,
694                    )
695                except Exception as e:
696                    logger.warning(
697                        "Failed to clean up evicted session '%s' for user '%s': %s",
698                        evicted_session_id,
699                        oldest_user,
700                        e,
701                    )
702
703        # Create a new session for this user.
704        try:
705            session = await _session_service().create_session(
706                app_name=APP_NAME,
707                user_id=user_id,
708            )
709            _user_sessions[user_id] = session.id
710            logger.debug("Created new session '%s' for user '%s'.", session.id, user_id)
711        except Exception as e:
712            logger.error(
713                "Failed to create session for user '%s': %s. Reusing existing session ID.",
714                user_id,
715                e,
716            )
717            _user_sessions[user_id] = f"session_{uuid.uuid4().hex[:12]}"
718
719    return _user_sessions[user_id]

787def _format_sources(sources: list[str | dict[str, Any]]) -> list[str]:
788    """Normalize tool sources into display-ready strings.
789
790    Policy tool returns dicts with 'source'/'section'; claim tools may return
791    plain strings for multi-source claims. Handles both.
792
793    Args:
794        sources: Raw source entries from a tool response.
795
796    Returns:
797        list[str]: Formatted source strings.
798    """
799    formatted = []
800    for item in sources:
801        if isinstance(item, dict):
802            source = item.get("source") or item.get("document", "")
803            section = item.get("section", "")
804            formatted.append(f"{section} ({source})" if section else source)
805        else:
806            formatted.append(item)  # already a string
807    return formatted

828def _bind_user_id(user_id: str) -> None:
829    """Bind the user id to the current async context for the tools to read.
830
831    The database layer scopes every tool query through ``current_user_id``, so
832    the agent -- which is also driven outside a request, by the evaluation
833    harness -- sets it from the identity it was given rather than relying on
834    the request having done it. An id that is not a UUID is left unbound and
835    reported, because failing loudly would cost the caller their turn while an
836    unbound context fails at the query, where it is actionable.
837
838    Args:
839        user_id: The user identifier driving this run.
840
841    Returns:
842        None.
843    """
844    try:
845        current_user_id.set(uuid.UUID(user_id))
846    except ValueError:
847        logger.warning("User id '%s' is not a UUID; tool queries will not be scoped.", user_id)
848
849
850# The span that represents one turn, whatever service ends up recording it.
851# Tracing is opt-in in the observability layer; with it off the decorator is
852# the function itself, so the agent is unchanged by observability that is not
853# deployed.
854@trace_turn("omnicare_agent_turn")
855async def run_agent(user_id: str, message: str) -> dict[str, Any]:
856    """Execute the ADK agent for a single user turn.
857
858    Args:
859        user_id: The authenticated user identifier (ADK user_id).
860        message: The user's natural-language message.
861
862    Returns:
863        dict: {
864            "response": str,      # synthesized agent text
865            "sources": list[str], # document citations
866            "tool_calls": list,   # tool invocation traces
867            "session_id": str     # ADK session id used
868        }
869    """
870    runner = get_runner()
871
872    # Ensure session exists (this is the idempotency/reuse layer).
873    session_id = await _ensure_session(user_id)
874
875    # Tools read the authenticated user through the context variable, which the
876    # auth dependency sets for HTTP requests; a direct call has not come through
877    # it, so it is bound here too.
878    _bind_user_id(user_id)
879
880    logger.debug("Running agent for user '%s' (session '%s').", user_id, session_id)
881
882    # Execute the agent via ADK Runner
883    final_text = ""
884    sources: list[str] = []
885    tool_calls: list[dict[str, Any]] = []
886
887    async for event in runner.run_async(
888        user_id=user_id,
889        session_id=session_id,
890        new_message=_new_user_content(message),
891    ):
892        # Collect tool call information for telemetry/debugging
893        if event.get_function_calls():
894            for fc in event.get_function_calls():
895                tool_calls.append({"name": fc.name, "arguments": fc.args})
896
897        # Collect function results (sources and citations)
898        if event.get_function_responses():
899            for fr in event.get_function_responses():
900                if not isinstance(fr.response, dict):
901                    continue
902                sources_raw = fr.response.get("sources", [])
903                if sources_raw:
904                    sources.extend(_format_sources(sources_raw))
905                # Claims tools may return 'citations' — accept both
906                citation = fr.response.get("citation")
907                if citation:
908                    sources.append(citation)
909
910                # Attach the result to the tool call it answers, newest first:
911                # the last unmatched call is the one that produced it. Without
912                # it the trace says a tool ran but not what came back.
913                tool_name = fr.name if hasattr(fr, "name") else ""
914                for tc in reversed(tool_calls):
915                    if tc.get("name") == tool_name:
916                        tc["result"] = fr.response
917                        break
918
919        # Final response
920        if event.is_final_response():
921            final_text = _extract_text(event)
922
923    if not final_text:
924        logger.error(
925            "No final response extracted for user '%s' (session '%s').", user_id, session_id
926        )
927
928    return {
929        "response": final_text,
930        "sources": sources,
931        "tool_calls": tool_calls,
932        "session_id": session_id,
933    }

1026@trace_turn("omnicare_agent_turn_stream")
1027async def run_agent_stream(
1028    user_id: str,
1029    message: str,
1030    result: _StreamResult | None = None,
1031) -> AsyncGenerator[str, None]:
1032    """Run the OmniCare agent and stream results as Server-Sent Events (SSE).
1033
1034    This is the streaming entry point used by the SSE chat endpoint. It runs
1035    the same agent and the same session as :func:`run_agent`; the difference is
1036    what comes back. Tool activity is emitted as typed events the client can
1037    render directly, and the model's own events are forwarded verbatim so the
1038    route can map them into the streaming contract.
1039
1040    Args:
1041        user_id: The authenticated user identifier (ADK user_id).
1042        message: The user's natural-language message.
1043        result: Optional container the caller supplies, populated with the
1044            session id, the answer fragments, the citations and the tool trace
1045            so the route can persist the turn once the stream closes.
1046
1047    Yields:
1048        str: Formatted SSE ``data:`` frames.
1049
1050    Returns:
1051        None. The final ``response_complete`` event is yielded by the route.
1052    """
1053    session_id = await _ensure_session(user_id)
1054
1055    # The tools scope their database reads through the context variable, which
1056    # the auth dependency sets for HTTP requests. The agent is also driven
1057    # directly -- by the evaluation harness and by these tests -- so the value
1058    # is set here as well rather than assumed to have come from the request.
1059    _bind_user_id(user_id)
1060
1061    user_content = types.Content(
1062        role="user",
1063        parts=[types.Part(text=message)],
1064    )
1065    final_text_parts: list[str] = []
1066    sources: list[str] = []
1067    tool_calls: list[dict[str, Any]] = []
1068
1069    runner = get_runner()
1070
1071    async for event in runner.run_async(
1072        user_id=user_id,
1073        session_id=session_id,
1074        new_message=user_content,
1075        run_config=RunConfig(streaming_mode=StreamingMode.SSE),
1076    ):
1077        # The final response carries the answer text; fragmentation is kept so
1078        # the caller can reconstruct the turn exactly as it was streamed.
1079        if event.is_final_response() and event.content and event.content.parts:
1080            for part in event.content.parts:
1081                if hasattr(part, "text") and part.text:
1082                    final_text_parts.append(part.text)
1083
1084        function_calls = event.get_function_calls()
1085        if function_calls:
1086            for fc in function_calls:
1087                tool_name = fc.name if hasattr(fc, "name") else str(fc)
1088                tool_arguments = fc.args if hasattr(fc, "args") else {}
1089                tool_calls.append({"name": tool_name, "arguments": tool_arguments})
1090                yield sse_frame(
1091                    StreamToolCall(tool=tool_name, arguments=_as_object(tool_arguments))
1092                )
1093
1094        function_responses = event.get_function_responses()
1095        if function_responses:
1096            for fr in function_responses:
1097                result_data = fr.response if hasattr(fr, "response") else {}
1098                tool_name = fr.name if hasattr(fr, "name") else ""
1099
1100                if isinstance(result_data, dict):
1101                    for src in result_data.get("sources", []):
1102                        if isinstance(src, dict):
1103                            section = src.get("section", "")
1104                            source_file = src.get("source", "")
1105                            sources.append(f"{section} ({source_file})")
1106                        elif isinstance(src, str):
1107                            sources.append(src)
1108
1109                    # Claim tools return a single citation alongside their
1110                    # result, which belongs in the same list.
1111                    if "citation" in result_data:
1112                        sources.append(result_data["citation"])
1113
1114                # Attach the result to the tool call it answers, newest first:
1115                # the last unmatched call is the one that produced it.
1116                for tc in reversed(tool_calls):
1117                    if tc.get("name") == tool_name:
1118                        tc["result"] = result_data
1119                        break
1120
1121                tool_error = _tool_failure_message(result_data)
1122                if tool_error is not None:
1123                    yield sse_frame(StreamToolError(tool=tool_name, error=tool_error))
1124                else:
1125                    yield sse_frame(
1126                        StreamToolResult(
1127                            tool=tool_name,
1128                            result=_as_object(result_data),
1129                        )
1130                    )
1131
1132        sse_event = event.model_dump_json(exclude_none=True, by_alias=True)
1133        yield f"data: {sse_event}\n\n"
1134
1135    if result is not None:
1136        result.session_id = session_id
1137        result.final_text_parts = final_text_parts
1138        result.sources = sources
1139        result.tool_calls = tool_calls

1142# [SUB-TASK] Conversation history rehydration.
1143#
1144# This section is independent of the shared session store and can be deleted in
1145# full -- along with the ``before_model_callback=`` argument in ``create_agent``
1146# -- without touching anything else in this module. It exists because
1147# ``conversation_messages`` was written on every turn but never read: a fresh
1148# ADK session (a restart, a new replica, or the first turn after switching the
1149# storage) started from nothing and the user had to re-explain the situation.
1150# This callback replays the recent turns into the model's history and folds the
1151# older ones into a summary, so the persisted store is read as well as written.
1152
1153
1154async def rehydrate_conversation_history(
1155    callback_context: CallbackContext,
1156    llm_request: LlmRequest,
1157) -> None:
1158    """Replay persisted turns into a session that has no history of its own.
1159
1160    Returning None hands the turn to the model unchanged; returning an
1161    ``LlmResponse`` would skip the model entirely, which this callback never
1162    does -- it only widens what the model reads.
1163
1164    Args:
1165        callback_context: ADK callback context, used to reach the user.
1166        llm_request: The request about to be sent, whose ``contents`` are
1167            extended in place when there is history worth replaying.
1168
1169    Returns:
1170        None. The model always runs; only its inputs change.
1171    """
1172    try:
1173        # ADK builds the request from the session's events plus the new user
1174        # message, so a single content means the session is empty and the user
1175        # is relying on the database to remember the conversation.
1176        if len(llm_request.contents) > 1:
1177            return
1178        user_id = _callback_user_id(callback_context)
1179        if not user_id:
1180            return
1181        history = await _load_recent_turns(user_id)
1182        if not history:
1183            return
1184        llm_request.contents = _merge_same_role_contents(history + list(llm_request.contents))
1185    except Exception:
1186        # A history problem must never cost the user their message.
1187        logger.warning("Could not rehydrate conversation history.", exc_info=True)

1196async def _load_recent_turns(user_id: str) -> list[types.Content]:
1197    """Load the recent turns for a user, oldest first, with a compaction summary.
1198
1199    Only the last ``session_history_turns`` turns are replayed in full. The
1200    turns older than that are represented by one summary content, because a
1201    long conversation has to fit the model's context window but the fact that
1202    it happened must not vanish from the history.
1203
1204    Args:
1205        user_id: The user whose turns to read.
1206
1207    Returns:
1208        list[types.Content]: Contents to prepend, oldest first. Empty when the
1209        user has no persisted turns.
1210    """
1211    from app.core.database import async_session_factory  # noqa: PLC0415
1212    from app.models.conversations.conversation import Conversation  # noqa: PLC0415
1213    from app.models.conversations.conversation_message import ConversationMessage  # noqa: PLC0415
1214
1215    try:
1216        user_uuid = uuid.UUID(str(user_id))
1217    except ValueError:
1218        # Session user ids that are not UUIDs cannot be matched to a user.
1219        return []
1220
1221    async with async_session_factory() as session:
1222        result = await session.execute(
1223            select(
1224                ConversationMessage.role, ConversationMessage.content, ConversationMessage.timestamp
1225            )
1226            .join(Conversation, ConversationMessage.conversation_id == Conversation.id)
1227            .where(Conversation.user_id == user_uuid)
1228            .order_by(ConversationMessage.timestamp.desc())
1229            .limit(settings.session_history_compaction_turns)
1230        )
1231        rows = list(result.all())
1232
1233    if not rows:
1234        return []
1235
1236    recent = list(reversed(rows[: settings.session_history_turns]))
1237    compacted = list(reversed(rows[settings.session_history_turns :]))
1238    contents: list[types.Content] = []
1239    if compacted:
1240        contents.append(_compaction_summary_content(compacted))
1241    contents.extend(_turn_contents(recent))
1242    return contents

1245def _turn_contents(rows: list[Any]) -> list[types.Content]:
1246    """Convert persisted turns into model/user contents.
1247
1248    Args:
1249        rows: ``(role, content, timestamp)`` tuples, oldest first.
1250
1251    Returns:
1252        list[types.Content]: One content per non-empty turn.
1253    """

1255    for role, content, _timestamp in rows:

1257        if not text:
1258            continue
1259        is_user = str(role).lower() == "user"
1260        contents.append(
1261            types.Content(
1262                role="user" if is_user else "model",
1263                parts=[types.Part.from_text(text=text)],
1264            )
1265        )

1269def _compaction_summary_content(rows: list[Any]) -> types.Content:
1270    """Summarize the turns that are too old to replay.
1271
1272    The summary is deliberately deterministic: it reports how much was folded
1273    away and quotes the user's earliest questions, which is the part of an old
1274    turn that still shapes what a follow-up question means.
1275
1276    Args:
1277        rows: ``(role, content, timestamp)`` tuples, oldest first.
1278
1279    Returns:
1280        types.Content: A user-role content holding the summary.
1281    """
1282    user_texts = [str(content).strip() for role, content, _ in rows if str(role).lower() == "user"]
1283    excerpts = [f"'{text[:80]}'" for text in user_texts[:3]]
1284    spans = f"{rows[0][2]:%Y-%m-%d %H:%M} to {rows[-1][2]:%Y-%m-%d %H:%M}"
1285    summary = (
1286        f"Summary of earlier conversation: {len(rows)} older turns from {spans} were "
1287        "compacted into this summary to preserve context. The user's earliest questions"
1288        f" were {', '.join(excerpts) if excerpts else 'not recorded'}."
1289    )
1290    return types.Content(role="user", parts=[types.Part.from_text(text=summary)])

1293def _merge_same_role_contents(contents: list[types.Content]) -> list[types.Content]:
1294    """Merge runs of contents that share a role into single contents.
1295
1296    The model API rejects a conversation where one side speaks twice in a row,
1297    and a compaction summary directly followed by a user turn would do exactly
1298    that.
1299
1300    Args:
1301        contents: Conversation contents in send order.
1302
1303    Returns:
1304        list[types.Content]: The same contents with neighbouring same-role
1305        contents combined.
1306    """
1307    merged: list[types.Content] = []
1308    for content in contents:
1309        if merged and merged[-1].role == content.role:
1310            merged[-1].parts.extend(content.parts or [])
1311            continue
