65def _normalize_text(text: str) -> str:
66    """Lowercase, strip punctuation, collapse whitespace."""
67    text = text.lower()
68    text = re.sub(r"[^\w\s]", " ", text)
69    return re.sub(r"\s+", " ", text).strip()
70

72def _token_overlap(text_a: str, text_b: str) -> float:
73    """Compute Jaccard-like token overlap between two texts."""
74    tokens_a = set(_normalize_text(text_a).split())
75    tokens_b = set(_normalize_text(text_b).split())
76    if not tokens_a or not tokens_b:
77        return 0.0
78    intersection = tokens_a & tokens_b
79    union = tokens_a | tokens_b
80    return len(intersection) / len(union)

83def _key_fact_recall(
84    answer: str,
85    gold_answer: str,
86    domain_terms: set[str] | None = None,
87) -> float:
88    """Check what fraction of gold answer's key facts appear in the answer.
89
90    Key facts are the numbers in the gold answer plus any
91    host-supplied domain vocabulary (``domain_terms``) that
92    appears in it.
93    """
94    # Extract key facts: numbers, and optional domain terms
95    number_pattern = re.compile(r"\b\d[\d,]*\b")
96
97    key_facts = set(number_pattern.findall(gold_answer))
98
99    if domain_terms:
100        gold_lower = gold_answer.lower()
101        key_facts.update(term for term in domain_terms if term in gold_lower)
102
103    if not key_facts:
104        return _token_overlap(answer, gold_answer)
105
106    answer_lower = answer.lower()
107    hits = sum(1 for fact in key_facts if fact.lower() in answer_lower or fact in answer)
108    return hits / len(key_facts)

111def evaluate_answer_heuristic(
112    query: str,
113    answer: str,
114    gold_answer: str,
115    retrieved_context: str = "",
116    domain_terms: set[str] | None = None,
117) -> AnswerScore:
118    """Score an answer using deterministic heuristics (no LLM needed).
119
120    Args:
121        query: The original user question.
122        answer: The generated answer to evaluate.
123        gold_answer: The reference/gold-standard answer.
124        retrieved_context: The retrieved context chunks (for faithfulness check).
125        domain_terms: Host-specific vocabulary whose presence in the
126            gold answer counts as a key fact. Defaults to None
127            (numbers only), so the judge is domain-neutral out of the box.
128
129    Returns:
130        AnswerScore with heuristic-based scores.
131    """
132    if not answer or not answer.strip():
133        return AnswerScore(reasoning="Empty answer")
134
135    # Faithfulness: how much of the answer is grounded in context
136    if retrieved_context:
137        faithfulness = _token_overlap(answer, retrieved_context)
138    else:
139        faithfulness = _token_overlap(answer, gold_answer)
140
141    # Relevance: overlap between answer and question terms
142    relevance = min(_token_overlap(answer, query) * 3.0, 1.0)
143
144    # Completeness: key fact recall from gold answer
145    completeness = _key_fact_recall(answer, gold_answer, domain_terms)
146
147    # Conciseness: penalize excessively long answers
148    answer_len = len(answer.split())
149    gold_len = max(len(gold_answer.split()), 1)
150    ratio = answer_len / gold_len
151    if ratio <= 2.0:
152        conciseness = 1.0
153    elif ratio <= 4.0:
154        conciseness = max(0.3, 1.0 - (ratio - 2.0) * 0.35)
155    else:
156        conciseness = 0.2
157
158    overall = 0.30 * faithfulness + 0.25 * relevance + 0.30 * completeness + 0.15 * conciseness
159
160    return AnswerScore(
161        faithfulness=round(faithfulness, 4),
162        relevance=round(relevance, 4),
163        completeness=round(completeness, 4),
164        conciseness=round(conciseness, 4),
165        overall=round(overall, 4),
166        reasoning="heuristic",
167    )

174_JUDGE_PROMPT = """\
175You are an expert evaluator for a Retrieval-Augmented Generation (RAG) system.
176Your task is to evaluate the quality of a generated answer.
177
178## Evaluation Criteria
179
180Score each dimension from 0.0 to 1.0:
181
1821. **Faithfulness** (0-1): Is the answer factually grounded in the retrieved context?
183   Does it avoid hallucinating information not present in the context?
184   - 1.0 = Fully faithful, all claims supported by context
185   - 0.0 = Completely hallucinated or contradicts context
186
1872. **Relevance** (0-1): Does the answer directly address the user's question?
188   - 1.0 = Directly and precisely answers the question
189   - 0.0 = Completely irrelevant
190
1913. **Completeness** (0-1): Does the answer cover all key points from the gold answer?
192   - 1.0 = Covers all key information
193   - 0.0 = Misses all key information
194
1954. **Conciseness** (0-1): Is the answer appropriately concise without unnecessary verbosity?
196   - 1.0 = Perfectly concise
197   - 0.0 = Extremely verbose or padded
198
199## Input
200
201**User Question:** {query}
202
203**Retrieved Context:**
204{context}
205
206**Generated Answer:** {answer}
207
208**Gold/Reference Answer:** {gold_answer}
209
210## Output Format
211
212Respond with ONLY a JSON object (no markdown, no extra text):
213{{
214  "faithfulness": <float>,
215  "relevance": <float>,
216  "completeness": <float>,
217  "conciseness": <float>,
218  "reasoning": "<brief explanation>"
219}}
220"""

223async def evaluate_answer_llm(
224    query: str,
225    answer: str,
226    gold_answer: str,
227    retrieved_context: str = "",
228    settings: RagSettings | None = None,
229) -> AnswerScore:
230    """Score an answer using the configured LLM as a judge.
231
232    Args:
233        query: The original user question.
234        answer: The generated answer to evaluate.
235        gold_answer: The reference/gold-standard answer.
236        retrieved_context: The retrieved context chunks.
237        settings: Host settings providing ``llm_model``; when the setting
238            is absent the judge cannot run and the heuristic fallback applies.
239
240    Returns:
241        AnswerScore with LLM-judged scores, or the heuristic
242        fallback when the judge fails.
243    """
244    import litellm  # noqa: PLC0415
245
246    prompt = _JUDGE_PROMPT.format(
247        query=query,
248        context=retrieved_context or "(no context provided)",
249        answer=answer,
250        gold_answer=gold_answer,
251    )
252
253    try:
254        if settings is None:
255            raise ValueError("LLM judge requires RagSettings with llm_model")
256
257        llm_model = getattr(settings, "llm_model", None)
258        if llm_model is None:
259            raise ValueError("LLM judge requires settings with llm_model")
260
261        response = await litellm.acompletion(
262            model=llm_model,
263            messages=[{"role": "user", "content": prompt}],
264            temperature=0.0,
265            max_tokens=500,
266        )
267
268        content = response.choices[0].message.content.strip()
269
270        # Try to extract JSON from the response
271        json_match = re.search(r"\{[^}]+\}", content, re.DOTALL)
272        scores = json.loads(json_match.group()) if json_match else json.loads(content)
273
274        faithfulness = float(scores.get("faithfulness", 0))
275        relevance = float(scores.get("relevance", 0))
276        completeness = float(scores.get("completeness", 0))
277        conciseness = float(scores.get("conciseness", 0))
278        reasoning = scores.get("reasoning", "")
279
280        overall = 0.30 * faithfulness + 0.25 * relevance + 0.30 * completeness + 0.15 * conciseness
281
282        return AnswerScore(
283            faithfulness=round(faithfulness, 4),
284            relevance=round(relevance, 4),
285            completeness=round(completeness, 4),
286            conciseness=round(conciseness, 4),
287            overall=round(overall, 4),
288            reasoning=reasoning,
289        )
290
291    except Exception as exc:
292        logger.error("LLM judge evaluation failed: %s", exc)
293        score = evaluate_answer_heuristic(
294            query=query,
295            answer=answer,
296            gold_answer=gold_answer,
297            retrieved_context=retrieved_context,
298        )
299        score.reasoning = f"llm_judge_unavailable ({type(exc).__name__}): {score.reasoning}"
300        return score
