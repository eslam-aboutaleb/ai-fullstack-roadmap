1"""Claims retrieval adapter over ragnor.
2
3The OmniCare claims binding — the ``claims`` table, its
4``description`` column, and the metadata columns returned
5in claim metadata — lives here as
6:data:`CLAIMS_RETRIEVER_SPEC`. The generic hybrid
7retrieval machinery is ragnor's
8:class:`ragnor.retrieval.HybridRetriever`.
9
10The horizontal privilege-escalation guard lives here too:
11:class:`ClaimsRetriever` always passes
12``owner_id=str(user_id)`` as an equality filter, so a
13caller can never see another user's claims. The filter is
14applied inside the adapter, not by callers.
15"""
16

45class ClaimsRetriever:
46    """Hybrid retrieval over the OmniCare claims table.
47
48    Wraps ragnor's :class:`ragnor.retrieval.HybridRetriever`
49    with the claims binding. Every retrieval is scoped to
50    the requesting user: ``owner_id`` is passed as an
51    equality filter on both search stages, so another
52    user's claims are never returned.
53    """
54
55    def __init__(self, settings: Any = None, session_factory: Any = None) -> None:
56        """Initialize the claims retriever.
57
58        Args:
59            settings: Application settings. Defaults to
60                ``get_settings()``.
61            session_factory: Session provider the store opens
62                its transactions with. Defaults to the
63                application's ``async_session_factory``.
64        """
65        self._settings = settings or get_settings()
66        self._session_factory = session_factory or async_session_factory
67        self._retriever = HybridRetriever(
68            spec=CLAIMS_RETRIEVER_SPEC,
69            settings=self._settings,
70            session_factory=self._session_factory,
71        )
72
73    async def retrieve(  # noqa: PLR0913
74        self,
75        query: str,
76        user_id: uuid.UUID,
77        n_results: int = 5,
78        distance_threshold: float | None = None,
79    ) -> list[dict[str, Any]]:
80        """Run hybrid search over the calling user's claims only.
81
82        Args:
83            query: Natural language query.
84            user_id: The authenticated user whose claims are
85                searched. The ``owner_id`` equality filter is
86                always applied — this is the horizontal
87                privilege-escalation guard.
88            n_results: Maximum number of results to return.
89            distance_threshold: Maximum L2 distance for the
90                vector search part. Defaults to
91                ``settings.rag_distance_threshold``.
92
93        Returns:
94            List of result dicts with ``document``,
95            ``metadata``, ``distance``, and ``_rrf_score`` keys.
96        """
97        if distance_threshold is None:
98            distance_threshold = self._settings.rag_distance_threshold
99        results = await self._retriever.retrieve(
100            query,
101            n_results=n_results,
102            distance_threshold=distance_threshold,
103            owner_id=str(user_id),
104        )
105        return as_dicts(results)

108async def retrieve_claims_hybrid(  # noqa: PLR0913
109    query: str,
110    user_id: uuid.UUID,
111    n_results: int = 5,
112    distance_threshold: float | None = None,
113) -> list[dict[str, Any]]:
114    """Run hybrid search over the calling user's claims only.
115
116    Module-level entry point. Any retrieval failure is logged and reported as an empty
117    result set rather than raised.
118
119    Args:
120        query: Natural language query.
121        user_id: The authenticated user whose claims are
122            searched. The ``owner_id`` equality filter is
123            always applied — this is the horizontal
124            privilege-escalation guard.
125        n_results: Maximum number of results to return.
126        distance_threshold: Maximum L2 distance for the
127            vector search part. Defaults to
128            ``settings.rag_distance_threshold``.
129
130    Returns:
131        List of result dicts, or an empty list on failure.
132    """
133    try:
134        return await ClaimsRetriever().retrieve(
135            query,
136            user_id=user_id,
137            n_results=n_results,
138            distance_threshold=distance_threshold,
139        )
140    except Exception as exc:
141        logger.error("Hybrid claims search failed: %s", exc)
142        return []
