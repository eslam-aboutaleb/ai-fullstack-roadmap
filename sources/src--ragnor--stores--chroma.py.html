1"""Chroma implementation of the VectorStore interface.
2
3Provides hybrid search (vector similarity + token-overlap
4keyword matching with Reciprocal Rank Fusion), document
5counting, lifecycle operations, and upsert operations against
6a Chroma collection.
7
8The store subclasses :class:`ragnor.stores.base.HybridVectorStore`:
9the vector stage lives in :meth:`vector_search` (Chroma's
10native nearest-neighbor search) and the keyword stage in
11:meth:`keyword_search` (an in-process token-overlap scan over
12the collection's documents, because Chroma has no built-in
13full-text rank). The inherited composite ``hybrid_search``
14runs both stages and fuses them with RRF, so the result shape
15matches the other backends.
16
17Keyword-search performance notice: the keyword stage is an
18in-process fallback that fetches every document in the
19collection and scores token overlap. It is intended for local
20development, small corpora, and testing -- not for
21production-scale keyword retrieval. Hosts at scale combine a
22pgvector :class:`VectorIndex` with an external lexical engine
23adapted to :class:`ragnor.stores.base.LexicalIndex`.
24
25The store talks to Chroma's synchronous client API and
26wraps every call in ``asyncio.to_thread`` so the event
27loop is never blocked; the sync client is Chroma's stable
28surface (``EphemeralClient``, ``PersistentClient``,
29``HttpClient``). A client can also be injected directly
30via ``client`` for hosts and tests that manage their own.
31
32Configuration (no host imports): the store receives
33everything through constructor arguments. The Chroma
34client is created lazily on first use -- an
35``EphemeralClient`` by default, or a
36``PersistentClient`` when ``path`` is given.
37
38Distance metric: the store accepts the uniform ``metric``
39constructor kwarg so the registry can build every backend
40the same way, but the cosine metric is NOT supported here
41-- only the ``l2`` default is. Constructing the store with
42the cosine metric raises ``ValueError``; cosine ranking is
43available from the pgvector, memory, and faiss backends.
44"""

72def _where_filter(filters: dict[str, Any]) -> dict[str, Any] | None:
73    """Build a Chroma ``where`` filter from equality filters."""
74    if not filters:
75        return None
76    return {key: {"$eq": value} for key, value in filters.items()}

88    def __init__(  # noqa: PLR0913, PLR0917
89        self,
90        table_name: str,
91        id_field: str = "id",
92        embedding_dim: int | None = None,
93        settings: RagSettings | None = None,
94        embedding_fn: EmbeddingFunction | None = None,
95        client: Any | None = None,
96        path: str | None = None,
97        session_factory: Any | None = None,
98        metric: str | None = None,
99    ):
100        """Initialize the Chroma store.
101
102        Args:
103            table_name: Name of the Chroma collection.
104            id_field: Accepted for constructor compatibility
105                with the other stores; Chroma stores the
106                document id as the record id.
107            embedding_dim: Dimensionality of the embedding
108                vectors. Required; pass the dimension
109                configured for the active embedding model.
110            settings: Host settings; accepted for
111                constructor compatibility with the other
112                stores, and read for the optional
113                ``rag_distance_metric`` field when ``metric``
114                is omitted.
115            embedding_fn: Async callable that embeds a list
116                of texts, used to auto-embed queries.
117            client: Pre-built Chroma client. When omitted,
118                one is created lazily -- an
119                ``EphemeralClient`` by default, or a
120                ``PersistentClient`` when ``path`` is given.
121            path: Directory for a persistent Chroma database.
122                When omitted, an in-memory ephemeral
123                database is used.
124            session_factory: Accepted for constructor
125                compatibility with the other stores; ignored
126                (Chroma manages its own connections).
127            metric: Distance metric for the vector stage. Only
128                ``l2`` (the default) is supported by this
129                backend.
130
131        Raises:
132            ValueError: If ``table_name`` contains unsafe
133                characters, if ``embedding_dim`` is omitted,
134                if ``metric`` is not a supported distance
135                metric, or if the cosine metric is requested
136                (unsupported by this backend).
137        """
138        super().__init__(
139            table_name=table_name,
140            id_field=id_field,
141            embedding_dim=embedding_dim,
142            settings=settings,
143            embedding_fn=embedding_fn,
144            session_factory=session_factory,
145            metric=metric,
146        )
147        if self.metric == "cosine":
148            raise ValueError(
149                "the 'cosine' distance metric is unsupported by the Chroma "
150                "backend; use l2 (the default), or switch to the pgvector, "
151                "memory, or faiss backend for cosine ranking"
152            )
153        if embedding_dim is None:
154            raise ValueError(
155                "embedding_dim is required; pass the dimension configured "
156                "for the active embedding model"
157            )
158        self._settings = settings
159        self._client = client
160        self._path = path
161        self._collection: Any | None = None

163    def _resolve_collection(self) -> Any:
164        """Return the Chroma collection, creating it on first use.
165
166        Raises:
167            RuntimeError: If no client was injected and
168                ``chromadb`` is not installed.
169        """
170        if self._collection is None:
171            if self._client is None:
172                import chromadb  # noqa: PLC0415
173
174                if self._path is not None:
175                    self._client = chromadb.PersistentClient(path=self._path)
176                else:
177                    self._client = chromadb.EphemeralClient()
178            self._collection = self._client.get_or_create_collection(name=self.table_name)
179        return self._collection

181    async def vector_search(  # noqa: PLR0913, PLR0917
182        self,
183        embedding: list[float],
184        n_results: int,
185        threshold: float,
186        text_field: str = "text",
187        metadata_fields: list[str] | None = None,
188        session: Any = None,
189        **filters: Any,
190    ) -> list[dict[str, Any]]:
191        """Native nearest-neighbor search over the collection.
192
193        Chroma returns L2 distances; the threshold is applied
194        in process, matching the other backends' semantics.
195
196        Args:
197            embedding: Query embedding vector (required; the
198                method does not auto-embed).
199            n_results: Maximum number of results to return.
200            threshold: Maximum L2 distance.
201            text_field: Accepted for interface compatibility;
202                Chroma stores document text as the record's
203                document.
204            metadata_fields: Metadata keys to include in the
205                result metadata dict.
206            session: Accepted for interface compatibility;
207                ignored (Chroma has no SQL session concept).
208            **filters: Equality filters applied to the search
209                (e.g., ``tenant_id="..."``).
210
211        Returns:
212            list[dict]: Result dicts ordered by distance ascending,
213            each carrying ``id``, ``document``, ``metadata``,
214            ``distance``, and ``keyword_score`` (``None``).
215
216        Raises:
217            ValueError: If identifiers are unsafe or the embedding
218                has the wrong dimension.
219            Exception: Re-raises Chroma errors.
220        """
221        for key in filters:
222            validate_identifier(key, "filter key")
223        metadata_fields = metadata_fields or []
224        for field in metadata_fields:
225            validate_identifier(field, "metadata_field")
226        validate_embedding(embedding, self.embedding_dim, label="query embedding")
227
228        collection = self._resolve_collection()
229        where = _where_filter(filters)
230
231        response = await asyncio.to_thread(
232            collection.query,
233            query_embeddings=[embedding],
234            n_results=n_results,
235            where=where,
236        )
237        results = []
238        for raw_id, document, distance, metadata in zip(
239            (response.get("ids") or [[]])[0],
240            (response.get("documents") or [[]])[0],
241            (response.get("distances") or [[]])[0],
242            (response.get("metadatas") or [[]])[0],
243            strict=False,
244        ):
245            if distance is not None and distance >= threshold:
246                continue
247            doc_id = str(raw_id)
248            results.append(
249                format_search_hit(
250                    doc_id=doc_id,
251                    document=document,
252                    metadata=project_metadata(dict(metadata or {}), metadata_fields),
253                    distance=float(distance),
254                )
255            )
256        return results

258    async def keyword_search(  # noqa: PLR0913, PLR0917
259        self,
260        query: str,
261        n_results: int,
262        text_field: str = "text",
263        metadata_fields: list[str] | None = None,
264        session: Any = None,
265        **filters: Any,
266    ) -> list[dict[str, Any]]:
267        """Token-overlap keyword search over the collection's documents.
268
269        Chroma has no native full-text rank, so the keyword
270        stage fetches every document in the collection and
271        scores token overlap in process -- an in-process
272        fallback intended for local development, small
273        corpora, and testing, not production-scale keyword
274        retrieval.
275
276        Args:
277            query: Natural language search query.
278            n_results: Maximum number of results to return.
279            text_field: Ignored for payload retrieval; Chroma stores
280                document text natively as the record's document.
281            metadata_fields: Keys projected into the returned result metadata.
282            session: Ignored (Chroma operations are non-transactional).
283            **filters: Metadata equality constraints passed to Chroma.
284
285        Returns:
286            list[dict]: Result dicts ordered by keyword score
287            descending (tie-broken by id), each carrying ``id``,
288            ``document``, ``metadata``, ``distance`` (``None``),
289            and ``keyword_score``.
290
291        Raises:
292            ValueError: If identifiers are unsafe.
293            Exception: Re-raises Chroma errors.
294        """
295        for key in filters:
296            validate_identifier(key, "filter key")
297        metadata_fields = metadata_fields or []
298        for field in metadata_fields:
299            validate_identifier(field, "metadata_field")
300
301        collection = self._resolve_collection()
302        where = _where_filter(filters)
303        query_tokens = set(tokenize(query))
304
305        all_records = await asyncio.to_thread(collection.get, where=where)
306        keyword_scores: dict[str, float] = {}
307        documents: dict[str, str] = {}
308        metadatas: dict[str, dict[str, Any]] = {}
309        for raw_id, document, metadata in zip(
310            all_records.get("ids") or [],
311            all_records.get("documents") or [],
312            all_records.get("metadatas") or [],
313            strict=False,
314        ):
315            doc_id = str(raw_id)
316            documents[doc_id] = document
317            metadatas[doc_id] = dict(metadata or {})
318            score = token_overlap(query_tokens, tokenize(str(document)))
319            if score > 0.0:
320                keyword_scores[doc_id] = score
321
322        keyword_ranked = sorted(
323            keyword_scores, key=lambda doc_id: (-keyword_scores[doc_id], doc_id)
324        )[:n_results]
325
326        results = []
327        for doc_id in keyword_ranked:
328            record_metadata = metadatas.get(doc_id, {})
329            results.append(
330                format_search_hit(
331                    doc_id=doc_id,
332                    document=documents.get(doc_id),
333                    metadata=project_metadata(record_metadata, metadata_fields),
334                    keyword_score=keyword_scores[doc_id],
335                )
336            )
337        return results

339    async def count(
340        self,
341        session: Any = None,
342        strict: bool = False,
343        **filters: Any,
344    ) -> int:
345        """Count documents in the collection.
346
347        Args:
348            session: Accepted for interface compatibility;
349                ignored.
350            strict: When ``True``, a failed count raises
351                :class:`ragnor.types.RetrievalError`.
352            **filters: Equality filters (e.g.,
353                ``tenant_id="..."``).
354
355        Returns:
356            int: Number of matching documents, or 0 on error.
357
358        Raises:
359            ValueError: If a filter key is unsafe.
360            ragnor.types.RetrievalError: If the count fails
361                and ``strict=True``.
362        """
363        for key in filters:
364            validate_identifier(key, "filter key")
365
366        try:
367            collection = self._resolve_collection()
368            if filters:
369                records = await asyncio.to_thread(collection.get, where=_where_filter(filters))
370                return len(records.get("ids") or [])
371            return int(await asyncio.to_thread(collection.count))
372        except Exception as exc:
373            return handle_count_error(self.table_name, exc, strict, logger)

375    async def get_embeddings(self, ids: list[str], session: Any = None) -> dict[str, list[float]]:
376        if not ids:
377            return {}
378        records = await asyncio.to_thread(
379            self._resolve_collection().get, ids=ids, include=["embeddings"]
380        )
381        vectors = records.get("embeddings")
382        if vectors is None:
383            return {}
384        return {
385            str(doc_id): list(vector)
386            for doc_id, vector in zip(records["ids"], vectors, strict=True)
387        }

389    async def delete(
390        self,
391        ids: list[Any] | None = None,
392        filters: dict[str, Any] | None = None,
393        session: Any = None,
394    ) -> int:
395        """Delete documents selected by ``ids`` and/or ``filters``.
396
397        At least one selector must be supplied; an empty ``ids``
398        list is rejected. When both are supplied, both
399        restrictions apply (conjunction): the matching records
400        are resolved first, then exactly those ids are deleted,
401        so the returned count is the number of documents that
402        existed and matched.
403
404        Args:
405            ids: Document ids to delete.
406            filters: Equality filters (e.g.,
407                ``tenant_id="..."``).
408            session: Accepted for interface compatibility;
409                ignored (Chroma manages its own connections).
410
411        Returns:
412            int: Number of deleted documents.
413
414        Raises:
415            ValueError: If no selector is supplied, ``ids`` is
416                empty, or a filter key is unsafe.
417            Exception: Re-raises Chroma errors.
418        """
419        validate_delete_selectors(ids, filters)
420
421        collection = self._resolve_collection()
422        where = _where_filter(filters) if filters else None
423        records = await asyncio.to_thread(
424            collection.get,
425            ids=[str(doc_id) for doc_id in ids] if ids is not None else None,
426            where=where,
427        )
428        matching_ids = [str(doc_id) for doc_id in (records.get("ids") or [])]
429        if matching_ids:
430            await asyncio.to_thread(collection.delete, ids=matching_ids)
431        return len(matching_ids)

433    async def get(  # noqa: PLR0913, PLR0917
434        self,
435        ids: list[Any] | None = None,
436        filters: dict[str, Any] | None = None,
437        session: Any = None,
438        text_field: str = "text",
439        metadata_fields: list[str] | None = None,
440    ) -> list[dict[str, Any]]:
441        """Fetch documents selected by ``ids`` and/or ``filters``.
442
443        The selection is resolved by Chroma's ``get`` (by record
444        ids and/or ``where`` equality filters). With no selector,
445        every document is returned.
446
447        Args:
448            ids: Document ids to fetch.
449            filters: Equality filters (e.g.,
450                ``tenant_id="..."``).
451            session: Accepted for interface compatibility;
452                ignored (Chroma has no SQL session concept).
453            text_field: Accepted for interface compatibility;
454                Chroma stores document text as the record's
455                document.
456            metadata_fields: Metadata keys to include in the
457                result metadata dict.
458
459        Returns:
460            list[dict]: ``id``/``document``/``metadata`` dicts
461            (score keys are meaningless for direct fetches).
462
463        Raises:
464            ValueError: If any identifier is unsafe.
465            Exception: Re-raises Chroma errors.
466        """
467        for key in filters or {}:
468            validate_identifier(key, "filter key")
469        metadata_fields = metadata_fields or []
470        for field in metadata_fields:
471            validate_identifier(field, "metadata_field")
472
473        collection = self._resolve_collection()
474        where = _where_filter(filters) if filters else None
475        records = await asyncio.to_thread(
476            collection.get,
477            ids=[str(doc_id) for doc_id in ids] if ids is not None else None,
478            where=where,
479        )
480        results = []
481        for raw_id, document, metadata in zip(
482            records.get("ids") or [],
483            records.get("documents") or [],
484            records.get("metadatas") or [],
485            strict=False,
486        ):
487            record_metadata = dict(metadata or {})
488            results.append(
489                {
490                    "id": str(raw_id),
491                    "document": document,
492                    "metadata": {
493                        field: record_metadata[field]
494                        for field in metadata_fields
495                        if field in record_metadata and record_metadata[field] is not None
496                    },
497                }
498            )
499        return results

501    async def upsert(  # noqa: PLR0913, PLR0917
502        self,
503        documents: list[dict[str, Any]],
504        session: Any = None,
505        text_field: str = "text",
506        embedding_field: str = "embedding",
507        id_field: str | None = None,
508        extra_fields: dict[str, Any] | None = None,
509    ) -> None:
510        """Upsert documents into the collection.
511
512        Args:
513            documents: List of document dicts with keys: id,
514                text, metadata, embedding. Metadata keys are
515                included as Chroma metadata; the document text
516                is stored as the record's document.
517            session: Accepted for interface compatibility;
518                ignored.
519            text_field: Document dict key holding the text.
520                Defaults to "text".
521            embedding_field: Document dict key holding the
522                embedding vector. Defaults to "embedding".
523            id_field: Accepted for interface compatibility;
524                Chroma stores the document id as the record
525                id.
526            extra_fields: Additional static fields to include
527                in every record's metadata (e.g.,
528                ``{"tenant_id": "..."}``).
529
530        Raises:
531            ValueError: If any metadata key contains unsafe
532                characters.
533            Exception: Re-raises Chroma errors after logging.
534        """
535        validate_upsert_documents(
536            documents,
537            text_field,
538            embedding_field,
539            "id",
540            self.id_field,
541            extra_fields,
542            self.embedding_dim,
543        )
544        if not documents:
545            return
546
547        ids = [str(doc.get("id")) for doc in documents]
548        texts = [doc[text_field] for doc in documents]
549        embeddings = [list(doc[embedding_field]) for doc in documents]
550        metadatas: list[dict[str, Any]] = []
551        for doc in documents:
552            metadata = dict(doc.get("metadata", {}))
553            if extra_fields:
554                metadata.update(extra_fields)
555            metadatas.append(metadata)
556
557        try:
558            collection = self._resolve_collection()
559            await asyncio.to_thread(
560                collection.upsert,
561                ids=ids,
562                documents=texts,
563                embeddings=embeddings,
564                metadatas=metadatas,
565            )
566            logger.info("Upserted %d documents to %s.", len(documents), self.table_name)
567        except Exception as exc:
568            logger.error("Upsert failed on %s: %s", self.table_name, exc)
569            raise
