1"""Claims ingestion adapter over ragnor.
2
3Claim ingestion writes directly into the ``claims``
4vector store (the binding lives in
5:mod:`app.domain.claims.retriever`): the claim text is
6embedded, the embedding is validated against the
7configured dimension, and the claim row is upserted
8with its metadata. This module is the permanent home for the binding.
9"""

28async def ingest_claim(  # noqa: PLR0913, PLR0917
29    claim_uuid: uuid.UUID,
30    claim_id: str,
31    owner_id: uuid.UUID,
32    claim_type: str,
33    description: str,
34    policy_number: str,
35    status: str,
36    amount: float,
37) -> None:
38    """Ingest a single claim into the vector store.
39
40    Args:
41        claim_uuid: Internal UUID of the claim row.
42        claim_id: Human-readable claim identifier.
43        owner_id: UUID of the user who owns the claim.
44        claim_type: Category of the claim.
45        description: Factual description of the incident.
46        policy_number: The policyholder's policy number.
47        status: Current processing status.
48        amount: Claimed amount in US dollars.
49    """
50    from app.core.config import get_settings  # noqa: PLC0415
51
52    embed_fn = EmbeddingFactory.get_embedding_function()
53    text_content = f"Claim {claim_id}: {claim_type} - {description}"
54    embedding = await embed_fn([text_content])
55    embedding = embedding[0]
56    validate_embedding(
57        embedding,
58        expected_dim=get_embedding_dimension(get_settings().embedding_model),
59        label="claim embedding",
60    )
61
62    store = get_vector_store(
63        table_name=CLAIMS_RETRIEVER_SPEC.table_name,
64        id_field=CLAIMS_RETRIEVER_SPEC.id_field,
65    )
66    await store.upsert(
67        documents=[
68            {
69                "id": str(claim_uuid),
70                "text": text_content,
71                "embedding": embedding,
72                "metadata": {
73                    "claim_id": claim_id,
74                    "policy_number": policy_number,
75                    "claim_type": claim_type,
76                    "status": status,
77                    "amount": amount,
78                    "description": description,
79                    "owner_id": str(owner_id),
80                },
81            }
82        ]
83    )

86async def ingest_all_claims() -> None:
87    """Ingest all claims into the vector store.
88
89    Reads every claim row and calls
90    :func:`ingest_claim` for each one. Used to
91    (re)build the claims index from the database.
92    """
93    async with async_session_factory() as session:
94        result = await session.execute(select(Claim))
95        claims = result.scalars().all()
96
97        for claim in claims:
98            await ingest_claim(
99                claim_uuid=claim.id,
100                claim_id=claim.claim_id,
101                owner_id=claim.owner_id,
102                claim_type=claim.claim_type,
103                description=claim.description,
104                policy_number=claim.policy_number,
105                status=claim.status,
106                amount=float(claim.amount),
107            )
