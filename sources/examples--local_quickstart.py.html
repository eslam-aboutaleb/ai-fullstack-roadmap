1"""Offline quickstart for ragnor's high-level convenience API.
2
3Three lines of application code -- build, ingest, search -- with
4every component overridable. The embedder here is a deterministic
5fake defined in this file (tokens hash into vector dimensions, so
6texts that share vocabulary get close L2 distance); ragnor ships no
7local embedder in core, so the builder requires an injected one.
8
9Run it with::
10
11    python examples/local_quickstart.py
12
13Everything runs in process: no network, no database, no API keys,
14no external services -- only the core install is required
15(``pip install git+https://github.com/eslam-aboutaleb/ragnor.git``).
16"""
17
18from __future__ import annotations
19
20import asyncio
21import hashlib
22import math
23import re
24
25from ragnor.builder import RagnorBuilder
26from ragnor.types import Document
27
28
29class HashingEmbedder:
30    """Deterministic fake embedder: tokens hash into vector dimensions.
31
32    Shared tokens land in shared dimensions, so texts that share
33    vocabulary get close L2 distance -- bag-of-words similarity
34    without any network calls.
35    """
36
37    def __init__(self, dim: int = 8) -> None:
38        """Initialize the embedder with a fixed dimension.
39
40        Args:
41            dim: Embedding dimensionality.
42        """
43        self._dim = dim
44
45    def embed(self, text: str) -> list[float]:
46        """Embed ``text`` into a deterministic vector.
47
48        Args:
49            text: Text to embed.
50
51        Returns:
52            The normalized bag-of-words embedding.
53        """
54        vector = [0.0] * self._dim
55        for token in re.findall(r"[a-z0-9]+", text.lower()):
56            digest = hashlib.sha256(token.encode("utf-8")).digest()
57            index = int.from_bytes(digest[:2], "big") % self._dim
58            vector[index] += 1.0
59        norm = math.sqrt(sum(value * value for value in vector))
60        if norm > 0.0:
61            vector = [value / norm for value in vector]
62        return vector
63
64    async def __call__(self, texts: list[str]) -> list[list[float]]:
65        """Embed a batch of texts.
66
67        Args:
68            texts: Texts to embed.
69
70        Returns:
71            One embedding vector per text.
72        """
73        return [self.embed(text) for text in texts]
74
75    async def embed_query(self, text: str) -> list[float]:
76        """Embed a query.
77
78        Args:
79            text: Query text.
80
81        Returns:
82            The query embedding vector.
83        """
84        return self.embed(text)
85
86    async def embed_documents(self, texts: list[str]) -> list[list[float]]:
87        """Embed a batch of documents.
88
89        Args:
90            texts: Document texts to embed.
91
92        Returns:
93            One embedding vector per document.
94        """
95        return [self.embed(text) for text in texts]
96

98async def main() -> None:
99    """Build a Ragnor instance, ingest two documents, and search."""
100    embedder = HashingEmbedder(dim=8)
101
102    # The three-line workflow: build, ingest, search.
103    rag = await RagnorBuilder.local(embedder, embedding_dim=8).build()
104    count = await rag.ingest(
105        [
106            Document(
107                id="refund-policy",
108                text="The refund policy covers water damage within thirty days of purchase.",
109            ),
110            Document(
111                id="office-hours",
112                text="Office hours are Monday to Friday, nine to five.",
113            ),
114        ]
115    )
116    print(f"ingested {count} chunks")
117
118    results = await rag.search("refund policy", n_results=3)
119    print("top results for 'refund policy':")
120    for result in results:
121        print(f"  [{result.id}] {result.document[:70]}")
122
123    print("quickstart complete: offline build, ingest, and search all work")
