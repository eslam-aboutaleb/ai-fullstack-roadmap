1"""In-process aggregates for LLM cost, tool latency, and limiter rejections.
2
3Events are held in bounded deques rather than plain counters so one store can
4answer both "over this window" and "since boot", and so a process that stays up
5for weeks cannot grow the store without limit. The store is deliberately
6synchronous and in-process: producing telemetry must never be able to block or
7fail the request that is being measured.
8"""

98def _mean_milliseconds(latencies: list[float]) -> float:
99    """Return the average of ``latencies`` in milliseconds, rounded for display."""
100    if not latencies:
101        return 0.0
102    return round(sum(latencies) * 1000.0 / len(latencies), 3)

105def _count_by(values: list[str]) -> dict[str, int]:
106    """Count ``values`` and return them sorted so output is stable across calls."""
107    return dict(sorted(Counter(values).items()))

110def _summarise_llm(calls: list[LlmCall]) -> dict[str, Any]:
111    """Aggregate LLM calls into the shape the health endpoint reports.
112
113    Args:
114        calls: Calls inside the reporting window.
115
116    Returns:
117        dict[str, Any]: Call counts, token and dollar totals, and average latency.
118    """
119    return {
120        "call_count": len(calls),
121        "error_count": sum(1 for call in calls if not call.success),
122        "prompt_tokens": sum(call.prompt_tokens for call in calls),
123        "completion_tokens": sum(call.completion_tokens for call in calls),
124        "total_cost_usd": round(sum(call.cost_usd for call in calls), 6),
125        "avg_latency_ms": _mean_milliseconds([call.latency_seconds for call in calls]),
126        "calls_by_model": _count_by([call.model for call in calls]),
127    }

130def _summarise_tools(calls: list[ToolCall]) -> dict[str, Any]:
131    """Aggregate tool invocations into per-tool counts and average latency.
132
133    Args:
134        calls: Invocations inside the reporting window.
135
136    Returns:
137        dict[str, Any]: Totals plus one entry per tool, ordered by tool name.
138    """
139    grouped: dict[str, list[ToolCall]] = {}
140    for call in calls:
141        grouped.setdefault(call.name, []).append(call)
142    by_tool = [
143        {
144            "name": name,
145            "call_count": len(grouped[name]),
146            "error_count": sum(1 for call in grouped[name] if not call.success),
147            "avg_latency_ms": _mean_milliseconds([call.latency_seconds for call in grouped[name]]),
148        }
149        for name in sorted(grouped)
150    ]
151    return {
152        "total_calls": len(calls),
153        "error_count": sum(1 for call in calls if not call.success),
154        "by_tool": by_tool,
155    }

158def _summarise_rejections(rejections: list[LimiterRejection]) -> dict[str, Any]:
159    """Aggregate limiter rejections into per-route counts.
160
161    Args:
162        rejections: Rejections inside the reporting window.
163
164    Returns:
165        dict[str, Any]: Total rejections plus a per-route breakdown.
166    """
167    return {
168        "rejections": len(rejections),
169        "by_route": _count_by([rejection.route for rejection in rejections]),
170    }

173class MetricsRegistry:
174    """Thread-safe, bounded store of telemetry events."""
175
176    def __init__(self, max_events: int = _MAX_EVENTS) -> None:
177        """Initialise the registry.
178
179        Args:
180            max_events: Number of events retained per event type.
181        """
182        self._llm_calls: deque[LlmCall] = deque(maxlen=max_events)
183        self._tool_calls: deque[ToolCall] = deque(maxlen=max_events)
184        self._limiter_rejections: deque[LimiterRejection] = deque(maxlen=max_events)
185        # LiteLLM fires its logging hooks from its own worker threads, so the
186        # store cannot assume a single-threaded event loop.
187        self._lock = threading.Lock()
188
189    def record_llm_call(self, call: LlmCall) -> None:
190        """Append one model completion to the store.
191
192        Args:
193            call: The completion to record.
194        """
195        with self._lock:
196            self._llm_calls.append(call)
197        logger.debug(
198            "LLM call recorded",
199            extra={
200                "event": "llm_call",
201                "model": call.model,
202                "prompt_tokens": call.prompt_tokens,
203                "completion_tokens": call.completion_tokens,
204                "cost_usd": call.cost_usd,
205                "latency_seconds": call.latency_seconds,
206                "success": call.success,
207            },
208        )
209
210    def record_tool_call(self, call: ToolCall) -> None:
211        """Append one tool invocation to the store.
212
213        Args:
214            call: The invocation to record.
215        """
216        with self._lock:
217            self._tool_calls.append(call)
218        logger.debug(
219            "Tool call recorded",
220            extra={
221                "event": "tool_call",
222                "tool": call.name,
223                "latency_seconds": call.latency_seconds,
224                "success": call.success,
225            },
226        )
227
228    def record_limiter_rejection(self, rejection: LimiterRejection) -> None:
229        """Append one rate-limit rejection to the store.
230
231        Args:
232            rejection: The rejection to record.
233        """
234        with self._lock:
235            self._limiter_rejections.append(rejection)
236        logger.debug(
237            "Rate limit rejection recorded",
238            extra={
239                "event": "rate_limit_rejection",
240                "route": rejection.route,
241            },
242        )
243
244    def snapshot(self, window_seconds: float = DEFAULT_WINDOW_SECONDS) -> dict[str, Any]:
245        """Summarise the events recorded in the last ``window_seconds``.
246
247        Args:
248            window_seconds: Length of the reporting window.
249
250        Returns:
251            dict[str, Any]: Window length plus LLM, tool, and limiter summaries.
252        """
253        cutoff = time.time() - window_seconds
254        with self._lock:
255            llm_calls = [call for call in self._llm_calls if call.timestamp >= cutoff]
256            tool_calls = [call for call in self._tool_calls if call.timestamp >= cutoff]
257            rejections = [
258                rejection for rejection in self._limiter_rejections if rejection.timestamp >= cutoff
259            ]
260        return {
261            "window_seconds": window_seconds,
262            "llm": {**_summarise_llm(llm_calls), "window_seconds": window_seconds},
263            "tools": {**_summarise_tools(tool_calls), "window_seconds": window_seconds},
264            "rate_limits": {**_summarise_rejections(rejections), "window_seconds": window_seconds},
265        }
266
267    def reset(self) -> None:
268        """Drop every recorded event."""
269        with self._lock:
270            self._llm_calls.clear()
271            self._tool_calls.clear()
272            self._limiter_rejections.clear()

288def build_llm_call(  # noqa: PLR0913 -- one field per billable reading of a call
289    *,
290    model: str = _UNKNOWN_MODEL,
291    prompt_tokens: int = 0,
292    completion_tokens: int = 0,
293    cost_usd: float = 0.0,
294    latency_seconds: float = 0.0,
