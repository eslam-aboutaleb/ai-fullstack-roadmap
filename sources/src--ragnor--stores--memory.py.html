1"""In-memory vector store for tests and zero-dependency development.
2
3Implements the :class:`VectorStore` interface entirely in process
4memory: brute-force vector search, a token-overlap approximation
5of full-text search, and the shared Reciprocal Rank Fusion merge
6(:mod:`ragnor.stores._scoring`).
7
8The vector stage ranks by the store's distance metric
9(:mod:`ragnor.metrics`): ``l2`` by default, ``cosine`` when the
10store is constructed with ``metric="cosine"``.
11
12The store subclasses :class:`ragnor.stores.base.HybridVectorStore`:
13the vector stage lives in :meth:`vector_search` and the keyword
14stage in :meth:`keyword_search` (both capability methods that
15always raise on failure), and the inherited composite
16``hybrid_search`` runs both stages and fuses them with RRF.
17
18Approximation notice: the keyword score is a token-overlap ratio,
19not a real ``ts_rank`` over a ``tsvector``. It is intended for
20tests and development, not as a production search backend.
21"""

51@vector_store_registry.register("memory")
52class InMemoryVectorStore(HybridVectorStore):
53    """In-memory vector store.
54
55    Brute-force vector search over the store's distance metric
56    (``l2`` by default, ``cosine`` opt-in via ``metric``) with
57    token-overlap keyword matching, merged with the shared RRF
58    formula (``1.0 / (60 + rank)`` per stage, 1-based ranks,
59    tie-break by id). Result dicts carry the same keys as the
60    other backends: ``id``, ``document``, ``metadata``,
61    ``distance`` (``None`` for keyword-only hits),
62    ``keyword_score`` (``None`` for vector-only hits), and
63    ``_rrf_score``.
64    """

117    def _matches_filters(self, record: dict[str, Any], filters: dict[str, Any]) -> bool:
118        """Return whether ``record`` satisfies every equality or IN filter."""
119        for key, value in filters.items():
120            record_val = record.get(key)
121            if isinstance(value, (list, tuple, set, frozenset)):
122                if record_val not in value:
123                    return False
124            elif record_val != value:
125                return False
126        return True

128    def _metadata_projection(

130    ) -> dict[str, Any]:
131        """Project ``metadata_fields`` out of in-memory ``record``."""
132        return project_metadata(record, metadata_fields)

134    async def vector_search(  # noqa: PLR0913, PLR0917
135        self,
136        embedding: list[float],
137        n_results: int,
138        threshold: float,
139        text_field: str = "text",
140        metadata_fields: list[str] | None = None,
141        session: Any = None,
142        **filters: Any,
143    ) -> list[dict[str, Any]]:
144        """Brute-force vector search over the record map.
145
146        The candidates are scored with the store's distance metric:
147        ``l2_distance`` for ``l2`` (the default) and
148        ``cosine_distance`` for ``cosine``.
149
150        Args:
151            embedding: Query embedding vector (required; the method
152                does not auto-embed).
153            n_results: Maximum number of results to return.
154            threshold: Maximum distance under the store's metric
155                (metric-relative; never comparable across metrics).
156            text_field: Name of the field containing the document text.
157            metadata_fields: Fields to include in the result metadata.
158            session: Accepted for interface compatibility; ignored
159                (no database).
160            **filters: Equality filters matched against document
161                fields (metadata and extra fields).
162
163        Returns:
164            list[dict]: Result dicts ordered by distance ascending
165                (tie-broken by id), each carrying ``id``, ``document``,
166                ``metadata``, ``distance``, and ``keyword_score``
167                (``None``).
168
169        Raises:
170            ValueError: If identifiers are unsafe or the embedding
171                has the wrong dimension.
172        """
173        validate_identifier(text_field, "text_field")
174        for key in filters:
175            validate_identifier(key, "filter key")
176        metadata_fields = metadata_fields or []
177        for field in metadata_fields:
178            validate_identifier(field, "metadata_field")
179        if self.embedding_dim is not None:
180            validate_embedding(embedding, self.embedding_dim, label="query embedding")
181
182        vector_distances: dict[str, float] = {}
183        for doc_id, record in self._records.items():
184            if not self._matches_filters(record, filters):
185                continue
186            doc_embedding = self._embeddings.get(doc_id)
187            if doc_embedding is not None:
188                if self.metric == "cosine":
189                    distance = cosine_distance(embedding, doc_embedding)
190                else:
191                    distance = l2_distance(embedding, doc_embedding)
192                if distance < threshold:
193                    vector_distances[doc_id] = distance
194
195        vector_ranked = [
196            doc_id
197            for doc_id, _ in sorted(vector_distances.items(), key=lambda item: (item[1], item[0]))[
198                :n_results
199            ]
200        ]
201        results = []
202        for doc_id in vector_ranked:
203            record = self._records[doc_id]
204            results.append(
205                format_search_hit(
206                    doc_id=str(doc_id) if doc_id is not None else None,
207                    document=record.get(text_field),
208                    metadata=self._metadata_projection(record, metadata_fields),
209                    distance=float(vector_distances[doc_id]),
210                )
211            )
212        return results

214    async def keyword_search(  # noqa: PLR0913, PLR0917
215        self,
216        query: str,
217        n_results: int,
218        text_field: str = "text",
219        metadata_fields: list[str] | None = None,
220        session: Any = None,
221        **filters: Any,
222    ) -> list[dict[str, Any]]:
223        """Token-overlap keyword search over the record map.
224
225        The keyword score is a token-overlap ratio, not a real
226        full-text rank -- an approximation intended for tests and
227        development.
228
229        Args:
230            query: Natural language search query.
231            n_results: Maximum number of results to return.
232            text_field: Name of the field containing the document text.
233            metadata_fields: Fields to include in the result metadata.
234            session: Accepted for interface compatibility; ignored
235                (no database).
236            **filters: Equality filters matched against document
237                fields (metadata and extra fields).
238
239        Returns:
240            list[dict]: Result dicts ordered by overlap ratio
241            descending (tie-broken by id), each carrying ``id``,
242            ``document``, ``metadata``, ``distance`` (``None``), and
243            ``keyword_score``.
244
245        Raises:
246            ValueError: If identifiers are unsafe.
247        """
248        validate_search_identifiers(text_field, filters, metadata_fields)
249        metadata_fields = metadata_fields or []
250        keyword_scores = score_records_overlap(
251            self._records, query, text_field, filters, self._matches_filters
252        )
253
254        keyword_ranked = [
255            doc_id
256            for doc_id, _ in sorted(keyword_scores.items(), key=lambda item: (-item[1], item[0]))[
257                :n_results
258            ]
259        ]
260        results = []
261        for doc_id in keyword_ranked:
262            record = self._records[doc_id]
263            results.append(
264                format_search_hit(
265                    doc_id=str(doc_id) if doc_id is not None else None,
266                    document=record.get(text_field),
267                    metadata=self._metadata_projection(record, metadata_fields),
268                    keyword_score=float(keyword_scores[doc_id]),
269                )
270            )
271        return results

273    async def count(
274        self,
275        session: Any = None,
276        strict: bool = False,
277        **filters: Any,
278    ) -> int:
279        """Count documents in the store.
280
281        Args:
282            session: Accepted for interface compatibility; ignored
283                (no database).
284            strict: Accepted for interface compatibility; this store
285                has no database failures to surface.
286            **filters: Equality filters matched against document
287                fields (metadata and extra fields).
288
289        Returns:
290            int: Number of matching documents.
291        """
292        for key in filters:
293            validate_identifier(key, "filter key")
294        return sum(1 for record in self._records.values() if self._matches_filters(record, filters))

296    async def upsert(  # noqa: PLR0913, PLR0917
297        self,
298        documents: list[dict[str, Any]],
299        session: Any = None,
300        text_field: str = "text",
301        embedding_field: str = "embedding",
302        id_field: str | None = None,
303        extra_fields: dict[str, Any] | None = None,
304    ) -> None:
305        """Upsert documents into the store.
306
307        The in-memory equivalent of ``ON CONFLICT (id) DO UPDATE``:
308        a document whose id already exists is replaced wholesale.
309
310        Args:
311            documents: List of document dicts with keys: id, text,
312                metadata, embedding.
313            session: Accepted for interface compatibility; ignored
314                (no database).
315            text_field: Name of the text field. Defaults to "text".
316            embedding_field: Name of the embedding field. Defaults
317                to "embedding".
318            id_field: Name of the primary key field. Defaults to the
319                store's configured ``id_field``.
320            extra_fields: Additional static fields to include in every
321                document (e.g., ``{"tenant_id": "..."}``).
322
323        Raises:
324            ValueError: If any field name contains unsafe characters.
325        """
326        id_field = validate_upsert_documents(
327            documents,
328            text_field,
329            embedding_field,
330            id_field,
331            self.id_field,
332            extra_fields,
333            self.embedding_dim,
334        )
335        if not documents:
336            return
337
338        for doc in documents:
339            doc_id = doc.get("id")
340            self._records[doc_id] = build_upsert_record(doc, id_field, text_field, extra_fields)
341            self._embeddings[doc_id] = doc[embedding_field]
342
343        logger.info("Upserted %d documents to in-memory store %s.", len(documents), self.table_name)
344
345    async def get_embeddings(self, ids: list[str], session: Any = None) -> dict[str, list[float]]:
346        return {
347            doc_id: list(self._embeddings[doc_id]) for doc_id in ids if doc_id in self._embeddings
348        }

350    async def delete(
351        self,
352        ids: list[Any] | None = None,
353        filters: dict[str, Any] | None = None,
354        session: Any = None,
355    ) -> int:
356        """Delete documents selected by ``ids`` and/or ``filters``.
357
358        At least one selector must be supplied; an empty ``ids``
359        list is rejected (an empty id list never means "delete
360        everything"). When both are supplied, both restrictions
361        apply (conjunction).
362
363        Args:
364            ids: Document ids to delete.
365            filters: Equality filters matched against document
366                fields (metadata and extra fields).
367            session: Accepted for interface compatibility; ignored
368                (no database).
369
370        Returns:
371            int: Number of deleted documents.
372
373        Raises:
374            ValueError: If no selector is supplied, ``ids`` is
375                empty, or a filter key is unsafe.
376        """
377        validate_delete_selectors(ids, filters)
378
379        deleted = 0
380        for doc_id in list(self._records):
381            if ids is not None and doc_id not in set(ids):
382                continue
383            if filters is not None and not self._matches_filters(self._records[doc_id], filters):
384                continue
385            del self._records[doc_id]
386            self._embeddings.pop(doc_id, None)
387            deleted += 1
388        logger.info("Deleted %d documents from in-memory store %s.", deleted, self.table_name)
389        return deleted

392        self,

399        """Fetch documents selected by ``ids`` and/or ``filters``.

401        With no selector, every document is returned. When both

404
405        Args:
406            ids: Document ids to fetch.
407            filters: Equality filters matched against document
408                fields (metadata and extra fields).
409            session: Accepted for interface compatibility; ignored

411            text_field: Name of the field containing the document
412                text. Defaults to "text".
413            metadata_fields: Fields to include in the result

416        Returns:

418            (score keys are meaningless for direct fetches).

420        Raises:

422        """

425            self._records,

427            filters=filters,
428            text_field=text_field,
429            metadata_fields=metadata_fields,
