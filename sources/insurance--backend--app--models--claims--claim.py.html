1"""
2SQLAlchemy ORM model for the ``claims`` table.
3
4Represents an insurance claim filed by an OmniCare policyholder. Each claim
5is linked to its owner via a foreign key, stores structured claim metadata,
6and includes a computed ``tsvector`` column for PostgreSQL full-text search and a
7``embedding`` column for pgvector hybrid retrieval.
8"""
9
10import uuid
11from datetime import date, datetime
12from decimal import Decimal
13
14from pgvector.sqlalchemy import Vector
15from sqlalchemy import Date, DateTime, ForeignKey, Index, Numeric, String, func
16from sqlalchemy.dialects.postgresql import TSVECTOR, UUID
17from sqlalchemy.orm import Mapped, mapped_column
18
19from app.models.base import Base
20
21
22class Claim(Base):
23    """Database model for an OmniCare insurance claim.
24
25    Attributes:
26        id: Primary key UUID.
27        claim_id: Human-readable claim identifier (e.g., "CLM-8821").
28        policy_number: The policyholder's policy number, indexed for lookup.
29        claim_type: Category of the claim (e.g., "Water Damage", "Personal Property").
30        status: Current processing status (e.g., "Submitted", "Approved", "Denied").
31        amount: Claimed amount in US dollars.
32        description: Factual description of the incident.
33        owner_id: Foreign key to the user who filed the claim.
34        incident_date: Calendar date the policyholder reported the incident occurred.
35            Nullable because claims filed before the column existed have no incident
36            date to backfill, and a submission may omit it.
37        created_at: Timestamp the claim row was written, generated by the database so
38            rows predating any application-level clock still order correctly.
39        updated_at: Timestamp of the last write to the row.
40        embedding: pgvector embedding of the claim text for hybrid search.
41        tsvector: PostgreSQL tsvector for PostgreSQL full-text search over the description.
42    """
43
44    __tablename__ = "claims"
45
46    id: Mapped[uuid.UUID] = mapped_column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
47    claim_id: Mapped[str] = mapped_column(String, unique=True, index=True)
48    policy_number: Mapped[str] = mapped_column(String, index=True)
49    claim_type: Mapped[str] = mapped_column(String)
50    status: Mapped[str] = mapped_column(String)
51    amount: Mapped[Decimal] = mapped_column(Numeric(12, 2))
52    description: Mapped[str] = mapped_column(String)
53    owner_id: Mapped[uuid.UUID] = mapped_column(
54        ForeignKey("users.id", ondelete="CASCADE"), index=True
55    )
56    embedding: Mapped[str] = mapped_column(Vector(1536), nullable=False, server_default="[0]")
57    text: Mapped[str | None] = mapped_column(String, nullable=True)
58    incident_date: Mapped[date | None] = mapped_column(Date, nullable=True)
59    created_at: Mapped[datetime] = mapped_column(
60        DateTime(timezone=True), nullable=False, server_default=func.now()
61    )
62    updated_at: Mapped[datetime] = mapped_column(
63        DateTime(timezone=True),
64        nullable=False,
65        server_default=func.now(),
66        onupdate=func.now(),
67    )
68    tsvector: Mapped[str] = mapped_column(TSVECTOR, nullable=False, server_default="''")
69
70    __table_args__ = (
71        Index("ix_claims_tsvector", tsvector, postgresql_using="gin"),
72        # The owner-scoped claim list filters on owner_id and orders by created_at
73        # DESC, so both columns belong in one index rather than two.
74        Index("ix_claims_owner_id_created_at", owner_id, created_at),
