41def run_migrations_offline() -> None:
42    """Run migrations in 'offline' mode.
43
44    This configures the context with just a URL and not an Engine, though an
45    Engine is acceptable here as well.  By skipping the Engine creation we
46    don't even need a DBAPI to be available.
47
48    Calls to context.execute() here emit the given string to the script output.
49    """
50    url = _get_database_url()
51    context.configure(
52        url=url,
53        target_metadata=target_metadata,
54        literal_binds=True,
55        dialect_opts={"paramstyle": "named"},
56    )
57
58    with context.begin_transaction():
59        context.run_migrations()

62def _run_sync_migrations(connection: Any) -> None:

64    context.configure(

66        target_metadata=target_metadata,
67        compare_type=True,
68        compare_server_default=True,

75async def run_migrations_online() -> None:
76    """Run migrations in 'online' mode.
77
78    Creates an async engine and delegates migration execution to a sync
79    function via connection.run_sync(), which is the SQLAlchemy-recommended
80    pattern for running sync-only Alembic operations over an async connection.
81    """
82    database_url = _get_database_url()
83    connectable = create_async_engine(
84        database_url,
85        poolclass=pool.NullPool,
86    )
87
88    async with connectable.connect() as connection:
89        await connection.run_sync(_run_sync_migrations)
90
91    await connectable.dispose()
