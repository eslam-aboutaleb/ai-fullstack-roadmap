1"""
2The audit trail: one append-only row per high-value action, and the hook that
3writes the claim-lifecycle half of it.
4
5Two call paths live here on purpose:
6
7  * ``record_audit`` is what a route calls directly, inside the transaction
8    that performs the audited action, so the entry commits with it.
9    Authentication events and feedback submissions are recorded this way.
10  * A ``ClaimEvent`` flush listener covers the claim lifecycle (submission,
11    status transitions, withdrawal). That half is recorded from the model layer
12    rather than the route because the claim lifecycle is owned by code this
13    module must not reach into, and a listener also catches a transition written
14    by any future caller -- a trail a future rewrite of the route could bypass
15    is not a trail.
16
17Both paths share one guarantee: an audit write that fails is logged and
18swallowed. An outage of the trail must not take the business action down with
19it, which means an entry can be missing during an incident -- the same trade
20every append-only log makes, and the reason this table is not a foreign key to
21success.
22"""

43# A user agent is a header the caller controls, so it is capped before it can
44# become a stored document; the truncated prefix is what distinguishes agents
45# in practice, so nothing that matters is lost.
46_MAX_USER_AGENT_LENGTH = 512

49def _request_scoped_identity(
50    actor_id: uuid.UUID | None,
51    request: Request | None,
52) -> tuple[uuid.UUID | None, str | None, str | None]:
53    """Resolve the actor, client IP, and user agent for one entry.
54
55    The actor falls back to the ``current_user_id`` context variable when the
56    caller does not name one, so an instrumented endpoint that already
57    authenticated the request does not have to thread the id through the call.
58
59    Args:
60        actor_id: The actor the caller named, if any.
61        request: The request being audited, if there is one.
62
63    Returns:
64        tuple[uuid.UUID | None, str | None, str | None]: The actor, the client
65        IP, and the user agent, each None when it is not knowable.
66    """
67    resolved_actor = actor_id
68    if resolved_actor is None:
69        # ``current_user_id`` is only set inside an authenticated request; a
70        # startup path or a worker sees LookupError and records a system actor.
71        try:
72            resolved_actor = current_user_id.get()
73        except LookupError:
74            resolved_actor = None
75
76    ip_address: str | None = None
77    user_agent: str | None = None
78    if request is not None:
79        ip_address = request.client.host if request.client is not None else None
80        raw_user_agent = request.headers.get("user-agent")
81        if raw_user_agent is not None:
82            user_agent = raw_user_agent[:_MAX_USER_AGENT_LENGTH]
83
84    return resolved_actor, ip_address, user_agent

87def _audit_values(  # noqa: PLR0913 - one field per column an entry carries
88    action: str,
89    entity_type: str,
90    entity_id: str | None,
91    *,
92    before: dict | None,
93    after: dict | None,
94    actor_id: uuid.UUID | None,
95    request: Request | None,
96) -> dict:
97    """Assemble the column values of one audit row, filling in the request context.
98
99    Args:
100        action: The verb performed, e.g. ``"auth.signin"``.
101        entity_type: The kind of thing acted on, e.g. ``"user"``.
102        entity_id: The identifier of the thing acted on, if known.
103        before: Fields as they were, or None when nothing was overwritten.
104        after: Fields as they are now, or None.
105        actor_id: The actor the caller named, if any.
106        request: The request being audited, if there is one.
107
108    Returns:
109        dict: The values for an ``AuditEvent`` row.
110    """
111    resolved_actor, ip_address, user_agent = _request_scoped_identity(actor_id, request)
112    return {
113        "actor_id": resolved_actor,
114        "action": action,
115        "entity_type": entity_type,
116        "entity_id": entity_id,
117        "before": before,
118        "after": after,
119        "ip_address": ip_address,
120        "user_agent": user_agent,
121        # The correlation id ties this row to its own line in the application
122        # log; empty string (no request in context) is stored as NULL.
123        "correlation_id": get_request_id() or None,
124    }

127async def record_audit(  # noqa: PLR0913 - one field per column an entry carries
128    session: AsyncSession,
129    action: str,
130    entity_type: str,
131    entity_id: str | None,
132    *,
133    before: dict | None = None,
134    after: dict | None = None,
135    actor_id: uuid.UUID | None = None,
136    request: Request | None = None,
137) -> None:
138    """Append one entry to the audit trail, never disturbing the caller.
139
140    The row is written inside a SAVEPOINT of the caller's transaction, so the
141    caller's own commit persists both together: an entry can no more outlive the
142    action failing than the action can fail because the entry did. Call this
143    before the commit you were going to make anyway -- no second transaction,
144    and nothing new to forget.
145
146    A write that fails rolls back to its own savepoint, which is why the
147    caller's session is safe whatever happens here: no rollback of the caller's
148    transaction, no expired objects, no poisoned connection.
149
150    Args:
151        session: The session performing the audited action.
152        action: The verb performed, e.g. ``"auth.signin"``.
153        entity_type: The kind of thing acted on, e.g. ``"user"``.
154        entity_id: The identifier of the thing acted on, if known.
155        before: Fields as they were, or None when nothing was overwritten.
156        after: Fields as they are now, or None.
157        actor_id: The actor, when the caller can name one. Falls back to the
158            authenticated user in context.
159        request: The request being audited, when one is in scope. Supplies the
160            client IP and user agent.
161
162    Returns:
163        None: This function never raises. A failed write is logged at warning
164        level and the request that produced it continues.
165    """
166    try:
167        values = _audit_values(
168            action,
169            entity_type,
170            entity_id,
171            before=before,
172            after=after,
173            actor_id=actor_id,
174            request=request,
175        )
176        async with session.begin_nested():
177            await session.execute(insert(AuditEvent).values(**values))
178    except Exception:
179        logger.warning(
180            "Audit trail entry for '%s' on '%s' could not be written.",
181            action,
182            entity_type,
183            exc_info=True,
184        )

187def _claim_event_action(from_status: str | None, to_status: str) -> str:
188    """Name the audit verb a lifecycle event deserves.
189
190    The filing event that opens a claim's history is a submission, the claimant
191    pulling their own claim back is a withdrawal, and everything in between is
192    a handler decision. Three verbs rather than one, because "the status
193    changed" is the question an auditor asks last and never first.
194
195    Args:
196        from_status: The status the claim left, or None for the filing event.
197        to_status: The status the claim moved to.
198
199    Returns:
200        str: ``"claim.submitted"``, ``"claim.withdrawn"``, or
201        ``"claim.status_changed"``.
202    """
203    if from_status is None:
204        return "claim.submitted"
205    if to_status == ClaimStatus.WITHDRAWN.value:
206        return "claim.withdrawn"
207    return "claim.status_changed"

210def _claim_labels(session: Session, events: list[ClaimEvent]) -> dict[uuid.UUID, str]:
211    """Resolve the human-readable claim id behind each lifecycle event.
212
213    ``claim_events`` stores the claim's primary key, but the trail is only
214    usable if a human can search it with the claim id they were given, so the
215    label is resolved here. The lookup runs inside the flush, where the session
216    serves it from the database it is already writing to.
217
218    Args:
219        session: The session flushing the events.
220        events: The lifecycle events being audited.
221
222    Returns:
223        dict[uuid.UUID, str]: Claim primary key to claim id, for every claim
224        that could be resolved. A claim that cannot be read is absent, and its
225        event is recorded against the primary key instead.
226    """
227    labels: dict[uuid.UUID, str] = {}
228    for event_row in events:
229        if event_row.claim_uuid in labels:
230            continue
231        try:
232            claim = session.get(Claim, event_row.claim_uuid)
233        except Exception:
234            logger.warning("Could not resolve a claim label for the audit trail.", exc_info=True)
235            continue
236        if claim is not None:
237            labels[event_row.claim_uuid] = claim.claim_id
238    return labels

241def _queue_claim_audit_rows(session: Session, flushed_events: list[ClaimEvent]) -> None:
242    """Queue one audit row per lifecycle event flushed in this session.
243
244    Rows are added to the same session rather than written directly, so they
245    join the audited transaction: a transition that commits leaves its entry,
246    and one that rolls back leaves nothing. This runs inside a flush, where no
247    ``await`` is possible, which is why the add is synchronous.
248
249    Args:
250        session: The session whose flush produced the events.
251        flushed_events: The lifecycle events this flush wrote.
252    """
253    claim_labels = _claim_labels(session, flushed_events)
254    for flushed_event in flushed_events:
255        entity_id = claim_labels.get(flushed_event.claim_uuid, str(flushed_event.claim_uuid))
256        # The request's IP and user agent are not reachable from a flush, where
257        # no request object is in scope; the actor is, because the lifecycle
258        # records it on the event itself.
259        session.add(
260            AuditEvent(
261                **_audit_values(
262                    _claim_event_action(flushed_event.from_status, flushed_event.to_status),
263                    "claim",
264                    entity_id,
265                    before={"status": flushed_event.from_status}
266                    if flushed_event.from_status
267                    else None,
268                    after={"status": flushed_event.to_status},
269                    actor_id=flushed_event.actor_id,
270                    request=None,
271                )
272            )
273        )

276def _after_flush(session: Session, flush_context: Any) -> None:
277    """Queue audit rows for every lifecycle event a flush wrote.
278
279    Args:
280        session: The session that finished flushing.
281        flush_context: SQLAlchemy's internal flush context, unused.
282    """
283    try:
284        flushed_events = [obj for obj in session.new if isinstance(obj, ClaimEvent)]
285    except Exception:
286        logger.warning("Could not inspect flushed claim events for the audit trail.", exc_info=True)
287        return
288
289    if not flushed_events:
290        return
291
292    try:
293        _queue_claim_audit_rows(session, flushed_events)
294    except Exception:
295        logger.warning("Audit trail rows for claim events could not be queued.", exc_info=True)

298# Listening on the Session class rather than one sessionmaker catches the
299# request sessions, the worker sessions, and the application's own startup
300# work alike: a lifecycle event should not be auditable only when it happens
301# inside a route handler.
302event.listen(Session, "after_flush", _after_flush)
