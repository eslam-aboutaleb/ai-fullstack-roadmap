1"""Procedure playbook version storage for the ragnor ingestion pipeline.
2
3The ``procedure_versions`` table is application-side storage: ragnor's
4:class:`ragnor.ingestion.IngestionPipeline` orchestrates
5load/hash/compare/retire/chunk/embed/upsert and talks to a
6:class:`ragnor.ingestion.VersionStore` protocol, which
7:class:`ProcedureVersionStore` implements here.
8
9One store instance is bound to one playbook file. The pipeline is deliberately
10per-source — its ``find_active`` returns "the active version" for the source it
11was handed — so binding the store to a single ``source_file`` is what lets one
12directory of playbooks share a single pipeline without their versions
13colliding.
14
15Unlike the policy adapter there is no historical snapshot version to pin: this
16table is new, so the chunker's own reported version can be recorded as-is and an
17unchanged corpus still compares equal on the next run.
18"""
19

40async def _find_active_version(session: Any, source_file: str) -> ProcedureVersion | None:
41    """Return the active version row for ``source_file``, or ``None``."""
42    return (
43        await session.execute(
44            select(ProcedureVersion)
45            .where(
46                ProcedureVersion.source_file == source_file,
47                ProcedureVersion.effective_to.is_(None),
48            )
49            .order_by(ProcedureVersion.effective_from.desc())
50            .limit(1)
51        )
52    ).scalar_one_or_none()
53

55class ProcedureVersionStore:
56    """``VersionStore`` implementation against the ``ProcedureVersion`` model.
57
58    Every method joins the pipeline's transaction: the session is the
59    pipeline's, so version retirement, chunk deletion, and version insertion
60    commit together with the chunk upsert.
61
62    Args:
63        source_file: Playbook file name this store versions, e.g.
64            ``coverage-review.md``.
65    """
66
67    def __init__(self, source_file: str) -> None:
68        self.source_file = source_file
69
70    async def _find_active(self, session: Any, source_file: str) -> ProcedureVersion | None:
71        return await _find_active_version(session, source_file)
72
73    async def find_active(self, session: Any) -> VersionRecord | None:
74        """Return the active version for this playbook, or ``None``.
75
76        Args:
77            session: Session the read runs on.
78
79        Returns:
80            The active :class:`VersionRecord`, or ``None`` when this
81            playbook has never been ingested.
82        """
83        active_version = await self._find_active(session, self.source_file)
84        if active_version is None:
85            return None
86        return VersionRecord(
87            version_id=str(active_version.version_id),
88            source_id=None,
89            snapshot=IngestionSnapshot(
90                source_hash=active_version.source_hash,
91                embedding_model=active_version.embedding_model or "",
92                embedding_dim=active_version.embedding_dim or 0,
93                chunker_version=active_version.chunker_version or "",
94                chunk_size=active_version.chunk_size or 0,
95                overlap=active_version.overlap or 0,
96                retrieval_schema_version=(active_version.retrieval_schema_version or ""),
97            ),
98        )
99
100    async def close_active(self, session: Any) -> None:
101        """Retire the active version of this playbook.
102
103        Args:
104            session: Session the retirement runs on.
105        """
106        active_version = await self._find_active(session, self.source_file)
107        if active_version is not None:
108            active_version.effective_to = datetime.now(UTC)
109            await session.flush()
110
111    async def create_version(self, session: Any, snapshot: IngestionSnapshot) -> VersionRecord:
112        """Insert a new version row carrying ``snapshot``.
113
114        Args:
115            session: Session the insert runs on.
116            snapshot: The snapshot to record.
117
118        Returns:
119            The created :class:`VersionRecord`.
120        """
121        version = ProcedureVersion(
122            source_file=self.source_file,
123            version="current",
124            effective_from=datetime.now(UTC),
125            source_hash=snapshot.source_hash,
126            embedding_model=snapshot.embedding_model,
127            embedding_dim=snapshot.embedding_dim,
128            chunker_version=snapshot.chunker_version,
129            chunk_size=snapshot.chunk_size,
130            overlap=snapshot.overlap,
131            retrieval_schema_version=snapshot.retrieval_schema_version,
132        )
133        session.add(version)
134        await session.flush()
135        return VersionRecord(
136            version_id=str(version.version_id),
137            source_id=None,
138            snapshot=snapshot,
139        )
140
141    async def delete_chunks(self, session: Any, version_id: str) -> None:
142        """Delete the chunks belonging to ``version_id``.
143
144        Args:
145            session: Session the delete runs on.
146            version_id: Owning version id.
147        """
148        await session.execute(
149            sa_text("DELETE FROM procedure_chunks WHERE procedure_version_id = :vid"),
150            {"vid": version_id},
151        )

154async def get_active_version_record(source_file: str) -> dict[str, Any] | None:
155    """Return the active version provenance for one playbook.
156
157    Opened outside the ingestion pipeline so the ``get_procedure`` tool can
158    report which version of a playbook it just returned without running a
159    pipeline.
160
161    Args:
162        source_file: Playbook file name, e.g. ``coverage-review.md``.
163
164    Returns:
165        A dict with ``version``, ``source_hash``, ``chunker_version``, and
166        ``effective_from`` keys, or ``None`` when the playbook has no active
167        version.
168    """
169    async with async_session_factory() as session:
170        active_version = await _find_active_version(session, source_file)
171        if active_version is None:
172            return None
173        return {
174            "version": active_version.version,
175            "source_hash": active_version.source_hash,
176            "chunker_version": active_version.chunker_version,
177            "effective_from": active_version.effective_from,
178        }
