1"""MCP (Model Context Protocol) server adapter for ragnor.
2
3Exposes KnowledgeBridge retrieval, source verification, and status tools
4over MCP stdio or remote transports.
5"""
6
7from __future__ import annotations
8
9import logging
10from collections.abc import Callable
11from typing import Any
12
13from ragnor.tools.bridge import KnowledgeBridge
14
15logger = logging.getLogger(__name__)
16
17# Detect available MCP SDK versions (mcp 2.x MCPServer vs mcp 1.x FastMCP)
18_MCPServerClass: Any = None
19try:
20    from mcp.server.mcpserver import MCPServer
21
22    _MCPServerClass = MCPServer
23except (ImportError, ModuleNotFoundError):
24    try:
25        from mcp.server.fastmcp import FastMCP  # type: ignore[no-redef]
26
27        _MCPServerClass = FastMCP
28    except (ImportError, ModuleNotFoundError):
29        _MCPServerClass = None
30

32def create_mcp_server(  # noqa: PLR0913
33    bridge: KnowledgeBridge,
34    name: str = "ragnor-knowledge",
35    *,
36    enable_management_tools: bool = False,
37    server_class: Any | None = None,
38    client_id: str | None = None,
39    client_id_provider: Callable[[], str | None] | None = None,
40) -> Any:
41    """Create and configure an MCP server instance wrapping a KnowledgeBridge.
42
43    By default, only read tools (search_knowledge, get_source, knowledge_status)
44    are registered. Ingestion/sync and revocation are separate opt-in management
45    tools enabled only when :paramref:`enable_management_tools` is True and the
46    bridge's allow_management flag is set.
47
48    Args:
49        bridge: The configured :class:`KnowledgeBridge`.
50        name: MCP server name reported during initialization.
51        enable_management_tools: If True, registers sync and deletion tools.
52        server_class: Optional override class for MCPServer / FastMCP.
53
54    Returns:
55        Configured MCPServer or FastMCP instance.
56
57    Raises:
58        ImportError: If the official 'mcp' SDK is not installed and no server_class is given.
59    """
60    cls = server_class or _MCPServerClass
61    if cls is None:
62        raise ImportError(
63            "The MCP server adapter requires the 'mcp' package. "
64            "Install it with: pip install 'ragnor[mcp]' or pip install 'mcp>=1.0.0'"
65        )
66
67    server = cls(name)
68
69    def identity() -> str | None:
70        # The provider is trusted host code, never a model-supplied tool argument.
71        if client_id_provider is not None:
72            return client_id_provider()
73        try:
74            from mcp.server.auth.middleware.auth_context import get_access_token  # noqa: PLC0415
75        except ImportError:
76            return client_id
77        token = get_access_token()
78        return token.client_id if token is not None else client_id
79
80    # search_knowledge tool
81    @server.tool(
82        name="search_knowledge",
83        description=(
84            "Execute authorized hybrid search across indexed knowledge collections. "
85            "Returns a structured EvidenceBundle containing deduplicated evidence items, "
86            "stable source/revision citations, and token budget accounting."
87        ),
88    )
89    async def search_knowledge(  # noqa: PLR0913, PLR0917
90        query: str,
91        collection: str = bridge.default_collection,
92        filters: dict[str, Any] | None = None,
93        candidate_k: int = 20,
94        final_k: int = 5,
95        budget_limit: int | None = None,
96        scope: str | None = None,
97    ) -> dict[str, Any]:
98        """Search knowledge collection with budget limits and tenant isolation."""
99        bundle = await bridge.search_knowledge(
100            client_id=identity(),
101            query=query,
102            collection=collection,
103            filters=filters,
104            candidate_k=candidate_k,
105            final_k=final_k,
106            budget_limit=budget_limit,
107            scope=scope,
108        )
109        return bundle.to_dict()
110
111    # get_source tool
112    @server.tool(
113        name="get_source",
114        description=(
115            "Fetch an authorized, bounded text span and revision metadata from an immutable "
116            "document snapshot. Prevents substituting newer content for old citations."
117        ),
118    )
119    async def get_source(  # noqa: PLR0913, PLR0917
120        source_ref: str,
121        document_id: str | None = None,
122        revision: str = "v1",
123        start_offset: int | None = None,
124        end_offset: int | None = None,
125        max_span_length: int | None = None,
126        collection: str = bridge.default_collection,
127        scope: str | None = None,
128    ) -> dict[str, Any]:
129        """Fetch bounded source span for citation verification."""
130        span_result = await bridge.get_source(
131            client_id=identity(),
132            source_ref=source_ref,
133            document_id=document_id,
134            revision=revision,
135            start_offset=start_offset,
136            end_offset=end_offset,
137            max_span_length=max_span_length,
138            collection=collection,
139            scope=scope,
140        )
141        return span_result.to_dict()
142
143    # knowledge_status tool
144    @server.tool(
145        name="knowledge_status",
146        description=(
147            "Inspect sanitized health, document count, and freshness summary for an "
148            "authorized knowledge collection. Does not expose internal connection credentials."
149        ),
150    )
151    async def knowledge_status(
152        collection: str = bridge.default_collection,
153        scope: str | None = None,
154        include_expensive: bool = False,
155    ) -> dict[str, Any]:
156        """Retrieve sanitized status summary for a collection."""
157        return await bridge.knowledge_status(
158            client_id=identity(),
159            collection=collection,
160            scope=scope,
161            include_expensive=include_expensive,
162        )
163
164    # Opt-in management tools
165    if enable_management_tools:
166        if not bridge.allow_management:
167            logger.warning(
168                "enable_management_tools=True was requested on MCP server, but "
169                "bridge.allow_management is False. Management calls will be rejected."
170            )
171
172        @server.tool(
173            name="sync_collection",
174            description=(
175                "Ingest and sync document batches into a knowledge collection. "
176                "Guarded by host management permissions."
177            ),
178        )
179        async def sync_collection(
180            collection: str,
181            source_id: str,
182            documents: list[dict[str, Any]],
183            scope: str | None = None,
184        ) -> dict[str, Any]:
185            """Ingest document batches into collection."""
186            return await bridge.sync_collection(
187                client_id=identity(),
188                collection=collection,
189                source_id=source_id,
190                documents=documents,
191                scope=scope,
192            )
193
194        @server.tool(
195            name="revoke_document",
196            description=(
197                "Authoritatively remove/revoke a document and its active chunks from search. "
198                "Guarded by host management permissions."
199            ),
200        )
201        async def revoke_document(
202            collection: str,
203            document_id: str,
204            source_id: str = "default_source",
205            scope: str | None = None,
206        ) -> dict[str, Any]:
207            """Revoke document from active search visibility."""
208            return await bridge.revoke_document(
209                client_id=identity(),
210                collection=collection,
211                document_id=document_id,
212                source_id=source_id,
213                scope=scope,
214            )
215
216    return server
