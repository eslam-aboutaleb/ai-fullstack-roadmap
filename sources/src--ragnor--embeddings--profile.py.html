1"""Embedding profile: identity, configuration, and compatibility rules.
2
3An :class:`EmbeddingProfile` captures every dimension of an embedding
4configuration that matters for compatibility: the provider, service
5endpoint, model, dimension, normalization, distance metric, task mode,
6and non-secret preprocessing options.  Two profiles that produce
7incompatible vector spaces have a different :meth:`EmbeddingProfile.canonical_hash`,
8so storing the hash alongside indexed vectors makes compatibility
9checks unambiguous.
10
11The same model name served through a different provider or service URL
12is *not* automatically equivalent -- ``provider_id`` and
13``service_url`` both feed the canonical hash.
14
15Usage::
16
17    profile = EmbeddingProfile(
18        provider_id="openai",
19        model_id="text-embedding-3-small",
20        dimension=1536,
21    )
22    h = profile.canonical_hash()  # stable, secrets-free SHA-256 hex
23
24    # Validate at startup (warns for unknown dimensions by default):
25    validate_profile_at_startup(profile)
26
27    # Compatible asymmetric pair for bi-encoder retrieval:
28    assert profiles_compatible_for_query(doc_profile, query_profile)
29"""
30

100@dataclass(frozen=True)
101class EmbeddingProfile:
102    """Describes a concrete embedding configuration.
103
104    All attributes that affect vector-space geometry feed
105    :meth:`canonical_hash`.  Secrets (API keys, tokens) must **never**
106    appear here; use ``preprocessing_config`` only for non-secret options
107    and pass it through :func:`~ragnor.ingestion.versioning.safe_config`
108    before storing.
109
110    Attributes:
111        provider_id: Stable identifier for the embedding provider
112            (e.g. ``"openai"``, ``"cohere"``, ``"voyage"``).  Distinct
113            from the model name: the same model hosted by two providers
114            is *not* automatically equivalent.
115        model_id: Embedding model identifier forwarded to the provider
116            (e.g. ``"text-embedding-3-small"``).
117        dimension: Output vector dimension.
118        normalization: Normalization the provider applies before returning
119            the vector.  Affects dot-product equivalence.
120        distance_metric: Metric the downstream vector index uses.
121        task_mode: Default task mode for this profile.  Per-call overrides
122            are supported by :class:`~ragnor.embeddings.executor.EmbeddingExecutor`.
123        model_revision: Optional provider-reported model revision / snapshot
124            tag.  When present, a revision change is treated as a model
125            change (different canonical hash).
126        service_url: Optional provider endpoint URL (without secrets).
127            ``None`` means the provider's default endpoint.  Two profiles
128            that differ only in ``service_url`` are not equivalent.
129        preprocessing_config: Non-secret preprocessing options forwarded
130            to the provider (e.g. instruction prefixes, truncation strategy).
131            Secrets are stripped automatically by :meth:`canonical_hash`.
132    """
133
134    provider_id: str
135    model_id: str
136    dimension: int
137    normalization: NormalizationMode = NormalizationMode.NONE
138    distance_metric: DistanceMetric = DistanceMetric.COSINE
139    task_mode: TaskMode = TaskMode.SYMMETRIC
140    model_revision: str | None = None
141    service_url: str | None = None
142    preprocessing_config: dict[str, Any] = field(default_factory=dict)
143
144    # ------------------------------------------------------------------
145    # Canonical hash
146    # ------------------------------------------------------------------
147
148    def canonical_hash(self) -> str:
149        """Return the stable, secrets-free SHA-256 hex digest of this profile.
150
151        The hash covers all semantically relevant fields.  ``safe_config``
152        is applied to ``preprocessing_config`` to strip secrets before
153        hashing.  The result is stable across Python interpreter restarts
154        (no hash-randomisation).
155
156        Returns:
157            40-character lower-case hex string.
158        """
159        payload: dict[str, Any] = {
160            "provider_id": self.provider_id,
161            "model_id": self.model_id,
162            "dimension": self.dimension,
163            "normalization": str(self.normalization),
164            "distance_metric": str(self.distance_metric),
165            "task_mode": str(self.task_mode),
166            "model_revision": self.model_revision,
167            "service_url": self.service_url,
168            "preprocessing_config": safe_config(self.preprocessing_config) or {},
169        }
170        canonical = json.dumps(payload, sort_keys=True, separators=(",", ":"))
171        return hashlib.sha256(canonical.encode("utf-8")).hexdigest()
172
173    # ------------------------------------------------------------------
174    # Convenience
175    # ------------------------------------------------------------------
176
177    def is_known_dimension(self) -> bool:
178        """Return ``True`` when the dimension matches the registry."""
179        registered = _DIMENSIONS.get(self.model_id)
180        return registered is None or registered == self.dimension
181
182    def __repr__(self) -> str:  # pragma: no cover
183        return (
184            f"EmbeddingProfile(provider={self.provider_id!r}, model={self.model_id!r}, "
185            f"dim={self.dimension}, task={self.task_mode}, "
186            f"hash={self.canonical_hash()[:8]}...)"
187        )

195def profiles_compatible_for_query(
196    doc_profile: EmbeddingProfile,
197    query_profile: EmbeddingProfile,
198) -> bool:
199    """Return whether ``query_profile`` is compatible with ``doc_profile``.
200
201    Asymmetric task modes (document vs. query) are *intentionally*
202    compatible -- bi-encoder retrieval requires this.  Two profiles are
203    compatible when they share the same provider, service endpoint, model,
204    model revision, dimension, normalization, and distance metric.  Task
205    mode differences are allowed.
206
207    Args:
208        doc_profile: Profile used to embed stored documents.
209        query_profile: Profile used to embed the query.
210
211    Returns:
212        ``True`` when the profiles can be used together safely.
213    """
214    return (
215        doc_profile.provider_id == query_profile.provider_id
216        and doc_profile.service_url == query_profile.service_url
217        and doc_profile.model_id == query_profile.model_id
218        and doc_profile.model_revision == query_profile.model_revision
219        and doc_profile.dimension == query_profile.dimension
220        and doc_profile.normalization == query_profile.normalization
221        and doc_profile.distance_metric == query_profile.distance_metric
222    )
223
224
225def profiles_require_reindex(
226    old_profile: EmbeddingProfile,
227    new_profile: EmbeddingProfile,
228) -> bool:
229    """Return whether switching from ``old_profile`` to ``new_profile`` requires reindexing.
230
231    Any change that alters the vector space geometry (provider, model,
232    dimension, normalization, distance metric, or service URL) requires
233    a new index generation.  Metadata-only changes (e.g. renaming
234    ``provider_id`` to an alias that produces identical vectors) are not
235    detected here and must be handled explicitly by the operator.
236
237    Args:
238        old_profile: Profile active on the existing index.
239        new_profile: Profile being evaluated for adoption.
240
241    Returns:
242        ``True`` when the active index cannot serve the new profile.
243    """
244    return old_profile.canonical_hash() != new_profile.canonical_hash()

252def validate_profile_at_startup(
253    profile: EmbeddingProfile,
254    *,
255    strict: bool = False,
256) -> None:
257    """Validate ``profile`` at application startup.
258
259    Warns about unknown dimensions in lenient mode (the default);
260    raises :class:`ProfileValidationError` in strict mode.  The
261    legacy :data:`~ragnor.embeddings.dimensions.FALLBACK_DIMENSION`
262    is not treated as an error unless the profile dimension diverges
263    from it *and* the model is known.
264
265    Args:
266        profile: Profile to validate.
267        strict: When ``True``, unknown dimensions or mismatched
268            registered dimensions raise instead of warning.
269
270    Raises:
271        ProfileValidationError: If ``strict=True`` and validation fails.
272    """
273    registered = _DIMENSIONS.get(profile.model_id)
274
275    if registered is None:
276        msg = (
277            f"No dimension registered for model '{profile.model_id}'; "
278            f"profile declares dimension={profile.dimension}. "
279            f"Register it with ragnor.embeddings.dimensions.register_dimension() "
280            f"if this is wrong (falling back to FALLBACK_DIMENSION={FALLBACK_DIMENSION} "
281            f"is unsafe in production)."
282        )
283        if strict:
284            raise ProfileValidationError(msg)
285        warnings.warn(msg, stacklevel=2)
286        return
287
288    if registered != profile.dimension:
289        msg = (
290            f"Profile dimension {profile.dimension} does not match the registered "
291            f"dimension {registered} for model '{profile.model_id}'. "
292            f"This will cause vector dimension mismatches at insert time."
293        )
294        if strict:
295            raise ProfileValidationError(msg)
296        warnings.warn(msg, stacklevel=2)
297
298    if profile.dimension <= 0:
299        raise ProfileValidationError(
300            f"Embedding dimension must be positive, got {profile.dimension}."
301        )
