1"""OmniCare RAG evaluation wiring over ragnor.
2
3The evaluation harness (retrieval metrics, answer judges,
4the two-phase runner) is domain-agnostic (ragnor); this
5module is the OmniCare binding: the curated evaluation
6dataset grounded in ``sample_policy.md`` and the
7``run_evaluation()`` entry point that wires the OmniCare
8hybrid retriever, the curated dataset, and the OmniCare
9answer generator into ragnor's runner.
10
11What each mode actually measures -- the names are only
12honest if this stays true:
13
14* ``retrieval-only`` -- retrieval metrics only.
15* ``context`` -- retrieval metrics plus answer metrics
16  computed against the *retrieved context*. This is a
17  diagnosis of the retriever, not of the model: a high
18  ``avg_faithfulness`` here says the answer is in the
19  chunks, nothing more.
20* ``end-to-end`` -- retrieval metrics plus answer metrics
21  computed against the *agent's* answer. ``avg_faithfulness``
22  here is the fraction of the model's own words that the
23  retrieved context supports, which is the number the name
24  promises.
25
26``end-to-end`` therefore always resolves an answer
27generator, in this order: the caller's injected
28``answer_fn``, a process-wide default installed with
29:func:`set_default_answer_fn`, the live agent when live
30evaluation is enabled, and finally an explicitly labelled
31retrieved-context substitute. Which one ran is reported in
32``answer_metrics["answer_source"]`` so a reader can never
33mistake a substitute's numbers for the model's.
34"""

93class ExpectedBehavior(StrEnum):
94    """What a correct assistant answer looks like for a sample.
95
96    Only ``ANSWER`` was representable before, which made the dataset a set of
97    questions the corpus can answer and left the failure mode that actually
98    costs money -- answering an unanswerable question with invented coverage --
99    completely unmeasured.
100    """
101
102    ANSWER = "answer"
103    REFUSE = "refuse"

106def expected_behavior_of(sample: EvalSample) -> ExpectedBehavior:
107    """Return the behaviour a sample expects, defaulting to :attr:`ExpectedBehavior.ANSWER`.
108
109    Args:
110        sample: The evaluation sample to inspect.
111
112    Returns:
113        The sample's declared behaviour, or ``answer`` when it declares none.
114    """
115    declared = sample.metadata.get(BEHAVIOR_METADATA_KEY)
116    if isinstance(declared, ExpectedBehavior):
117        return declared
118    if isinstance(declared, str):
119        try:
120            return ExpectedBehavior(declared.lower())
121        except ValueError:
122            logger.warning(
123                "Sample %r declares unknown expected_behavior %r; treating it as 'answer'.",
124                sample.query,
125                declared,
126            )
127    return ExpectedBehavior.ANSWER

400# ---------------------------------------------------------------------------
401# Refusal behaviour: the invented-coverage detector
402# ---------------------------------------------------------------------------
403
404# A claim about coverage terms. Only assertions count: a genuine refusal often
405# has to *name* a deductible ("no deductible can be quoted for it") to explain
406# why it is declining, and penalising that would make the metric punish the
407# behaviour it exists to reward.
408_COVERAGE_CLAIM = re.compile(
409    # a dollar figure, the single most common way an invented limit appears
410    r"\$\s?\d[\d,]*(?:\.\d+)?"
411    # a percentage, written either as a sign or spelled out
412    r"|\b\d[\d,]*(?:\.\d+)?\s?(?:%|percent(?:age)?)"
413    # a deductible tied to an amount, e.g. "deductible of $500"
414    r"|\bdeductible\b[^.;]{0,24}?\$?\s?\d"
415    # an asserted exclusion, e.g. "is strictly excluded" / "are not covered"
416    r"|\b(?:is|are|was|were)\s+(?:strictly\s+|explicitly\s+|generally\s+)?"
417    r"(?:excluded|not covered|never covered)\b"
418    # an asserted cap
419    r"|\b(?:up to|capped at|limited to|maximum of)\s+\$?\s?\d",
420    re.IGNORECASE,
421)
422
423# A cue that the answer is declining rather than answering. The list is
424# deliberately broad: a false "answer" verdict on a refusal is the expensive
425# error, because it hides the failure this metric was added to expose.
426_REFUSAL_CUE = re.compile(
427    r"\bi\s+(?:do\s+not|don't)\s+have\s+(?:that|this|the)\b"
428    r"|\bi\s+(?:can\s?not|can't)\s+(?:answer|help|provide|determine|confirm|say|tell)\b"
429    r"|\bi\s+(?:am|'m)\s+unable\b"
430    r"|\bnot\s+(?:addressed|mentioned|covered|discussed|included)\b"
431    r"|\b(?:do|does|did)\s+not\s+(?:address|mention|cover|state|say|include|apply)\b"
432    r"|\bno\s+(?:information|mention)\s+(?:about|of|on)\b"
433    r"|\boutside\s+(?:the\s+scope\s+of\s+)?my\b"
434    r"|\bunrelated\s+to\b"
435    r"|\bplease\s+contact\s+(?:our\s+)?support\b"
436    r"|\bconnecting\s+you\s+with\b"
437    r"|\bi'?m\s+sorry\b[^.]{0,60}\b(?:cannot|can't|unable|do\s+not|don't)\b"
438    r"|\bwould\s+be\s+(?:inventing|making\s+(?:it\s+)?up|guessing)\b"
439    r"|\bcan\s+only\s+help\s+with\b"
440    r"|\bso\s+i\s+cannot\b"
441    r"|\bnot\s+something\s+i\s+can\b"
442    r"|\bno\s+such\s+(?:coverage|thing)\b",
443    re.IGNORECASE,
444)

447def states_coverage_claim(answer: str) -> bool:
448    """Return True when ``answer`` asserts a coverage limit, deductible, or exclusion.
449
450    This is the invented-coverage detector. It is intentionally narrow -- only
451    assertions match -- because a correct refusal on an unanswerable question
452    still has to talk about the thing it is declining to state.
453
454    Args:
455        answer: The generated answer text.
456
457    Returns:
458        True when the answer makes a claim the corpus has to support.
459    """
460    return bool(_COVERAGE_CLAIM.search(answer or ""))

463def looks_like_refusal(answer: str) -> bool:
464    """Return True when ``answer`` declines to answer rather than answering.
465
466    Args:
467        answer: The generated answer text.
468
469    Returns:
470        True when the answer carries a refusal cue.
471    """
472    return bool(_REFUSAL_CUE.search(answer or ""))

475def refusal_verdict(answer: str, behavior: ExpectedBehavior) -> bool:
476    """Return True when ``answer`` behaves the way ``behavior`` requires.
477
478    Args:
479        answer: The generated answer text.
480        behavior: The behaviour the sample expected.
481
482    Returns:
483        True when the answer met the expectation. A sample expecting a refusal
484        passes when it states no coverage claim. A sample expecting an answer
485        passes when it either states coverage or says nothing about coverage at
486        all, and fails only when it carries a refusal cue -- a decline with no
487        claim in it is the over-refusal direction, and a decline that still
488        quotes the policy is a contradictory answer worth counting as answered.
489    """
490    text = (answer or "").strip()
491    if not text:
492        return False
493    if behavior == ExpectedBehavior.REFUSE:
494        return not states_coverage_claim(text)
495    is_refusal = looks_like_refusal(text) and not states_coverage_claim(text)
496    return not is_refusal

504@dataclass(frozen=True)
505class EvalFloors:
506    """Minimum acceptable metrics for an evaluation run.
507
508    Floors exist so a regression is a build failure rather than a number
509    somebody notices a week later. ``None`` disables a floor, which is how a
510    caller declares "this subset is not scored here" instead of asserting a
511    floor of zero that no run could ever break.
512    """
513
514    recall_at_k: float = 0.9
515    mrr: float = 0.8
516    refusal_accuracy: float | None = None
517    false_refusal_rate: float | None = None
518
519
520OFFLINE_FLOORS = EvalFloors(recall_at_k=0.9, mrr=0.8, refusal_accuracy=1.0, false_refusal_rate=0.35)
521"""Floors for a deterministic run, where the answer double is ours and exact.
522
523The refusal direction is exact here: the double is a known quantity, so it has
524to be perfect. The false-refusal tripwire is loose, because a deterministic
525double can only reach the questions the corpus answers lexically and declining
526the ones that need the embedding model's generalisation is correct behaviour
527rather than a regression. It is not zero, because an answer generator that
528refused *everything* would otherwise sail through the refusal floor above.
529"""
530
531LIVE_FLOORS = EvalFloors(recall_at_k=0.9, mrr=0.8, refusal_accuracy=0.8, false_refusal_rate=0.2)
532"""Floors for a live run, where a real model is allowed some imperfection."""
533
534#: The same floors with the retrieval checks switched off. A caller that scored
535#: one thing -- an injected answer generator over a fixed corpus, say -- must not
536#: inherit a retrieval verdict about a retrieval it never ran.
537OFFLINE_REFUSAL_FLOORS = replace(OFFLINE_FLOORS, recall_at_k=0.0, mrr=0.0)
538LIVE_REFUSAL_FLOORS = replace(LIVE_FLOORS, recall_at_k=0.0, mrr=0.0)

545LEXICAL_STOPWORDS = frozenset(
546    """a an the and or of to in on at for with from by is are was were be been being do does did
547    have has had i me my we our you your he she it its they them this that these those what which
548    who whom when where why how not no yes if then than so as up out off over under again very
549    am can could should would will shall may might must want need get got go going make made
550    there here about into onto per via""".split()
551)
552"""Function words a query and a section can share without the query being about that section.
553
554The list exists so "reachability" cannot be satisfied by the pronoun or the
555preposition every English sentence carries. A sample counts as lexically
556reachable only when some word that carries its meaning also appears in the text
557of a section it expects.
558"""
559
560
561def lexically_reachable(sample: EvalSample, section_texts: Mapping[str, str]) -> bool:
562    """Return True when some content word of ``sample`` appears in an expected section.
563
564    A retrieval floor can only be asserted over questions the retriever has a
565    chance of answering *by the mechanism under test*. A query that shares no
566    content word with any expected section -- "what types of claims does OmniCare
567    handle?" against a policy that never lists claim types -- is answerable only
568    by the embedding model's generalisation, so measuring it here would score the
569    double rather than the retrieval machinery and would fail the same way no
570    matter what broke.
571
572    The test is mechanical on purpose: it is computed from the corpus, not chosen
573    by hand, so a sample cannot be quietly dropped from the floor it fails.
574
575    Args:
576        sample: The evaluation sample to test.
577        section_texts: The text of every section in the corpus, keyed by section.
578
579    Returns:
580        True when at least one content word of the query appears in the text of
581        at least one expected section.
582    """
583    content_words = {
584        token
585        for token in re.findall(r"[a-z0-9]+", sample.query.lower())
586        if token not in LEXICAL_STOPWORDS and len(token) > 2 and any(ch.isalpha() for ch in token)
587    }
588    if not content_words:
589        return False
590
591    for section in sample.expected_sections:
592        text = section_texts.get(section, "")
593        if not text:
594            continue
595        if content_words & set(re.findall(r"[a-z0-9]+", text.lower())):
596            return True
597    return False

600async def load_policy_section_texts() -> dict[str, str]:
601    """Return the text of every indexed policy section, keyed by section.
602
603    Args:
604        None.
605
606    Returns:
607        A mapping of section name to chunk text. Empty when the policy has not
608        been ingested, in which case the caller should not filter on
609        reachability rather than treat every sample as unreachable.
610    """
611    from sqlalchemy import text as sql_text  # noqa: PLC0415
612
613    from app.core.database import async_session_factory  # noqa: PLC0415
614
615    async with async_session_factory() as session:
616        rows = (await session.execute(sql_text("SELECT section, text FROM policy_chunks"))).all()
617
618    sections: dict[str, str] = {}
619    for section, body in rows:
620        sections[section] = f"{sections.get(section, '')}\n{body or ''}"
621    return sections

627
628    Attributes:
629        metric: Name of the metric, as it appears in the result.
630        floor: The value the metric had to reach.
631        actual: The value it reached.
632        detail: Which samples contributed the shortfall, when they can be named.
633    """
634
635    metric: str
636    floor: float
637    actual: float
638    detail: str = ""
639
640    def __str__(self) -> str:
641        """Render the breach as a single diff-friendly line."""
642        base = f"{self.metric}: {self.actual:.4f} < floor {self.floor:.4f}"
643        return f"{base}\n{self.detail}" if self.detail else base

680async def agent_answer_fn(query: str, context: str) -> str:
681    """Answer one evaluation query with the real OmniCare agent.
682
683    The agent is the product, so scoring anything else would score a stand-in.
684
685    Args:
686        query: The evaluation question, passed to the agent as a user message.
687        context: The context ragnor retrieved. Ignored: the agent runs the real
688            ``query_policy`` tool and therefore retrieves its own. Conditioning
689            it on a context it never saw would make the faithfulness score
690            measure an answer that does not exist.
691
692    Returns:
693        The agent's final text response.
694    """
695    # Imported here rather than at module scope so that importing the evaluation
696    # module does not pull the ADK runtime into every process that reads it.
697    from app.agent.agent import reset_user_session, run_agent  # noqa: PLC0415
698
699    # A cold session per sample: a session reused across samples would carry the
700    # previous question's answer into this one, and the second sample would be
701    # scored on a conversation nobody designed.
702    await reset_user_session(str(EVAL_USER_ID))
703    result = await run_agent(str(EVAL_USER_ID), query)
704    return str(result.get("response") or "")

707async def context_proxy_answer_fn(query: str, context: str) -> str:
708    """Stand in for a model by returning the retrieved context.
709
710    Only reached when no model can be run. The result it produces is
711    indistinguishable from ``context`` mode, which is exactly why the caller is
712    required to label the run with :data:`ANSWER_SOURCE_CONTEXT_PROXY`: an
713    unlabelled number from here is the defect this module exists to remove.
714
715    Args:
716        query: Ignored; nothing is generated, only relayed.
717        context: The context ragnor retrieved for this sample.
718
719    Returns:
720        The retrieved context, or the assistant's own no-information sentence
721        when nothing was retrieved.
722    """

724        return context
725    return "I don't have that information in my policy documents. Please contact our support team."

728def resolve_answer_function(
729    injected: AnswerFunction | None = None,
730) -> tuple[AnswerFunction | None, str]:
731    """Decide which answer generator an ``end-to-end`` run will use.
732
733    Args:
734        injected: A caller-supplied generator, typically a deterministic double.
735
736    Returns:
737        A ``(answer_fn, source_label)`` pair. ``answer_fn`` is None only when no
738        generator can be resolved at all, which cannot happen through the
739        documented path.
740    """
741    if injected is not None:
742        return injected, ANSWER_SOURCE_INJECTED
743    if _default_answer_fn is not None:
744        return _default_answer_fn, ANSWER_SOURCE_DEFAULT
745    if live_evaluation_enabled():
746        return agent_answer_fn, ANSWER_SOURCE_AGENT
747    return context_proxy_answer_fn, ANSWER_SOURCE_CONTEXT_PROXY

755async def run_evaluation(
756    config: EvalConfig | None = None,
757    answer_fn: AnswerFunction | None = None,
758    dataset: list[EvalSample] | None = None,
759) -> EvalResult:
760    """Run a complete RAG evaluation.
761
762    Injects the OmniCare hybrid retriever, the curated evaluation dataset, and
763    the resolved answer generator into ragnor's runner. ``context`` mode scores
764    the retrieved context (a retrieval diagnosis); ``end-to-end`` mode scores
765    what the answer generator produced (the model, when one is available).
766
767    Args:
768        config: Evaluation configuration. If None, uses ragnor's defaults.
769        answer_fn: Answer generator ``async (query, context) -> answer`` used
770            for ``end-to-end`` mode. When omitted, the generator is resolved by
771            :func:`resolve_answer_function` and recorded in the result.
772        dataset: Samples to score. Defaults to the curated dataset filtered by
773            the config's category and difficulty. Passing it explicitly lets a
774            caller score a subset -- the unanswerable samples alone, say --
775            without the run's config and its dataset disagreeing.
776
777    Returns:
778        EvalResult with all metrics and per-sample details.
779    """
780    if config is None:
781        config = EvalConfig()
782
783    if dataset is None:
784        dataset = get_eval_dataset(category=config.category, difficulty=config.difficulty)
785
786    async def _retrieval_fn(query: str) -> list[dict[str, Any]]:
787        return await retrieve_hybrid(query=query, n_results=config.n_results)
788
789    answer_source = ANSWER_SOURCE_RETRIEVED_CONTEXT if config.mode == "context" else ""
790    resolved_fn: AnswerFunction | None = None
791    if config.mode == "end-to-end":
792        resolved_fn, answer_source = resolve_answer_function(answer_fn)
793
794    result = await _ragnor_run_evaluation(
795        config,
796        _retrieval_fn,
797        dataset,
798        settings=get_settings(),
799        answer_fn=resolved_fn,
800    )
801
802    _annotate_result(result, dataset, answer_source)
803    await _persist_result(result, config)
804    return result

807def _annotate_result(
808    result: EvalResult,
809    dataset: list[EvalSample],
810    answer_source: str,
811) -> None:
812    """Add the refusal-behaviour metrics and provenance to a finished result.
813
814    ragnor reports faithfulness, relevance, completeness, and conciseness. None of
815    those notice an answer that invents coverage for a question the corpus
816    cannot answer, so a run can look excellent while the model does the one
817    thing that costs the company money. These metrics close that hole.
818
819    Args:
820        result: The ragnor result to annotate, modified in place.
821        dataset: The dataset the run scored, in the runner's sample order.
822        answer_source: Which answer generator produced the answers.
823    """
824    if not result.answer_metrics:
825        return
826
827    refusal_verdicts: list[bool] = []
828    answer_verdicts: list[bool] = []
829    regressions: list[str] = []
830
831    for index, sample_result in enumerate(result.per_sample_results):
832        behavior = (
833            expected_behavior_of(dataset[index])
834            if index < len(dataset)
835            else _behavior_from_sample(sample_result)
836        )
837        sample_result[BEHAVIOR_METADATA_KEY] = behavior.value
838        verdict = refusal_verdict(sample_result.get("generated_answer", ""), behavior)
839        if behavior == ExpectedBehavior.REFUSE:
840            refusal_verdicts.append(verdict)
841        else:
842            answer_verdicts.append(verdict)
843        if not verdict:
844            regressions.append(f"{behavior.value}: {sample_result.get('query', '')}")
845
846    result.answer_metrics["answer_source"] = answer_source
847    result.answer_metrics["refusal_expected"] = len(refusal_verdicts)
848    result.answer_metrics["answer_expected"] = len(answer_verdicts)
849    result.answer_metrics["refusal_accuracy"] = _ratio(refusal_verdicts)
850    result.answer_metrics["false_refusal_rate"] = _ratio([not v for v in answer_verdicts])
851
852    result.summary = "\n".join(
853        [
854            _answer_source_note(answer_source),
855            result.summary,
856            "",
857            "Refusal Behaviour:",
858            f"  Unanswerable samples: {len(refusal_verdicts)}",
859            f"  Refusal accuracy: {result.answer_metrics['refusal_accuracy']:.4f}",
860            f"  False refusal rate: {result.answer_metrics['false_refusal_rate']:.4f}",
861        ]
862    )
863
864    if regressions:
865        result.summary += "\n  Samples failing their expected behaviour:\n" + "\n".join(
866            f"    - {entry}" for entry in regressions
867        )
868

870def _behavior_from_sample(sample_result: dict[str, Any]) -> ExpectedBehavior:
871    """Recover a sample's expectation from its annotated result entry.
872
873    Only reached when the result carries more entries than the dataset, which
874    the runner does not currently do; the fallback keeps the annotation total
875    rather than raising on a future runner change.
876
877    Args:
878        sample_result: One per-sample entry from an annotated result.
879
880    Returns:
881        The recorded behaviour, or ``answer`` when the entry carries none.
882    """
883    declared = sample_result.get(BEHAVIOR_METADATA_KEY)
884    if isinstance(declared, ExpectedBehavior):
885        return declared
886    if isinstance(declared, str):
887        try:
888            return ExpectedBehavior(declared.lower())
889        except ValueError:
890            pass
891    return ExpectedBehavior.ANSWER

894def _ratio(flags: list[bool]) -> float:
895    """Return the fraction of ``flags`` that are True, or 0.0 for an empty list.
896
897    An empty denominator returns zero rather than one on purpose: "no unanswerable
898    samples were scored" must not read as "every unanswerable sample behaved
899    correctly", because that is the direction that hides a missing subset.
900    """
901    return round(sum(1 for flag in flags if flag) / len(flags), 4) if flags else 0.0

904def _answer_source_note(answer_source: str) -> str:
905    """Return the provenance line that leads an annotated summary."""
906    if answer_source == ANSWER_SOURCE_CONTEXT_PROXY:
907        return (
908            "NOTE: no answer generator was available, so the retrieved context was "
909            f"scored in place of the model's answer (answer_source={answer_source}). "
910            "These answer metrics describe the retriever, not the model. Set "
911            f"{LIVE_EVAL_ENV}=1 or inject an answer_fn to score the model."
912        )
913    return f"Answer source: {answer_source}"

916def check_floors(
917    result: EvalResult,
918    floors: EvalFloors,
919    dataset: list[EvalSample] | None = None,
920    behavior: ExpectedBehavior | None = None,
921    sample_predicate: Callable[[EvalSample], bool] | None = None,
922) -> list[FloorBreach]:
923    """Compare a finished result against its floors.
924
925    Args:
926        result: The evaluation result to check.
927        floors: The minimum acceptable metrics.
928        dataset: The dataset the run scored. Supplies the expected behaviour per
929            query, which a retrieval-only run has not annotated into its result,
930            and the sample objects a ``sample_predicate`` is applied to.
931        behavior: Restrict the retrieval floors to samples expecting this
932            behaviour. Retrieval quality is only meaningful on questions the
933            corpus can answer, so the gate checks the answerable subset and lets
934            the unanswerable subset be scored by the refusal floors.
935        sample_predicate: Optional filter on the dataset, applied before the
936            behaviour filter. Used to narrow the retrieval floors to the samples
937            the retrieval mechanism can reach.
938
939    Returns:
940        One :class:`FloorBreach` per metric that fell short, empty when the run
941        passes.
942    """
943    breaches: list[FloorBreach] = []
944
945    dataset = list(dataset) if dataset else []
946    if sample_predicate is not None:
947        dataset = [sample for sample in dataset if sample_predicate(sample)]
948
949    subset = _behaviour_subset(result, dataset, behavior)
950    if not subset:
951        wanted = behavior.value if behavior is not None else "any"
952        return [FloorBreach("dataset", 1.0, 0.0, f"no samples expect {wanted!r}")]
953
954    recalls = [float(sample.get("recall", 0.0)) for sample in subset]
955    mrrs = [float(sample.get("mrr", 0.0)) for sample in subset]
956    recall = sum(recalls) / len(recalls)
957    mrr = sum(mrrs) / len(mrrs)
958
959    if recall < floors.recall_at_k:
960        breaches.append(
961            FloorBreach(
962                "recall_at_k",
963                floors.recall_at_k,
964                round(recall, 4),
965                _below_floor(subset, floors.recall_at_k, "recall"),
966            )
967        )
968    if mrr < floors.mrr:
969        breaches.append(
970            FloorBreach(
971                "mrr",
972                floors.mrr,
973                round(mrr, 4),
974                _below_floor(subset, floors.mrr, "mrr"),
975            )
976        )
977
978    # Refusal behaviour can only be judged on a run that generated answers. A
979    # retrieval-only run has none, so its refusal accuracy is not zero -- it is
980    # absent, and reporting it as a failure would send a reader hunting for a
981    # regression that does not exist.
982    has_answers = any("generated_answer" in sample for sample in result.per_sample_results)
983    answer_metrics = result.answer_metrics if has_answers else {}
984    if floors.refusal_accuracy is not None and has_answers:
985        actual = float(answer_metrics.get("refusal_accuracy", 0.0))
986        if actual < floors.refusal_accuracy:
987            breaches.append(
988                FloorBreach(
989                    "refusal_accuracy",
990                    floors.refusal_accuracy,
991                    actual,
992                    "samples that stated a limit, deductible, or exclusion on an "
993                    "unanswerable question:\n  "
994                    + "\n  ".join(_misbehaving(result, dataset, ExpectedBehavior.REFUSE)),
995                )
996            )
997    if floors.false_refusal_rate is not None and has_answers:
998        actual = float(answer_metrics.get("false_refusal_rate", 0.0))
999        if actual > floors.false_refusal_rate:
1000            breaches.append(
1001                FloorBreach(
1002                    "false_refusal_rate",
1003                    floors.false_refusal_rate,
1004                    actual,
1005                    "answerable samples the answer declined to answer:\n  "
1006                    + "\n  ".join(_misbehaving(result, dataset, ExpectedBehavior.ANSWER)),
1007                )
1008            )
1009
1010    return breaches

1013def _behaviour_subset(
1014    result: EvalResult,
1015    dataset: list[EvalSample],
1016    behavior: ExpectedBehavior | None,
1017) -> list[dict[str, Any]]:
1018    """Return the per-sample entries the caller selected and wants judged.
1019
1020    The dataset is the selection. A per-sample entry whose query is not in it
1021    was deliberately left out -- by a behaviour filter, or by a reachability
1022    predicate -- so it must not fall back to a default expectation and slip into
1023    the numbers. Only when no dataset is supplied at all does the annotated
1024    result become the source, which is what a caller checking a result it did
1025    not build needs.
1026
1027    Args:
1028        result: The evaluation result to filter.
1029        dataset: The samples the run scored, or an empty list when the caller
1030            supplied none.
1031        behavior: The behaviour to keep, or None to keep every selected sample.
1032
1033    Returns:
1034        The per-sample entries that were selected and match ``behavior``.
1035    """
1036    if not dataset:
1037        if behavior is None:
1038            return list(result.per_sample_results)
1039        selected = result.per_sample_results
1040        return [sample for sample in selected if _behavior_from_sample(sample) == behavior]
1041
1042    expected_by_query = {sample.query: expected_behavior_of(sample) for sample in dataset}
1043    subset = []
1044    for sample in result.per_sample_results:
1045        query = str(sample.get("query"))
1046        if query not in expected_by_query:
1047            continue
1048        if behavior is None or expected_by_query[query] == behavior:
1049            subset.append(sample)
1050    return subset
1051
1052
1053def _below_floor(subset: list[dict[str, Any]], floor: float, key: str) -> str:
1054    """Render the samples that pulled a metric below its floor."""
1055    regressing = sorted(
1056        f"{sample.get('query')!r} {key}={sample.get(key)}"
1057        for sample in subset
1058        if float(sample.get(key, 0.0)) < floor
1059    )
1060    if not regressing:
1061        return ""
1062    return "regressions:\n  " + "\n  ".join(regressing)
1063
1064
1065def _misbehaving(
1066    result: EvalResult,
1067    dataset: list[EvalSample],
1068    behavior: ExpectedBehavior,
1069) -> list[str]:
1070    """Name the samples that did not meet ``behavior``, for a failure message.
1071
1072    A floor breach that does not name the regressing samples is a number nobody
1073    can act on; each entry carries the query, what went wrong, and an excerpt of
1074    the answer so the distance between the failure message and its cause is zero.
1075    """
1076    expected_by_query = {sample.query: expected_behavior_of(sample) for sample in dataset}
1077    named = []
1078    for sample in result.per_sample_results:
1079        query = str(sample.get("query"))
1080        if expected_by_query.get(query, _behavior_from_sample(sample)) != behavior:
1081            continue
1082        answer = str(sample.get("generated_answer", ""))
1083        # A retrieval-only run has no answer to judge; it is not a refusal.
1084        if "generated_answer" not in sample:
1085            continue
1086        if refusal_verdict(answer, behavior):
1087            continue
1088        reason = (
1089            "stated a coverage claim"
1090            if states_coverage_claim(answer)
1091            else ("declined to answer" if looks_like_refusal(answer) else "no usable answer")
1092        )
1093        excerpt = " ".join(answer.split())[:160]
1094        named.append(f"{query!r}: {reason}; answer={excerpt!r}")
1095    return sorted(named)

1098async def _persist_result(result: EvalResult, config: EvalConfig) -> None:
1099    """Write one row per completed run, never failing the run over it.
1100
1101    A run whose metrics cannot be stored still produced the metrics; aborting
1102    here would trade the measurement for its archive.
1103
1104    Args:
1105        result: The finished result to store.
1106        config: The configuration the run used, stored alongside it.
1107    """
1108    from app.domain.evaluation_runs import persistence_enabled  # noqa: PLC0415
1109
1110    if not persistence_enabled():
1111        return
1112
1113    metrics = {
1114        "samples": len(result.per_sample_results),
1115        "retrieval": result.retrieval_metrics,
1116        "answer": result.answer_metrics,
1117        "per_sample": [
1118            {
1119                "query": sample.get("query"),
1120                "expected_behavior": str(sample.get(BEHAVIOR_METADATA_KEY, "answer")),
1121                "recall": sample.get("recall"),
1122                "mrr": sample.get("mrr"),
1123                "answer_faithfulness": sample.get("answer_faithfulness"),
1124                "answer_overall": sample.get("answer_overall"),
1125            }
1126            for sample in result.per_sample_results
1127        ],
1128    }
1129
1130    try:
1131        await record_eval_run(
1132            mode=str(config.mode),
1133            config=dict(result.config or {}),
1134            metrics=metrics,
1135        )
1136    except Exception:
1137        logger.warning("Could not persist the evaluation run.", exc_info=True)
