75    def __init__(  # noqa: PLR0913
76        self,
77        retriever: Any | None = None,
78        *,
79        collections: Mapping[str, CollectionConfig | Any] | None = None,
80        assembler: EvidenceAssembler | None = None,
81        source_lookup: SourceLookup | None = None,
82        health_checker: HealthChecker | None = None,
83        authorizer: ScopeAuthorizer | None = None,
84        allow_management: bool = False,
85        default_collection: str = "default",
86        default_budget_tokens: int = 4000,
87    ) -> None:
88        """Initialize KnowledgeBridge.
89
90        Args:
91            retriever: Default retriever for single-collection setups.
92            collections: Optional mapping of collection names to retrievers or configs.
93            assembler: Injected EvidenceAssembler (defaults to ApproximateTokenizer).
94            source_lookup: Injected SourceLookup (defaults to MemorySnapshotStore).
95            health_checker: Injected HealthChecker for status probes.
96            authorizer: Injected ScopeAuthorizer for tenant validation.
97            allow_management: Opt-in flag to enable management operations (sync, delete).
98            default_collection: Name of the fallback default collection.
99            default_budget_tokens: Default token budget for evidence assembly.
100        """
101        self.default_collection = default_collection
102        self.default_budget_tokens = default_budget_tokens
103        self.allow_management = allow_management
104        self.authorizer = authorizer or ScopeAuthorizer(default_allow=True)
105        self.assembler = assembler or EvidenceAssembler(tokenizer=ApproximateTokenizer())
106        self.source_lookup = source_lookup or SourceLookup(MemorySnapshotStore())
107        self.health_checker = health_checker
108
109        # Registry of collections
110        self._collections: dict[str, CollectionConfig] = {}
111        if collections:
112            for name, item in collections.items():
113                if isinstance(item, CollectionConfig):
114                    if item.name != name:
115                        raise ValueError("Collection registry key must match its configured name")
116                    self._collections[name] = item
117                else:
118                    self._collections[name] = CollectionConfig(
119                        name=name,
120                        retriever=item,
121                        source_lookup=self.source_lookup,
122                    )
123
124        if retriever is not None and default_collection not in self._collections:
125            self._collections[default_collection] = CollectionConfig(
126                name=default_collection,
127                retriever=retriever,
128                source_lookup=self.source_lookup,
129            )

149    def _get_collection(self, collection_name: str) -> CollectionConfig:
150        """Look up collection configuration or raise KeyError."""
151        if collection_name in self._collections:
152            return self._collections[collection_name]
153        raise KeyError(
154            f"Collection {collection_name!r} not found. "
155            f"Available collections: {list(self._collections.keys())}"
156        )

158    def _authorize(
159        self, collection: str, scope: str | None, client_id: str | None
160    ) -> CollectionConfig:
161        cfg = self._get_collection(collection)
162        self.authorizer.enforce(client_id, cfg.name, scope)
163        if "*" not in cfg.allowed_scopes and scope not in cfg.allowed_scopes:
164            raise ScopeAuthorizationError("Collection does not permit the requested scope")
165        return cfg

177    async def search_knowledge(  # noqa: PLR0913, PLR0917, PLR0915
178        self,
179        query: str,
180        collection: str | None = None,
181        filters: dict[str, Any] | None = None,
182        candidate_k: int = 20,
183        final_k: int = 5,
184        budget_limit: int | ContextBudget | None = None,
185        scope: str | None = None,
186        client_id: str | None = None,
187        distance_threshold: float | None = None,
188    ) -> EvidenceBundle:
189        """Execute authorized retrieval and evidence assembly.
190
191        Args:
192            query: User search query.
193            collection: Target knowledge collection selector.
194            filters: Narrowing metadata filters.
195            candidate_k: Number of raw candidates to retrieve before assembly.
196            final_k: Maximum number of evidence items in final bundle.
197            budget_limit: Total token budget or preconfigured ContextBudget.
198            scope: Isolation scope (e.g. tenant_id) to enforce.
199            client_id: Client identifier for authentication/scope checks.
200            distance_threshold: Maximum vector distance cutoff.
201
202        Returns:
203            Structured :class:`EvidenceBundle` with deduplicated, bounded evidence.
204        """
205        # Security validation on inputs
206        validate_safe_string(query, "query")
207        collection = self.default_collection if collection is None else collection
208        validate_safe_identifier(collection, "collection")
209        if scope is not None:
210            validate_safe_identifier(scope, "scope")
211        validate_safe_filters(filters)
212
213        # Scope authorization
214        cfg = self._authorize(collection, scope, client_id)
215        retriever = cfg.retriever
216
217        # Retrieve candidates from retriever
218        merged_filters = dict(filters or {})
219        scope_field = self._scope_field(cfg)
220        if scope is not None:
221            merged_filters[scope_field] = scope
222        for count in (candidate_k, final_k):
223            if isinstance(count, bool) or not isinstance(count, int) or count < 1:
224                raise ValueError("candidate_k and final_k must be positive integers")
225
226        raw_results: Any = []
227        try:
228            if isinstance(
229                retriever, (HybridRetriever, DenseRetrieval, HybridRetrieval, KeywordRetrieval)
230            ):
231                threshold = distance_threshold
232                if threshold is None and not isinstance(retriever, HybridRetriever):
233                    threshold = 1.5
234                request = RetrievalRequest(
235                    query,
236                    final_k=candidate_k,
237                    candidate_k=candidate_k,
238                    threshold=threshold,
239                    filters=merged_filters,
240                    isolation_scope=scope,
241                    scope_field=scope_field,
242                    strict=True,
243                    metadata_fields=[
244                        "source_id",
245                        "document_id",
246                        "doc_id",
247                        "revision",
248                        "page",
249                        "section",
250                        "line",
251                        "title",
252                        "uri",
253                        "start_offset",
254                        "end_offset",
255                        "freshness",
256                        scope_field,
257                    ],
258                )
259                raw_results = (await retriever.retrieve_typed(request)).to_dicts()
260            elif hasattr(retriever, "retrieve"):
261                parameters = inspect.signature(retriever.retrieve).parameters
262                options = dict(merged_filters)
263                if "distance_threshold" in parameters:
264                    options["distance_threshold"] = distance_threshold
265                elif "threshold" in parameters and distance_threshold is not None:
266                    options["threshold"] = distance_threshold
267                raw_results = await retriever.retrieve(
268                    query=query, n_results=candidate_k, **options
269                )
270            elif hasattr(retriever, "hybrid_search"):
271                # VectorStore direct call
272                thresh = distance_threshold if distance_threshold is not None else 1.5
273                std_meta = [
274                    "source_id",
275                    "document_id",
276                    "revision",
277                    "page",
278                    "section",
279                    "line",
280                    "title",
281                    "uri",
282                    "start_offset",
283                    "end_offset",
284                    "freshness",
285                    scope_field,
286                    "doc_id",
287                    "provenance",
288                ]
289                if filters:
290                    for k in filters:
291                        if k not in std_meta:
292                            std_meta.append(k)
293                raw_results = await retriever.hybrid_search(
294                    query=query,
295                    embedding=[],
296                    n_results=candidate_k,
297                    threshold=thresh,
298                    metadata_fields=std_meta,
299                    strict=True,
300                    **merged_filters,
301                )
302
303            elif callable(retriever):
304                raw_results = await retriever(query=query, n_results=candidate_k, **merged_filters)
305        except Exception as exc:  # noqa: BLE001
306            logger.warning("Retrieval execution failed for collection %r: %s", collection, exc)
307            budget = (
308                budget_limit
309                if isinstance(budget_limit, ContextBudget)
310                else ContextBudget(
311                    total_budget=budget_limit or self.default_budget_tokens,
312                    max_evidence_items=final_k,
313                )
314            )
315            return EvidenceBundle(
316                items=[],
317                status=EvidenceStatus.BACKEND_FAILURE,
318                total_tokens=0,
319                budget=budget,
320                status_detail=f"Backend retrieval failure: {exc}",
321            )
322
323        # Standardize candidate results into SearchHits
324        hits: list[SearchHit] = []
325        if isinstance(raw_results, Sequence):
326            for res in raw_results:
327                if isinstance(res, SearchHit):
328                    hits.append(res)
329                elif hasattr(res, "to_search_hit"):
330                    hits.append(res.to_search_hit())
331                elif hasattr(res, "document") and hasattr(res, "id"):
332                    hits.append(SearchHit.from_retrieved_document(res))
333                elif isinstance(res, dict):
334                    hits.append(SearchHit.from_dict(res))
335
336        # Build budget
337        if isinstance(budget_limit, ContextBudget):
338            budget = budget_limit
339        else:
340            budget = ContextBudget(
341                total_budget=budget_limit or self.default_budget_tokens,
342                max_evidence_items=final_k,
343            )
344
345        # Assemble evidence
346        return self.assembler.assemble(hits, budget=budget)

348    async def get_source(  # noqa: PLR0913, PLR0917
349        self,
350        source_ref: str,
351        document_id: str | None = None,
352        revision: str = "v1",
353        location: dict[str, Any] | tuple[int, int] | None = None,
354        collection: str | None = None,
355        scope: str | None = None,
356        client_id: str | None = None,
357        start_offset: int | None = None,
358        end_offset: int | None = None,
359        max_span_length: int | None = None,
360    ) -> BoundedSpanResult:
361        """Fetch an authorized, bounded text span and revision metadata.
362
363        Args:
364            source_ref: Evidence source reference or source repository identifier.
365            document_id: Document identifier (parsed from source_ref if omitted).
366            revision: Document revision snapshot identifier.
367            location: Optional location map or offset tuple.
368            collection: Knowledge collection selector.
369            scope: Isolation scope.
370            client_id: Client identifier for authorization.
371            start_offset: Character start offset.
372            end_offset: Character end offset.
373            max_span_length: Maximum allowed span length.
374
375        Returns:
376            :class:`BoundedSpanResult` containing verified slice and provenance.
377        """
378        validate_safe_string(source_ref, "source_ref")
379        if document_id is not None:
380            validate_safe_identifier(document_id, "document_id")
381        validate_safe_identifier(revision, "revision")
382        collection = self.default_collection if collection is None else collection
383        validate_safe_identifier(collection, "collection")
384        if scope is not None:
385            validate_safe_identifier(scope, "scope")
386
387        cfg = self._authorize(collection, scope, client_id)
388        lookup = cfg.source_lookup or self.source_lookup
389
390        resolved_source_id = source_ref
391        resolved_doc_id = document_id
392        resolved_revision = revision
393        parsed_start = parsed_end = None
394        if document_id is None:
395            (resolved_source_id, resolved_doc_id, parsed_revision, parsed_start, parsed_end) = (
396                parse_source_reference(source_ref)
397            )
398            resolved_revision = parsed_revision or revision
399
400        # Parse location parameters
401        start = parsed_start if start_offset is None else start_offset
402        end = parsed_end if end_offset is None else end_offset
403
404        if isinstance(location, tuple) and len(location) == 2:
405            start = location[0]
406            end = location[1]
407        elif isinstance(location, dict):
408            if "start_offset" in location:
409                start = location.get("start_offset")
410            if "end_offset" in location:
411                end = location.get("end_offset")
412
413        auth_scope = scope if scope is not None else "default"
414        return lookup.lookup_span(
415            source_id=resolved_source_id,
416            document_id=resolved_doc_id,
417            revision=resolved_revision,
418            authorization_scope=auth_scope,
419            start_offset=start,
420            end_offset=end,
421            max_span_length=max_span_length,
422            strict=False,
423        )

425    async def knowledge_status(
426        self,
427        collection: str | None = None,
428        scope: str | None = None,
429        client_id: str | None = None,
430        include_expensive: bool = False,
431    ) -> dict[str, Any]:
432        """Inspect health, count, and freshness summary for a collection.
433
434        Output is sanitized to exclude sensitive credentials, local paths, or internal URIs.
435
436        Args:
437            collection: Authorized collection selector.
438            scope: Isolation scope to inspect.
439            client_id: Client identifier for authorization check.
440            include_expensive: Whether to run deep consistency verification.
441
442        Returns:
443            Sanitized dictionary containing state, document counts, and component health.
444        """
445        collection = self.default_collection if collection is None else collection
446        validate_safe_identifier(collection, "collection")
447        if scope is not None:
448            validate_safe_identifier(scope, "scope")
449
450        cfg = self._authorize(collection, scope, client_id)
451        retriever = cfg.retriever
452
453        # Run health check if checker is available
454        if self.health_checker is not None:
455            report: HealthReport = await self.health_checker.check_health(
456                scope=scope,
457                include_expensive=include_expensive,
458            )
459            raw_dict = report.to_dict()
460            # Sanitize components to avoid leaking internal paths or connection strings
461            sanitized_components = {}
462            for name, comp in raw_dict.get("components", {}).items():
463                details = comp.get("details", {})
464                sanitized_details = {
465                    k: v
466                    for k, v in details.items()
467                    if k not in ("connection_string", "password", "url", "dsn", "secret")
468                }
469                sanitized_components[name] = {
470                    "state": comp.get("state"),
471                    "reason": comp.get("reason"),
472                    "latency_ms": comp.get("latency_ms"),
473                    "details": sanitized_details,
474                }
475
476            return {
477                "collection": collection,
478                "scope": scope,
479                "state": raw_dict.get("state", HealthState.READY.value),
480                "timestamp": raw_dict.get("timestamp"),
481                "reasons": raw_dict.get("reasons", []),
482                "components": sanitized_components,
483                "is_healthy": report.is_healthy,
484            }
485
486        # Fallback baseline probe on the retriever
487        doc_count = 0
488        state = HealthState.READY.value
489        reason = ""
490        try:
491            if hasattr(retriever, "count"):
492                doc_count = await retriever.count()
493        except Exception as exc:  # noqa: BLE001
494            state = HealthState.DEGRADED.value
495            reason = f"Retrieval probe warning: {exc}"
496
497        return {
498            "collection": collection,
499            "scope": scope,
500            "state": state,
501            "reason": reason,
502            "document_count": doc_count,
503            "is_healthy": state == HealthState.READY.value,
504        }

510    async def sync_collection(  # noqa: PLR0913, PLR0915
511        self,
512        collection: str,
513        source_id: str,
514        documents: Sequence[Any],
515        scope: str | None = None,
516        client_id: str | None = None,
517    ) -> dict[str, Any]:
518        """Ingest or sync documents into a collection.
519
520        Guarded by :attr:`allow_management`.
521
522        Raises:
523            PermissionError: If management tools are not permitted by host.
524        """
525        if not self.allow_management:
526            raise PermissionError(
527                "Management operations (sync/ingestion) are disabled by host security rules. "
528                "Enable them explicitly with allow_management=True."
529            )
530
531        collection = self.default_collection if collection is None else collection
532        validate_safe_identifier(collection, "collection")
533        validate_safe_string(source_id, "source_id")
534        if scope is not None:
535            validate_safe_identifier(scope, "scope")
536
537        cfg = self._authorize(collection, scope, client_id)
538        retriever = cfg.retriever
539        lookup = cfg.source_lookup or self.source_lookup
540
541        # If retriever is or has a vector store, upsert
542        store = getattr(retriever, "store", retriever)
543        ingested_count = 0
544
545        # Also register snapshots in source lookup for immutable citation tracking
546        snapshot_store = getattr(lookup, "_store", None)
547
548        normalized: list[dict[str, Any]] = []
549        snapshots: list[SnapshotRecord] = []
550        scope_field = self._scope_field(cfg)
551        for doc in documents:
552            row = dict(doc) if isinstance(doc, Mapping) else vars(doc).copy()
553            doc_id = str(row.get("id") or row.get("external_id") or row.get("source_name") or "")
554            text = row.get("text", row.get("document", row.get("content", "")))
555            metadata = dict(row.get("metadata") or {})
556            revision = str(row.get("revision") or metadata.get("revision") or "v1")
557            parent_id = str(metadata.get("document_id") or doc_id)
558            validate_safe_identifier(doc_id, "document id")
559            validate_safe_string(text, "document text")
560            validate_safe_identifier(revision, "document revision")
561            for key in {scope_field, "tenant_id", "scope", "isolation_scope"}:
562                if key in metadata and metadata[key] != scope:
563                    raise ScopeAuthorizationError(
564                        "Document ownership conflicts with authorized scope"
565                    )
566            if "source_id" in metadata and metadata["source_id"] != source_id:
567                raise ScopeAuthorizationError("Document source conflicts with authorized source")
568            metadata.update(source_id=source_id, document_id=parent_id, revision=revision)
569            if scope is not None:
570                metadata[scope_field] = scope
571            key = json.dumps([cfg.name, scope, source_id, doc_id], separators=(",", ":"))
572            normalized.append(
573                {
574                    **row,
575                    "id": hashlib.sha256(key.encode()).hexdigest(),
576                    "text": text,
577                    "metadata": metadata,
578                }
579            )
580            snapshots.append(SnapshotRecord(source_id, parent_id, revision, text, scope, metadata))
581
582        # Validate all immutable revisions before mutating vectors or snapshots.
583        pending: dict[tuple[str, str], SnapshotRecord] = {}
584        for snapshot in snapshots:
585            identity = (snapshot.document_id, snapshot.revision)
586            previous = pending.get(identity)
587            if previous is not None and (
588                previous.content != snapshot.content or previous.metadata != snapshot.metadata
589            ):
590                raise ValueError("Conflicting document revisions in the same batch")
591            pending[identity] = snapshot
592            if snapshot_store is not None:
593                if hasattr(snapshot_store, "get_scoped"):
594                    existing = snapshot_store.get_scoped(
595                        source_id, snapshot.document_id, snapshot.revision, scope
596                    )
597                else:
598                    existing = snapshot_store.get(
599                        source_id, snapshot.document_id, snapshot.revision
600                    )
601                if (
602                    existing is not None
603                    and existing.isolation_scope == scope
604                    and (
605                        existing.content != snapshot.content
606                        or existing.metadata != snapshot.metadata
607                    )
608                ):
609                    raise ValueError("Cannot replace an immutable document revision")
610
611        if hasattr(store, "upsert"):
612            await store.upsert(normalized)
613            ingested_count = len(normalized)
614        if snapshot_store is not None:
615            for snapshot in snapshots:
616                snapshot_store.put(snapshot)
617
618        return {
619            "status": "success",
620            "collection": collection,
621            "source_id": source_id,
622            "documents_synced": ingested_count,
623        }

625    async def revoke_document(  # noqa: PLR0913
626        self,
627        collection: str,
628        document_id: str,
629        source_id: str = "default_source",
630        scope: str | None = None,
631        client_id: str | None = None,
632    ) -> dict[str, Any]:
633        """Revoke/delete a document and its chunks from the collection.
634
635        Guarded by :attr:`allow_management`.
636
637        Raises:
638            PermissionError: If management tools are not permitted by host.
639        """
640        if not self.allow_management:
641            raise PermissionError(
642                "Management operations (deletion/revocation) are disabled by host security rules. "
643                "Enable them explicitly with allow_management=True."
644            )
645
646        collection = self.default_collection if collection is None else collection
647        validate_safe_identifier(collection, "collection")
648        validate_safe_identifier(document_id, "document_id")
649        validate_safe_string(source_id, "source_id")
650        if scope is not None:
651            validate_safe_identifier(scope, "scope")
652
653        cfg = self._authorize(collection, scope, client_id)
654        retriever = cfg.retriever
655        store = getattr(retriever, "store", retriever)
656
657        lookup = cfg.source_lookup or self.source_lookup
658        lookup.revoke(source_id, document_id, scope)
659        deleted_count = 0
660        if hasattr(store, "delete"):
661            filters: dict[str, Any] = {"document_id": document_id, "source_id": source_id}
662            if scope is not None:
663                filters[self._scope_field(cfg)] = scope
664            deleted_count = await store.delete(filters=filters)
665
666        return {
667            "status": "revoked",
668            "collection": collection,
669            "document_id": document_id,
670            "chunks_removed": deleted_count,
671        }
