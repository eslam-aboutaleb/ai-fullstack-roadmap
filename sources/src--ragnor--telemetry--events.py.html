35def sanitize_error_category(exc: Any) -> str:  # noqa: PLR0911
36    """Classify an exception into a sanitized, secret-free category string.
37
38    Prevents sensitive strings (queries, passwords, file paths, connection URLs)
39    from leaking into logs or event streams.
40    """
41    if exc is None:
42        return "none"
43
44    type_name = type(exc).__name__.lower()
45    msg = str(exc).lower()
46
47    if any(k in type_name for k in ("auth", "credential", "permission")) or any(
48        k in msg for k in ("unauthorized", "401", "forbidden", "403", "api key", "apikey", "sk-")
49    ):
50        return "authentication_error"
51    if "rate" in msg or "429" in msg or "quota" in msg:
52        return "rate_limit_error"
53    if "timeout" in type_name or "timeout" in msg:
54        return "timeout_error"
55    if (
56        any(k in type_name for k in ("connect", "network", "socket"))
57        or "connection" in msg
58        or "refused" in msg
59    ):
60        return "connection_error"
61    if "dimension" in type_name or "dimension" in msg:
62        return "dimension_error"
63    if "notfound" in type_name or "not found" in msg or "404" in msg:
64        return "not_found_error"
65    if any(k in type_name for k in ("validation", "valueerror", "typeerror")) or "invalid" in msg:
66        return "validation_error"
67    if any(k in msg for k in ("fence", "lease", "stale", "superseded")):
68        return "lease_or_fence_error"
69    if "transient" in type_name:
70        return "transient_error"
71
72    return type(exc).__name__

75@dataclass
76class RedactionConfig:
77    """Privacy configuration for telemetry attribute redaction.
78
79    By default, raw text, query strings, credentials, and user identifiers
80    are excluded from all telemetry events. Hosts can explicitly opt-in to
81    capturing specific fields with designated redaction rules.
82    """
83
84    allow_queries: bool = False
85    allow_document_text: bool = False
86    allow_user_identifiers: bool = False
87    allow_unmasked_errors: bool = False
88    mask_credentials: bool = True
89
90    def __init__(  # noqa: PLR0913
91        self,
92        allow_queries: bool = False,
93        allow_document_text: bool = False,
94        allow_user_identifiers: bool = False,
95        allow_unmasked_errors: bool = False,
96        mask_credentials: bool = True,
97        *,
98        allow_raw_query: bool | None = None,
99        allow_raw_text: bool | None = None,
100        allow_user_id: bool | None = None,
101        scrub_credentials: bool | None = None,
102    ) -> None:
103        self.allow_queries = allow_raw_query if allow_raw_query is not None else allow_queries
104        self.allow_document_text = (
105            allow_raw_text if allow_raw_text is not None else allow_document_text
106        )
107        self.allow_user_identifiers = (
108            allow_user_id if allow_user_id is not None else allow_user_identifiers
109        )
110        self.allow_unmasked_errors = allow_unmasked_errors
111        self.mask_credentials = (
112            scrub_credentials if scrub_credentials is not None else mask_credentials
113        )
114
115    _SENSITIVE_KEY_PATTERNS: tuple[str, ...] = (
116        "password",
117        "secret",
118        "token",
119        "key",
120        "auth",
121        "credential",
122        "dsn",
123        "database_url",
124    )
125
126    def scrub_text(self, text: str) -> str:
127        """Scrub known credential patterns from arbitrary text."""
128        # Replace Bearer tokens, passwords, keys
129        pattern = re.compile(
130            r"(?i)(bearer\s+[A-Za-z0-9_\-\.]+)|"
131            r"((?:password|token|secret|key)\s*[=:]\s*['\"]?[^\s'\"]+['\"]?)"
132        )
133        return pattern.sub("[REDACTED]", text)
134
135    def sanitize_attributes(self, attributes: dict[str, Any]) -> dict[str, Any]:
136        """Sanitize an attributes dictionary according to redaction rules."""
137        sanitized: dict[str, Any] = {}
138        for k, v in attributes.items():
139            k_lower = k.lower()
140
141            # Mask credentials
142            if self.mask_credentials and any(p in k_lower for p in self._SENSITIVE_KEY_PATTERNS):
143                sanitized[k] = "[REDACTED]"
144                continue
145
146            # Query text
147            if "query" in k_lower and not self.allow_queries:
148                sanitized[k] = "[QUERY_REDACTED]"
149                continue
150
151            # Document raw text
152            if (
153                ("text" in k_lower or "document" in k_lower or "content" in k_lower)
154                and not self.allow_document_text
155                and isinstance(v, str)
156                and len(v) > 20
157            ):
158                sanitized[k] = f"[TEXT_REDACTED_LEN_{len(v)}]"
159                continue
160
161            # User identifiers
162            if (
163                any(u in k_lower for u in ("user_id", "email", "username", "account"))
164                and not self.allow_user_identifiers
165            ):
166                sanitized[k] = "[USER_REDACTED]"
167                continue
168
169            sanitized[k] = v
170        return sanitized

176@dataclass
177class TelemetryEvent:
178    """A strongly typed event emitted by ragnor components.
179
180    Attributes:
181        event_kind: Classification from :class:`EventKind` or custom string.
182        timestamp: Unix epoch timestamp in seconds.
183        event_id: Unique event UUID string.
184        run_id: Pipeline or sync execution run identifier.
185        request_id: Search or query request identifier.
186        stage: Component stage (e.g. "sync", "embedding", "retrieval").
187        source_id: Scoped source identifier.
188        document_id: Scoped document identifier.
189        isolation_scope: Tenant or namespace scope.
190        counts: Stage-specific numeric measurements (e.g. batch size, tokens).
191        duration_ms: Duration in milliseconds for timed operations.
192        error_category: Sanitized error classification.
193        raw_text: Optional document text, excluded by default.
194        raw_query: Optional query text, excluded by default.
195        raw_credentials: Always excluded or None.
196        user_id: Optional user identifier, excluded by default.
197        attributes: Additional sanitized event key-values.
198    """
199
200    event_kind: str
201    timestamp: float = field(default_factory=time.time)
202    event_id: str = field(default_factory=lambda: str(uuid.uuid4()))
203    run_id: str | None = None
204    request_id: str | None = None
205    stage: str | None = None
206    source_id: str | None = None
207    document_id: str | None = None
208    isolation_scope: str | None = None
209    generation: int | None = None
210    counts: dict[str, int] = field(default_factory=dict)
211    duration_ms: float | None = None
212    error_category: str | None = None
213    error: Any = None
214    raw_text: str | None = None
215    raw_query: str | None = None
216    raw_credentials: str | None = None
217    user_id: str | None = None
218    attributes: dict[str, Any] = field(default_factory=dict)
219
220    def __post_init__(self) -> None:
221        """Sanitize error to error_category and scrub error object."""
222        if self.error is not None and self.error_category is None:
223            self.error_category = sanitize_error_category(self.error)
224        self.error = None
225
226    @property
227    def sanitized_error_category(self) -> str | None:
228        """Alias for error_category for clarity."""
229        return self.error_category
230
231    @classmethod
232    def create(  # noqa: PLR0913
233        cls,
234        event_kind: str | EventKind,
235        *,
236        run_id: str | None = None,
237        request_id: str | None = None,
238        stage: str | None = None,
239        source_id: str | None = None,
240        document_id: str | None = None,
241        isolation_scope: str | None = None,
242        generation: int | None = None,
243        counts: dict[str, int] | None = None,
244        duration_ms: float | None = None,
245        error: Any = None,
246        raw_text: str | None = None,
247        raw_query: str | None = None,
248        raw_credentials: str | None = None,
249        user_id: str | None = None,
250        attributes: dict[str, Any] | None = None,
251        redaction: RedactionConfig | None = None,
252        **extra_attrs: Any,
253    ) -> TelemetryEvent:
254        """Create a TelemetryEvent with automatic sanitization and redaction."""
255        cfg = redaction or DEFAULT_REDACTION_CONFIG
256        err_cat = sanitize_error_category(error) if error is not None else None
257
258        merged_attrs = dict(attributes or {})
259        merged_attrs.update(extra_attrs)
260        safe_attrs = cfg.sanitize_attributes(merged_attrs)
261
262        # Apply default exclusions & redactions
263        safe_query: str | None = None
264        if cfg.allow_queries and raw_query is not None:
265            safe_query = cfg.scrub_text(raw_query) if cfg.mask_credentials else raw_query
266
267        safe_text: str | None = None
268        if cfg.allow_document_text and raw_text is not None:
269            safe_text = cfg.scrub_text(raw_text) if cfg.mask_credentials else raw_text
270
271        safe_user: str | None = None
272        if cfg.allow_user_identifiers and user_id is not None:
273            safe_user = user_id
274
275        return cls(
276            event_kind=str(event_kind),
277            run_id=run_id,
278            request_id=request_id,
279            stage=stage,
280            source_id=source_id,
281            document_id=document_id,
282            isolation_scope=isolation_scope,
283            counts=dict(counts or {}),
284            duration_ms=duration_ms,
285            error_category=err_cat,
286            raw_text=safe_text,
287            raw_query=safe_query,
288            raw_credentials=None,  # credentials never allowed
289            user_id=safe_user,
290            attributes=safe_attrs,
291        )

310class MemoryEventSink:
311    """In-memory EventSink capturing events for inspection, assertions, and testing."""
312
313    def __init__(self) -> None:
314        self.events: list[TelemetryEvent] = []
315
316    def emit(self, event: TelemetryEvent) -> None:
317        """Record event in memory."""
318        self.events.append(event)
319
320    def clear(self) -> None:
321        """Clear recorded events."""
322        self.events.clear()
323
324    def get_by_kind(self, event_kind: str | EventKind) -> list[TelemetryEvent]:
325        """Return all recorded events matching ``event_kind``."""
326        target = str(event_kind)
327        return [e for e in self.events if e.event_kind == target]
328
329    def get_by_stage(self, stage: str) -> list[TelemetryEvent]:
330        """Return all recorded events matching ``stage``."""
331        return [e for e in self.events if e.stage == stage]

334class LoggingEventSink:
335    """EventSink emitting event summaries via Python's standard logging module."""
336
337    def __init__(self, logger_name: str = "ragnor.events", level: int = logging.INFO) -> None:
338        self._logger = logging.getLogger(logger_name)
339        self._level = level
340
341    def emit(self, event: TelemetryEvent) -> None:
342        """Log event summary."""
343        parts = [f"[{event.event_kind}]"]
344        if event.stage:
345            parts.append(f"stage={event.stage}")
346        if event.run_id:
347            parts.append(f"run_id={event.run_id}")
348        if event.request_id:
349            parts.append(f"request_id={event.request_id}")
350        if event.duration_ms is not None:
351            parts.append(f"duration_ms={event.duration_ms:.2f}")
352        if event.error_category:
353            parts.append(f"error={event.error_category}")
354        if event.counts:
355            parts.append(f"counts={event.counts}")
356
357        self._logger.log(self._level, " ".join(parts))

360class CompositeEventSink:
361    """EventSink forwarding emitted events to multiple underlying sinks."""
362
363    def __init__(self, sinks: Sequence[EventSink]) -> None:
364        self._sinks = list(sinks)
365
366    def emit(self, event: TelemetryEvent) -> None:
367        """Forward event to all configured child sinks."""
368        for sink in self._sinks:
369            try:
370                sink.emit(event)
371            except Exception as exc:  # noqa: BLE001
372                logger.warning("EventSink %r failed to emit event: %s", sink, exc)
