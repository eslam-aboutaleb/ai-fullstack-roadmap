1"""LiteLLM callback that records tokens, cost, and latency for every model call.
2
3LiteLLM runs its logging hooks on the same completion path as the chat and
4streaming endpoints, which makes this the one place that sees every billable
5call without touching either endpoint. Everything here is defensive on purpose:
6the hooks are invoked from LiteLLM's own error handling, so raising out of them
7would turn an already failed model call into a failed request. Telemetry is
8best-effort -- a chat request never fails because of it.
9"""

29def _lookup(source: Any, key: str, default: Any = None) -> Any:
30    """Read ``key`` from a mapping, a LiteLLM response object, or a model.
31
32    LiteLLM hands the hooks the logging payload as a plain dict and the response
33    as a pydantic object, so the same field has to be readable from either. A
34    ``None`` result is normalised to ``default`` because LiteLLM leaves optional
35    response fields absent rather than populated.
36
37    Args:
38        source: Object, mapping, or pydantic model to read from. ``None`` is
39            accepted and yields ``default``.
40        key: Field name to read.
41        default: Value returned when the field is missing or ``None``.
42
43    Returns:
44        Any: The field value, or ``default``.
45    """
46    if source is None:
47        return default
48    if isinstance(source, Mapping):
49        value = source.get(key)
50    else:
51        value = getattr(source, key, None)
52        if value is None:
53            getter = getattr(source, "get", None)
54            value = getter(key) if callable(getter) else None
55    return default if value is None else value

58def _as_float(value: Any) -> float | None:
59    """Coerce a numeric ``value`` to float, returning ``None`` for anything else."""
60    if isinstance(value, int | float) and not isinstance(value, bool):
61        return float(value)
62    return None

65def _resolve_tokens(standard: Any, usage: Any, key: str) -> int:
66    """Return a token count from the logging payload or the response usage.
67
68    Args:
69        standard: LiteLLM's ``standard_logging_object`` payload, when present.
70        usage: Token usage object from the response, when present.
71        key: Either ``"prompt_tokens"`` or ``"completion_tokens"``.
72
73    Returns:
74        int: The token count, or 0 when no source reports one -- cache hits and
75        embeddings legitimately have none.
76    """
77    value = _lookup(standard, key)
78    if value is None:
79        value = _lookup(usage, key)
80    parsed = _as_float(value)
81    return int(parsed) if parsed is not None else 0

84def _resolve_cost(kwargs: Any, response_obj: Any, standard: Any) -> float:
85    """Price one call, preferring LiteLLM's own calculator.
86
87    ``litellm.completion_cost`` is authoritative, but it raises for models that
88    are absent from its price map and for responses it cannot read. In both
89    cases the cost LiteLLM already stamped on the logging payload is the next
90    best thing, because a call still costs *something*.
91
92    Args:
93        kwargs: The logging payload LiteLLM passed to the hook.
94        response_obj: The provider response, if the call produced one.
95        standard: LiteLLM's ``standard_logging_object``, when present.
96
97    Returns:
98        float: Cost in USD; 0.0 when no source could price the call.
99    """
100    if response_obj is not None:
101        try:
102            return float(litellm.completion_cost(completion_response=response_obj))
103        except Exception:
104            logger.debug("litellm.completion_cost could not price this call", exc_info=True)
105    for source in (standard, kwargs):
106        cost = _as_float(_lookup(source, "response_cost"))
107        if cost is not None:
108            return cost
109    return 0.0

112def _resolve_model(kwargs: Any, response_obj: Any, standard: Any) -> str:
113    """Return the model name from the first source that reports one.
114
115    Args:
116        kwargs: The logging payload LiteLLM passed to the hook.
117        response_obj: The provider response, if the call produced one.
118        standard: LiteLLM's ``standard_logging_object``, when present.
119
120    Returns:
121        str: The model identifier, or ``"unknown"``.
122    """
123    for source in (standard, kwargs, response_obj):
124        model = _lookup(source, "model")
125        if isinstance(model, str) and model:
126            return model
127    return _UNKNOWN_MODEL

130def _elapsed_seconds(start_time: Any, end_time: Any) -> float | None:
131    """Return the seconds between two timestamps given as datetimes or numbers.
132
133    Args:
134        start_time: When the call began, as reported by LiteLLM.
135        end_time: When the call ended, as reported by LiteLLM.
136
137    Returns:
138        float | None: Elapsed seconds, or ``None`` when the timestamps cannot be
139        subtracted -- mixing a naive with an aware datetime raises, and which of
140        the two LiteLLM reports depends on the provider.
141    """
142    if isinstance(start_time, datetime) and isinstance(end_time, datetime):
143        try:
144            return max(0.0, (end_time - start_time).total_seconds())
145        except TypeError:
146            return None
147    start = _as_float(start_time)
148    end = _as_float(end_time)
149    if start is None or end is None:
150        return None
151    return max(0.0, end - start)

154def _resolve_latency(start_time: Any, end_time: Any, standard: Any) -> float:
155    """Return the call duration in seconds.
156
157    Args:
158        start_time: When the call began, as reported by LiteLLM.
159        end_time: When the call ended, as reported by LiteLLM.
160        standard: LiteLLM's ``standard_logging_object``, which already measures
161            the round trip and is the fallback for cache hits and providers that
162            call the hook without timing information.
163
164    Returns:
165        float: Duration in seconds, never negative.
166    """
167    latency = _elapsed_seconds(start_time, end_time)
168    if latency is not None:
169        return latency
170    return _as_float(_lookup(standard, "response_time")) or 0.0

173def _resolve_error(kwargs: Any) -> str:
174    """Describe why a call failed, for the failed-call record.
175
176    Args:
177        kwargs: The logging payload LiteLLM passed to the hook.
178
179    Returns:
180        str: The exception LiteLLM captured, or its message, or a placeholder.
181    """
182    exception = _lookup(kwargs, "exception") or _lookup(kwargs, "original_exception")
183    if exception is not None:
184        return exception if isinstance(exception, str) else repr(exception)
185    return _lookup(kwargs, "error_information") or _UNKNOWN_ERROR

188class LlmCostCallback(CustomLogger):
189    """``CustomLogger`` that appends one :class:`LlmCall` per model completion.
190
191    Both the synchronous and the asynchronous hooks are implemented because
192    LiteLLM routes a completion through whichever pair matches the caller, and
193    the agent drives LiteLLM asynchronously.
194    """
195
196    def log_success_event(
197        self, kwargs: Any, response_obj: Any, start_time: Any, end_time: Any
198    ) -> None:
199        """Record a successful synchronous completion.
200
201        Args:
202            kwargs: LiteLLM logging payload.
203            response_obj: The provider response.
204            start_time: Call start reported by LiteLLM.
205            end_time: Call end reported by LiteLLM.
206        """
207        self._record(kwargs, response_obj, start_time, end_time, success=True)
208
209    def log_failure_event(
210        self, kwargs: Any, response_obj: Any, start_time: Any, end_time: Any
211    ) -> None:
212        """Record a failed synchronous completion.
213
214        Args:
215            kwargs: LiteLLM logging payload.
216            response_obj: The provider response, when one was produced.
217            start_time: Call start reported by LiteLLM.
218            end_time: Call end reported by LiteLLM.
219        """
220        self._record(kwargs, response_obj, start_time, end_time, success=False)
221
222    async def async_log_success_event(
223        self, kwargs: Any, response_obj: Any, start_time: Any, end_time: Any
224    ) -> None:
225        """Record a successful asynchronous completion.
226
227        Args:
228            kwargs: LiteLLM logging payload.
229            response_obj: The provider response.
230            start_time: Call start reported by LiteLLM.
231            end_time: Call end reported by LiteLLM.
232        """
233        self._record(kwargs, response_obj, start_time, end_time, success=True)
234
235    async def async_log_failure_event(
236        self, kwargs: Any, response_obj: Any, start_time: Any, end_time: Any
237    ) -> None:
238        """Record a failed asynchronous completion.
239
240        Args:
241            kwargs: LiteLLM logging payload.
242            response_obj: The provider response, when one was produced.
243            start_time: Call start reported by LiteLLM.
244            end_time: Call end reported by LiteLLM.
245        """
246        self._record(kwargs, response_obj, start_time, end_time, success=False)
247
248    def _record(
249        self,
250        kwargs: Any,
251        response_obj: Any,
252        start_time: Any,
253        end_time: Any,
254        *,
255        success: bool,
256    ) -> None:
257        """Write one completion event, swallowing every failure.
258
259        Args:
260            kwargs: LiteLLM logging payload.
261            response_obj: The provider response, when one was produced.
262            start_time: Call start reported by LiteLLM.
263            end_time: Call end reported by LiteLLM.
264            success: Whether the call returned a response.
265        """
266        try:
267            standard = _lookup(kwargs, "standard_logging_object")
268            usage = _lookup(response_obj, "usage")
269            get_metrics().record_llm_call(
270                build_llm_call(
271                    model=_resolve_model(kwargs, response_obj, standard),
272                    prompt_tokens=_resolve_tokens(standard, usage, "prompt_tokens"),
273                    completion_tokens=_resolve_tokens(standard, usage, "completion_tokens"),
274                    cost_usd=_resolve_cost(kwargs, response_obj, standard),
275                    latency_seconds=_resolve_latency(start_time, end_time, standard),
276                    success=success,
277                    error="" if success else _resolve_error(kwargs),
278                )
279            )
280        except Exception:
281            logger.debug("LLM telemetry dropped", exc_info=True)

284def register_llm_cost_callback() -> LlmCostCallback:
285    """Add the cost callback to LiteLLM's process-wide callback list.
286
287    ``litellm.callbacks`` is copied per call by LiteLLM's own request setup, so
288    appending here takes effect without touching a request path. The membership
289    check keeps repeated startups -- the test suite re-runs the application
290    lifespan -- from stacking duplicate callbacks that would double every count.
291
292    Returns:
293        LlmCostCallback: The active cost callback, newly registered or already
294        present.
295    """
296    for callback in litellm.callbacks:
297        if isinstance(callback, LlmCostCallback):
298            return callback
299
300    callback = LlmCostCallback()
301    litellm.callbacks = [*litellm.callbacks, callback]
302    logger.info("LiteLLM cost and latency callback registered")
303    return callback
