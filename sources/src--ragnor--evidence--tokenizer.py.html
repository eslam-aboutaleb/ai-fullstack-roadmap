34class ApproximateTokenizer:
35    """Conservative heuristic token estimator.
36
37    NOTICE: This is an APPROXIMATE estimator. It uses a conservative
38    character-to-token and word-to-token ratio to estimate token counts without
39    requiring third-party model tokenizers or network calls.
40
41    WARNING: Never claim this approximate estimate guarantees a model's context
42    limit. For strict compliance with model-specific context windows, inject an
43    :class:`ExactTokenizerAdapter`.
44    """
45
46    @property
47    def is_exact(self) -> bool:
48        """Always False for heuristic estimators."""
49        return False
50
51    @property
52    def name(self) -> str:
53        return "approximate_heuristic"
54
55    def count_tokens(self, text: str) -> int:
56        """Conservatively estimate token count for ``text``.
57
58        Uses the upper bound of ~3.3 characters per token and ~1.35 tokens per word
59        to avoid underestimating token usage on punctuation-heavy or technical texts.
60        """
61        if not text:
62            return 0
63        char_est = math.ceil(len(text) / 3.3)
64        words = text.split()
65        word_est = math.ceil(len(words) * 1.35) if words else 0
66        return max(char_est, word_est, 1)

69class ExactTokenizerAdapter:
70    """Adapter wrapping an exact model-specific token counter.
71
72    Enforces exact model-specific context limits.
73
74    Can wrap any callable `Callable[[str], int]`, or objects with an
75    `encode(text)` method (such as tiktoken or HuggingFace tokenizers).
76    """
77
78    def __init__(
79        self,
80        tokenizer_or_fn: Any,
81        name: str = "exact_model_tokenizer",
82    ) -> None:
83        """Initialize the exact tokenizer adapter.
84
85        Args:
86            tokenizer_or_fn: A callable taking a string and returning int,
87                or a tokenizer object with an `encode(text)` method.
88            name: Identifier for the model or tokenizer backend.
89        """
90        self._name = name
91        if callable(tokenizer_or_fn):
92            self._count_fn: Callable[[str], int] = tokenizer_or_fn
93        elif hasattr(tokenizer_or_fn, "encode"):
94            self._count_fn = lambda text: len(tokenizer_or_fn.encode(text))
95        else:
96            raise TypeError(
97                "Expected a callable or an object with an 'encode' method, "
98                f"got {type(tokenizer_or_fn)}"
99            )
100
101    @property
102    def is_exact(self) -> bool:
103        """Always True for exact adapters."""
104        return True
105
106    @property
107    def name(self) -> str:
108        return self._name
109
110    def count_tokens(self, text: str) -> int:
111        """Count exact tokens using the underlying model tokenizer."""
112        if not text:
113            return 0
114        return int(self._count_fn(text))
