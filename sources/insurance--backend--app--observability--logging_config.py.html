1"""Structured log output with the correlation id on every record.
2
3``logging.basicConfig`` writes one free-text line per record and knows nothing
4about the request that produced it. Replacing the handler with a formatter that
5emits a fixed JSON schema gives every record the same fields -- including the
6correlation id, which is stamped at record-creation time so it belongs to the
7context that logged it, not the context that later formats it.
8"""
9

19# Attributes every LogRecord carries. Anything outside this set was passed by a
20# caller as ``logging``'s ``extra=`` and is forwarded into the JSON payload, which
21# is what lets a record carry structured fields such as a model name or a cost.
22_RESERVED_RECORD_ATTRS = frozenset(vars(logging.makeLogRecord({}))) | {
23    "asctime",
24    "message",
25    "request_id",
26    "taskName",
27}
28
29_TEXT_FORMAT = "%(asctime)s - %(name)s - %(levelname)s - [%(request_id)s] %(message)s"
30
31_logging_configured = False
32_original_record_factory = logging.getLogRecordFactory()

35def _record_factory(*args: Any, **kwargs: Any) -> logging.LogRecord:
36    """Create a LogRecord that already carries the current correlation id.
37
38    Stamping the record when it is created rather than when a handler formats it
39    means a record belongs to the context that emitted it, which stays correct
40    even for handlers that format later or elsewhere.
41
42    Args:
43        args: Positional arguments forwarded to the previous record factory.
44        kwargs: Keyword arguments forwarded to the previous record factory.
45
46    Returns:
47        logging.LogRecord: The record with ``request_id`` set.
48    """
49    record = _original_record_factory(*args, **kwargs)
50    record.request_id = display_request_id()
51    return record

54class ObservabilityLogHandler(logging.StreamHandler):

57    The type exists so reconfiguring logging replaces *this* handler instead of

63class JsonLogFormatter(logging.Formatter):
64    """Render one log record as a single JSON object."""
65
66    def format(self, record: logging.LogRecord) -> str:
67        """Serialise ``record`` into a JSON object.
68
69        Args:
70            record: The record to render.
71
72        Returns:
73            str: One-line JSON with the fixed fields plus any structured
74            ``extra`` attributes the caller supplied.
75        """
76        payload: dict[str, Any] = {
77            "timestamp": datetime.fromtimestamp(record.created, tz=UTC).isoformat(),
78            "level": record.levelname,
79            "logger": record.name,
80            "message": record.getMessage(),
81            "request_id": getattr(record, "request_id", "-"),
82        }
83        attributes = {
84            key: value
85            for key, value in record.__dict__.items()
86            if key not in _RESERVED_RECORD_ATTRS and not key.startswith("_")
87        }
88        if attributes:
89            payload["attributes"] = attributes
90        if record.exc_info:
91            payload["exception"] = self.formatException(record.exc_info)
92        if record.stack_info:
93            payload["stack"] = self.formatStack(record.stack_info)
94        return json.dumps(payload, default=str)

97class TextLogFormatter(logging.Formatter):
98    """Render one log record on a single human-readable line.
99
100    The layout keeps the original field order and adds the correlation id in
101    brackets, so an operator reading a local log can still eyeball a trace.
102    """
103
104    def format(self, record: logging.LogRecord) -> str:
105        """Render ``record`` with its correlation id guaranteed to be present.
106
107        Args:
108            record: The record to render.
109
110        Returns:
111            str: One formatted line.
112        """
113        if not getattr(record, "request_id", ""):
114            # A record emitted before logging was configured has no id, and the
115            # format string references the field directly.
116            record.request_id = display_request_id()
117        return super().format(record)

120def build_formatter(log_format: str) -> logging.Formatter:
121    """Return the formatter for ``log_format``.
122
123    Args:
124        log_format: Either ``"json"`` or ``"text"``; anything else falls back to
125            the text layout rather than failing at import time.
126
127    Returns:
128        logging.Formatter: The requested formatter.
129    """
130    if log_format == "json":
131        return JsonLogFormatter()
132    return TextLogFormatter(_TEXT_FORMAT)

135def configure_logging(log_level: str, log_format: str) -> None:
136    """Install the process-wide log handler, formatter, and record factory.
137
138    Args:
139        log_level: Standard level name, e.g. ``"INFO"``; an unrecognised value is
140            treated as ``INFO``.
141        log_format: ``"json"`` or ``"text"``; see :func:`build_formatter`.
142    """
143    global _logging_configured  # noqa: PLW0603
144
145    handler = ObservabilityLogHandler()
146    handler.setFormatter(build_formatter(log_format))
147
148    root = logging.getLogger()
149    for installed in [h for h in root.handlers if isinstance(h, ObservabilityLogHandler)]:
150        root.removeHandler(installed)
151    root.addHandler(handler)
152    root.setLevel(getattr(logging, log_level.upper(), logging.INFO))
153
154    if not _logging_configured:
155        logging.setLogRecordFactory(_record_factory)
156        _logging_configured = True
