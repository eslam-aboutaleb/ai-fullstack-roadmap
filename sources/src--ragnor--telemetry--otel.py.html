1"""Optional OpenTelemetry bridge for event tracing and metric emission.
2
3This module provides an optional bridge from :class:`~ragnor.telemetry.events.TelemetryEvent`
4to OpenTelemetry spans, span events, and metrics. It is an optional extra
5(``pip install 'ragnor[opentelemetry]'``) and never a core dependency.
6"""
7
8from __future__ import annotations
9
10import logging
11from typing import Any
12
13from ragnor.telemetry.events import TelemetryEvent
14
15logger = logging.getLogger(__name__)
16
17
18class OpenTelemetryEventSink:
19    """EventSink forwarding ragnor events to active OpenTelemetry spans and meters.
20
21    Requires the ``opentelemetry`` extra (``pip install 'ragnor[opentelemetry]'``).
22    """
23
24    def __init__(
25        self,
26        tracer: Any = None,
27        meter: Any = None,
28        *,
29        service_name: str = "ragnor",
30    ) -> None:
31        """Initialize the OpenTelemetry bridge sink.
32
33        Args:
34            tracer: Optional opentelemetry.trace.Tracer. If None, acquires the
35                global tracer.
36            meter: Optional opentelemetry.metrics.Meter. If None, acquires the
37                global meter.
38            service_name: Instrumentation scope name.
39
40        Raises:
41            ImportError: If ``opentelemetry-api`` is not installed.
42        """
43        try:
44            from opentelemetry import metrics, trace  # noqa: PLC0415
45        except ImportError as exc:
46            raise ImportError(
47                "OpenTelemetry integration requires the 'opentelemetry' extra: "
48                "pip install 'ragnor[opentelemetry]'"
49            ) from exc
50
51        self._trace = trace
52        self._metrics = metrics
53        self._tracer = tracer or trace.get_tracer(service_name)
54        self._meter = meter or metrics.get_meter(service_name)
55
56        # Initialize standard instruments
57        self._event_counter = self._meter.create_counter(
58            name="ragnor.events.total",
59            description="Total count of ragnor telemetry events emitted",
60            unit="1",
61        )
62        self._duration_histogram = self._meter.create_histogram(
63            name="ragnor.duration_ms",
64            description="Duration of timed operations in milliseconds",
65            unit="ms",
66        )
67
68    def emit(self, event: TelemetryEvent) -> None:
69        """Forward a TelemetryEvent to OpenTelemetry span events and metrics."""
70        base_attributes: dict[str, Any] = {
71            "ragnor.event_kind": event.event_kind,
72        }
73        if event.stage:
74            base_attributes["ragnor.stage"] = event.stage
75        if event.run_id:
76            base_attributes["ragnor.run_id"] = event.run_id
77        if event.request_id:
78            base_attributes["ragnor.request_id"] = event.request_id
79        if event.source_id:
80            base_attributes["ragnor.source_id"] = event.source_id
81        if event.document_id:
82            base_attributes["ragnor.document_id"] = event.document_id
83        if event.isolation_scope:
84            base_attributes["ragnor.isolation_scope"] = event.isolation_scope
85        if event.error_category:
86            base_attributes["ragnor.error_category"] = event.error_category
87
88        # Add event to active OpenTelemetry span if available
89        span = self._trace.get_current_span()
90        if span is not None and span.is_recording():
91            span_attrs = dict(base_attributes)
92            for k, v in event.attributes.items():
93                if isinstance(v, (str, int, float, bool)):
94                    span_attrs[f"ragnor.attr.{k}"] = v
95            span.add_event(name=event.event_kind, attributes=span_attrs)
96
97        # Record metrics
98        metric_attrs = {
99            "event_kind": event.event_kind,
100            "stage": event.stage or "unknown",
101            "error_category": event.error_category or "none",
102        }
103        self._event_counter.add(1, attributes=metric_attrs)
104
105        if event.duration_ms is not None:
106            self._duration_histogram.record(event.duration_ms, attributes=metric_attrs)
