1"""High-level convenience API: the RagnorBuilder and the Ragnor facade.
2
3The builder sits *on top* of the low-level seams -- it wires defaults
4into :class:`ragnor.ingestion.pipeline.IngestionPipeline` and
5:class:`ragnor.retrieval.retriever.HybridRetriever` without forking
6their logic, so a host goes from zero to a working index in three
7lines::
8
9    rag = await RagnorBuilder.local(embedder=my_embedder).build()
10    await rag.ingest([Document(id="doc", text="...")])
11    results = await rag.search("query")
12
13Every component is overridable (``with_store`` / ``with_embedder`` /
14``with_chunker`` / ``with_version_store`` / ``with_settings``), and
15the low-level seams stay reachable through the facade's ``.store``,
16``.pipeline``, and ``.retriever`` attributes.
17
18ragnor ships no local embedder in core (the ``local-embeddings`` extra
19is reserved for a future local-model backend), so the builder requires
20an injected embedder -- the offline example
21(``examples/local_quickstart.py``) defines its own deterministic fake.
22The default wiring is fully offline: the in-memory vector store, the
23in-memory version store, and a null session provider need no database,
24no driver, and no API key.
25"""

86class InMemoryVersionStore:
87    """In-memory :class:`VersionStore` -- the first built-in implementation.
88
89    Tracks the active version, its snapshot, and the chunk ids of
90    retired versions in process memory. ``delete_chunks`` deletes a
91    retired version's chunks through the wired store's delete
92    capability (``await store.delete(filters={version_id_key:
93    version_id})``): the pipeline stamps ``version_id`` into chunk
94    metadata, so the metadata-filtered delete removes exactly the
95    retired version's chunks. The session argument is ignored -- the
96    memory store and this version store share no database, so the
97    pipeline's owned-session path runs on the null session provider.
98    """
99
100    def __init__(
101        self,
102        store: DocumentStore,
103        version_id_key: str = "version_id",
104    ) -> None:
105        """Initialize the version store wired to ``store``.
106
107        Args:
108            store: Store whose delete capability retires chunks.
109            version_id_key: Metadata key the pipeline stamps the
110                version id under.
111        """
112        self._store = store
113        self._version_id_key = version_id_key
114        self._active: VersionRecord | None = None
115        self._snapshots: dict[str, IngestionSnapshot] = {}
116        self._chunk_ids: dict[str, list[str]] = {}
117        self._next_version = 0
118
119    @property
120    def active(self) -> VersionRecord | None:
121        """The active version, or ``None`` when nothing is ingested."""
122        return self._active
123
124    async def find_active(self, session: Any) -> VersionRecord | None:
125        """Return the active version (the session is ignored).
126
127        Args:
128            session: Pipeline session (unused; storage is in process).
129
130        Returns:
131            The active version record, or ``None``.
132        """
133        return self._active
134
135    async def close_active(self, session: Any) -> None:
136        """Retire the active version (the session is ignored).
137
138        Args:
139            session: Pipeline session (unused; storage is in process).
140        """
141        self._active = None
142
143    async def create_version(
144        self,
145        session: Any,
146        snapshot: IngestionSnapshot,
147    ) -> VersionRecord:
148        """Insert a new version row carrying ``snapshot``.
149
150        Args:
151            session: Pipeline session (unused; storage is in process).
152            snapshot: The ingestion snapshot to store.
153
154        Returns:
155            The newly created version record.
156        """
157        self._next_version += 1
158        version_id = f"v{self._next_version}"
159        record = VersionRecord(version_id=version_id, snapshot=snapshot)
160        self._snapshots[version_id] = snapshot
161        self._active = record
162        return record
163
164    async def delete_chunks(self, session: Any, version_id: str) -> None:
165        """Delete the chunks belonging to ``version_id``.
166
167        The retired version's chunk ids are recorded (read through the
168        store's get capability) and the chunks are then deleted through
169        the store's delete capability, filtered by the version id the
170        pipeline stamps into chunk metadata.
171
172        Args:
173            session: Pipeline session (unused; storage is in process).
174            version_id: Identifier of the version whose chunks are
175                deleted.
176        """
177        rows = await self._store.get(filters={self._version_id_key: version_id})
178        self._chunk_ids[version_id] = [row["id"] for row in rows]
179        await self._store.delete(filters={self._version_id_key: version_id})
180
181    def snapshot(self, version_id: str) -> IngestionSnapshot | None:
182        """Return the snapshot recorded for ``version_id``.
183
184        Args:
185            version_id: Identifier of the version.
186
187        Returns:
188            The stored snapshot, or ``None`` when unknown.
189        """
190        return self._snapshots.get(version_id)
191
192    def chunk_ids(self, version_id: str) -> list[str]:
193        """Return the chunk ids recorded for a retired ``version_id``.
194
195        Chunk ids are recorded when a version is retired (its chunks
196        are deleted); the active version's chunks are read from the
197        store directly.
198
199        Args:
200            version_id: Identifier of the retired version.
201
202        Returns:
203            The chunk ids that belonged to the version.
204        """
205        return list(self._chunk_ids.get(version_id, []))

217    """Yield a no-op session (the builder runs without a database).

227class RagnorBuilder:
228    """Builder for the :class:`Ragnor` convenience facade.
229
230    ``RagnorBuilder.local(embedder)`` wires offline defaults -- the
231    in-memory vector store, the sliding-window chunker, the in-memory
232    version store, and :class:`BuilderSettings` -- and every component
233    is overridable before ``build()``.
234    """
235
236    def __init__(  # noqa: PLR0913
237        self,
238        embedder: EmbeddingFunction | None = None,
239        *,
240        embedding_model: str = DEFAULT_EMBEDDING_MODEL,
241        embedding_dim: int | None = None,
242        chunker: Chunker | None = None,
243        store: VectorStore | None = None,
244        version_store: VersionStore | None = None,
245        settings: RagSettings | None = None,
246    ) -> None:
247        """Initialize the builder.
248
249        Args:
250            embedder: Embedding function for chunk texts and queries.
251                ragnor ships no local embedder in core, so this is
252                required (``build()`` raises ``ValueError`` without
253                one).
254            embedding_model: Embedding model name recorded in
255                ingestion snapshots.
256            embedding_dim: Embedding dimensionality. When given, it is
257                registered for ``embedding_model`` via
258                :func:`ragnor.embeddings.register_dimension` and the
259                default store validates embeddings against it.
260            chunker: Chunker that splits document content. Defaults to
261                ``SlidingWindowChunker(chunk_size=600, overlap=100)``.
262            store: Vector store the chunks are upserted into. Defaults
263                to :class:`ragnor.stores.InMemoryVectorStore`.
264            version_store: Version storage. Defaults to
265                :class:`InMemoryVersionStore` wired to the store.
266            settings: Settings for the pipeline and retriever.
267                Defaults to :class:`BuilderSettings`.
268        """
269        self._embedder = embedder
270        self._embedding_model = embedding_model
271        self._embedding_dim = embedding_dim
272        self._chunker = chunker
273        self._store = store
274        self._version_store = version_store
275        self._settings = settings
276
277    @classmethod
278    def local(  # noqa: PLR0913
279        cls,
280        embedder: EmbeddingFunction | None,
281        *,
282        embedding_model: str = DEFAULT_EMBEDDING_MODEL,
283        embedding_dim: int | None = None,
284        chunker: Chunker | None = None,
285        store: VectorStore | None = None,
286        version_store: VersionStore | None = None,
287        settings: RagSettings | None = None,
288    ) -> RagnorBuilder:
289        """Build a builder with offline defaults.
290
291        Args:
292            embedder: Embedding function for chunk texts and queries
293                (required -- ragnor ships no local embedder in core).
294            embedding_model: Embedding model name recorded in
295                ingestion snapshots.
296            embedding_dim: Embedding dimensionality; registered for
297                ``embedding_model`` when given.
298            chunker: Chunker override.
299            store: Vector store override.
300            version_store: Version storage override.
301            settings: Settings override.
302
303        Returns:
304            A :class:`RagnorBuilder` ready for further customization
305            and ``build()``.
306        """
307        return cls(
308            embedder,
309            embedding_model=embedding_model,
310            embedding_dim=embedding_dim,
311            chunker=chunker,
312            store=store,
313            version_store=version_store,
314            settings=settings,
315        )

377    async def build(self) -> Ragnor:
378        """Wire the components into a :class:`Ragnor` facade.
379
380        The defaults are resolved here (not at ``local()`` time) so a
381        ``with_store`` override is also the store the default version
382        store is wired to. The retriever is wired through an injected
383        :class:`ragnor.retrieval.strategies.HybridRetrieval` strategy
384        owning the same store instance the pipeline writes to -- the
385        retriever's settings-driven construction would build a second,
386        disconnected store and a LiteLLM embedder, which the
387        zero-dependency core install cannot do.
388
389        Returns:
390            The wired :class:`Ragnor` facade.
391
392        Raises:
393            ValueError: If no embedder is configured.
394        """
395        if self._embedder is None:
396            raise ValueError(
397                "no embedder configured: pass embedder= to "
398                "RagnorBuilder.local() or call with_embedder() "
399                "(ragnor ships no local embedder in core)"
400            )
401
402        settings = self._settings
403        if settings is None:
404            settings = BuilderSettings(embedding_model=self._embedding_model)
405
406        if self._embedding_dim is not None:
407            register_dimension(settings.embedding_model, self._embedding_dim)
408
409        store = self._store
410        if store is None:
411            store = InMemoryVectorStore(
412                embedding_dim=self._embedding_dim,
413                embedding_fn=self._embedder,
414            )
415
416        chunker = self._chunker
417        if chunker is None:
418            chunker = SlidingWindowChunker(chunk_size=DEFAULT_CHUNK_SIZE, overlap=DEFAULT_OVERLAP)
419
420        version_store = self._version_store
421        if version_store is None:
422            version_store = InMemoryVersionStore(store)
423
424        session_factory = _null_session_factory
425        pipeline = IngestionPipeline(
426            store=store,
427            embedder=self._embedder,
428            chunker=chunker,
429            session_factory=session_factory,
430            settings=settings,
431        )
432        strategy = HybridRetrieval(
433            vector_index=store,
434            lexical_index=store,
435            embedder=self._embedder,
436        )
437        retriever = HybridRetriever(
438            spec=RetrieverSpec(
439                table_name=getattr(store, "table_name", "documents"),
440                metadata_fields=["source", "version_id"],
441            ),
442            settings=settings,
443            session_factory=session_factory,
444            strategy=strategy,
445        )
446        return Ragnor(
447            store=store,
448            pipeline=pipeline,
449            retriever=retriever,
450            version_store=version_store,
451            settings=settings,
452        )

455class Ragnor:
456    """High-level facade over the ragnor seams.
457
458    Exposes the three-line workflow (``ingest`` / ``search`` /
459    ``count``) while keeping the low-level seams reachable through
460    ``.store``, ``.pipeline``, and ``.retriever``.
461    """
462
463    def __init__(  # noqa: PLR0913, PLR0917
464        self,
465        store: VectorStore,
466        pipeline: IngestionPipeline,
467        retriever: HybridRetriever,
468        version_store: VersionStore,
469        settings: RagSettings,
470    ) -> None:
471        """Initialize the facade.
472
473        Args:
474            store: Vector store the chunks are upserted into.
475            pipeline: Ingestion pipeline the facade runs.
476            retriever: Hybrid retriever the facade searches through.
477            version_store: Version storage the pipeline records
478                snapshots in.
479            settings: Settings the pipeline and retriever read.
480        """
481        self.store = store
482        self.pipeline = pipeline
483        self.retriever = retriever
484        self._version_store = version_store
485        self._settings = settings
486
487    async def ingest(
488        self,
489        documents: list[Document],
490        *,
491        purge_on_empty: bool = False,
492    ) -> int:
493        """Ingest documents and return the stored chunk count.
494
495        The documents are wrapped in a
496        :class:`ragnor.ingestion.sources.ListDocumentSource` and run
497        through the ingestion pipeline: an unchanged document set is
498        skipped via snapshot comparison (idempotent), and changed
499        content replaces the active version -- the retired version's
500        chunks are deleted from the store.
501
502        Args:
503            documents: Documents to ingest.
504            purge_on_empty: Treat an empty list as authoritative: the
505                active version is retired and its chunks deleted from
506                the search index.
507
508        Returns:
509            The number of chunks stored, or the stored chunk count
510            when the documents were unchanged.
511        """
512        source = ListDocumentSource(documents)
513        return await self.pipeline.run(
514            source,
515            self._version_store,
516            purge_on_empty=purge_on_empty,
517        )
518
519    async def search(
520        self,
521        query: str,
522        n_results: int = 5,
523        **filters: Any,
524    ) -> list[SearchResult]:
525        """Search for documents matching ``query``.
526
527        Args:
528            query: Natural language search query.
529            n_results: Maximum number of results to return.
530            **filters: Equality filters applied to both search stages
531                (e.g. ``source="policy"``).
532
533        Returns:
534            Search results sorted by RRF score.
535        """
536        return await self.retriever.retrieve(query, n_results=n_results, **filters)
537
538    async def count(self, **filters: Any) -> int:
539        """Count documents in the store.
540
541        Args:
542            **filters: Equality filters matched against document
543                fields.
544
545        Returns:
546            Number of matching documents.
547        """
548        return await self.store.count(**filters)
