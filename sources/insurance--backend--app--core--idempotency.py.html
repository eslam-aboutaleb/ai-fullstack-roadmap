1"""
2Request idempotency cache for the OmniCare chat endpoint.
3
4Industry standard: clients supply an ``Idempotency-Key`` header (UUID or any
5opaque string). If the server has already successfully processed a request
6with that key it returns the cached response instead of executing the agent
7again. This prevents duplicate LLM calls when clients retry on network errors.
8
9If the client omits the header, a deterministic key is derived from
10``sha256(user_id + ":" + message)`` — ensuring that byte-identical retries
11of the same message from the same user are naturally deduplicated.
12
13Design:
14- Pluggable backends: an in-process TTL cache (the default, correct for a
15  single-process deployment) or Redis, which shares one cache across every
16  replica so a retry replays regardless of which instance receives it.
17- Configurable TTL (default: 5 min) and max size (default: 1 000 entries).
18- Entries are evicted in FIFO order when the cache is full, in addition to
19  TTL expiry on access.
20- Thread-safe for async use (single event-loop, no cross-thread mutations).
21- Keys are namespaced by user_id to prevent cross-user cache hits.
22- Request body hash is stored and verified to prevent body-mismatch replays.
23- Per-key asyncio.Lock prevents check-then-act races. With Redis, a claim key
24  extends the same guarantee across processes: the instance that wins the
25  claim computes the response and the others wait for and replay it.
26- Redis problems never fail a request. A missing or unreachable Redis leaves
27  the in-process cache in place, which merely costs the deduplication that
28  Redis existed to provide.
29"""

51# Maximum number of cached responses. When full, the oldest entry is evicted.
52_MAX_SIZE: int = 1_000
53
54# Seconds a cached response is considered valid. After this window the key is
55# expired and the request is processed fresh.
56_TTL_SECONDS: int = 300  # 5 minutes
57
58# Namespace for every key this module writes to Redis, so an operator sharing a
59# Redis database between environments can see (and if necessary clear) exactly
60# what belongs to this service.
61_KEY_PREFIX = "omnicare:idempotency:"
62
63# How long a claim on a key is honoured and how long a losing replica waits for
64# the winner's response before giving up and computing it itself. Both are
65# deliberately short: a stalled winner must not turn a client retry into a hang.

67_CLAIM_POLL_SECONDS = 0.05

80class IdempotencyBackend:
81    """Storage for the chat responses a replay can return."""
82
83    async def lookup(self, key: str) -> dict[str, Any] | None:
84        """Return the stored entry for a key, or None when there is none."""
85        raise NotImplementedError
86
87    async def store(self, key: str, response: Any, request_hash: str) -> None:
88        """Keep a response under a key so a later replay can return it."""
89        raise NotImplementedError
90
91    async def claim(self, key: str) -> bool:
92        """Reserve the right to compute a response nobody has stored yet.
93
94        Returns:
95            bool: True when the caller must compute the response itself. The
96            in-process backend never claims because the per-key asyncio lock
97            already serialises its callers.
98        """
99        return True
100
101    async def await_claimed_result(self, key: str) -> dict[str, Any] | None:
102        """Wait for another replica to publish the response it claimed."""
103        return None
104
105    async def release_claim(self, key: str) -> None:
106        """Give up a claim, whether or not the response was published."""

119class RedisIdempotencyBackend(IdempotencyBackend):
120    """Shared idempotency cache backed by Redis.
121
122    The claim step is what makes this worth the round-trips: without it, two
123    replicas that receive the same retry at the same moment would both miss the
124    cache and both call the model, which is the duplicate work Redis was
125    introduced to prevent. The first replica to claim the key computes and
126    publishes the response; the others wait for it and replay it.
127
128    Every Redis failure degrades instead of raising: a cache that is not
129    answering must not turn a client retry into a 500.
130
131    Args:
132        client: Connected-on-first-use Redis client.
133    """
134
135    def __init__(  # noqa: PLR0913 - one knob per behaviour worth tuning
136        self,
137        client: AsyncRedisClient,
138        *,
139        key_prefix: str = _KEY_PREFIX,
140        ttl_seconds: int = _TTL_SECONDS,
141        claim_ttl_seconds: float = _CLAIM_TTL_SECONDS,
142        poll_seconds: float = _CLAIM_POLL_SECONDS,
143        max_wait_seconds: float = _CLAIM_MAX_WAIT_SECONDS,
144    ) -> None:
145        self._client = client
146        self._key_prefix = key_prefix
147        self._ttl_seconds = ttl_seconds
148        self._claim_ttl_seconds = claim_ttl_seconds
149        self._poll_seconds = poll_seconds
150        self._max_wait_seconds = max_wait_seconds
151
152    def _key(self, key: str) -> str:
153        return f"{self._key_prefix}{key}"
154
155    def _claim_key(self, key: str) -> str:
156        return f"{self._key(key)}:claim"
157
158    async def lookup(self, key: str) -> dict[str, Any] | None:
159        """Return the stored entry for a key, or None on a miss or an outage."""
160        raw = await self._get(self._key(key))
161        if raw is None:
162            return None
163        try:
164            payload = json.loads(raw)
165        except json.JSONDecodeError:
166            logger.warning("Discarding unreadable idempotency payload for key '%s'.", key)
167            return None
168        if not isinstance(payload, dict):
169            logger.warning("Discarding non-object idempotency payload for key '%s'.", key)
170            return None
171        return {
172            "response": payload.get("response"),
173            "request_hash": payload.get("request_hash", ""),
174        }
175
176    async def store(self, key: str, response: Any, request_hash: str) -> None:
177        """Publish a completed response under the key, overwriting a claim."""
178        payload = json.dumps({"response": response, "request_hash": request_hash})
179        await self._set(self._key(key), payload)
180
181    async def claim(self, key: str) -> bool:
182        """Claim the key, or False when another replica already holds it."""
183        try:
184            return await self._client.set(
185                self._claim_key(key), "1", ttl=self._claim_ttl_seconds, nx=True
186            )
187        except RedisUnavailable:
188            # Behave as if the claim succeeded: computing the response locally
189            # is safer than refusing the request.
190            return True
191
192    async def await_claimed_result(self, key: str) -> dict[str, Any] | None:
193        """Poll for the response another replica claimed until it publishes.
194
195        Giving up as soon as the claim disappears means a replica that died
196        mid-computation cannot stall the retry; the waiter computes the
197        response itself instead.
198        """
199        deadline = time.monotonic() + self._max_wait_seconds
200        while True:
201            entry = await self.lookup(key)
202            if entry is not None:
203                return entry
204            try:
205                still_claimed = await self._client.exists(self._claim_key(key))
206            except RedisUnavailable:
207                return None
208            if not still_claimed or time.monotonic() >= deadline:
209                return None
210            await asyncio.sleep(self._poll_seconds)
211
212    async def release_claim(self, key: str) -> None:
213        """Drop the claim so a later retry does not wait on this one."""
214        try:
215            await self._client.delete(self._claim_key(key))
216        except RedisUnavailable:
217            logger.debug("Could not release idempotency claim for key '%s'.", key)
218
219    async def ping(self) -> bool:
220        """Report whether the backing Redis answers; used when wiring startup."""
221        return await self._client.ping()
222
223    async def close(self) -> None:
224        """Drop the connection, leaving the next command to open a new one.
225
226        The startup probe must not leave a connection open in the loop that
227        happened to run it: whatever context issues the first real command is
228        the one that should own the connection.
229        """
230        await self._client.close()
231
232    async def _get(self, key: str) -> str | None:
233        try:
234            return await self._client.get(key)
235        except RedisUnavailable:
236            logger.warning("Redis unreachable while reading key '%s'.", key)
237            return None
238
239    async def _set(self, key: str, payload: str) -> None:
240        try:
241            await self._client.set(key, payload, ttl=self._ttl_seconds)
242        except RedisUnavailable:
243            logger.warning("Redis unreachable while writing key '%s'.", key)

288async def configure_idempotency_storage() -> None:
289    """Select the idempotency backend from the runtime settings.
290
291    Called once from the application lifespan. Redis is used only when the
292    deployment asks for it *and* the configured Redis answers a PING; anything
293    else leaves the in-process cache in place, because a store that is not
294    there must not stop the chat endpoint from accepting requests.
295    """
296    if settings.idempotency_storage == "redis" and settings.redis_enabled:
297        try:
298            backend = build_redis_backend(
299                settings.redis_url,
300                connect_timeout=settings.redis_connect_timeout_seconds,
301                socket_timeout=settings.redis_socket_timeout_seconds,
302            )
303            if await backend.ping():
304                await backend.close()
305                _active[0] = backend
306                logger.info(
307                    "Idempotency responses are cached in Redis at %s and shared "
308                    "across replicas.",
309                    settings.redis_url,
310                )
311                return
312            logger.warning(
313                "Redis at %s did not answer; the idempotency cache stays in-process, "
314                "so a retry that lands on another replica is not deduplicated.",
315                settings.redis_url,
316            )
317        except ValueError as exc:
318            logger.warning(
319                "Configured REDIS_URL could not be used (%s); the idempotency cache "
320                "stays in-process.",
321                exc,
322            )
323    _active[0] = _MemoryIdempotencyBackend()

336def _compute_request_hash(body: Any) -> str:
337    """Compute SHA-256 hash of the canonical JSON representation of a request body."""
338    canonical = _canonical_json(body)
339    return hashlib.sha256(canonical.encode()).hexdigest()

342def make_idempotency_key(user_id: str, message: str) -> str:
343    """
344    Derive a deterministic idempotency key from user and message content.
345
346    Used as the fallback when the client does not supply an explicit
347    ``Idempotency-Key`` header.  The SHA-256 digest ensures:
348    - Identical (user_id, message) pairs always map to the same key.
349    - Different users with the same message text get different keys.
350    - The key is a fixed-length hex string safe for use as a dict key.
351
352    Args:
353        user_id: The authenticated user identifier.
354        message: The raw message text.
355
356    Returns:
357        64-character hex SHA-256 digest.
358    """
359    payload = f"{user_id}:{message}".encode()
360    return hashlib.sha256(payload).hexdigest()

363def get_cached_entry(key: str) -> dict[str, Any] | None:
364    """Return the full cache entry for the key if it exists and has not expired.
365
366    Callers that need the stored request hash must use this rather than
367    :func:`get_cached_response`, which returns only the response body and would make
368    the body-mismatch check unreachable.
369
370    Args:
371        key: The idempotency key to look up.
372
373    Returns:
374        The entry dict (``response``, ``request_hash``, ``cached_at``), or ``None`` if
375        absent or expired.
376    """
377    entry = _cache.get(key)
378    if entry is None:
379        return None
380
381    age = time.monotonic() - entry["cached_at"]
382    if age > _TTL_SECONDS:
383        _cache.pop(key, None)
384        _locks.pop(key, None)
385        logger.debug("Idempotency key '%s' expired after %.1fs.", key, age)
386        return None
387
388    logger.info(
389        "Cache HIT for idempotency key '%s' (age=%.1fs). Returning cached response.",
390        key,
391        age,
392    )
393    return entry
394

396def get_cached_response(key: str) -> dict[str, Any] | None:
397    """
398    Return a cached response if the key exists and has not expired.
399
400    Args:
401        key: The idempotency key to look up.
402
403    Returns:
404        The cached response dict, or ``None`` if absent or expired.
405    """
406    entry = get_cached_entry(key)
407    return None if entry is None else entry["response"]

410def store_response(key: str, response: dict[str, Any], request_hash: str) -> None:
411    """
412    Store a successfully computed response under the given idempotency key.
413
414    When the cache is at capacity the oldest entry is evicted before inserting
415    the new one (FIFO, relying on dict insertion-order guarantee in Python 3.7+).
416
417    Args:
418        key: The idempotency key.
419        response: The agent response dict to cache.
420        request_hash: SHA-256 hash of the request body for integrity checking.
421    """
422    if len(_cache) >= _MAX_SIZE:
423        oldest_key = next(iter(_cache))
424        _cache.pop(oldest_key)
425        _locks.pop(oldest_key, None)
426        logger.debug("Idempotency cache full. Evicted oldest key '%s'.", oldest_key)
427
428    _cache[key] = {
429        "response": response,
430        "request_hash": request_hash,
431        "cached_at": time.monotonic(),
432    }
433    logger.debug("Stored response under idempotency key '%s'.", key)

454def idempotent_endpoint() -> Callable[[F], F]:
455    """
456    Decorator for FastAPI route handlers to abstract idempotency cache checking.
457
458    It expects the route handler to have `idempotency_key` (str), `response`
459    (Response), `current_user_id` (str), and `payload` (request body) as
460    injected keyword arguments. The decorator:
461    1. Namespaces the cache key by user_id.
462    2. Verifies the request body hash matches the cached hash.
463    3. Uses per-key asyncio.Lock to prevent check-then-act races, extended to
464       Redis claim keys so replicas cannot both miss and both run the agent.
465    """
466
467    def decorator(func: F) -> F:
468        @wraps(func)
469        async def wrapper(*args: Any, **kwargs: Any) -> Any:
470            idempotency_key = kwargs.get("idempotency_key")
471            response = kwargs.get("response")
472            current_user_id = kwargs.get("current_user_id")
473            payload = kwargs.get("payload")
474
475            if not idempotency_key and current_user_id and payload is not None:
476                idempotency_key = make_idempotency_key(
477                    current_user_id,
478                    payload.message if hasattr(payload, "message") else str(payload),
479                )
480
481            if idempotency_key and response and current_user_id:
482                namespaced_key = f"{current_user_id}:{idempotency_key}"
483                backend = _backend()
484
485                request_body = (
486                    payload.model_dump() if payload and hasattr(payload, "model_dump") else payload
487                )
488                current_request_hash = (
489                    _compute_request_hash(request_body) if request_body is not None else ""
490                )
491
492                async def replay(entry: dict[str, Any]) -> Any:
493                    """Return a stored response, refusing a key reused with a new body."""
494                    cached_hash = entry.get("request_hash", "")
495                    if cached_hash and current_request_hash and cached_hash != current_request_hash:
496                        logger.warning(
497                            "Idempotency key '%s' reused with different body. Returning 409.",
498                            namespaced_key,
499                        )
500                        raise HTTPException(
501                            status_code=status.HTTP_409_CONFLICT,
502                            detail="Idempotency key reused with a different request body.",
503                        )
504                    response.headers["X-Idempotent-Replayed"] = "true"
505                    response.headers["X-Idempotency-Key"] = idempotency_key
506                    return entry.get("response")
507
508                async with _locks[namespaced_key]:
509                    cached_entry = await backend.lookup(namespaced_key)
510                    if cached_entry is not None:
511                        return await replay(cached_entry)
512
513                    if await backend.claim(namespaced_key):
514                        try:
515                            result = await func(*args, **kwargs)
516                        finally:
517                            await backend.release_claim(namespaced_key)
518                    else:
519                        published = await backend.await_claimed_result(namespaced_key)
520                        if published is not None:
521                            return await replay(published)
522                        result = await func(*args, **kwargs)
523
524                    response.headers["X-Idempotency-Key"] = idempotency_key
525                    stored = result.model_dump() if hasattr(result, "model_dump") else result
526                    await backend.store(namespaced_key, stored, current_request_hash)
527            else:
528                result = await func(*args, **kwargs)
529
530            return result
531
532        return wrapper  # type: ignore
533
534    return decorator
