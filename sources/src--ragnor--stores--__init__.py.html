1"""Vector store abstractions and providers for ragnor.
2
3The package is vector-database agnostic: the
4:class:`VectorStore` ABC defines the legacy fused
5contract, the capability Protocols
6(:class:`VectorIndex`, :class:`LexicalIndex`,
7:class:`DocumentStore`) describe what a backend can
8do structurally, and :class:`HybridVectorStore`
9composes the two search capabilities into the fused
10contract. Each backend lives in its own module that
11registers itself with the provider registry on
12import. Backend modules are loaded lazily (by
13:class:`ragnor.stores.store_factory.RegistryVectorStoreFactory`
14or the ``__getattr__`` below) so every backend's driver
15dependency stays optional.
16"""

45_LAZY_BACKENDS: dict[str, tuple[str, str, str]] = {
46    # Exported name -> (module, attribute, optional extra).
47    "PgVectorStore": ("ragnor.stores.pgvector", "PgVectorStore", "pgvector"),
48    "QdrantStore": ("ragnor.stores.qdrant", "QdrantStore", "qdrant"),

54def __getattr__(name: str) -> Any:
55    """Lazily import a database backend so its driver stays optional."""
56    if name in _LAZY_BACKENDS:
57        module_name, attribute, extra = _LAZY_BACKENDS[name]
58        try:
59            module = importlib.import_module(module_name)
60        except ImportError as exc:
61            raise ImportError(
62                f"The {extra} backend requires the '{extra}' extra: pip install 'ragnor[{extra}]'"
63            ) from exc
64        return getattr(module, attribute)
65    raise AttributeError(f"module {__name__!r} has no attribute {name!r}")
