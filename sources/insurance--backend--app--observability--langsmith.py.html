1"""The LangSmith implementation behind the observability layer's tracing API.
2
3This module is the only one in the application that names the tracing vendor.
4Everything outside the observability package imports ``tracing.py`` instead,
5which delegates here; nothing here is anyone else's business, and a change of
6tracing vendor is a change confined to this module.
7
8What it does: one trace per agent turn, with the calls the turn makes nested
9inside. LangSmith gets its picture of a run from two writers that have to agree
10on the shape of the trace tree. The ``langsmith`` SDK writes the chain run that
11represents one agent turn, and LiteLLM's own LangSmith callback writes the
12model calls that turn makes. The two are stitched together in one place: the
13metadata LiteLLM attaches to every completion. LiteLLM's callback does not
14read the SDK's run context, so without help each model call lands as its own
15orphan trace and the turn it belongs to is invisible; reading the current run
16tree and injecting its id, trace id, and dotted order into the completion
17metadata is what makes the model call a child of the turn
18(:func:`install_llm_trace_context`).
19
20The pieces, in the order startup assembles them:
21
22* :func:`configure_langsmith` exports the environment the SDK and LiteLLM
23  read, once, before the first traced call.
24* :func:`register_langsmith_callback` attaches LiteLLM's callback to the
25  process-wide callback list.
26* :func:`install_llm_trace_context` patches ADK's ``LiteLLMClient`` -- the
27  single seam every model completion passes through -- to inject the trace
28  fields.
29* :func:`trace_turn` and :func:`trace_tool` decorate the agent's turn and
30  tool entry points with the SDK spans.
31
32Everything here follows the same contract as the rest of the observability
33package: tracing is opt-in, and a failure anywhere in it is logged and
34skipped, because a chat request must never fail because of telemetry.
35"""

105def configure_langsmith() -> bool:
106    """Export the LangSmith environment variables the SDK and LiteLLM read.
107
108    Called from the application lifespan before the first traced call, so the
109    SDK's default client -- created lazily on first use -- and LiteLLM's
110    callback both resolve the configured key, project, and endpoint.
111
112    Returns:
113        bool: Whether tracing is active, with the reason logged when it is
114        not.
115    """
116    if not settings.langsmith_tracing_enabled:
117        return False
118    if not settings.langsmith_api_key.strip():
119        logger.warning(
120            "LangSmith tracing is enabled but LANGSMITH_API_KEY is empty; tracing stays off."
121        )
122        return False
123
124    os.environ["LANGSMITH_TRACING"] = "true"
125    os.environ["LANGSMITH_API_KEY"] = settings.langsmith_api_key
126    os.environ["LANGSMITH_PROJECT"] = settings.langsmith_project
127    endpoint = settings.langsmith_endpoint.strip()
128    if endpoint:
129        # LiteLLM's callback and the SDK name the same variable differently.
130        os.environ["LANGSMITH_ENDPOINT"] = endpoint
131        os.environ["LANGSMITH_BASE_URL"] = endpoint
132    workspace_id = settings.langsmith_workspace_id.strip()
133    if workspace_id:
134        os.environ["LANGSMITH_WORKSPACE_ID"] = workspace_id
135    return True

138def register_langsmith_callback() -> Any:
139    """Attach LiteLLM's LangSmith callback to the process-wide callback list.
140
141    A constructed instance rather than the ``"langsmith"`` callback name, for
142    two reasons: the credentials come from settings rather than from whatever
143    the ambient environment happens to hold, and the membership check keeps
144    repeated startups -- the test suite re-runs the application lifespan --
145    from stacking callbacks that would double every trace.
146
147    Returns:
148        LangsmithLogger: The active callback, newly registered or already
149        present.
150    """
151    from litellm.integrations.langsmith import LangsmithLogger
152
153    for callback in litellm.callbacks:
154        if isinstance(callback, LangsmithLogger):
155            return callback
156
157    callback = LangsmithLogger(
158        langsmith_api_key=settings.langsmith_api_key,
159        langsmith_project=settings.langsmith_project,
160        langsmith_base_url=settings.langsmith_endpoint.strip() or None,
161        langsmith_tenant_id=settings.langsmith_workspace_id.strip() or None,
162    )
163    litellm.callbacks = [*litellm.callbacks, callback]
164    logger.info(
165        "LangSmith tracing callback registered (project '%s').",
166        settings.langsmith_project,
167    )
168    return callback

171def _dotted_order_segment(run_id: str) -> str:
172    """Format one LangSmith dotted-order segment for ``run_id``.
173
174    LangSmith writes a segment as a UTC timestamp immediately followed by the
175    run id, and joins a run's ancestry with dots; a child's segment has to
176    follow the parent's so the trace renders as a tree.
177
178    Args:
179        run_id: The run id the segment belongs to.
180
181    Returns:
182        str: The formatted segment.
183    """
184    return datetime.now(UTC).strftime("%Y%m%dT%H%M%S%fZ") + run_id

187def trace_fields_for(run_tree: Any, identity: Mapping[str, str]) -> dict[str, str]:
188    """Return the fields a model call needs to join the current turn's trace.
189
190    These are exactly the fields LiteLLM's LangSmith callback reads back out
191    of a completion's metadata: the model call's own id, the trace and parent
192    run it belongs to, the dotted order that places it inside that trace, and
193    the thread id that groups the user's turns.
194
195    Args:
196        run_tree: The SDK run for the turn in progress, or ``None`` when the
197            call happens outside one -- startup ingestion, the drain workers.
198        identity: The turn identity bound by :func:`bind_turn_context`.
199
200    Returns:
201        dict[str, str]: The metadata fields to merge into the completion, or
202        an empty mapping when there is no turn to join.
203    """
204    if run_tree is None:
205        return {}
206    run_id = str(uuid.uuid4())
207    fields = {
208        "id": run_id,
209        "trace_id": str(run_tree.trace_id),
210        "parent_run_id": str(run_tree.id),
211        "dotted_order": f"{run_tree.dotted_order}.{_dotted_order_segment(run_id)}",
212    }
213    if session_id := identity.get("session_id"):
214        fields["session_id"] = session_id
215    return fields

233def _inject_trace_fields(kwargs: dict[str, Any]) -> None:
234    """Merge the current turn's trace fields into a completion's metadata.
235
236    A failure here costs the run its place in the trace tree and nothing
237    else, so every exception is swallowed: a broken trace context must not
238    break the model call it was labelling.
239
240    Args:
241        kwargs: The completion call's keyword arguments, updated in place.
242    """
243    try:
244        fields = trace_fields_for(_current_run_tree(), _TURN_IDENTITY.get())
245        if not fields:
246            return
247        metadata = dict(kwargs.get("metadata") or {})
248        metadata.update(fields)
249        kwargs["metadata"] = metadata
250    except Exception:
251        logger.debug("LangSmith trace context not attached to this model call", exc_info=True)

254def _patch_completion(client_cls: type, method_name: str) -> None:
255    """Wrap one ``LiteLLMClient`` completion method with trace injection.
256
257    Args:
258        client_cls: The class owning the method -- ADK's ``LiteLLMClient``.
259        method_name: ``"acompletion"`` or ``"completion"``; the pair covers
260            the asynchronous and the streaming paths.
261    """
262    original = getattr(client_cls, method_name)
263
264    if getattr(original, _TRACE_CONTEXT_MARKER, False):
265        return
266
267    if inspect.iscoroutinefunction(original):
268
269        @wraps(original)
270        async def traced_async(self: Any, *args: Any, **kwargs: Any) -> Any:
271            _inject_trace_fields(kwargs)
272            return await original(self, *args, **kwargs)
273
274        traced = traced_async
275    else:
276
277        @wraps(original)
278        def traced_sync(self: Any, *args: Any, **kwargs: Any) -> Any:
279            _inject_trace_fields(kwargs)
280            return original(self, *args, **kwargs)
281
282        traced = traced_sync
283
284    setattr(traced, _TRACE_CONTEXT_MARKER, True)
285    setattr(client_cls, method_name, traced)

288def install_llm_trace_context() -> None:
289    """Make every ADK model call join the LangSmith trace of its turn.
290
291    ADK's ``LiteLlm`` funnels every completion -- asynchronous for the plain
292    chat path, synchronous iteration for the streaming path -- through
293    ``LiteLLMClient``, so patching that pair is the one seam that sees them
294    all. The patch adds no behaviour of its own: it merges the trace fields
295    into metadata the completion already carries and forwards the call
296    unchanged, and it is installed once.
297    """
298    from google.adk.models.lite_llm import LiteLLMClient
299
300    for method_name in ("acompletion", "completion"):
301        _patch_completion(LiteLLMClient, method_name)
302    logger.info("ADK model calls will nest under the LangSmith trace of their turn.")

305def _turn_identity(
306    func: Callable[..., Any], args: tuple[Any, ...], kwargs: dict[str, Any]
307) -> dict[str, str]:
308    """Describe the turn an entry point is starting, for run metadata.
309
310    Read from the entry point's own arguments rather than from context a
311    request may already have set, because the evaluation harness calls the
312    same entry points outside a request. Missing values are simply absent:
313    a trace is still useful without them.
314
315    Args:
316        func: The entry point being called.
317        args: Its positional arguments.
318        kwargs: Its keyword arguments.
319
320    Returns:
321        dict[str, str]: ``user_id``, ``session_id``, and the correlation
322        ``request_id``, whichever are known.
323    """
324    identity: dict[str, str] = {}
325    try:
326        bound = inspect.signature(func).bind_partial(*args, **kwargs)
327    except TypeError:
328        bound = None
329    user_id = bound.arguments.get("user_id") if bound else None
330    if user_id:
331        identity["user_id"] = str(user_id)
332        # One thread per user: the conversation is the unit a reader follows,
333        # and a user's turns belong beside each other in LangSmith's view.
334        identity["session_id"] = str(user_id)
335    if request_id := get_request_id():
336        identity["request_id"] = request_id
337    return identity

340@contextmanager
341def bind_turn_context(identity: Mapping[str, str]) -> Iterator[None]:
342    """Bind the identity of the turn in progress for its duration.
343
344    Args:
345        identity: The turn identity to bind.
346
347    Yields:
348        None.
349    """
350    token = _TURN_IDENTITY.set(dict(identity))
351    try:
352        yield
353    finally:
354        _TURN_IDENTITY.reset(token)

370def _build_turn_span(run_name: str, *, streaming: bool) -> Callable[..., Any]:
371    """Build the SDK span that represents one agent turn.
372
373    Args:
374        run_name: Name the trace is listed under in LangSmith.
375        streaming: Whether the decorated entry point is a generator, whose
376            yields are reduced to the turn's shape rather than logged whole.
377
378    Returns:
379        Callable[..., Any]: The ``traceable``-wrapped callable.
380    """
381    from langsmith import traceable
382
383    kwargs: dict[str, Any] = {
384        "run_type": "chain",
385        "name": run_name,
386        "project_name": settings.langsmith_project,
387        "tags": [settings.app_name, settings.environment],
388        "metadata": {"model": settings.llm_model, "component": "adk-agent"},
389    }
390    if streaming:
391        kwargs["reduce_fn"] = _summarize_stream_frames
392    return traceable(**kwargs)

395def _summarize_stream_frames(frames: list[str]) -> dict[str, Any]:
396    """Compact the SSE frames a streamed turn yielded into trace output.
397
398    A streamed turn's natural output is its frames, and a trace that logged
399    every one of them would be the stream copied verbatim. The frames are
400    re-read into the shape the turn is understood by instead: how many events
401    it carried, which tools ran, and the answer the client received.
402
403    Args:
404        frames: The SSE frames the decorated generator yielded, in order.
405
406    Returns:
407        dict[str, Any]: Event count, the tool names invoked, and the
408        assembled answer text.
409    """
410    summary: dict[str, Any] = {"event_count": len(frames), "tools": [], "response": ""}
411    for frame in frames:
412        try:
413            payload = json.loads(frame.removeprefix("data: ").strip())
414        except (json.JSONDecodeError, TypeError):
415            continue
416        if not isinstance(payload, dict):
417            continue
418        if payload.get("type") == "tool_call":
419            if tool := payload.get("tool"):
420                summary["tools"].append(tool)
421        elif payload.get("type") == "text_delta":
422            summary["response"] += payload.get("content", "")
423    summary["response"] = summary["response"].strip()
424    return summary

427def trace_turn(run_name: str) -> Callable[[F], F]:
428    """Wrap an agent-turn entry point in a LangSmith chain run.
429
430    The chain run is the root of one trace: the model calls the turn makes
431    and the tool calls it triggers nest under it through the run context, so
432    one turn of a conversation is one trace in LangSmith. Tracing off returns
433    the function untouched -- a deployment that has not configured tracing
434    pays nothing and sees no change in behaviour.
435
436    Args:
437        run_name: Name the trace is listed under in LangSmith.
438
439    Returns:
440        Callable[[F], F]: The decorator.
441    """
442
443    def decorator(func: F) -> F:
444        if not langsmith_tracing_active():
445            return func
446        is_stream = inspect.isasyncgenfunction(func)
447        span = _build_turn_span(run_name, streaming=is_stream)
448
449        if is_stream:
450
451            @wraps(func)
452            async def stream_wrapper(*args: Any, **kwargs: Any) -> Any:
453                identity = _turn_identity(func, args, kwargs)
454                with bind_turn_context(identity):
455                    async for chunk in span(func)(
456                        *args, **kwargs, langsmith_extra=_run_extra(identity)
457                    ):
458                        yield chunk
459
460            return stream_wrapper  # type: ignore[return-value]
461
462        @wraps(func)
463        async def wrapper(*args: Any, **kwargs: Any) -> Any:
464            identity = _turn_identity(func, args, kwargs)
465            with bind_turn_context(identity):
466                return await span(func)(*args, **kwargs, langsmith_extra=_run_extra(identity))
467
468        return wrapper  # type: ignore[return-value]
469
470    return decorator

473def trace_tool(name: str) -> Callable[[F], F]:
474    """Wrap a tool handler in a LangSmith tool span.
475
476    The span nests under the agent turn through the run context, so a trace
477    shows which tool ran, with what arguments, and what it returned. Tracing
478    off returns the handler unchanged.
479
480    Args:
481        name: Tool name the span is recorded under.
482
483    Returns:
484        Callable[[F], F]: The decorator.
485    """
486    if not langsmith_tracing_active():
487        return lambda handler: handler
488
489    from langsmith import traceable
490
491    span = traceable(
492        run_type="tool",
493        name=name,
494        project_name=settings.langsmith_project,
495    )
496
497    def decorator(handler: F) -> F:
498        return span(handler)
499
500    return decorator
