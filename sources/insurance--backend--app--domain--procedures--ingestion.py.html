1"""Procedure playbook ingestion over the ragnor pipeline.
2
3"Skills" here means domain procedure playbooks retrieved as documents through a
4second index — not Claude-style on-disk skill bundles (ADK 1.3 has no native
5loader for those), and not the OpenAI Skills API (the app drives models through
6``LiteLlm`` on ADK's chat-completions path rather than the Responses API, so a
7``skill_reference`` is not consumable today). That choice matters because the
8system prompt is already dense and every added rule costs tokens on every turn:
9with the playbooks in the procedures index, a compliance update is an edit to a
10markdown file with no deploy and no prompt change.
11
12Ingestion reuses ragnor's :class:`ragnor.ingestion.IngestionPipeline` unchanged.
13One pipeline instance is shared across the playbooks in the directory and run
14once per file, each run bound to a
15:class:`~app.domain.procedures.store.ProcedureVersionStore` for that one file,
16so the per-source snapshot comparison, version retirement, and chunk deletion
17all stay per-playbook. Nothing here forks or reimplements the pipeline.
18
19The startup gate is read from ``OMNICARE_INGEST_PROCEDURES`` rather than from
20``app.core.config.Settings``: the settings module is owned by another track, and a
21flag read inside this package is the non-invasive way to make boot cost
22optional. Ingestion is idempotent — an unchanged corpus is detected by snapshot
23comparison and skips embedding entirely — so the default is enabled.
24"""
25

61def procedures_ingestion_enabled() -> bool:
62    """Return whether procedure ingestion should run at startup.
63
64    The gate is an environment variable read here rather than a field on
65    ``Settings``: the settings module is owned by another track. Ingestion is
66    idempotent (an unchanged corpus skips embedding), so the default is on.
67
68    Returns:
69        ``True`` unless the variable is explicitly set to a falsy value.
70    """
71    raw = os.environ.get(PROCEDURES_INGESTION_ENV)
72    if raw is None or not raw.strip():
73        return _DEFAULT_PROCEDURES_INGESTION
74    return raw.strip().lower() in _TRUTHY

77def _build_pipeline() -> IngestionPipeline:
78    """Build the shared pipeline bound to the procedure chunk table.
79
80    Returns:
81        An :class:`IngestionPipeline` writing into ``procedure_chunks``.
82    """
83    from app.domain.embeddings import EmbeddingFactory, get_vector_store  # noqa: PLC0415
84
85    return IngestionPipeline(
86        store=get_vector_store(
87            table_name="procedure_chunks",
88            id_field="id",
89        ),
90        embedder=EmbeddingFactory.get_embedding_function(),
91        # The pipeline overwrites each chunk's ``source`` metadata with the
92        # document it actually came from, so the chunker's own source name is
93        # never what ends up stored.
94        chunker=MarkdownSectionChunker(SlidingWindowChunker()),
95        session_factory=async_session_factory,
96        settings=get_settings(),
97        version_id_key="procedure_version_id",
98    )

101async def ingest_procedures(procedures_dir: str | None = None) -> int:
102    """Ingest the procedure playbook corpus into the vector store.
103
104    Delegates each playbook to the ragnor
105    :class:`~ragnor.ingestion.IngestionPipeline`: it loads the file, hashes its
106    content, compares the ingestion snapshot against the active version for
107    that file, and — only when the source changed — retires the old version,
108    deletes its chunks, chunks and embeds the new content, and upserts the
109    chunks, all in one transaction serialized by a PostgreSQL advisory lock
110    keyed by the file path.
111
112    Args:
113        procedures_dir: Directory holding the playbook markdown files. If
114            ``None``, uses ``DEFAULT_PROCEDURES_DIR``.
115
116    Returns:
117        The number of chunks the procedures index holds for this corpus after
118        the run, or ``0`` when the run was skipped. The count is read from the
119        index rather than summed per playbook, because ragnor's skip path
120        reports the store-wide chunk count for an unchanged source: summing
121        that would inflate the number on every warm boot.
122    """
123    if not procedures_ingestion_enabled():
124        logger.info(
125            "Procedure ingestion disabled (%s=false); serving the procedures "
126            "index already present in the database.",
127            PROCEDURES_INGESTION_ENV,
128        )
129        return 0
130
131    directory = Path(procedures_dir) if procedures_dir else DEFAULT_PROCEDURES_DIR
132    if not directory.is_dir():
133        logger.warning("Procedures directory not found at '%s'; skipping ingestion.", directory)
134        return 0
135
136    playbooks = sorted(directory.glob("*.md"))
137    if not playbooks:
138        logger.warning("No procedure playbooks found in '%s'; skipping ingestion.", directory)
139        return 0
140
141    pipeline = _build_pipeline()
142    for playbook in playbooks:
143        # One store per file: the pipeline's find_active/close_active are
144        # per-source, so a shared store would retire one playbook's version
145        # when another changed.
146        store = ProcedureVersionStore(playbook.name)
147        await pipeline.run(
148            FileDocumentSource(str(playbook)),
149            store,
150            lock_key=str(playbook),
151        )
152
153    total = await _corpus_chunk_count()
154    logger.info("Procedure ingestion complete: %d chunks indexed.", total)
155    return total

158async def _corpus_chunk_count() -> int:
159    """Return the number of chunks the procedures index holds."""
160    async with async_session_factory() as session:
161        return (
162            await session.execute(sa_text("SELECT count(*) FROM procedure_chunks"))
163        ).scalar_one()
164
165
166if __name__ == "__main__":
167    asyncio.run(ingest_procedures())
