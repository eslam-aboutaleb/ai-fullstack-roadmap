21async def run_stdio_server(
22    bridge: KnowledgeBridge,
23    *,
24    name: str = "ragnor-knowledge",
25    enable_management_tools: bool = False,
26    client_id: str | None = None,
27    client_id_provider: Callable[[], str | None] | None = None,
28) -> None:
29    """Run MCP server over standard input and standard output.
30
31    Uses the official MCP SDK if available, or falls back to a built-in
32    zero-dependency JSON-RPC 2.0 stdio server for read tools.
33
34    Args:
35        bridge: The KnowledgeBridge providing retrieval and source verification.
36        name: MCP server name.
37        enable_management_tools: Whether to enable management tools.
38    """
39    if _MCPServerClass is not None:
40        server = create_mcp_server(
41            bridge=bridge,
42            name=name,
43            enable_management_tools=enable_management_tools,
44            client_id=client_id,
45            client_id_provider=client_id_provider,
46        )
47        if hasattr(server, "run_stdio_async"):
48            await server.run_stdio_async()
49            return
50
51    # Lightweight built-in JSON-RPC stdio loop (zero external dependencies)
52    await _run_builtin_stdio_loop(
53        bridge,
54        name=name,
55        enable_management=enable_management_tools,
56        client_id=client_id,
57        client_id_provider=client_id_provider,
58    )

61async def _run_builtin_stdio_loop(
62    bridge: KnowledgeBridge,
63    name: str,
64    enable_management: bool,
65    client_id: str | None = None,
66    client_id_provider: Callable[[], str | None] | None = None,
67) -> None:
68    """Zero-dependency JSON-RPC 2.0 stdio protocol handler."""
69    loop = asyncio.get_running_loop()
70    reader = asyncio.StreamReader()
71    protocol = asyncio.StreamReaderProtocol(reader)
72    await loop.connect_read_pipe(lambda: protocol, sys.stdin)
73
74    tool_definitions = [
75        {
76            "name": "search_knowledge",
77            "description": "Execute authorized hybrid search across indexed knowledge collections.",
78            "inputSchema": {
79                "type": "object",
80                "properties": {
81                    "query": {"type": "string"},
82                    "collection": {"type": "string", "default": bridge.default_collection},
83                    "filters": {"type": "object"},
84                    "candidate_k": {"type": "integer", "default": 20},
85                    "final_k": {"type": "integer", "default": 5},
86                    "budget_limit": {"type": "integer"},
87                    "scope": {"type": "string"},
88                },
89                "required": ["query"],
90            },
91        },
92        {
93            "name": "get_source",
94            "description": "Fetch authorized, bounded text span and revision metadata.",
95            "inputSchema": {
96                "type": "object",
97                "properties": {
98                    "source_ref": {"type": "string"},
99                    "document_id": {"type": "string"},
100                    "revision": {"type": "string", "default": "v1"},
101                    "start_offset": {"type": "integer"},
102                    "end_offset": {"type": "integer"},
103                    "collection": {"type": "string", "default": bridge.default_collection},
104                    "scope": {"type": "string"},
105                },
106                "required": ["source_ref"],
107            },
108        },
109        {
110            "name": "knowledge_status",
111            "description": "Inspect sanitized health, document count, and freshness summary.",
112            "inputSchema": {
113                "type": "object",
114                "properties": {
115                    "collection": {"type": "string", "default": bridge.default_collection},
116                    "scope": {"type": "string"},
117                },
118            },
119        },
120    ]
121
122    while not reader.at_eof():
123        line = await reader.readline()
124        if not line:
125            break
126
127        line_str = line.decode("utf-8").strip()
128        if not line_str:
129            continue
130
131        req_id = None
132        try:
133            request = json.loads(line_str)
134            req_id = request.get("id")
135            method = request.get("method")
136            params = request.get("params", {})
137
138            if method == "initialize":
139                response = {
140                    "jsonrpc": "2.0",
141                    "id": req_id,
142                    "result": {
143                        "protocolVersion": "2024-11-05",
144                        "capabilities": {"tools": {}},
145                        "serverInfo": {"name": name, "version": "0.4.0"},
146                    },
147                }
148            elif method == "tools/list":
149                response = {
150                    "jsonrpc": "2.0",
151                    "id": req_id,
152                    "result": {"tools": tool_definitions},
153                }
154            elif method == "tools/call":
155                tool_name = params.get("name")
156                arguments = params.get("arguments", {})
157                # Override any untrusted identity in raw JSON-RPC arguments.
158                arguments = {
159                    **arguments,
160                    "client_id": (
161                        client_id_provider() if client_id_provider is not None else client_id
162                    ),
163                }
164
165                if tool_name == "search_knowledge":
166                    res = await bridge.search_knowledge(**arguments)
167                    out = res.to_dict()
168                elif tool_name == "get_source":
169                    res = await bridge.get_source(**arguments)
170                    out = res.to_dict()
171                elif tool_name == "knowledge_status":
172                    out = await bridge.knowledge_status(**arguments)
173                else:
174                    response = {
175                        "jsonrpc": "2.0",
176                        "id": req_id,
177                        "error": {"code": -32601, "message": f"Method {tool_name!r} not found"},
178                    }
179                    sys.stdout.write(json.dumps(response) + "\n")
180                    sys.stdout.flush()
181                    continue
182
183                response = {
184                    "jsonrpc": "2.0",
185                    "id": req_id,
186                    "result": {
187                        "content": [{"type": "text", "text": json.dumps(out, indent=2)}],
188                        "isError": False,
189                    },
190                }
191            else:
192                response = {
193                    "jsonrpc": "2.0",
194                    "id": req_id,
195                    "error": {"code": -32601, "message": f"Unknown method {method!r}"},
196                }
197
198            sys.stdout.write(json.dumps(response) + "\n")
199            sys.stdout.flush()
200        except Exception as exc:  # noqa: BLE001
201            err_response = {
202                "jsonrpc": "2.0",
203                "id": req_id if "req_id" in locals() else None,
204                "error": {"code": -32603, "message": f"Internal error: {exc}"},
205            }
206            sys.stdout.write(json.dumps(err_response) + "\n")
207            sys.stdout.flush()
