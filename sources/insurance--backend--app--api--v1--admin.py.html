1"""
2Admin namespace: the operator's read surface over data no policyholder may see.
3
4Four endpoints, every one behind ``require_role("admin")``:
5
6  - ``GET /admin/claims``          -- the cross-owner claims queue.
7  - ``GET /admin/embedding-jobs``  -- embedding-outbox depth and dead letters.
8  - ``GET /admin/audit``           -- the paged audit trail.
9  - ``GET /admin/feedback``        -- the feed of down-rated answers.
10
11The routes answer the questions an operator previously could not ask without
12shelling into the database: whose claims are stuck where, how deep the
13embedding queue is and which jobs exhausted their retries, what every user did
14to every entity, and which real answers users marked wrong. There is
15deliberately no admin UI and no write surface here -- this is the read API a
16follow-up console consumes, and nothing a browser can drive.
17
18The response envelopes live in this module rather than in
19``app.schemas.models`` because they describe rows owned by other modules: the
20shared module is the frozen contract surface, and an admin page shape has no
21reason to appear in it.
22"""

336async def list_embedding_jobs(
337    status: str | None = Query(
338        default=None,
339        description="Filter to jobs in this status (e.g. 'failed', 'dead_letter').",
340    ),
341    limit: int = Query(default=_DEFAULT_PAGE_SIZE, ge=1, le=_MAX_PAGE_SIZE),
342    offset: int = Query(default=0, ge=0),
343    _admin: str = Depends(require_role("admin")),
344    session: AsyncSession = Depends(get_db),
345) -> EmbeddingJobPage:
346    """List embedding outbox jobs with queue depth and dead letters.
347
348    Args:
349        status: Optional status filter for the paged job list.
350        limit: Page size for both the job list and the dead-letter list.
351        offset: Number of newest jobs to skip.
352        _admin: The authorization dependency, which resolves the caller's id
353            once it has proven the caller is an admin.
354        session: Async database session.
355
356    Returns:
357        EmbeddingJobPage: One page of jobs, the per-status counts, and the
358        dead-letter list.
359    """
360    stmt = select(EmbeddingJob)
361    if status is not None:
362        stmt = stmt.where(EmbeddingJob.status == status)
363    # claim_id breaks ties so a page ordered on a same-microsecond created_at is
364    # still deterministic across repeated requests.
365    stmt = stmt.order_by(EmbeddingJob.created_at.desc(), EmbeddingJob.claim_id.desc())
366    stmt = stmt.limit(limit).offset(offset)
367
368    jobs = (await session.execute(stmt)).scalars().all()
369
370    # Counted unfiltered: queue depth is the shape of the whole queue, and an
371    # operator filtering to one status still needs to see where the rest stands.
372    counts_rows = (
373        await session.execute(
374            select(EmbeddingJob.status, func.count()).group_by(EmbeddingJob.status)
375        )
376    ).all()
377    counts = dict(counts_rows)
378
379    dead_letter_rows = (
380        (
381            await session.execute(
382                select(EmbeddingJob)
383                .where(EmbeddingJob.status == _DEAD_LETTER_STATUS)
384                .order_by(EmbeddingJob.created_at.desc(), EmbeddingJob.claim_id.desc())
385                .limit(limit)
386            )
387        )
388        .scalars()
389        .all()
390    )
391    dead_letter = [_embedding_job_item(job) for job in dead_letter_rows]
392
393    return EmbeddingJobPage(
394        jobs=[_embedding_job_item(job) for job in jobs],
395        counts=counts,
396        dead_letter=dead_letter,
397        count=len(jobs),
398        limit=limit,
399        offset=offset,
400    )
