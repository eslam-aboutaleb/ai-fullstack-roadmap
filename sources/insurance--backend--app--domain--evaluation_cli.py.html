1"""Command-line entry point for the OmniCare RAG evaluation.
2
3CI and a developer's terminal run the *same* code path through this module.
4When the gate test builds its own runner wiring, the numbers CI checks stop
5being the numbers the CLI reports, and a regression can pass one while failing
6the other; removing that duplication is what this module is for.
7
8    python -m app.domain.evaluation_cli --mode retrieval --dataset stratified
9    python -m app.domain.evaluation_cli --mode end-to-end --dataset unanswerable
10
11``end-to-end`` is opt-in (``--live`` or ``OMNICARE_EVAL_LIVE=1``) because it
12makes one real model call per sample. ``retrieval`` is the default and needs no
13API key, so it is what CI runs.
14"""

68def select_dataset(
69    name: str,
70    category: QuestionCategory | None = None,
71    difficulty: Difficulty | None = None,
72) -> list[EvalSample]:
73    """Return the samples a ``--dataset`` selector names.
74
75    Args:
76        name: One of :data:`DATASET_SELECTORS`.
77        category: Optional category filter, applied before the selector.
78        difficulty: Optional difficulty filter, applied before the selector.
79
80    Returns:
81        The selected samples, in curated order.
82
83    Raises:
84        ValueError: When the selector or a filter value is unknown.
85    """
86    selected = list(EVAL_DATASET)
87    if category is not None:
88        selected = [sample for sample in selected if sample.category == category]
89    if difficulty is not None:
90        selected = [sample for sample in selected if sample.difficulty == difficulty]
91
92    if name == "stratified":
93        return selected
94    if name == "answerable":
95        return [s for s in selected if expected_behavior_of(s) == ExpectedBehavior.ANSWER]
96    if name == "unanswerable":
97        return [s for s in selected if expected_behavior_of(s) == ExpectedBehavior.REFUSE]
98    raise ValueError(f"unknown dataset selector {name!r}; valid: {list(DATASET_SELECTORS)}")

101def parse_category(value: str | None) -> QuestionCategory | None:
102    """Convert a ``--category`` value into a :class:`QuestionCategory`.
103
104    Args:
105        value: The raw flag value, or None when the flag was omitted.
106
107    Returns:
108        The category, or None when no filter was requested.
109
110    Raises:
111        ValueError: When the value is not a known category.
112    """
113    if not value:
114        return None
115    try:
116        return QuestionCategory(value.lower())
117    except ValueError:
118        valid = [category.value for category in QuestionCategory]
119        raise ValueError(f"unknown category {value!r}; valid: {valid}") from None

203def format_summary(result: EvalResult, config: EvalConfig) -> str:
204    """Render the stable, diff-friendly report for a finished run.
205
206    Stability is the point: the field order is fixed, nothing volatile is
207    printed (no wall-clock timestamps, no latency, no duration), and every value
208    is rounded, so two runs of the same inputs produce two reports that diff as
209    empty.
210
211    Args:
212        result: The finished evaluation result.
213        config: The configuration the run executed with.
214
215    Returns:
216        The report text.
217    """
218    retrieval = result.retrieval_metrics or {}
219    answer = result.answer_metrics or {}
220
221    lines = [
222        f"mode: {config.mode}",
223        f"samples: {len(result.per_sample_results or [])}",
224        f"judge: {config.judge}",
225        f"retrieval_n_results: {config.n_results}",
226        f"recall@{config.n_results}: {retrieval.get('recall_at_k', 0.0):.4f}",
227        f"precision@{config.n_results}: {retrieval.get('precision_at_k', 0.0):.4f}",
228        f"mrr: {retrieval.get('mrr', 0.0):.4f}",
229        f"retrieval_errors: {retrieval.get('errors', 0)}",
230    ]
231
232    if answer:
233        answer_source = str(answer.get("answer_source", "unknown"))
234        lines.extend(
235            [
236                f"answer_source: {answer_source}",
237                "answer metrics:",
238                f"avg_faithfulness: {answer.get('avg_faithfulness', 0.0):.4f}",
239                f"avg_relevance: {answer.get('avg_relevance', 0.0):.4f}",
240                f"avg_completeness: {answer.get('avg_completeness', 0.0):.4f}",
241                f"avg_conciseness: {answer.get('avg_conciseness', 0.0):.4f}",
242                f"avg_overall: {answer.get('avg_overall', 0.0):.4f}",
243                "refusal metrics:",
244                f"unanswerable_samples: {answer.get('refusal_expected', 0)}",
245                f"refusal_accuracy: {answer.get('refusal_accuracy', 0.0):.4f}",
246                f"false_refusal_rate: {answer.get('false_refusal_rate', 0.0):.4f}",
247                *_answer_source_note(answer_source),
248            ]
249        )
250
251    return "\n".join(lines)

254def _answer_source_note(answer_source: str) -> list[str]:
255    """Return the provenance note that follows a run's answer metrics.
256
257    An unlabelled number from the retrieved-context substitute is exactly the
258    defect this module was written to remove, so the note is not optional.
259    """
260    if answer_source == ANSWER_SOURCE_CONTEXT_PROXY:
261        return [
262            "",
263            "note: no model was run; the retrieved context was scored in place of the "
264            "model's answer. avg_faithfulness here describes the retriever. Re-run with "
265            "--live to score the agent.",
266        ]
267    return ["", f"note: answers were generated by {answer_source}."]

270def format_samples(result: EvalResult, dataset: list[EvalSample], limit: int = 8) -> str:
271    """Render the per-sample section of the report, weakest sample first.
272
273    Args:
274        result: The finished evaluation result.
275        dataset: The dataset the run scored, supplying the expected behaviour of
276            samples whose result entries are not annotated.
277        limit: Maximum number of samples to list.
278
279    Returns:
280        The per-sample report text, or an empty string when there is nothing to
281        report.
282    """
283    samples = result.per_sample_results or []
284    if not samples:
285        return ""
286
287    expected_by_query = {sample.query: expected_behavior_of(sample) for sample in dataset}
288    ranked = sorted(samples, key=lambda s: (float(s.get("recall", 0.0)), float(s.get("mrr", 0.0))))
289    lines = ["", f"per-sample (weakest {min(limit, len(ranked))} of {len(ranked)}):"]
290    for sample in ranked[:limit]:
291        query = str(sample.get("query"))
292        behavior = expected_by_query.get(query, ExpectedBehavior.ANSWER).value
293        lines.append(
294            f"  [{behavior}] recall={sample.get('recall')} mrr={sample.get('mrr')} "
295            f"query={query!r}"
296        )
297        answer = sample.get("generated_answer")
298        if isinstance(answer, str) and answer.strip():
299            lines.append(f"      answer: {' '.join(answer.split())[:200]}")
300    return "\n".join(lines)

303async def evaluate(args: argparse.Namespace) -> int:
304    """Run the evaluation described by ``args`` and report it.
305
306    Args:
307        args: Parsed command-line arguments.
308
309    Returns:
310        The process exit code: 0 on success, 1 on a floor breach, 2 on a usage
311        error or a run that could not execute.
312    """
313    if args.live:
314        enable_live_evaluation()
315    if args.no_persist:
316        # Read through the environment rather than a module flag so the CLI and
317        # every other caller honour the same switch.
318        os.environ["OMNICARE_EVAL_PERSIST"] = "0"
319
320    try:
321        config = EvalConfig(
322            mode=MODE_ALIASES[args.mode],
323            judge=args.judge,
324            n_results=args.n_results,
325            category=parse_category(args.category),
326            difficulty=Difficulty(args.difficulty) if args.difficulty else None,
327        )
328        dataset = select_dataset(
329            args.dataset,
330            category=config.category,
331            difficulty=config.difficulty,
332        )
333    except ValueError as exc:
334        print(f"error: {exc}", file=sys.stderr)
335        return EXIT_USAGE
336
337    if config.mode == "end-to-end" and not live_evaluation_enabled():
338        # A run that silently scores the context while claiming end-to-end is
339        # the defect this module exists to remove, so the CLI refuses the
340        # ambiguous case instead of producing numbers nobody can trust.
341        print(
342            "error: --mode end-to-end spends one real model call per sample. "
343            f"Pass --live or set {LIVE_EVAL_ENV}=1 to confirm that is intended.",
344            file=sys.stderr,
345        )
346        return EXIT_USAGE
347
348    logger.info(
349        "Running OmniCare evaluation: mode=%s dataset=%s samples=%d",
350        config.mode,
351        args.dataset,
352        len(dataset),
353    )
354
355    try:
356        result = await run_evaluation(config, dataset=dataset)
357    except Exception as exc:
358        logger.error("Evaluation run failed: %s", exc, exc_info=True)
359        print(f"error: evaluation run failed: {exc}", file=sys.stderr)
360        return EXIT_USAGE
361
362    print("\n" + format_summary(result, config))
363    if not args.quiet:
364        print(format_samples(result, dataset))
365
366    if args.output:
367        _write_json(result, args.output)
368
369    if args.no_floors:
370        return EXIT_OK
371
372    floors = LIVE_FLOORS if live_evaluation_enabled() else OFFLINE_FLOORS
373    predicate = None if args.all_samples else await _reachability_predicate(dataset)
374    breaches = check_floors(
375        result,
376        floors,
377        dataset,
378        behavior=ExpectedBehavior.ANSWER,
379        sample_predicate=predicate,
380    )
381    if breaches:
382        print("\nfloors: FAILED")
383        for breach in breaches:
384            print(f"  {breach}")
385        return EXIT_FLOOR_BREACH
386
387    print("\nfloors: OK")
388    return EXIT_OK

391async def _reachability_predicate(
392    dataset: list[EvalSample],
393) -> Callable[[EvalSample], bool] | None:
394    """Return the filter that drops samples the corpus cannot reach lexically.
395
396    Returns None when the policy is not indexed: with no corpus text to compare
397    against, every sample would look unreachable and the floor would be dropped
398    silently, which is worse than checking all of them.
399
400    Args:
401        dataset: The samples the run scored.
402
403    Returns:
404        The predicate, or None when reachability cannot be determined.
405    """
406    section_texts = await load_policy_section_texts()
407    if not section_texts:
408        return None
409    return lambda sample: lexically_reachable(sample, section_texts)

412def _write_json(result: EvalResult, path: str) -> None:
413    """Write the full result to ``path`` as JSON.
414
415    The per-sample context is dropped: it is the bulk of the payload, it is
416    already in the database as ``policy_chunks``, and keeping it makes the file
417    unreadable next to the numbers it documents.
418    """
419    payload: dict[str, Any] = asdict(result)
420    for sample in payload.get("per_sample_results", []):
421        sample.pop("context", None)
422
423    with open(path, "w", encoding="utf-8") as handle:
424        json.dump(payload, handle, indent=2, default=str)
425    print(f"\ndetailed results written to {path}")

428def main(argv: Sequence[str] | None = None) -> int:
429    """Run the evaluation CLI.
430
431    Args:
432        argv: Argument vector, defaulting to ``sys.argv[1:]``.
433
434    Returns:
435        The process exit code.
436    """
437    logging.basicConfig(level=logging.INFO, format="%(levelname)s %(name)s: %(message)s")
438    args = build_parser().parse_args(argv)
