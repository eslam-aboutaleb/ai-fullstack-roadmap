1"""
2Persistence helpers for chat turns and their conversation threads.
3
4A turn is persisted transactionally on its own session: the conversation is
5located by ``session_id`` (created if it does not yet exist) and a matched pair
6of user + assistant ``ConversationMessage`` rows is appended. The two message
7ids are generated here rather than by the database so they can be handed back to
8the caller: the chat and streaming endpoints echo them in their response frames
9and the frontend adopts them in place of any locally-generated id. The feedback
10endpoint authorises on these ids, so the id a client rates must be the id the
11database actually persisted -- not a client-side guess that matches nothing.
12
13The save is fire-and-forget from the caller's perspective: every exception is
14caught and logged so a persistence failure can never break the chat response.
15When it fails, the returned :class:`SavedMessageIds` carries no ids, which the
16callers render as ``null`` in the response while still returning the answer.
17"""
18
19from __future__ import annotations
20
21import logging
22import uuid
23from dataclasses import dataclass
24from datetime import UTC, datetime
25
26from sqlalchemy import select
27
28from app.core.database import async_session_factory
29from app.models.conversations.conversation import Conversation
30from app.models.conversations.conversation_message import ConversationMessage
31
32logger = logging.getLogger(__name__)
33
34_TITLE_MAX_LENGTH = 50
35
36
37@dataclass(frozen=True)
38class SavedMessageIds:
39    """The primary keys the backend assigned to a persisted turn's messages.
40
41    Attributes:
42        user_message_id: The persisted user message's id, or ``None`` when the
43            turn could not be saved.
44        assistant_message_id: The persisted assistant answer's id, or ``None``
45            when the turn could not be saved. This is the id the feedback
46            endpoint authorises against, so a client rates this message.
47    """
48
49    user_message_id: str | None = None
50    assistant_message_id: str | None = None
51
52
53async def save_conversation_turn(  # noqa: PLR0913
54    user_id: str,
55    session_id: str,
56    message: str,
57    response_text: str,
58    sources: list[str],
59    tool_calls: list[dict[str, object]],
60) -> SavedMessageIds:
61    """Persist a chat turn and report the ids the database assigned to it.
62
63    The conversation is looked up by ``session_id`` and created on first sight,
64    using the first characters of the user's message as a title. Two message
65    ids are generated up front and passed explicitly to their
66    ``ConversationMessage`` rows, so the same values are both stored and
67    returned: the caller can tell the client the id the backend will accept for
68    feedback, and the two can never drift apart the way a client-generated id
69    and a database default would.
70
71    Args:
72        user_id: The authenticated user's id (UUID string).
73        session_id: The agent session id that groups this conversation's turns.
74        message: The user's message text.
75        response_text: The assistant's complete answer text.
76        sources: The citations the answer was built from.
77        tool_calls: The tool activity recorded for the turn.
78
79    Returns:
80        SavedMessageIds: The user and assistant message ids, or an instance
81            with both set to ``None`` when persistence failed (the failure is
82            logged, never raised).
83    """
84    try:
85        async with async_session_factory() as db_session:
86            conv = (
87                await db_session.execute(
88                    select(Conversation).where(Conversation.session_id == session_id)
89                )
90            ).scalar_one_or_none()
91
92            if conv is None:
93                title = message[:_TITLE_MAX_LENGTH] or "New conversation"
94                conv = Conversation(
95                    user_id=uuid.UUID(user_id),
96                    session_id=session_id,
97                    title=title,
98                )
99                db_session.add(conv)
100                await db_session.flush()
101
102            user_message_id = uuid.uuid4()
103            assistant_message_id = uuid.uuid4()
104
105            user_message = ConversationMessage(
106                id=user_message_id,
107                conversation_id=conv.id,
108                role="user",
109                content=message,
110                timestamp=datetime.now(UTC),
111                message_metadata=None,
112            )
113            assistant_message = ConversationMessage(
114                id=assistant_message_id,
115                conversation_id=conv.id,
116                role="assistant",
117                content=response_text,
118                timestamp=datetime.now(UTC),
119                message_metadata={
120                    "sources": sources or [],
121                    "tool_calls": tool_calls or [],
122                },
123            )
124
125            db_session.add(user_message)
126            db_session.add(assistant_message)
127            await db_session.commit()
128
129            return SavedMessageIds(
130                user_message_id=str(user_message_id),
131                assistant_message_id=str(assistant_message_id),
132            )
133    except Exception as exc:
134        logger.error("Failed to save conversation turn: %s", exc, exc_info=True)
135        return SavedMessageIds()
