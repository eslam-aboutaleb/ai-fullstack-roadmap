1"""Retrieval strategies over storage capability interfaces.
2
3Strategies compose capability Protocols
4(:class:`ragnor.stores.base.VectorIndex` and
5:class:`ragnor.stores.base.LexicalIndex`) instead of
6the fused :class:`ragnor.stores.base.VectorStore`
7contract, so retrieval topologies the fused contract
8cannot express become possible:
9
10* :class:`DenseRetrieval` -- vector search only;
11* :class:`KeywordRetrieval` -- keyword search only,
12  with **no embedding dependency**;
13* :class:`HybridRetrieval` -- both stages, fused
14  with Reciprocal Rank Fusion, optionally reranked.
15
16Enhancements in this release:
17* Typed :class:`~ragnor.retrieval.types.RetrievalRequest` and
18  :class:`~ragnor.retrieval.types.RetrievalResponse` wrappers with
19  full bidirectional bridges to the legacy dict API.
20* Separate ``candidate_k`` from ``final_k`` so an optional
21  reranker stage has enough candidate documents before final slicing.
22* Backend capability declarations for filter operators, lexical search,
23  document fetch, generation visibility, deletion, and transaction
24  guarantees. Unsupported mandatory constraints fail closed.
25* Push filters directly into each dense and lexical search stage.
26* Scope validation before evidence assembly as defense in depth.
27* Optional host-injected :class:`~ragnor.retrieval.query_transform.QueryTransformer`
28  with hard limits and timeouts.
29"""

87def _coerce_request(
88    query_or_request: str | RetrievalRequest,
89    n_results: int,
90    kwargs: dict[str, Any],
91) -> tuple[RetrievalRequest, dict[str, Any]]:
92    """Convert arguments to a canonical :class:`RetrievalRequest`."""
93    clean_kwargs = dict(kwargs)
94    if isinstance(query_or_request, RetrievalRequest):
95        req = query_or_request
96        return req, clean_kwargs
97
98    candidate_k = clean_kwargs.pop("candidate_k", None)
99    threshold = clean_kwargs.pop("threshold", None)
100    strict = clean_kwargs.pop("strict", False)
101    isolation_scope = clean_kwargs.pop("isolation_scope", None)
102    scope_field = clean_kwargs.pop("scope_field", "tenant_id")
103    mandatory_capabilities = clean_kwargs.pop("mandatory_capabilities", None)
104    expected_generation = clean_kwargs.pop("expected_generation", None)
105    text_field = clean_kwargs.pop("text_field", "text")
106    metadata_fields = clean_kwargs.pop("metadata_fields", None)
107    session = clean_kwargs.pop("session", None)
108
109    req = RetrievalRequest(
110        query=query_or_request,
111        final_k=n_results,
112        candidate_k=candidate_k,
113        threshold=threshold,
114        filters=clean_kwargs,
115        mandatory_capabilities=set(mandatory_capabilities or ()),
116        isolation_scope=isolation_scope,
117        scope_field=scope_field,
118        expected_generation=expected_generation,
119        text_field=text_field,
120        metadata_fields=metadata_fields or [],
121        session=session,
122        strict=strict,
123    )
124    return req, clean_kwargs

127def _validate_scope_defense_in_depth(
128    results: list[dict[str, Any]],
129    req: RetrievalRequest,
130) -> list[dict[str, Any]]:
131    """Validate that every result belongs to the requested isolation scope."""
132    validated: list[dict[str, Any]] = []
133    for row in results:
134        meta = row.get("metadata") or {}
135        row_scope = meta.get(req.scope_field, row.get(req.scope_field))
136        scope_invalid = req.isolation_scope is not None and row_scope != req.isolation_scope
137        generation_invalid = (
138            req.expected_generation is not None
139            and meta.get("generation") != req.expected_generation
140        )
141        if scope_invalid or generation_invalid:
142            msg = (
143                f"Scope violation: result {row.get('id')!r} belongs to scope "
144                f"{row_scope!r}, expected {req.isolation_scope!r}; "
145                f"generation {meta.get('generation')!r}, expected {req.expected_generation!r}."
146            )
147            logger.warning(msg)
148            if req.strict:
149                raise ScopeViolationError(msg)
150            continue
151        validated.append(row)
152    return validated

155def _required_metadata(req: RetrievalRequest) -> list[str]:
156    fields = list(req.metadata_fields)
157    if req.isolation_scope is not None:
158        fields.append(req.scope_field)
159    if req.expected_generation is not None:
160        fields.append("generation")
161    return list(dict.fromkeys(fields))

164def _prepare_retrieval(backend: Any, req: RetrievalRequest) -> dict[str, Any]:
165    caps = get_backend_capabilities(backend)
166    caps.validate_constraints(req.mandatory_capabilities)
167    effective_filters = req.effective_filters()
168    caps.validate_filters(effective_filters)
169    return effective_filters

171
172def _dedupe_and_finalize(
173    results: list[dict[str, Any]], req: RetrievalRequest
174) -> list[dict[str, Any]]:
175    seen_ids: set[str] = set()
176    deduped: list[dict[str, Any]] = []
177    for r in results:
178        rid = str(r.get("id"))
179        if rid not in seen_ids:
180            seen_ids.add(rid)
181            deduped.append(r)
182    validated = _validate_scope_defense_in_depth(deduped, req)
183    return validated[: req.final_k]

186async def _query_vector(embedder: EmbeddingFunction, query: str) -> list[float]:
187    if hasattr(embedder, "embed_query"):
188        return await embedder.embed_query(query)
189    return (await embedder([query]))[0]

192async def _expand_queries(query: str, transformer: QueryTransformer | None) -> list[str]:
193    if transformer is not None:
194        return await transformer.transform(query)
195    return [query]

198class _BaseRetrievalStrategy:
199    """Base class providing typed response wrapping across retrieval strategies."""
200
201    async def retrieve(
202        self,
203        query: str | RetrievalRequest,
204        n_results: int = 5,
205        **kwargs: Any,
206    ) -> list[dict[str, Any]]:
207        raise NotImplementedError
208
209    async def retrieve_typed(self, request: RetrievalRequest) -> RetrievalResponse:
210        """Execute retrieval and return a typed :class:`RetrievalResponse`."""
211        rows = await self.retrieve(request)
212        return RetrievalResponse.from_dicts(
213            rows,
214            query=request.query,
215            candidate_k=request.get_candidate_k(),
216            final_k=request.final_k,
217        )

220def _handle_retrieval_exception(exc: Exception, stage: str, strict: bool) -> list[dict[str, Any]]:
221    if isinstance(exc, (CapabilityUnsupportedError, ScopeViolationError)):
222        raise exc
223    logger.error("%s retrieval failed: %s", stage, exc)
224    if strict:
225        raise RetrievalError(f"{stage} retrieval failed: {exc}") from exc
226    return []

229class DenseRetrieval(_BaseRetrievalStrategy):
230    """Dense-only retrieval over a :class:`VectorIndex`.
231
232    Embeds the query with the injected embedder and
233    calls the index's ``vector_search``.
234    """
235
236    def __init__(
237        self,
238        vector_index: VectorIndex,
239        embedder: EmbeddingFunction,
240        *,
241        query_transformer: QueryTransformer | None = None,
242    ) -> None:
243        self._vector_index = vector_index
244        self._embedder = embedder
245        self._query_transformer = (
246            BoundedQueryTransformer(query_transformer) if query_transformer is not None else None
247        )
248
249    async def retrieve(
250        self,
251        query: str | RetrievalRequest,
252        n_results: int = 5,
253        **kwargs: Any,
254    ) -> list[dict[str, Any]]:
255        """Retrieve documents by vector similarity."""
256        req, remaining_kwargs = _coerce_request(query, n_results, kwargs)
257        if req.threshold is None:
258            raise ValueError("dense retrieval requires a distance threshold; pass threshold=...")
259
260        effective_filters = _prepare_retrieval(self._vector_index, req)
261
262        try:
263            queries = await _expand_queries(req.query, self._query_transformer)
264            all_results: list[dict[str, Any]] = []
265            candidate_k = req.get_candidate_k()
266
267            for q in queries:
268                query_vector = await _query_vector(self._embedder, q)
269                results = await self._vector_index.vector_search(
270                    embedding=query_vector,
271                    n_results=candidate_k,
272                    threshold=req.threshold,
273                    text_field=req.text_field,
274                    metadata_fields=_required_metadata(req),
275                    session=req.session,
276                    **effective_filters,
277                )
278                all_results.extend(results)
279
280            return _dedupe_and_finalize(all_results, req)
281        except Exception as exc:
282            return _handle_retrieval_exception(exc, "Dense", req.strict)

334class HybridRetrieval(_BaseRetrievalStrategy):
335    """Hybrid retrieval over two capability stages, RRF-fused and optionally reranked.
336
337    Supports candidate_k separately from final_k, filter pushdown,
338    backend capability verification, scope validation, and optional query transformation.
339    """
340
341    def __init__(  # noqa: PLR0913, PLR0917
342        self,
343        vector_index: VectorIndex,
344        lexical_index: LexicalIndex,
345        embedder: EmbeddingFunction,
346        reranker: Reranker | None = None,
347        *,
348        query_transformer: QueryTransformer | None = None,
349    ) -> None:
350        self._vector_index = vector_index
351        self._lexical_index = lexical_index
352        self._embedder = embedder
353        self._reranker = reranker
354        self._query_transformer = (
355            BoundedQueryTransformer(query_transformer) if query_transformer is not None else None
356        )
357
358    async def retrieve(
359        self,
360        query: str | RetrievalRequest,
361        n_results: int = 5,
362        **kwargs: Any,
363    ) -> list[dict[str, Any]]:
364        """Retrieve documents by hybrid vector + keyword search."""
365        req, remaining_kwargs = _coerce_request(query, n_results, kwargs)
366        if req.threshold is None:
367            raise ValueError("hybrid retrieval requires a distance threshold; pass threshold=...")
368
369        # Capability validation on both storage backends
370        v_caps = get_backend_capabilities(self._vector_index)
371        l_caps = get_backend_capabilities(self._lexical_index)
372        v_caps.validate_constraints(req.mandatory_capabilities)
373        l_caps.validate_constraints(req.mandatory_capabilities)
374
375        effective_filters = req.effective_filters()
376        v_caps.validate_filters(effective_filters)
377        l_caps.validate_filters(effective_filters)
378
379        try:
380            candidate_k = req.get_candidate_k()
381            queries = [req.query]
382            if self._query_transformer is not None:
383                queries = await self._query_transformer.transform(req.query)
384
385            all_fused: list[dict[str, Any]] = []
386
387            for q in queries:
388                query_vector = await _query_vector(self._embedder, q)
389                vector_results = await self._vector_index.vector_search(
390                    embedding=query_vector,
391                    n_results=candidate_k,
392                    threshold=req.threshold,
393                    text_field=req.text_field,
394                    metadata_fields=_required_metadata(req),
395                    session=req.session,
396                    **effective_filters,
397                )
398                keyword_results = await self._lexical_index.keyword_search(
399                    query=q,
400                    n_results=candidate_k,
401                    text_field=req.text_field,
402                    metadata_fields=_required_metadata(req),
403                    session=req.session,
404                    **effective_filters,
405                )
406                fused = fuse_hybrid(vector_results, keyword_results, candidate_k)
407                all_fused.extend(fused)
408
409            # Deduplicate by id if multi-query generated multiple lists
410            seen_ids: set[str] = set()
411            candidates: list[dict[str, Any]] = []
412            for item in all_fused:
413                item_id = str(item.get("id"))
414                if item_id not in seen_ids:
415                    seen_ids.add(item_id)
416                    candidates.append(item)
417
418            # Rerank candidate list if a reranker is provided
419            if self._reranker is not None and candidates:
420                candidates = await self._reranker.rerank(req.query, candidates)
421
422            # Scope defense-in-depth validation
423            validated = _validate_scope_defense_in_depth(candidates, req)
424
425            # Return sliced to final_k
426            return validated[: req.final_k]
427
428        except CapabilityUnsupportedError:
429            raise
430        except ScopeViolationError:
431            raise
432        except Exception as exc:
433            logger.error("Hybrid retrieval failed: %s", exc)
434            if req.strict:
435                raise RetrievalError(f"Hybrid retrieval failed: {exc}") from exc
436            return []
