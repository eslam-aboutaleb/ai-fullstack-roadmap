1"""Policy ingestion adapter over ragnor.
2
3The OmniCare policy version tables (``policies``,
4``policy_versions``, ``policy_ingestion_meta``) are
5application-side storage: ragnor's
6:class:`ragnor.ingestion.IngestionPipeline` orchestrates
7load/hash/compare/retire/chunk/embed/upsert and talks to a
8:class:`ragnor.ingestion.VersionStore` protocol, which
9:class:`PolicyVersionStore` implements here against the
10OmniCare tables. This module is the permanent home for the binding.
11
12The stored snapshot carries ``chunker_version="v1"`` — the
13historical value — even though the chunker itself reports
14``section-aware-v1``. The wrapper below pins the snapshot
15version so stored snapshots keep comparing equal; a mismatch
16would re-ingest every policy on the next run.
17"""
18

58"""Chunker version recorded in ingestion snapshots.
59
60The :class:`MarkdownSectionChunker` reports
61``"section-aware-v1"``, but the snapshots stored by
62existing deployments carry ``chunker_version="v1"``.
63The adapter wraps the chunker so the snapshot
64comparison stays stable and existing policies are not
65spuriously re-ingested. This is the ``chunker_version``
66trap flagged across ingestion and chunking; this module
67is the permanent home for the mapping.

71class _HistoricalSnapshotChunker:
72    """Chunker adapter that reports the historical snapshot version.
73
74    Delegates chunking to the wrapped chunker but reports
75    the snapshot version the version table already stores,
76    so ``build_snapshot`` produces a snapshot that compares
77    equal to the stored one.
78    """
79
80    def __init__(self, chunker: Chunker, snapshot_version: str) -> None:
81        self._chunker = chunker
82        self._snapshot_version = snapshot_version
83
84    def chunk(self, text: str) -> list[Chunk]:
85        return self._chunker.chunk(text)
86
87    @property
88    def version(self) -> str:
89        return self._snapshot_version
90
91    @property
92    def config(self) -> dict[str, Any]:
93        return self._chunker.config

96class PolicyVersionStore:
97    """``VersionStore`` implementation against the Policy/PolicyVersion models.
98
99    Every method joins the pipeline's transaction: the
100    session is the pipeline's, so version retirement,
101    chunk deletion, and version insertion commit together
102    with the chunk upsert.
103    """
104
105    _PRODUCT = "omnicare_base"
106    _JURISDICTION = "US"
107
108    async def _find_or_create_policy(self, session: Any) -> Policy:
109        policy = (
110            await session.execute(
111                select(Policy).where(
112                    Policy.product == self._PRODUCT,
113                    Policy.jurisdiction == self._JURISDICTION,
114                )
115            )
116        ).scalar_one_or_none()
117        if policy is None:
118            policy = Policy(product=self._PRODUCT, jurisdiction=self._JURISDICTION)
119            session.add(policy)
120            await session.flush()
121        return policy
122
123    async def _find_active_version(self, session: Any, policy: Policy) -> PolicyVersion | None:
124        return (
125            await session.execute(
126                select(PolicyVersion)
127                .where(
128                    PolicyVersion.policy_id == policy.policy_id,
129                    PolicyVersion.effective_to.is_(None),
130                )
131                .order_by(PolicyVersion.effective_from.desc())
132                .limit(1)
133            )
134        ).scalar_one_or_none()
135
136    async def find_active(self, session: Any) -> VersionRecord | None:
137        policy = await self._find_or_create_policy(session)
138        active_version = await self._find_active_version(session, policy)
139        if active_version is None:
140            return None
141        return VersionRecord(
142            version_id=str(active_version.version_id),
143            source_id=str(policy.policy_id),
144            snapshot=IngestionSnapshot(
145                source_hash=active_version.source_hash,
146                embedding_model=active_version.embedding_model or "",
147                embedding_dim=active_version.embedding_dim or 0,
148                chunker_version=active_version.chunker_version or "",
149                chunk_size=active_version.chunk_size or 0,
150                overlap=active_version.overlap or 0,
151                retrieval_schema_version=(active_version.retrieval_schema_version or ""),
152            ),
153        )
154
155    async def close_active(self, session: Any) -> None:
156        policy = await self._find_or_create_policy(session)
157        active_version = await self._find_active_version(session, policy)
158        if active_version is not None:
159            active_version.effective_to = datetime.now(UTC)
160            await session.flush()
161
162    async def create_version(self, session: Any, snapshot: IngestionSnapshot) -> VersionRecord:
163        policy = await self._find_or_create_policy(session)
164        version = PolicyVersion(
165            policy_id=policy.policy_id,
166            version="current",
167            effective_from=datetime.now(UTC),
168            source_hash=snapshot.source_hash,
169            embedding_model=snapshot.embedding_model,
170            embedding_dim=snapshot.embedding_dim,
171            chunker_version=snapshot.chunker_version,
172            chunk_size=snapshot.chunk_size,
173            overlap=snapshot.overlap,
174            retrieval_schema_version=snapshot.retrieval_schema_version,
175        )
176        session.add(version)
177        await session.flush()
178        return VersionRecord(
179            version_id=str(version.version_id),
180            source_id=str(policy.policy_id),
181            snapshot=snapshot,
182        )
183
184    async def delete_chunks(self, session: Any, version_id: str) -> None:
185        await session.execute(
186            sa_text("DELETE FROM policy_chunks WHERE policy_version_id = :vid"),
187            {"vid": version_id},
188        )

191def chunk_policy_document(filepath: str) -> list[dict[str, Any]]:
192    """Parse a Markdown policy file into structured, overlapping chunks.
193
194    .. deprecated::
195        Use the ragnor ingestion pipeline with a
196        :class:`ragnor.chunking.MarkdownSectionChunker`
197        instead. The pipeline owns ingestion now; this thin
198        wrapper exists only for callers that want the chunk
199        dicts from a file path on disk -- the unit tests that
200        pin the chunker's section boundaries, mainly. Nothing
201        in the application's ingestion path calls it, so it
202        carries no snapshot or version state and cannot be
203        relied on to stay in step with what
204        :func:`ingest_policy` writes to the database.
205
206    Args:
207        filepath: Path to the Markdown policy file.
208
209    Returns:
210        List of chunk dicts with keys: id, text, metadata.
211        Metadata includes: chunk_id, section, source,
212        chunk_index, sub_chunk_index.
213    """
214    try:
215        with open(filepath, encoding="utf-8") as f:
216            content = f.read()
217    except FileNotFoundError:
218        logger.error("Policy document not found at: %s", filepath)
219        return []
220
221    source_filename = filepath.rsplit("/", maxsplit=1)[-1]
222    chunker = MarkdownSectionChunker(
223        SlidingWindowChunker(),
224        source_name=source_filename,
225    )
226    return [
227        {
228            "id": chunk.id,
229            "text": chunk.text,
230            "metadata": chunk.metadata,
231        }
232        for chunk in chunker.chunk(content)
233    ]

236async def ingest_policy(policy_path: str | None = None) -> int:
237    """Ingest the policy document into the vector store.
238
239    Delegates to the ragnor :class:`IngestionPipeline`:
240    loads the policy file, hashes its content, compares
241    the ingestion snapshot against the active version,
242    and -- when the source changed -- retires the old
243    version, deletes its chunks, chunks and embeds the
244    new content, validates every embedding, and upserts
245    the chunks with their policy and version ownership,
246    all in one transaction serialized by a PostgreSQL
247    advisory lock keyed by the source path.
248
249    The chunk metadata carries ``policy_id`` and
250    ``policy_version_id`` — the upsert column derivation
251    reads metadata keys only, so the ownership columns are
252    stamped inside the metadata dict by the pipeline.
253
254    Args:
255        policy_path: Path to the policy document. If None,
256            uses ``settings.policy_file_path``.
257
258    Returns:
259        The number of chunks ingested, or the stored chunk
260        count when the source was unchanged.
261    """
262    settings = get_settings()
263    policy_path = policy_path or settings.policy_file_path
264
265    if not policy_path:
266        logger.warning("No policy_path configured; skipping ingestion.")
267        return 0
268
269    from app.domain.embeddings import EmbeddingFactory, get_vector_store  # noqa: PLC0415
270
271    source = FileDocumentSource(policy_path)
272    source_name = policy_path.rsplit("/", maxsplit=1)[-1]
273    chunker = _HistoricalSnapshotChunker(
274        MarkdownSectionChunker(SlidingWindowChunker(), source_name=source_name),
275        snapshot_version=POLICY_CHUNKER_SNAPSHOT_VERSION,
276    )
277    pipeline = IngestionPipeline(
278        store=get_vector_store(
279            table_name=POLICY_RETRIEVER_SPEC.table_name,
280            id_field=POLICY_RETRIEVER_SPEC.id_field,
281        ),
282        embedder=EmbeddingFactory.get_embedding_function(),
283        chunker=chunker,
284        session_factory=async_session_factory,
285        settings=settings,
286        version_id_key="policy_version_id",
287        source_id_key="policy_id",
288    )
289    return await pipeline.run(source, PolicyVersionStore(), lock_key=policy_path)
