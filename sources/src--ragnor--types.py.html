1"""Shared value types for ragnor.
2
3Dataclasses are the canonical boundary across chunking, embedding,
4storage, and retrieval. They replace the raw ``dict[str, Any]``
5plumbing used by earlier iterations; the dict-based store API keeps
6working during the transition, and later plans adapt it to these
7types.
8"""
9
10from __future__ import annotations
11
12from dataclasses import dataclass, field
13from typing import Any
14
15
16@dataclass
17class Chunk:
18    """A text fragment produced by a chunker."""
19
20    id: str
21    text: str
22    metadata: dict[str, Any] = field(default_factory=dict)
23
24
25@dataclass
26class Document:
27    """A document with an optional embedding vector."""
28
29    id: str
30    text: str
31    metadata: dict[str, Any] = field(default_factory=dict)
32    embedding: list[float] | None = None
33
34
35@dataclass
36class SearchResult:
37    """A single hit from a (hybrid) search.
38
39    Mirrors the dict keys the pgvector store's hybrid search returns
40    today: ``document``, ``metadata``, ``distance``,
41    ``keyword_score``, ``_rrf_score``, and ``id``. ``distance`` is
42    ``None`` for keyword-only hits and ``keyword_score`` is ``None``
43    for vector-only hits.
44    """
45
46    id: str | None
47    document: str
48    metadata: dict[str, Any] = field(default_factory=dict)
49    distance: float | None = None
50    keyword_score: float | None = None
51    rrf_score: float = 0.0
52
53
54class RetrievalError(Exception):
55    """Typed error for retrieval failures (used by strict-mode search)."""
