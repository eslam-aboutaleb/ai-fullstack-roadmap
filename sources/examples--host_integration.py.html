1"""Host-application integration template for ragnor.
2
3This example wires ragnor's seams the way a host application
4does: a domain embedding provider registered through
5``embedding_registry``, a custom vector backend registered
6through ``vector_store_registry`` (backed by a shared
7in-memory index), the ingestion pipeline over a bundled
8document, hybrid retrieval, and the retrieval evaluation
9harness.
10
11Run it with::
12
13    python examples/host_integration.py
14
15Everything runs in process: no network, no database, no API
16keys, no external services.

63class DomainSettings:
64    """Domain settings satisfying the :class:`RagSettings` protocol.
65
66    ragnor ships no concrete settings base: the
67    :class:`RagSettings` protocol is the contract, and a host
68    subclasses its own settings base (e.g. pydantic
69    ``BaseSettings``) and adds these fields. Override the
70    domain fields (embedding model, provider names, distance
71    threshold, ``database_url``) to adapt the template.
72    ``database_url`` stays empty for the offline demo — point
73    it at PostgreSQL when using the pgvector backend. Validate
74    at start-up with ``isinstance(settings, RagSettings)``.
75    """
76
77    embedding_model: str = DOMAIN_EMBEDDING_MODEL
78    embedding_provider: str = "domain-provider"
79    rag_distance_threshold: float = 1.3
80    vector_store_provider: str = "domain-backend"
81    openai_api_key: str = ""
82    openai_api_base: str = ""
83    embedding_drain_interval_seconds: float = 5.0
84    embedding_drain_batch_size: int = 10
85    job_stale_after_seconds: float = 300.0
86    worker_id: str = "domain-worker"
87    ingest_on_startup: bool = False
88    database_url: str = ""

91def _fake_embed(text: str) -> list[float]:
92    """Deterministic hash embedding (no API key required)."""
93    digest = hashlib.sha256(text.encode("utf-8")).digest()
94    vector: list[float] = []
95    while len(vector) < FALLBACK_DIMENSION:
96        digest = hashlib.sha256(digest).digest()
97        vector.extend(byte / 255.0 for byte in digest)
98    return vector[:FALLBACK_DIMENSION]

101class FakeDomainEmbedder:
102    """Deterministic domain embedder implementing ``EmbeddingFunction``.
103
104    The user-supplied embedder: every method delegates to
105    :func:`_fake_embed`, so embeddings are deterministic and
106    need no API key.
107    """
108
109    async def __call__(self, input: list[str]) -> list[list[float]]:
110        """Embed a batch of texts.
111
112        Args:
113            input: Texts to embed.
114
115        Returns:
116            One embedding vector per input text.
117        """
118        return [_fake_embed(text) for text in input]
119
120    async def embed_query(self, input: str | list[str]) -> list[float]:
121        """Embed a query.
122
123        Args:
124            input: Query text (or a list whose first element is
125                the query).
126
127        Returns:
128            The query embedding vector.
129        """
130        return _fake_embed(input if isinstance(input, str) else input[0])
131
132    async def embed_documents(self, input: list[str]) -> list[list[float]]:
133        """Embed a batch of documents.
134
135        Args:
136            input: Document texts to embed.
137
138        Returns:
139            One embedding vector per document.
140        """
141        return [_fake_embed(text) for text in input]

169@vector_store_registry.register("domain-backend")
170class DomainVectorBackend(VectorStore):
171    """Demo vector backend over a shared in-memory index.
172
173    One shared :class:`InMemoryVectorStore` per table name
174    keeps the pipeline's writes visible to the retriever's
175    reads within this process; a production backend shares a
176    database server instead, and the host wiring code is
177    identical. ``session``, ``strict``, and ``settings`` are
178    accepted for interface compatibility.
179    """
180
181    _indexes: dict[str, InMemoryVectorStore] = {}
182
183    def __init__(  # noqa: PLR0913, PLR0917
184        self,
185        table_name: str = "documents",
186        id_field: str = "id",
187        embedding_dim: int | None = None,
188        session_factory: Any = None,
189        settings: RagSettings | None = None,
190        embedding_fn: EmbeddingFunction | None = None,
191    ) -> None:
192        """Initialize the backend, reusing the shared index for ``table_name``.
193
194        Args:
195            table_name: Logical table name; the shared index is
196                keyed by it.
197            id_field: Name of the primary key field.
198            embedding_dim: Expected embedding dimensionality.
199            session_factory: Accepted for interface compatibility.
200            settings: Accepted for interface compatibility.
201            embedding_fn: Embedding function for automatic query
202                embedding.
203        """
204        backend = DomainVectorBackend._indexes.get(table_name)
205        if backend is None:
206            backend = InMemoryVectorStore(
207                table_name=table_name,
208                id_field=id_field,
209                embedding_dim=embedding_dim,
210                embedding_fn=embedding_fn,
211            )
212            DomainVectorBackend._indexes[table_name] = backend
213        self._backend = backend
214        self.table_name = table_name
215        self.id_field = id_field
216        self.embedding_dim = embedding_dim
217        self._embedding_fn = embedding_fn
218
219    async def hybrid_search(  # noqa: PLR0913, PLR0917
220        self,
221        query: str,
222        embedding: list[float],
223        n_results: int,
224        threshold: float,
225        text_field: str = "text",
226        metadata_fields: list[str] | None = None,
227        session: Any = None,
228        strict: bool = False,
229        **filters: Any,
230    ) -> list[dict[str, Any]]:
231        """Hybrid search delegated to the shared in-memory index.
232
233        Args:
234            query: Natural language search query.
235            embedding: Query embedding vector.
236            n_results: Maximum number of results to return.
237            threshold: Maximum distance for the vector search part.
238            text_field: Name of the field containing the document text.
239            metadata_fields: Fields to include in the result metadata.
240            session: Accepted for interface compatibility.
241            strict: Accepted for interface compatibility.
242            **filters: Equality filters matched against document fields.
243
244        Returns:
245            Result dicts sorted by RRF score.
246        """
247        return await self._backend.hybrid_search(
248            query=query,
249            embedding=embedding,
250            n_results=n_results,
251            threshold=threshold,
252            text_field=text_field,
253            metadata_fields=metadata_fields,
254            session=session,
255            strict=strict,
256            **filters,
257        )

391@dataclass
392class HostApp:
393    """The wired host application."""
394
395    settings: DomainSettings
396    store: VectorStore
397    embedder: EmbeddingFunction
398    pipeline: IngestionPipeline
399    retriever: HybridRetriever
400    version_store: MemoryVersionStore
401    harness: RagEvaluationHarness
402
403    async def ingest(self, source: DocumentSource) -> int:
404        """Ingest a document source.
405
406        Args:
407            source: Document source to ingest.
408
409        Returns:
410            The number of chunks stored.
411        """
412        return await self.pipeline.run(source, self.version_store)
413
414    async def retrieve(self, query: str, n_results: int = 5) -> list[dict[str, Any]]:
415        """Retrieve documents matching a query.
416
417        Args:
418            query: Natural language search query.
419            n_results: Maximum number of results to return.
420
421        Returns:
422            Result dicts sorted by RRF score.
423        """
424        return as_dicts(await self.retriever.retrieve(query, n_results=n_results))
425
426    async def evaluate(self) -> RetrievalMetrics:
427        """Evaluate retrieval quality against the bundled test set.
428
429        The harness adapter surfaces the chunker's
430        position-based ``chunk_id`` (carried in each
431        result's metadata) as the result id: the store
432        keys rows by UUID, while the curated test set is
433        keyed by the stable ``chunk_N`` ids.
434
435        Returns:
436            Aggregated retrieval metrics.
437        """
438        return await self.harness.evaluate(self._retrieve_by_chunk_id)
439
440    async def _retrieve_by_chunk_id(
441        self,
442        query: str,
443    ) -> list[dict[str, Any]]:
444        """Retrieve result dicts keyed by position-based chunk id.
445
446        Args:
447            query: Natural language search query.
448
449        Returns:
450            Result dicts whose ``id`` is the chunker's
451            ``chunk_id`` when the metadata carries one.
452        """
453        rows = await self.retrieve(query)
454        for row in rows:
455            chunk_id = row.get("metadata", {}).get("chunk_id")
456            if chunk_id:
457                row["id"] = chunk_id
458        return rows

461def build_host(table_name: str = "domain_chunks") -> HostApp:
462    """Wire the domain application together.
463
464    ``get_embedding_function`` resolves ``"domain-provider"``
465    through ``embedding_registry`` and ``get_vector_store``
466    resolves ``"domain-backend"`` through ``vector_store_registry``;
467    a host may alternatively construct its embedder directly
468    (``FakeDomainEmbedder()``) and pass it to ``IngestionPipeline``
469    — both paths satisfy the ``EmbeddingFunction`` protocol.
470
471    Args:
472        table_name: Logical table name for the chunk index.
473
474    Returns:
475        The wired :class:`HostApp`.
476    """
477    settings = DomainSettings()
478    embedder = get_embedding_function(settings)
479    embedding_dim = get_embedding_dimension(settings.embedding_model)
480    store = get_vector_store(
481        table_name=table_name,
482        embedding_dim=embedding_dim,
483        settings=settings,
484        embedding_fn=embedder,
485    )
486    chunker = MarkdownSectionChunker(SlidingWindowChunker())
487    version_store = MemoryVersionStore()
488    pipeline = IngestionPipeline(
489        store=store,
490        embedder=embedder,
491        chunker=chunker,
492        session_factory=null_session_factory,
493        settings=settings,
494    )
495    retriever = HybridRetriever(
496        spec=RetrieverSpec(
497            table_name=table_name,
498            metadata_fields=["chunk_id", "section", "source"],
499        ),
500        settings=settings,
501        session_factory=null_session_factory,
502    )
503    harness = RagEvaluationHarness(TEST_SET)
504    return HostApp(
505        settings=settings,
506        store=store,
507        embedder=embedder,
508        pipeline=pipeline,
509        retriever=retriever,
510        version_store=version_store,
511        harness=harness,
512    )

515async def run_demo() -> HostApp:
516    """Run the offline demo: ingest, re-ingest, retrieve, and evaluate.
517
518    Returns:
519        The wired :class:`HostApp`, ready to serve queries.
520    """
521    app = build_host()
522    document = Path(__file__).with_name("sample_document.md")
523    source = FileDocumentSource(str(document))
524
525    count = await app.ingest(source)
526    print(f"ingested {count} chunks from {document.name}")
527
528    count = await app.ingest(source)
529    print(
530        f"re-ingest returned {count} stored chunks "
531        "(unchanged source skipped via snapshot comparison)"
532    )
533
534    results = await app.retrieve("What is the refund policy?", n_results=3)
535    print("top results for 'What is the refund policy?':")
536    for result in results:
537        print(f"  [{result['id']}] {result['document'][:70]}")
538
539    metrics = await app.evaluate()
540    print(
541        f"retrieval quality: recall@5={metrics.recall_at_k:.4f} "
542        f"precision@5={metrics.precision_at_k:.4f} mrr={metrics.mrr:.4f} "
543        f"errors={metrics.errors}"
544    )
545    print("host integration demo complete: retrieval quality validated, ready to serve")
546    return app
