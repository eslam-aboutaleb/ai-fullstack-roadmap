51def _error_record(error: Any) -> dict[str, Any]:
52    """Do not retain exception messages, which may contain source credentials."""
53    if isinstance(error, dict):
54        # Connector-owned structured errors remain useful without copying
55        # arbitrary provider messages or payloads into public diagnostics.
56        record: dict[str, Any] = {
57            key: value
58            for key, value in error.items()
59            if key in {"reason", "error_type", "code"} and isinstance(value, str)
60        }
61        for key in ("status_code", "retry_after"):
62            value = error.get(key)
63            if key in error and (value is None or isinstance(value, (int, float))):
64                record[key] = value
65        return record
66    return {"reason": "source_failure", "error_type": type(error).__name__}

207async def collect_pages(
208    connector: SourceConnector, max_pages: int | None = None, page_size: int | None = None
209) -> LoadResult:
210    """Collect from the beginning; never bless a partial or failed traversal."""
211    if max_pages is not None and max_pages < 1:
212        raise ValueError("max_pages must be positive")
213    if page_size is not None and page_size < 1:
214        raise ValueError("page_size must be positive")
215    result = LoadResult(complete=False)
216    cursor = None
217    seen: list[Any] = []
218    while True:
219        try:
220            page = await connector.load_page(cursor=cursor, limit=page_size)
221        except Exception as exc:
222            result.errors.append(_error_record(exc))
223            result.next_cursor = cursor
224            result.has_more = True
225            result.status = SourceStatus.INCOMPLETE
226            return result
227        result.extend(page)
228        result.skipped.extend(page.skipped)
229        result.errors.extend(_error_record(error) for error in page.errors)
230        result.pages += 1
231        result.total_items += page.total_items
232        result.incremental = result.incremental or page.incremental
233        result.checkpoint = page.checkpoint
234        result.next_cursor = page.next_cursor
235        result.has_more = page.has_more
236        result.metadata.update(page.metadata)
237        if page.errors or page.status not in {"ok", "empty", "success", "empty_success"}:
238            result.status = (
239                SourceStatus.INCOMPLETE
240                if page.status
241                in {
242                    "ok",
243                    "empty",
244                    "success",
245                    "empty_success",
246                }
247                else page.status
248            )
249            return result
250        if not page.has_more:
251            # A terminal page reached after a cursor cannot itself be complete;
252            # the collector can certify the traversal that began at None.
253            result.complete = cursor is not None or page.complete
254            result.status = SourceStatus.OK if result else SourceStatus.EMPTY
255            if not result.complete:
256                result.status = SourceStatus.INCOMPLETE
257            return result
258        token = page.next_cursor
259        if token is None or token == cursor or token in seen:
260            result.errors.append(
261                {"reason": "invalid_continuation", "error_type": "IncompleteScanError"}
262            )
263            result.status = SourceStatus.INCOMPLETE
264            return result
265        if max_pages is not None and result.pages >= max_pages:
266            result.status = SourceStatus.INCOMPLETE
267            return result
268        seen.append(token)
269        cursor = token

272def extract_documents(  # noqa: PLR0913, PLR0917
273    data: bytes,
274    source_name: str,
275    external_id: str | None,
276    metadata: dict[str, Any] | None,
277    provenance: dict[str, Any] | None,
278    extractor: Any = None,
279) -> LoadResult:
280    """Extract text or page documents, preserving the actual extraction config."""
281    from ragnor.ingestion.extractors import (  # noqa: PLC0415
282        HTMLExtractor,
283        PDFExtractor,
284        TextExtractor,
285    )
286    from ragnor.embeddings._safe_config import safe_config  # noqa: PLC0415
287
288    metadata = dict(metadata or {})
289    provenance = dict(provenance or {})
290    media_type = str(metadata.get("media_type", "")).split(";", 1)[0].lower()
291    suffix = PurePosixPath(source_name.split("?", 1)[0]).suffix.lower()
292    if extractor is None:
293        if media_type == "application/pdf" or suffix == ".pdf":
294            extractor = PDFExtractor()
295        elif media_type in {"text/html", "application/xhtml+xml"} or suffix in {".html", ".htm"}:
296            extractor = HTMLExtractor()
297        else:
298            extractor = TextExtractor()
299    identity = str(getattr(extractor, "id", type(extractor).__name__))
300    metadata["extraction"] = {
301        "id": identity,
302        "version": str(getattr(extractor, "version", "0")),
303        "options": safe_config(getattr(extractor, "options", {})),
304    }
305    result = LoadResult()
306    try:
307        if identity == "text":
308            # Decode using the actual text encoding, not a UTF-8-only byte
309            # heuristic: UTF-16 and other explicitly configured encodings work.
310            try:
311                decoded = data.decode(getattr(extractor, "encoding", "utf-8"))
312            except UnicodeDecodeError:
313                decoded = "\x00"
314            controls = sum(ord(char) < 32 and char not in "\n\r\t\f\b" for char in decoded)
315            if "\x00" in decoded or controls / max(1, len(decoded)) > 0.1:
316                result.skipped.append(SkipRecord(source_name, "binary"))
317                return result
318        pages = (
319            extractor.extract_pages(data)
320            if hasattr(extractor, "extract_pages")
321            else [(None, extractor.extract(data, name=source_name))]
322        )
323        for page, text in pages:
324            location = {"page": page} if page is not None else None
325            if not text.strip():
326                result.skipped.append(SkipRecord(source_name, "no_text", location=location))
327                continue
328            result.append(
329                RawDocument(
330                    source_name,
331                    text,
332                    external_id,
333                    dict(metadata),
334                    dict(provenance),
335                    identity,
336                    location,
337                )
338            )
339    except Exception as exc:
340        result.errors.append(_error_record(exc))
341        result.complete = False
342        result.status = SourceStatus.ERROR
343    return result

346def parse_checkpoint_since(checkpoint: Any) -> float | None:
347    """Parse a checkpoint into an epoch timestamp float, or None."""
348    if checkpoint is None:
349        return None
350    if isinstance(checkpoint, (int, float)):
351        return float(checkpoint)
352    if isinstance(checkpoint, str):
353        try:
354            return float(checkpoint)
355        except ValueError:
356            from datetime import datetime  # noqa: PLC0415
357
358            try:
359                return datetime.fromisoformat(checkpoint).timestamp()
360            except (ValueError, TypeError) as exc:
361                raise ValueError(f"Invalid checkpoint: {checkpoint}") from exc
362    if isinstance(checkpoint, dict):
363        return float(checkpoint.get("mtime", checkpoint.get("since", 0)))
364    raise ValueError(f"Invalid checkpoint: {checkpoint}")
