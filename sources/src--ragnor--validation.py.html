1"""Input validation helpers shared across ragnor providers.
2
3The stores rely on these validators: SQL identifiers are
4checked against a strict allowlist before interpolation, and
5embedding vectors are checked for the expected dimension and
6finite values before they reach the database.
7"""
8
9from __future__ import annotations
10
11from collections.abc import Iterable, Mapping, Sequence
12import math
13import re
14from typing import Any

16_IDENTIFIER_PATTERN = re.compile(r"[A-Za-z_][A-Za-z0-9_]*")

19def validate_identifier(name: str, label: str) -> None:
20    """Validate that a string is a safe SQL identifier (table, column, etc.).
21
22    Only allows alphanumeric characters and underscores, starting with a
23    letter or underscore. This is a strict whitelist that prevents SQL
24    injection through identifier interpolation.
25
26    Args:
27        name: The identifier string to validate.
28        label: Human-readable label for error messages.
29
30    Raises:
31        ValueError: If the identifier contains unsafe characters.
32    """
33    if not _IDENTIFIER_PATTERN.fullmatch(name):
34        raise ValueError(f"Unsafe {label}: {name!r}")

37def validate_embedding(
38    embedding: list[float],
39    expected_dim: int,
40    label: str = "embedding",
41) -> None:
42    """Validate that an embedding vector has the expected dimension and contains only finite values.
43
44    Args:
45        embedding: The embedding vector to validate.
46        expected_dim: Expected dimensionality of the embedding.
47        label: Human-readable label for error messages.
48
49    Raises:
50        ValueError: If the embedding has the wrong dimension or contains non-finite values.
51        TypeError: If a value in the embedding is not a number.
52    """
53    if len(embedding) != expected_dim:
54        raise ValueError(f"{label} has {len(embedding)} dimensions; expected {expected_dim}")
55    for i, value in enumerate(embedding):
56        try:
57            if not math.isfinite(value):
58                raise ValueError(f"{label}[{i}] is not finite: {value!r}")
59        except TypeError as exc:
60            raise TypeError(f"{label}[{i}] is not a number: {value!r}") from exc

63def validate_upsert_identifiers(
64    text_field: str,
65    embedding_field: str,
66    id_field: str | None = None,
67    default_id_field: str = "id",
68    extra_fields: Iterable[str] | None = None,
69) -> str:
70    """Validate identifiers used in upsert operations and return the resolved id field."""
71    validate_identifier(text_field, "text_field")
72    validate_identifier(embedding_field, "embedding_field")
73    resolved_id = id_field if id_field is not None else default_id_field
74    validate_identifier(resolved_id, "id_field")
75    if extra_fields:
76        for key in extra_fields:
77            validate_identifier(key, "extra_field")
78    return resolved_id

81def validate_search_identifiers(
82    text_field: str,
83    filters: Iterable[str] | None = None,
84    metadata_fields: Iterable[str] | None = None,
85) -> None:
86    """Validate identifiers used in search operations."""
87    validate_identifier(text_field, "text_field")
88    if filters:
89        for key in filters:
90            validate_identifier(key, "filter key")
91    if metadata_fields:
92        for field in metadata_fields:
93            validate_identifier(field, "metadata_field")

96def validate_delete_selectors(
97    ids: Sequence[Any] | None,
98    filters: Mapping[str, Any] | None,
99) -> None:
100    """Validate deletion selectors (ids and/or filters)."""
101    if ids is None and filters is None:
102        raise ValueError("delete requires at least one of ids or filters")
103    if ids is not None and len(ids) == 0:
104        raise ValueError("ids must not be empty; an empty id list never means delete everything")
105    if filters is not None:
106        for key in filters:
107            validate_identifier(key, "filter key")
