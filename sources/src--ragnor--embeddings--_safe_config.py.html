1"""Shared secret-stripping utility for canonical hash computation.
2
3This module exists solely to break the circular import between
4:mod:`ragnor.embeddings.profile` and :mod:`ragnor.ingestion.versioning`.
5Both modules need :func:`safe_config` but neither can import the other
6at module load time.
7
8Do not import from :mod:`ragnor.embeddings` or :mod:`ragnor.ingestion`
9from this module.
10"""
11
12from __future__ import annotations
13
14import math
15from typing import Any
16
17#: Keys (lowercased) that are stripped from dicts before hashing.
18_SECRET_KEYS = {
19    "token",
20    "headers",
21    "cookies",
22    "secret",
23    "password",
24    "credential",
25    "authorization",
26    "api_key",
27    "apikey",
28}
29
30_SECRET_SUFFIXES = ("_token",)
31_SECRET_SUBSTRINGS = ("secret", "password", "credential", "authorization", "api_key", "apikey")
32
33
34def _is_secret_key(key: str) -> bool:
35    key_lower = key.lower()
36    if key_lower in _SECRET_KEYS:
37        return True
38    if any(key_lower.endswith(suffix) for suffix in _SECRET_SUFFIXES):
39        return True
40    return any(part in key_lower for part in _SECRET_SUBSTRINGS)
41
42
43def safe_config(value: Any) -> Any:
44    """Copy JSON configuration only, omitting secrets and arbitrary objects.
45
46    Never use repr/default=str: clients and credentials may expose secrets there.
47
48    This is a standalone copy of the same function in
49    :mod:`ragnor.ingestion.versioning`; keep them in sync.
50    """
51    if value is None or isinstance(value, (str, bool, int)):
52        return value
53    if isinstance(value, float):
54        return value if math.isfinite(value) else None
55    if isinstance(value, dict):
56        return {
57            key: safe_config(item)
58            for key, item in value.items()
59            if isinstance(key, str)
60            and not _is_secret_key(key)
61            and isinstance(item, (str, bool, int, float, dict, list, tuple, type(None)))
62        }
63    if isinstance(value, (list, tuple)):
64        return [
65            safe_config(item)
66            for item in value
67            if isinstance(item, (str, bool, int, float, dict, list, tuple, type(None)))
68        ]
69    return None
