67class MemorySnapshotStore:
68    """In-memory implementation of :class:`SnapshotStore`."""
69
70    def __init__(self) -> None:
71        self._snapshots: dict[tuple[str, str, str, str | None], SnapshotRecord] = {}
72        self._revoked: set[tuple[str, str, str | None]] = set()
73
74    def put(self, snapshot: SnapshotRecord) -> None:
75        """Save a snapshot record."""
76        key = (
77            snapshot.source_id,
78            snapshot.document_id,
79            snapshot.revision,
80            snapshot.isolation_scope,
81        )
82        existing = self._snapshots.get(key)
83        if existing is not None:
84            if existing.content != snapshot.content or existing.metadata != snapshot.metadata:
85                raise ValueError("Cannot replace an immutable document revision")
86            return
87        self._snapshots[key] = deepcopy(snapshot)
88
89    def get(self, source_id: str, document_id: str, revision: str) -> SnapshotRecord | None:
90        """Get snapshot by source, document, and revision."""
91        matches = [
92            record
93            for key, record in self._snapshots.items()
94            if key[:3] == (source_id, document_id, revision)
95        ]
96        if len(matches) > 1:
97            raise ScopeAuthorizationError("Snapshot lookup requires an isolation scope")
98        return deepcopy(matches[0]) if matches else None
99
100    def get_scoped(
101        self, source_id: str, document_id: str, revision: str, scope: str | None
102    ) -> SnapshotRecord | None:
103        """Select an exact scoped revision, or an explicitly public revision."""
104        for candidate_scope in (scope, None):
105            key = (source_id, document_id, revision, candidate_scope)
106            if key in self._snapshots:
107                return deepcopy(self._snapshots[key])
108        return None
109
110    def revoke(self, source_id: str, document_id: str, scope: str | None) -> None:
111        """Retain historical bytes while denying access to every revision."""
112        self._revoked.add((source_id, document_id, scope))
113
114    def is_revoked(self, source_id: str, document_id: str, scope: str | None) -> bool:
115        return (source_id, document_id, scope) in self._revoked or (
116            source_id,
117            document_id,
118            None,
119        ) in self._revoked
120
121    def has_revision(self, source_id: str, document_id: str, revision: str) -> bool:
122        """Check if revision exists."""
123        return any(key[:3] == (source_id, document_id, revision) for key in self._snapshots)

167class SourceLookup:
168    """Authorized source document lookup engine.
169
170    Requires the same authorization scope as search and returns only
171    bounded authorized spans from immutable retained snapshots.
172    """
173
174    def __init__(
175        self,
176        snapshot_store: SnapshotStore,
177        default_max_span_length: int = 4000,
178    ) -> None:
179        """Initialize SourceLookup.
180
181        Args:
182            snapshot_store: Store holding immutable revision snapshots.
183            default_max_span_length: Maximum allowed character length for returned spans.
184        """
185        self._store = snapshot_store
186        self._validate_span_limit(default_max_span_length)
187        self.default_max_span_length = default_max_span_length
188
189    @staticmethod
190    def _validate_span_limit(limit: int) -> None:
191        if isinstance(limit, bool) or not isinstance(limit, int) or limit < 1:
192            raise ValueError("max_span_length must be a positive integer")
193
194    def revoke(self, source_id: str, document_id: str, scope: str | None) -> None:
195        """Persist revocation through a store supporting document access control."""
196        if not all(hasattr(self._store, method) for method in ("revoke", "is_revoked")):
197            raise TypeError("Snapshot store must implement revoke and is_revoked for management")
198        self._store.revoke(source_id, document_id, scope)
199
200    def lookup_span(  # noqa: PLR0913, PLR0917
201        self,
202        source_id: str,
203        document_id: str,
204        revision: str,
205        authorization_scope: str | None = None,
206        start_offset: int | None = None,
207        end_offset: int | None = None,
208        max_span_length: int | None = None,
209        strict: bool = False,
210    ) -> BoundedSpanResult:
211        """Retrieve a bounded span from an immutable revision snapshot.
212
213        Args:
214            source_id: Source repository identifier.
215            document_id: Document identifier.
216            revision: Stated revision identifier.
217            authorization_scope: Caller's isolation scope (e.g. tenant_id).
218            start_offset: Optional character start offset.
219            end_offset: Optional character end offset.
220            max_span_length: Maximum allowed span length in characters.
221            strict: If True, raises exceptions instead of returning error statuses.
222
223        Returns:
224            :class:`BoundedSpanResult` with the slice of text and provenance.
225
226        Raises:
227            RevisionUnavailableError: If exact revision is not retained and strict=True.
228            ScopeAuthorizationError: If scope check fails and strict=True.
229        """
230        span_limit = self.default_max_span_length if max_span_length is None else max_span_length
231        self._validate_span_limit(span_limit)
232        span_limit = min(span_limit, self.default_max_span_length)
233        try:
234            if hasattr(self._store, "is_revoked") and self._store.is_revoked(
235                source_id, document_id, authorization_scope
236            ):
237                raise ScopeAuthorizationError("Document access has been revoked")
238            if hasattr(self._store, "get_scoped"):
239                snapshot = self._store.get_scoped(
240                    source_id, document_id, revision, authorization_scope
241                )
242                if snapshot is None and self._store.has_revision(source_id, document_id, revision):
243                    raise ScopeAuthorizationError("Document revision belongs to a different scope")
244            else:
245                snapshot = self._store.get(source_id, document_id, revision)
246        except ScopeAuthorizationError:
247            if strict:
248                raise
249            return BoundedSpanResult(
250                source_id,
251                document_id,
252                revision,
253                "",
254                0,
255                0,
256                0,
257                False,
258                EvidenceStatus.BACKEND_FAILURE,
259            )
260
261        # Historical revision availability check
262        if snapshot is None:
263            msg = (
264                f"Historical revision {revision!r} is unavailable for document {document_id!r} "
265                f"in source {source_id!r}. Never substituting newer content for old citations."
266            )
267            logger.warning(msg)
268            if strict:
269                raise RevisionUnavailableError(msg)
270            return BoundedSpanResult(
271                source_id=source_id,
272                document_id=document_id,
273                revision=revision,
274                text="",
275                start_offset=start_offset or 0,
276                end_offset=end_offset or 0,
277                total_document_length=0,
278                is_truncated=False,
279                status=EvidenceStatus.REVISION_UNAVAILABLE,
280            )
281
282        # Scope authorization check
283        if snapshot.isolation_scope is not None and snapshot.isolation_scope != authorization_scope:
284            msg = (
285                f"Scope authorization failure for {document_id!r}: document scope is "
286                f"{snapshot.isolation_scope!r}, caller presented {authorization_scope!r}."
287            )
288            logger.warning(msg)
289            if strict:
290                raise ScopeAuthorizationError(msg)
291            return BoundedSpanResult(
292                source_id=source_id,
293                document_id=document_id,
294                revision=revision,
295                text="",
296                start_offset=start_offset or 0,
297                end_offset=end_offset or 0,
298                total_document_length=len(snapshot.content),
299                is_truncated=False,
300                status=EvidenceStatus.BACKEND_FAILURE,
301            )
302
303        # Extract bounded span
304        content = snapshot.content
305        doc_len = len(content)
306
307        start = max(0, start_offset) if start_offset is not None else 0
308        end = min(doc_len, end_offset) if end_offset is not None else doc_len
309
310        start = min(start, doc_len)
311        end = max(end, start)
312
313        is_truncated = False
314        if end - start > span_limit:
315            end = start + span_limit
316            is_truncated = True
317
318        span_text = content[start:end]
319
320        return BoundedSpanResult(
321            source_id=source_id,
322            document_id=document_id,
323            revision=revision,
324            text=span_text,
325            start_offset=start,
326            end_offset=end,
327            total_document_length=doc_len,
328            is_truncated=is_truncated,
329            status=EvidenceStatus.SUCCESS,
330        )
