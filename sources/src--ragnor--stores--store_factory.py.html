1"""Abstract factory for vector store backends.
2
3:class:`VectorStoreFactory` is the abstraction that creates
4vector stores. ragnor's default implementation,
5:class:`RegistryVectorStoreFactory`, selects a backend from
6the provider registry (``memory``, ``pgvector``, ``qdrant``,
7``chroma``, ``faiss``, or any host-registered backend), so
8the vector database is chosen through a factory instead of
9hardcoded imports -- including the zero-SQL in-memory store,
10which the same factory produces without touching any database.
11"""
12
13from __future__ import annotations

15import importlib

18from abc import ABC, abstractmethod
19from typing import TYPE_CHECKING, Any

28class VectorStoreFactory(ABC):
29    """Abstract factory that creates vector stores for any backend.
30
31    Subclass this to create stores for a vector database ragnor
32    does not know about, or to customize backend construction.
33    The factory is the single seam between ragnor and the
34    vector database.
35    """
36
37    @abstractmethod
38    def create(
39        self,
40        settings: RagSettings | None = None,
41        metric: str | None = None,
42        **kwargs: Any,
43    ) -> VectorStore:
44        """Create a vector store.
45
46        Args:
47            settings: Host settings; ``vector_store_provider``
48                selects the backend.
49            metric: Distance metric for the vector stage
50                (``"l2"`` by default, ``"cosine"`` opt-in).
51                Forwarded to the backend constructor only when it
52                is not ``None``, so custom backends whose
53                constructor does not accept the kwarg keep
54                working. ``settings`` is forwarded only to
55                backends that declare an explicit ``settings``
56                parameter.
57            **kwargs: Backend constructor arguments
58                (``table_name``, ``id_field``, ``embedding_dim``,
59                ``session_factory``, ``settings``, ``embedding_fn``).
60
61        Returns:
62            A configured :class:`VectorStore`.
63        """
64

66class RegistryVectorStoreFactory(VectorStoreFactory):
67    """Creates vector stores from the provider registry.
68
69    Reads ``settings.vector_store_provider`` (default
70    ``"memory"``), imports the matching backend module on
71    demand so its driver dependency stays optional, and
72    instantiates the provider with the standard kwargs.
73    """
74
75    def __init__(
76        self,
77        registry: Registry[VectorStore] | None = None,
78        default_provider: str = "memory",
79    ) -> None:
80        """Create the factory.
81
82        Args:
83            registry: Provider registry to select from; defaults
84                to ragnor's ``vector_store_registry``.
85            default_provider: Provider used when ``settings`` is
86                omitted or carries no ``vector_store_provider``.
87        """
88        if registry is None:
89            from ragnor.stores.registry import vector_store_registry  # noqa: PLC0415
90
91            registry = vector_store_registry
92        self._registry = registry
93        self._default_provider = default_provider
94
95    def create(
96        self,
97        settings: RagSettings | None = None,
98        metric: str | None = None,
99        **kwargs: Any,
100    ) -> VectorStore:
101        """Create a vector store from the registry.
102
103        ``metric`` is forwarded to the selected backend only when
104        it is not ``None``, so a custom registered backend whose
105        constructor does not accept the kwarg keeps working.
106        ``settings`` is forwarded only to backends whose
107        constructor declares an explicit ``settings`` parameter,
108        so a backend can read its optional host fields (such as
109        ``rag_distance_metric`` or the pgvector
110        ``database_url``); a constructor that only accepts
111        ``**kwargs`` receives the documented registry kwargs
112        unchanged.
113        """
114        provider = self._default_provider
115        if settings is not None:
116            provider = getattr(settings, "vector_store_provider", self._default_provider)
117        self._load_builtin_provider(provider)
118        store_class = self._registry.get(provider)
119        if metric is not None:
120            kwargs["metric"] = metric
121        if (
122            settings is not None
123            and "settings" not in kwargs
124            and self._accepts_settings(store_class)
125        ):
126            kwargs["settings"] = settings
127        return store_class(**kwargs)
128
129    @staticmethod
130    def _accepts_settings(store_class: type) -> bool:
131        """Return whether ``store_class.__init__`` declares a ``settings`` parameter.
132
133        Only an explicit parameter counts: a constructor accepting
134        ``**kwargs`` is treated as not declaring it, so the
135        registry kwargs a custom backend receives stay exactly
136        the documented ones.
137        """
138        try:
139            signature = inspect.signature(store_class)
140        except (TypeError, ValueError):  # pragma: no cover - exotic builtins
141            return False
142        return "settings" in signature.parameters
143
144    def _load_builtin_provider(self, name: str) -> None:
145        """Import a backend module so its registration runs.
146
147        ragnor ships one module per built-in backend, named
148        after its provider (``ragnor.stores.memory``,
149        ``ragnor.stores.pgvector``, ``ragnor.stores.qdrant``,
150        ...). Backend modules register themselves on import, and
151        importing them lazily by convention keeps every
152        backend's driver dependency out of ragnor's import graph
153        until the backend is actually selected -- so adding a
154        backend is a matter of dropping in a module, not
155        editing a chain of imports. A provider already in the
156        registry (for example a host-registered override) is
157        left untouched, and a name with no matching module is
158        left to the registry's ``KeyError``.
159
160        Raises:
161            ImportError: If the backend module exists but its
162                optional driver dependency is not installed.
163        """
164        key = name.lower()
165        if key in self._registry.list():
166            return
167        if not key.isidentifier():
168            return
169        module_name = f"ragnor.stores.{key}"
170        if importlib.util.find_spec(module_name) is None:
171            return
172        try:
173            importlib.import_module(module_name)
174        except ImportError as exc:
175            raise ImportError(
176                f"The {name!r} vector store backend could not be loaded. "
177                f"Built-in backends ship as optional extras -- install the "
178                f"backend's extra, e.g. pip install 'ragnor[{key}]'. "
179                f"Underlying error: {exc}"
180            ) from exc
