1"""
2SQLAlchemy ORM model for the ``conversations`` table.
3
4Represents a single chat conversation between a user and the OmniCare agent.
5Messages are stored as a JSONB array to preserve the original structure
6(role, content, timestamps, sources, tool calls) without requiring a separate
7message table.
8"""
9
10import uuid
11from datetime import UTC, datetime
12
13from sqlalchemy import Column, DateTime, ForeignKey, String
14from sqlalchemy.dialects.postgresql import JSONB, UUID
15
16from app.models.base import Base
17
18
19class Conversation(Base):
20    """Database model for a chat conversation.
21
22    Attributes:
23        id: Primary key UUID.
24        user_id: Foreign key to the user who owns this conversation.
25        session_id: Unique string identifier used by the ADK agent runtime
26            to track the in-memory session. Distinct from the database PK.
27        title: Human-readable conversation title, defaulting to "New Chat".
28        messages: JSONB array of message objects. Each message contains at
29            least ``id``, ``role``, ``content``, and ``timestamp``, plus
30            optional ``sources`` and ``tool_calls`` fields.
31        created_at: Timestamp of conversation creation.
32        updated_at: Timestamp of last message or metadata update.
33    """
34
35    __tablename__ = "conversations"
36
37    id = Column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
38    user_id = Column(UUID(as_uuid=True), ForeignKey("users.id", ondelete="CASCADE"), nullable=False)
39    session_id = Column(String(128), unique=True, index=True, nullable=False)
40    title = Column(String(255), nullable=False, default="New Chat")
41    # JSONB is used instead of a normalized messages table because:
42    #   1. Messages are always read/written together with the conversation.
43    #   2. The schema is append-only and managed by the agent runtime.
44    #   3. It simplifies queries for full conversation history retrieval.
45    messages = Column(JSONB, nullable=False, default=list)
46    created_at = Column(DateTime(timezone=True), default=lambda: datetime.now(UTC))
47    updated_at = Column(
48        DateTime(timezone=True),
49        default=lambda: datetime.now(UTC),
50        onupdate=lambda: datetime.now(UTC),
51    )
