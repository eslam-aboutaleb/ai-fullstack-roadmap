37@dataclass
38class PDFExtractor(Extractor):
39    """Extract PDF pages with optional pypdf; image-only pages need external OCR.
40
41    ``options['max_pages']`` bounds accepted documents (default 1000). Exceeding
42    the bound raises rather than returning a silently truncated document.
43    """
44
45    id: str = "pdf"
46    version: str = "2"
47    options: dict[str, Any] = field(default_factory=dict)
48
49    def __post_init__(self) -> None:
50        self.options = {"max_pages": 1000, **self.options}
51        limit = self.options["max_pages"]
52        if isinstance(limit, bool) or not isinstance(limit, int) or limit < 1:
53            raise ValueError("PDF max_pages must be a positive integer")
54
55    def extract_pages(self, data: bytes) -> list[tuple[int, str]]:
56        """Return every real page as ``(1-based number, text)``, including blanks."""
57        try:
58            from pypdf import PdfReader  # noqa: PLC0415
59        except ImportError as exc:
60            raise ImportError(
61                "PDF extraction requires pypdf; install 'ragnor[pdf]' "
62                "or 'pypdf'. OCR for image-only PDFs is not implemented."
63            ) from exc
64
65        reader = PdfReader(BytesIO(data))
66        count = len(reader.pages)
67        limit = self.options["max_pages"]
68        if count > limit:
69            raise ValueError(f"PDF has {count} pages, exceeding max_pages={limit}")
70        return [
71            (number, (page.extract_text() or "").strip())
72            for number, page in enumerate(reader.pages, start=1)
73        ]
74
75    def extract(self, data: bytes, *, name: str = "") -> str:
76        return "\n\n".join(text for _, text in self.extract_pages(data) if text)

79class _HTMLTextParser(HTMLParser):
80    """Render visible prose, suppressing boilerplate before emitting markup."""
81
82    _ignored = {"head", "script", "style", "nav", "footer", "aside", "noscript", "template"}
83    _roles = {"navigation", "banner", "contentinfo", "complementary"}
84    _void = {
85        "area",
86        "base",
87        "br",
88        "col",
89        "embed",
90        "hr",
91        "img",
92        "input",
93        "link",
94        "meta",
95        "param",
96        "source",
97        "track",
98        "wbr",
99    }
100    _blocks = {
101        "address",
102        "article",
103        "blockquote",
104        "br",
105        "dd",
106        "div",
107        "dl",
108        "dt",
109        "figcaption",
110        "figure",
111        "header",
112        "hr",
113        "li",
114        "main",
115        "ol",
116        "p",
117        "pre",
118        "section",
119        "table",
120        "tr",
121        "ul",
122        "h1",
123        "h2",
124        "h3",
125        "h4",
126        "h5",
127        "h6",
128    }
129    _headings = {"h1", "h2", "h3", "h4", "h5", "h6"}
130
131    def __init__(self) -> None:
132        super().__init__(convert_charrefs=True)
133        self.parts: list[str] = []
134        self.stack: list[tuple[str, bool, str | None]] = []
135
136    def handle_starttag(self, tag: str, attrs: list[tuple[str, str | None]]) -> None:
137        attributes = dict(attrs)
138        suppressed = (
139            bool(self.stack and self.stack[-1][1])
140            or tag in self._ignored
141            or bool(set((attributes.get("role") or "").lower().split()) & self._roles)
142            or "hidden" in attributes
143            or (attributes.get("aria-hidden") or "").lower() == "true"
144        )
145        href = attributes.get("href") if tag == "a" else None
146        if not suppressed:
147            if tag in self._blocks:
148                self.parts.append("\n")
149            if tag in self._headings:
150                self.parts.append("#" * int(tag[1]) + " ")
151            if tag in {"td", "th"}:
152                self.parts.append(" ")
153        if tag not in self._void:
154            self.stack.append((tag, suppressed, href))
155
156    def handle_startendtag(self, tag: str, attrs: list[tuple[str, str | None]]) -> None:
157        self.handle_starttag(tag, attrs)
158        if tag not in self._void:
159            self.handle_endtag(tag)
160
161    def handle_endtag(self, tag: str) -> None:
162        # Match the actual open element so stray closing tags cannot unhide text.
163        index = next(
164            (i for i in range(len(self.stack) - 1, -1, -1) if self.stack[i][0] == tag),
165            None,
166        )
167        if index is None:
168            return
169        closing = self.stack[index:]
170        del self.stack[index:]
171        for closed_tag, suppressed, href in reversed(closing):
172            if suppressed:
173                continue
174            if href and href.strip():
175                self.parts.append(f" ({href.strip()})")
176            if closed_tag in self._blocks:
177                self.parts.append("\n")
178
179    def handle_data(self, data: str) -> None:
180        if not self.stack or not self.stack[-1][1]:
181            self.parts.append(re.sub(r"\s+", " ", data))

184@dataclass
185class HTMLExtractor(Extractor):
186    """Conservative HTML boilerplate remover preserving headings and link targets."""
187
188    id: str = "html"
189    version: str = "2"
190    options: dict[str, Any] = field(default_factory=dict)
191
192    def extract(self, data: bytes, *, name: str = "") -> str:
193        parser = _HTMLTextParser()
194        parser.feed(data.decode("utf-8", errors="replace"))
195        parser.close()
196        lines = (
197            re.sub(r"[^\S\n]+", " ", line).strip() for line in "".join(parser.parts).splitlines()
198        )
199        return "\n".join(line for line in lines if line)
