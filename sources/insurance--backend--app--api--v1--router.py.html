1"""
2API v1 Router - mounts all v1 endpoint routers under the /api/v1 prefix.
3
4This router acts as the single aggregation point for all version-1 API
5endpoints. Each sub-router is responsible for its own domain:
6
7  - ``/health``        -- Service health and dependency checks.
8  - ``/chat``          -- Synchronous chat endpoint with idempotency.
9  - ``/chat/stream``   -- Streaming SSE chat endpoint.
10  - ``/chat/conversations`` -- Conversation history CRUD operations.
11  - ``/claims``        -- Claim lifecycle: prepare, confirm, cancel, list,
12                          read, status transition, withdraw.
13  - ``/feedback``      -- Answer quality ratings on the caller's own answers.
14  - ``/notifications``-- In-app notification feed for claim status changes.
15  - ``/admin``         -- Administrative queue, audit and feedback views.
16  - ``/rag``           -- RAG evaluation and dataset endpoints.
17  - ``/auth``          -- Public authentication endpoints (signup/signin/logout).
18
19Including this router in ``app/main.py`` makes all v1 endpoints available
20at their documented paths without requiring individual route registrations.
21The ``/admin`` routes are role-gated in their own module; they are mounted here
22so they share the versioned prefix, not to make them public.
23
24``GET /api/v1/health/detailed`` is not routed through here: it is registered
25straight on the app from ``app.observability.routes`` so the pinned
26``/health`` contract and its response shape stay untouched.
27"""
28
29from fastapi import APIRouter

31from app.api.v1.admin import router as admin_router
32from app.api.v1.auth import router as auth_router

34from app.api.v1.chat_stream import router as chat_stream_router
35from app.api.v1.claims import router as claims_router
36from app.api.v1.conversations import router as conversations_router
