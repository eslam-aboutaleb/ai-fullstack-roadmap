1"""
2Custom middleware for the OmniCare Financial backend.
3
4This module provides reusable Starlette middleware components that enforce
5cross-cutting concerns such as request payload size limits. These middleware
6are registered globally on the FastAPI application instance in ``app/main.py``
7and apply to every incoming request.
8"""
9
10from fastapi import Request, Response
11from fastapi.responses import JSONResponse
12from starlette.middleware.base import BaseHTTPMiddleware
13
14
15class RequestSizeLimitMiddleware(BaseHTTPMiddleware):
16    """Enforce a maximum request body size to protect against oversized payloads.
17
18    This middleware checks the ``Content-Length`` header first. If the header is
19    missing or the request uses chunked transfer encoding, the middleware reads
20    the request body up to the configured limit and returns an HTTP 413
21    (Payload Too Large) response if the actual body exceeds the threshold.
22    This prevents unnecessary memory allocation and protects downstream
23    processing from malformed or maliciously large requests.
24
25    Attributes:
26        max_upload_size: Maximum allowed request body size in bytes.
27    """
28
29    def __init__(self, app, max_upload_size: int = 5 * 1024 * 1024):
30        """Initialize the middleware with the target ASGI app and size limit.
31
32        Args:
33            app: The downstream ASGI application to wrap.
34            max_upload_size: Maximum allowed request body size in bytes.
35                Defaults to 5 MiB.
36        """
37        super().__init__(app)
38        self.max_upload_size = max_upload_size
39
40    async def dispatch(self, request: Request, call_next) -> Response:
41        """Intercept the request and enforce the size limit.
42
43        Reads the ``Content-Length`` header and compares it against
44        ``max_upload_size``. If the header is missing or unparseable, the
45        request body is read and measured. If the body exceeds the limit,
46        a 413 response is returned. Otherwise, the downstream application
47        receives the request with the cached body.
48
49        Args:
50            request: The incoming ASGI request.
51            call_next: The downstream middleware or application callable.
52
53        Returns:
54            Response: Either a 413 JSON error response or the downstream
55            response.
56        """
57        content_length = request.headers.get("content-length")
58        exceeded = False
59
60        if content_length:
61            try:
62                if int(content_length) > self.max_upload_size:
63                    exceeded = True
64            except ValueError:
65                exceeded = False
66        else:
67            body = await request.body()
68            if len(body) > self.max_upload_size:
69                exceeded = True
70
71        if exceeded:
72            return JSONResponse(
73                status_code=413,
74                content={
75                    "error": {
76                        "code": "PAYLOAD_TOO_LARGE",
77                        "message": (
78                            f"Request body exceeds the maximum allowed size of "
79                            f"{self.max_upload_size} bytes."
80                        ),
81                    }
82                },
83            )
84
85        return await call_next(request)
