1"""Explicit score semantics for retrieval results.
2
3Distance, similarity, Reciprocal Rank Fusion (RRF), and reranker
4scores represent fundamentally different concepts with different
5scales, ordering, and distributions:
6
7* **Distance** (e.g. L2, cosine distance): lower is closer/better.
8  Distance values are unbounded or in [0, 2]; they are metric distances,
9  NOT probabilities.
10* **Similarity** (e.g. cosine similarity, dot product): higher is
11  closer/better. Dot products are unbounded; cosine similarities are
12  in [-1, 1] or [0, 1]. They are NOT calibrated probabilities.
13* **RRF** (Reciprocal Rank Fusion score): higher rank is better.
14  Computed as ``sum(1 / (k + rank))``, typically in [0, 0.05]. It is a
15  rank-order statistic reflecting reciprocal ranking, NOT a probability.
16* **Lexical** (e.g. BM25, token overlap): higher is better. Unbounded
17  term match score, NOT a probability.
18* **Reranker** (e.g. cross-encoder logit, relevance score): higher is
19  better. Model-specific and uncalibrated unless explicitly converted
20  with a calibrated sigmoid.
21
22These scores are **not interchangeable confidence probabilities**.
23Comparing or thresholding across score kinds without explicit conversion
24will produce invalid rankings or false confidence.
25"""
26
27from __future__ import annotations
28
29from dataclasses import dataclass
30from enum import StrEnum
31
32
33class ScoreKind(StrEnum):
34    """Semantic classification of a retrieval score."""
35
36    DISTANCE = "distance"
37    SIMILARITY = "similarity"
38    RRF = "rrf"
39    LEXICAL = "lexical"
40    RERANKER = "reranker"
41
42
43@dataclass(frozen=True)
44class RawScore:
45    """An explicit retrieval score with its semantic kind and ordering.
46
47    Attributes:
48        value: Numeric score value.
49        kind: Semantic kind of the score (distance, similarity, RRF, etc.).
50        metric: Optional metric or model name (e.g. ``"cosine"``,
51            ``"l2"``, ``"cohere/rerank-v3.5"``).
52        lower_is_better: When ``True``, smaller values indicate closer /
53            better matches (e.g. distances). When ``False``, larger values
54            indicate better matches (e.g. similarities, RRF, reranker).
55    """
56
57    value: float
58    kind: ScoreKind
59    metric: str | None = None
60    lower_is_better: bool = False
61
62    @classmethod
63    def distance(cls, value: float, metric: str = "l2") -> RawScore:
64        """Create a distance score where lower is better."""
65        return cls(value=value, kind=ScoreKind.DISTANCE, metric=metric, lower_is_better=True)
66
67    @classmethod
68    def similarity(cls, value: float, metric: str = "cosine") -> RawScore:
69        """Create a similarity score where higher is better."""
70        return cls(value=value, kind=ScoreKind.SIMILARITY, metric=metric, lower_is_better=False)
71
72    @classmethod
73    def rrf(cls, value: float) -> RawScore:
74        """Create an RRF rank score where higher is better."""
75        return cls(value=value, kind=ScoreKind.RRF, metric="rrf", lower_is_better=False)
76
77    @classmethod
78    def lexical(cls, value: float, metric: str = "token_overlap") -> RawScore:
79        """Create a lexical score where higher is better."""
80        return cls(value=value, kind=ScoreKind.LEXICAL, metric=metric, lower_is_better=False)
81
82    @classmethod
83    def reranker(cls, value: float, model: str | None = None) -> RawScore:
84        """Create a reranker score where higher is better."""
85        return cls(value=value, kind=ScoreKind.RERANKER, metric=model, lower_is_better=False)
