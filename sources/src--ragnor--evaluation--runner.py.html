1"""RAG evaluation runner.
2
3Orchestrates end-to-end evaluation of
4a RAG pipeline by running a dataset through retrieval and
5answer generation, then computing metrics across all
6dimensions.
7
8The retrieval function and the dataset are **injected** by
9the caller: ragnor never imports a host application's
10retriever or dataset module. The host application wires
11its own hybrid retriever and curated dataset in a thin
12runner adapter.
13
14Supports three modes:
15  - **retrieval-only**: evaluates retrieval quality (Recall@K,
16    Precision@K, MRR)
17  - **context**: scores the joined retrieved context against
18    the gold answer (the legacy behavior under an honest
19    name -- it measures whether the retrieved chunks contain
20    the answer, not whether a model answers faithfully)
21  - **end-to-end**: evaluates both retrieval and answer
22    quality (faithfulness, relevance, completeness,
23    conciseness) on a real model output produced by an
24    injected answer generator
25
26Can be run as a CLI script or imported for use in tests and
27the eval API.
28"""
29

88async def _run_retrieval_eval(
89    retrieval_fn: RetrievalFunction,
90    dataset: list[EvalSample],
91) -> tuple[RetrievalMetrics, list[dict[str, Any]]]:
92    """Run retrieval evaluation using the existing RagEvaluationHarness.
93
94    Maps EvalSample entries to LabeledQuery format and runs them
95    through the injected retrieval function.
96
97    Returns:
98        Tuple of (aggregated metrics, per-sample details).
99    """
100    per_sample: list[dict[str, Any]] = []
101
102    for sample in dataset:
103        start = time.monotonic()
104        try:
105            results = await retrieval_fn(sample.query)
106            latency_ms = (time.monotonic() - start) * 1000
107
108            # Extract retrieved sections
109            retrieved_sections = set()
110            for r in results:
111                section = r.get("metadata", {}).get("section", "")
112                if section:
113                    retrieved_sections.add(section)
114
115            expected = set(sample.expected_sections)
116            hits = expected & retrieved_sections
117            recall = len(hits) / len(expected) if expected else 0.0
118            precision = len(hits) / max(len(retrieved_sections), 1)
119
120            # MRR: find rank of first relevant section
121            mrr = 0.0
122            for rank, r in enumerate(results, 1):
123                section = r.get("metadata", {}).get("section", "")
124                if section in expected:
125                    mrr = 1.0 / rank
126                    break
127
128            # Build context for answer evaluation
129            context_parts = [r.get("document", "") for r in results]
130            context = "\n---\n".join(context_parts)
131
132            per_sample.append(
133                {
134                    "query": sample.query,
135                    "category": sample.category,
136                    "difficulty": sample.difficulty.value,
137                    "expected_sections": sample.expected_sections,
138                    "retrieved_sections": list(retrieved_sections),
139                    "recall": round(recall, 4),
140                    "precision": round(precision, 4),
141                    "mrr": round(mrr, 4),
142                    "latency_ms": round(latency_ms, 2),
143                    "chunks_found": len(results),
144                    "context": context,
145                }
146            )
147        except Exception as exc:
148            logger.error("Retrieval eval failed for '%s': %s", sample.query, exc)
149            per_sample.append(
150                {
151                    "query": sample.query,
152                    "category": sample.category,
153                    "difficulty": sample.difficulty.value,
154                    "error": str(exc),
155                }
156            )
157
158    # Compute aggregate retrieval metrics
159    valid = [s for s in per_sample if "error" not in s]
160    n = len(valid)
161    metrics = RetrievalMetrics(
162        recall_at_k=sum(s["recall"] for s in valid) / n if n else 0.0,
163        precision_at_k=sum(s["precision"] for s in valid) / n if n else 0.0,
164        mrr=sum(s["mrr"] for s in valid) / n if n else 0.0,
165        avg_latency_ms=sum(s["latency_ms"] for s in valid) / n if n else 0.0,
166        total_queries=len(dataset),
167        errors=len(per_sample) - n,
168    )
169
170    return metrics, per_sample

178async def _run_answer_eval(
179    dataset: list[EvalSample],
180    per_sample_retrieval: list[dict[str, Any]],
181    answer_fn: AnswerFunction | None = None,
182    judge: str = "heuristic",
183    settings: RagSettings | None = None,
184) -> tuple[AnswerMetrics, list[dict[str, Any]]]:
185    """Run answer quality evaluation on the RAG pipeline output.
186
187    In ``context`` mode (``answer_fn is None``) the joined
188    retrieved context is scored against the gold answer --
189    the legacy behavior under an honest name. In
190    ``end-to-end`` mode the injected ``answer_fn`` generates
191    a real answer from ``(query, context)`` and the judges
192    receive the actual generated answer and the actual
193    retrieved context as separate inputs.
194
195    Args:
196        dataset: The evaluation samples.
197        per_sample_retrieval: Retrieval results for each sample (provides context).
198        answer_fn: Injected async answer generator
199            ``(query, context) -> answer``. When ``None``
200            (context mode) the retrieved context itself is
201            scored against the gold answer.
202        judge: "heuristic" or "llm" evaluation strategy.
203        settings: Host settings for the LLM judge (only used when
204            ``judge == "llm"``).
205
206    Returns:
207        Tuple of (aggregated metrics, per-sample answer details).
208    """
209    metrics = AnswerMetrics(total_evaluated=len(dataset))
210    answer_details: list[dict[str, Any]] = []
211
212    for i, sample in enumerate(dataset):
213        try:
214            retrieval_data = per_sample_retrieval[i] if i < len(per_sample_retrieval) else {}
215            context = retrieval_data.get("context", "")
216            chunks_found = retrieval_data.get("chunks_found", 0)
217
218            if answer_fn is not None:
219                generated_answer = await answer_fn(sample.query, context)
220            else:
221                generated_answer = context
222                if not generated_answer:
223                    generated_answer = "(no context retrieved)"
224
225            if judge == "llm":
226                score = await evaluate_answer_llm(
227                    query=sample.query,
228                    answer=generated_answer,
229                    gold_answer=sample.gold_answer,
230                    retrieved_context=context,
231                    settings=settings,
232                )
233            else:
234                score = evaluate_answer_heuristic(
235                    query=sample.query,
236                    answer=generated_answer,
237                    gold_answer=sample.gold_answer,
238                    retrieved_context=context,
239                )
240
241            metrics.avg_faithfulness += score.faithfulness
242            metrics.avg_relevance += score.relevance
243            metrics.avg_completeness += score.completeness
244            metrics.avg_conciseness += score.conciseness
245            metrics.avg_overall += score.overall
246
247            answer_details.append(
248                {
249                    "query": sample.query,
250                    "gold_answer": sample.gold_answer,
251                    "generated_answer": generated_answer[:500],
252                    "chunks_found": chunks_found,
253                    "faithfulness": score.faithfulness,
254                    "relevance": score.relevance,
255                    "completeness": score.completeness,
256                    "conciseness": score.conciseness,
257                    "overall": score.overall,
258                    "reasoning": score.reasoning,
259                }
260            )
261
262        except Exception as exc:
263            logger.error("Answer eval failed for '%s': %s", sample.query, exc)
264            metrics.errors += 1
265            answer_details.append(
266                {
267                    "query": sample.query,
268                    "error": str(exc),
269                }
270            )
271
272    # Average the metrics
273    n = metrics.total_evaluated - metrics.errors
274    if n > 0:
275        metrics.avg_faithfulness = round(metrics.avg_faithfulness / n, 4)
276        metrics.avg_relevance = round(metrics.avg_relevance / n, 4)
277        metrics.avg_completeness = round(metrics.avg_completeness / n, 4)
278        metrics.avg_conciseness = round(metrics.avg_conciseness / n, 4)
279        metrics.avg_overall = round(metrics.avg_overall / n, 4)
280
281    metrics.per_sample = answer_details
282    return metrics, answer_details

290async def run_evaluation(
291    config: EvalConfig | None = None,
292    retrieval_fn: RetrievalFunction | None = None,
293    dataset: list[EvalSample] | None = None,
294    settings: RagSettings | None = None,
295    answer_fn: AnswerFunction | None = None,
296) -> EvalResult:
297    """Run a complete RAG evaluation.
298
299    Args:
300        config: Evaluation configuration. If None, uses defaults
301            (end-to-end mode with heuristic judge).
302        retrieval_fn: Injected async retrieval function mapping a
303            query string to a list of result dicts. The host
304            application supplies its hybrid retriever here.
305        dataset: Injected evaluation samples, already filtered to
306            the config's category/difficulty.
307        settings: Host settings, read by the LLM judge when
308            ``config.judge == "llm"``.
309        answer_fn: Injected async answer generator
310            ``(query, context) -> generated answer``. Required
311            when ``config.mode == "end-to-end"``; the host
312            application supplies its answer model here.
313
314    Returns:
315        EvalResult with all metrics and per-sample details.
316
317    Raises:
318        ValueError: If the retrieval function or dataset is
319            missing, the mode is unknown, or the mode is
320            ``end-to-end`` without an ``answer_fn``.
321    """
322    if config is None:
323        config = EvalConfig()
324    if retrieval_fn is None or dataset is None:
325        raise ValueError(
326            "run_evaluation requires an injected retrieval_fn and dataset; "
327            "the host application wires its retriever and dataset in"
328        )
329    if config.mode not in ("retrieval-only", "context", "end-to-end"):
330        raise ValueError(
331            f"invalid evaluation mode {config.mode!r}; valid modes are "
332            "'retrieval-only', 'context', and 'end-to-end'"
333        )
334    if config.mode == "end-to-end" and answer_fn is None:
335        raise ValueError(
336            "mode='end-to-end' requires an answer generator; pass "
337            "answer_fn=... (an async (query, context) -> answer "
338            "callable) or use one of the other modes: 'retrieval-only', "
339            "'context', 'end-to-end'"
340        )
341
342    start_time = time.monotonic()
343    result = EvalResult(
344        timestamp=datetime.now(tz=UTC).isoformat(),
345        config={
346            "mode": config.mode,
347            "judge": config.judge,
348            "n_results": config.n_results,
349            "category": config.category,
350            "difficulty": config.difficulty.value if config.difficulty else None,
351        },
352    )
353
354    if not dataset:
355        result.summary = "No evaluation samples matched the filters."
356        return result
357
358    logger.info(
359        "Starting RAG evaluation: mode=%s, judge=%s, samples=%d",
360        config.mode,
361        config.judge,
362        len(dataset),
363    )
364
365    # Step 1: Retrieval evaluation
366    retrieval_metrics, per_sample_retrieval = await _run_retrieval_eval(
367        retrieval_fn=retrieval_fn,
368        dataset=dataset,
369    )
370    result.retrieval_metrics = asdict(retrieval_metrics)
371
372    # Step 2: Answer quality evaluation (context and end-to-end modes)
373    if config.mode in ("context", "end-to-end"):
374        answer_metrics, answer_details = await _run_answer_eval(
375            dataset=dataset,
376            per_sample_retrieval=per_sample_retrieval,
377            answer_fn=answer_fn,
378            judge=config.judge,
379            settings=settings,
380        )
381        result.answer_metrics = {
382            k: v for k, v in asdict(answer_metrics).items() if k != "per_sample"
383        }
384
385        # Merge per-sample results: the retrieved context, the
386        # generated answer, and the gold answer stay under their
387        # own names so each sample carries the full triple; the
388        # metric scores keep the answer_ prefix.
389        for i, retrieval_data in enumerate(per_sample_retrieval):
390            merged = {**retrieval_data}
391            if i < len(answer_details):
392                detail = answer_details[i]
393                merged["generated_answer"] = detail.get("generated_answer", "")
394                merged["gold_answer"] = detail.get("gold_answer", "")
395                merged.update(
396                    {
397                        f"answer_{k}": v
398                        for k, v in detail.items()
399                        if k
400                        not in (
401                            "query",
402                            "context",
403                            "generated_answer",
404                            "gold_answer",
405                            "chunks_found",
406                        )
407                    }
408                )
409            result.per_sample_results.append(merged)
410    else:
411        result.per_sample_results = per_sample_retrieval
412
413    result.duration_seconds = round(time.monotonic() - start_time, 2)
414
415    # Build summary
416    lines = [
417        f"RAG Evaluation Complete ({config.mode} mode)",
418        f"  Samples: {len(dataset)}",
419        f"  Duration: {result.duration_seconds}s",
420        "",
421        "Retrieval Metrics:",
422        f"  Recall@{config.n_results}: {retrieval_metrics.recall_at_k:.4f}",
423        f"  Precision@{config.n_results}: {retrieval_metrics.precision_at_k:.4f}",
424        f"  MRR: {retrieval_metrics.mrr:.4f}",
425        f"  Avg Latency: {retrieval_metrics.avg_latency_ms:.1f}ms",
426        f"  Errors: {retrieval_metrics.errors}",
427    ]
428
429    if config.mode in ("context", "end-to-end"):
430        lines.extend(
431            [
432                "",
433                "Answer Quality Metrics:",
434                f"  Faithfulness: {answer_metrics.avg_faithfulness:.4f}",
435                f"  Relevance: {answer_metrics.avg_relevance:.4f}",
436                f"  Completeness: {answer_metrics.avg_completeness:.4f}",
437                f"  Conciseness: {answer_metrics.avg_conciseness:.4f}",
438                f"  Overall: {answer_metrics.avg_overall:.4f}",
439                f"  Errors: {answer_metrics.errors}",
440            ]
441        )
442
443    result.summary = "\n".join(lines)
444    logger.info("\n%s", result.summary)
445
446    return result

454def main(
455    retrieval_fn: RetrievalFunction | None = None,
456    dataset: list[EvalSample] | None = None,
457    settings: RagSettings | None = None,
458    runner: Any = None,
459) -> None:
460    """CLI entry point for running RAG evaluation.
461
462    Args:
463        retrieval_fn: Injected retrieval function (the host
464            application wires its retriever here).
465        dataset: Injected evaluation dataset.
466        settings: Host settings for the LLM judge.
467        runner: Async callable invoked as
468            ``runner(config, retrieval_fn, dataset, settings)``.
469            Defaults to :func:`run_evaluation`; a host
470            application passes its own runner to inject
471            domain dependencies behind the CLI flags --
472            including the ``answer_fn`` answer generator
473            that ``mode="end-to-end"`` requires::
474
475                async def host_runner(config, retrieval_fn,
476                                    dataset, settings):
477                    return await run_evaluation(
478                        config, retrieval_fn, dataset, settings,
479                        answer_fn=host_answer_fn,
480                    )
481
482                main(retrieval_fn, dataset, settings,
483                     runner=host_runner)
484    """
485    import argparse  # noqa: PLC0415
486
487    parser = argparse.ArgumentParser(description="Run RAG evaluation")
488    parser.add_argument(
489        "--mode",
490        choices=["retrieval-only", "context", "end-to-end"],
491        default="end-to-end",
492        help="Evaluation mode (default: end-to-end)",
493    )
494    parser.add_argument(
495        "--judge",
496        choices=["heuristic", "llm"],
497        default="heuristic",
498        help="Answer evaluation strategy (default: heuristic)",
499    )
500    parser.add_argument(
501        "--n-results",
502        type=int,
503        default=5,
504        help="Number of retrieval results (default: 5)",
505    )
506    parser.add_argument(
507        "--category",
508        default=None,
509        help="Filter by question category (free-form)",
510    )
511    parser.add_argument(
512        "--difficulty",
513        choices=[d.value for d in Difficulty],
514        default=None,
515        help="Filter by difficulty level",
516    )
517    parser.add_argument(
518        "--output",
519        type=str,
520        default=None,
521        help="Output JSON file path for detailed results",
522    )
523
524    args = parser.parse_args()
525
526    config = EvalConfig(
527        mode=args.mode,
528        judge=args.judge,
529        n_results=args.n_results,
530        category=args.category or None,
531        difficulty=Difficulty(args.difficulty) if args.difficulty else None,
532    )
533
534    logging.basicConfig(level=logging.INFO)
535    run = runner or run_evaluation
536    result = asyncio.run(run(config, retrieval_fn, dataset, settings))
537
538    print("\n" + result.summary)
539
540    if args.output:
541        output_data = asdict(result)
542        # Remove large context fields from per-sample to keep output clean
543        for sample in output_data.get("per_sample_results", []):
544            sample.pop("context", None)
545
546        with open(args.output, "w") as f:
547            json.dump(output_data, f, indent=2, default=str)
548        print(f"\nDetailed results saved to: {args.output}")
549
550
551if __name__ == "__main__":  # pragma: no cover
552    main()
