1"""
2Outbox drainer for the ``notification_jobs`` table.
3
4Thin app-side entry point: wires the OmniCare notification store and processor into
5ragnor's generic drainer. The drain loop itself — per-pass stale reclaim,
6``FOR UPDATE SKIP LOCKED`` claiming, failure retry on the next tick, graceful
7SIGINT/SIGTERM — lives in :mod:`ragnor.jobs.drainer`.
8
9Run it directly:
10
11    python -m app.workers.notification_drain
12
13Configuration (all optional, validated by ``app.core.config.Settings``):
14
15    NOTIFICATION_DRAIN_INTERVAL_SECONDS  seconds to sleep when the queue is empty
16    NOTIFICATION_DRAIN_BATCH_SIZE       jobs delivered per pass
17
18Multiple drainers can run concurrently: ``SKIP LOCKED`` keeps two workers from
19delivering the same job, and the unique key on the feed's ``job_id`` keeps a
20redelivery after a reclaimed lock from notifying a claimant twice.
21
22Delivery here never touches the request path: the claim endpoints enqueue a job row
23in the claim transaction, and this process is what turns it into a notification.
24"""
25
26from __future__ import annotations
27
28import sys
29
30from app.core.config import settings
31from app.domain.notifications.outbox import NotificationJobProcessor, NotificationJobStore
32from ragnor.jobs.drainer import main as _ragnor_main
33
34
35def main() -> int:
36    """Console entry point: drain the notification outbox until signaled."""
37    store = NotificationJobStore()
38    processor = NotificationJobProcessor()
39    return _ragnor_main(
40        store,
41        processor,
42        interval=settings.notification_drain_interval_seconds,
43        batch_size=settings.notification_drain_batch_size,
44        worker_id=settings.worker_id,
45    )
46
47
48if __name__ == "__main__":
49    sys.exit(main())
