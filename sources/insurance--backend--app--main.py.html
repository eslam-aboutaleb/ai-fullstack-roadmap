84async def _run_alembic_migrations() -> None:
85    """Apply pending Alembic migrations during application startup.
86
87    Runs ``alembic upgrade head`` in a subprocess so that the migration
88    environment can safely import application settings and models without
89    conflicting with the running Uvicorn event loop.
90
91    Raises:
92        RuntimeError: If migrations fail. This causes the application to
93            fail fast rather than running with a mismatched schema.
94    """
95    backend_dir = Path(__file__).resolve().parents[1]
96    try:
97        # Run in a worker thread: subprocess.run blocks for up to
98        # the timeout, and this runs inside the async lifespan.
99        result = await asyncio.to_thread(  # noqa: S603
100            subprocess.run,
101            [sys.executable, "-m", "alembic", "upgrade", "head"],
102            cwd=backend_dir,
103            capture_output=True,
104            text=True,
105            timeout=120,
106            check=False,
107        )
108        if result.returncode == 0:
109            logger.info("Database migrations applied successfully.")
110            if result.stdout.strip():
111                logger.debug("Alembic output: %s", result.stdout.strip())
112        else:
113            error_msg = (
114                "Alembic migration failed with exit code %s: %s",
115                result.returncode,
116                result.stderr.strip(),
117            )
118            logger.error(*error_msg)
119            raise RuntimeError(f"Database migration failed: {result.stderr.strip()}")
120    except Exception as e:
121        logger.error("Failed to run database migrations: %s", e)
122        raise RuntimeError(f"Failed to run database migrations: {e}") from e

125async def _configure_distributed_state() -> None:
126    """Wire the session store and the Redis-backed caches from settings.
127
128    Both stores degrade rather than fail: Redis that is not configured, not
129    answering, or misconfigured leaves the in-process behaviour in place, which
130    costs replay deduplication and global rate limits but never the endpoints
131    themselves. Wired here, once, so the first request does not pay for it.
132    """
133    try:
134        await configure_idempotency_storage()
135    except Exception:
136        logger.warning(
137            "Idempotency storage could not be configured; staying in-process.",
138            exc_info=True,
139        )
140    try:
141        await warm_session_service()
142    except Exception:
143        logger.warning(
144            "Session storage could not be warmed; this process keeps in-memory sessions.",
145            exc_info=True,
146        )

149@asynccontextmanager
150async def lifespan(app: FastAPI):
151    """
152    Application lifespan handler.
153    On startup:
154      - Applies pending database migrations via Alembic.
155      - Configures LiteLLM environment variables.
156      - Registers the cost/latency telemetry hooks, and brings tracing up
157        when it is configured.
158      - Selects session storage and the shared caches (Redis when configured).
159      - Ingests the policy document into pgvector store (idempotent).
160      - Ingests the procedure playbooks into their own index, gated separately
161        from the policy corpus by OMNICARE_INGEST_PROCEDURES.
162    On shutdown:
163      - Gracefully terminates running sessions and resources.
164
165    Both ingests are best-effort: a failure is logged and startup continues,
166    so a broken policy file cannot take the claim endpoints down with it.
167    """
168    logger.info(
169        "Starting %s (Environment: %s, Version: %s)...",
170        settings.app_name,
171        settings.environment,
172        settings.app_version,
173    )
174
175    await _run_alembic_migrations()
176
177    # Configure LiteLLM environment variables from centralized settings
178    try:
179        configure_llm()
180        logger.info("LiteLLM configuration applied")
181    except Exception as e:
182        logger.warning(f"LiteLLM configuration skipped: {e}")
183
184    if settings.observability_enabled:
185        _install_observability_hooks()
186    else:
187        logger.info(
188            "Observability hooks disabled (OBSERVABILITY_ENABLED=false); no telemetry recorded."
189        )
190
191    await _configure_distributed_state()
192
193    # Ingest policy documents into vector store on startup
194    if settings.ingest_on_startup:
195        try:
196            count = await ingest_policy()
197            logger.info(f"Policy ingestion complete: {count} chunks indexed")
198        except Exception as e:
199            logger.error(f"Policy ingestion failed: {e}")
200            # Server continues running so claim status/submission endpoints remain available
201    else:
202        logger.info(
203            "Startup ingestion disabled (INGEST_ON_STARTUP=false); "
204            "serving the policy index already present in the database."
205        )
206
207    # Ingest the procedure playbooks into their own index. Gated separately from
208    # the policy corpus because the two are operated independently: compliance
209    # ships a new playbook without touching policies, and the call is idempotent
210    # so a warm boot costs a file read plus a snapshot compare. As above, a
211    # failure here must not take the chat endpoints down with it.
212    if procedures_ingestion_enabled():
213        try:
214            count = await ingest_procedures()
215            logger.info(f"Procedure ingestion complete: {count} chunks indexed")
216        except Exception as e:
217            logger.error(f"Procedure ingestion failed: {e}")
218
219    yield  # Server is running and receiving traffic
220
221    logger.info(f"Shutting down {settings.app_name}...")
222    from app.core.database import engine  # noqa: PLC0415
223
224    await engine.dispose()

243# Types that may appear in a Pydantic error ``ctx`` and are safe to pass through.
244# Anything else is rendered with ``repr`` so an unexpected object cannot expand into a
245# response body that dumps its attributes.
246_JSON_SAFE_ERROR_TYPES = (str, int, float, bool, type(None))
247_JSON_CONVERTED_ERROR_TYPES = (Decimal, UUID, datetime, date)

250def _jsonify_validation_value(value: Any) -> Any:  # noqa: PLR0911
251    """Convert one value from ``exc.errors()`` into a JSON-serialisable primitive.
252
253    Each branch maps one input shape to its JSON form. The table is explicit so an
254    unexpected type degrades to ``repr`` rather than being expanded attribute by
255    attribute, which is what ``jsonable_encoder`` would do.
256    """
257    if isinstance(value, bytes | bytearray):
258        return value.decode("utf-8", errors="replace")
259    if isinstance(value, _JSON_SAFE_ERROR_TYPES):
260        return value
261    if isinstance(value, Decimal):
262        # Keep integral values as int so the wire format is unchanged for whole-number
263        # constraints (0 stays 0, not 0.0); fractional constraints become float.
264        return int(value) if value == value.to_integral_value() else float(value)
265    if isinstance(value, UUID | datetime | date):
266        return value.isoformat()
267    if isinstance(value, dict):
268        return {str(key): _jsonify_validation_value(item) for key, item in value.items()}
269    return (
270        [_jsonify_validation_value(item) for item in value]
271        if isinstance(value, list | tuple | set)
272        else repr(value)
273    )

276def _sanitize_error_detail(value: Any) -> Any:
277    """Coerce a Pydantic error payload into JSON-serialisable primitives.
278
279    ``exc.errors()`` carries raw constraint values inside ``ctx``, and those are not
280    limited to JSON types: a ``gt`` constraint on a ``Decimal`` field puts
281    ``Decimal("0")`` in ``ctx``, and a UUID constraint puts a ``UUID``. Embedding them
282    verbatim makes ``JSONResponse`` raise ``TypeError``, which turns a validation failure
283    into an opaque 500.
284
285    This uses an explicit conversion table rather than ``jsonable_encoder`` on purpose.
286    The encoder expands unknown objects through ``dict()`` then ``vars()``, so a custom
287    validator's exception would be serialised attribute by attribute into the response
288    body, and it still raises when neither succeeds -- reintroducing the very failure
289    this exists to prevent.
290    """
291    return _jsonify_validation_value(value)

294@app.exception_handler(RequestValidationError)
295async def request_validation_exception_handler(
296    request: Request,
297    exc: RequestValidationError,
298) -> JSONResponse:
299    """
300    Convert FastAPI/Pydantic RequestValidationError into a standardized
301    ErrorResponse envelope with HTTP 422 Unprocessable Content.
302
303    This ensures the actual response schema matches the OpenAPI declaration
304    in the /chat endpoint (responses[422] = ErrorResponse).
305    """
306    logger.warning(
307        "Validation error on %s %s: %s",
308        request.method,
309        request.url.path,
310        _sanitize_error_detail(exc.errors()),
311    )
312
313    error = ErrorDetail(
314        code="VALIDATION_ERROR",
315        message="Request validation failed. See details for field-level errors.",
316        details=_sanitize_error_detail(exc.errors()),
317    )
318    return JSONResponse(
319        status_code=422,
320        content=ErrorResponse(error=error).model_dump(),
321    )

371@app.exception_handler(HTTPException)
372async def http_exception_handler(
373    request: Request,
374    exc: HTTPException,
375) -> JSONResponse:
376    """
377    Convert all HTTPExceptions into a standardized ErrorResponse envelope.
378
379    This ensures 401, 403, 404, 422, 500, etc. all return the same
380    {"error": {"code": "...", "message": "...", "details": ...}} shape.
381    """
382    logger.warning(
383        "HTTP %s on %s %s: %s",
384        exc.status_code,
385        request.method,
386        request.url.path,
387        exc.detail,
388    )
389    # Map status codes to error codes
390    code_map = {
391        400: "BAD_REQUEST",
392        401: "UNAUTHORIZED",
393        403: "FORBIDDEN",
394        404: "NOT_FOUND",
395        409: "CONFLICT",
396        422: "VALIDATION_ERROR",
397        429: "RATE_LIMIT_EXCEEDED",
398        500: "INTERNAL_ERROR",
399        503: "SERVICE_UNAVAILABLE",
400    }
401    error = ErrorDetail(
402        code=code_map.get(exc.status_code, "HTTP_ERROR"),
403        message=exc.detail if isinstance(exc.detail, str) else str(exc.detail),
404        details=None,
405    )
406    return JSONResponse(
407        status_code=exc.status_code,
408        content=ErrorResponse(error=error).model_dump(),
409    )
