1"""
2Pydantic schemas for the OmniCare Financial backend.
3
4This module defines the request/response envelopes, validation models,
5and error payloads used across all API v1 endpoints. Keeping all schemas
6in a single module ensures:
7
8  - Consistent validation rules (e.g., claim amounts must be positive).
9  - A single source of truth for the OpenAPI contract exposed at /docs.
10  - Reusable models between the REST layer and the agent tool layer.
11
12Security-sensitive schemas (``ClaimSubmission``) enforce strict field
13constraints to prevent malformed or malicious payloads from reaching
14the database.
15"""

71class ChatResponse(BaseModel):
72    """Response schema for the ``POST /api/v1/chat`` endpoint.
73
74    Attributes:
75        response: The agent's synthesized text response.
76        sources: Document citation sources and policy sections referenced
77            in the response.
78        tool_calls: List of tools invoked with arguments and execution results,
79            providing an auditable trace of the agent's reasoning path.
80        message_id: The backend-assigned id of the persisted assistant answer.
81            This is the id the feedback endpoint accepts, so a client must prefer

95        description="Document citation sources and policy sections referenced",
96    )
97    tool_calls: list[dict[str, Any]] = Field(
98        default_factory=list,
99        description="List of tools invoked with arguments and execution results",
100    )
101    message_id: str | None = Field(
102        default=None,
103        description=(
104            "Backend-assigned id of the persisted assistant answer; the id the "
105            "feedback endpoint accepts. None when the turn was not persisted."
106        ),
107    )
108    user_message_id: str | None = Field(
109        default=None,
110        description=(
111            "Backend-assigned id of the persisted user message. None when the "
112            "turn was not persisted."
113        ),
114    )
115
116    model_config = ConfigDict(
117        json_schema_extra={
118            "example": {
119                "response": "Water damage caused by sudden pipe bursts is covered up to $25,000.",
120                "sources": ["Section 1: Home Water Damage Coverage (sample_policy.md)"],
121                "tool_calls": [
122                    {
123                        "name": "query_policy",

376class ConversationRenameRequest(BaseModel):
377    """Request schema for renaming a conversation.
378
379    Attributes:
380        title: The new title, stored trimmed. The 1-255 bounds mirror the
381            ``conversations.title`` column, so an over-long title is refused by
382            validation instead of by the database.
383    """
384
385    title: str = Field(..., min_length=1, max_length=255)
386
387    @field_validator("title", mode="before")
388    @classmethod
389    def _trim_title(cls, value: str) -> str:
390        """Trim the title before the length bounds are applied.
391
392        Trimming first means the stored title is what the user sees rather
393        than the string they typed, and a title made only of whitespace is
394        refused as empty instead of surviving as a blank row.
395
396        Args:
397            value: The raw title from the request body.
398
399        Returns:
400            str: The title without surrounding whitespace.
401        """
402        return value.strip() if isinstance(value, str) else value
