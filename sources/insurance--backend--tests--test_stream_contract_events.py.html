36def _sse_events(payload: str) -> list[dict[str, Any]]:
37    """Parse an SSE body into the decoded JSON payload of every ``data:`` frame.
38
39    Args:
40        payload: The raw ``text/event-stream`` response body.
41
42    Returns:
43        list[dict[str, Any]]: One decoded payload per frame, in order.
44    """
45    events: list[dict[str, Any]] = []
46    for raw_block in payload.split("\n\n"):
47        block = raw_block.strip()
48        if not block:
49            continue
50        data = block.removeprefix("data:").strip()
51        events.append(json.loads(data))
52    return events

55def _payloads(frames: list[str]) -> list[dict[str, Any]]:
56    """Decode SSE frames into their JSON objects."""
57    return [json.loads(frame.removeprefix("data:").strip()) for frame in frames]

60_PRISTINE_AGENT_MODEL: dict[str, Any] = {"model": None}
61"""The agent's real model, captured once on first use.
62
63The agent is a process-wide singleton that several suites swap for a scripted
64double. Restoring to *this* value rather than to whatever is installed when a
65suite happens to run is what stops a leaked double from becoming the next
66suite's baseline. Held in a dict so the lazy capture needs no ``global``.
67"""

70def _pristine_agent_model(agent_module: Any) -> Any:
71    """Return the agent's real model, capturing it on first use."""
72    if _PRISTINE_AGENT_MODEL["model"] is None:
73        _PRISTINE_AGENT_MODEL["model"] = agent_module.omnicare_agent.model
74    return _PRISTINE_AGENT_MODEL["model"]

79    """Install a scripted model into the real agent and restore the original.
80
81    Yields:
82        Callable[[ScriptedTurn], ScriptedLlm]: Installer for one scripted turn.

89    def _install(turn: ScriptedTurn) -> ScriptedLlm:
90        double = ScriptedLlm(turn=turn)
91        agent_module.omnicare_agent.model = double
92        return double
93
94    yield _install
95
96    # Restore to the pristine model, not to whatever happens to be installed
97    # right now. Capturing the "original" per test chains: if an earlier test
98    # leaves a scripted double behind, this test would capture that double as
99    # its baseline and every later test in the process would silently run
100    # against the wrong script, which shows up as a tool trace that does not
101    # match the turn under test.
102    agent_module.omnicare_agent.model = original

105async def _collect_stream(turn: ScriptedTurn) -> tuple[list[str], _StreamResult]:
106    """Run one scripted turn through ``run_agent_stream``.
107
108    Args:
109        turn: The scripted model behaviour for this turn.
110
111    Returns:
112        tuple[list[str], _StreamResult]: The SSE frames yielded by the agent
113        and the result container they were collected into.
114    """
115    double = ScriptedLlm(turn=turn)
116    with patch("app.agent.agent.omnicare_agent.model", double):
117        result = _StreamResult()
118        frames = [
119            frame
120            async for frame in run_agent_stream(
121                user_id=str(uuid.uuid4()),
122                message="Is a burst pipe covered?",
123                result=result,
124            )
125        ]
126    return frames, result

354def test_stream_endpoint_emits_json_error_frame_then_failed_terminal(
355    test_client, mock_current_user
356):
357    """A failed turn emits an ``error`` frame and still ends the stream.
358
359    The ``error`` frame must be parseable JSON and be followed immediately by
360    the terminal ``response_complete`` with ``status == "error"``, so the
361    client can render the failure instead of a silently truncated answer.
362    """
363
364    async def failing_run(user_id, message, result):
365        result.session_id = "session-contract-error"
366        yield 'data: {"type": "text_delta", "content": "Partial answer"}\n\n'
367        raise RuntimeError("upstream model blew up in a very specific way")
368
369    with patch("app.api.v1.chat_stream.run_agent_stream") as mock_run_agent_stream:
370        mock_run_agent_stream.side_effect = failing_run
371
372        response = test_client.post(
373            "/api/v1/chat/stream",

375            headers=mock_current_user,
376        )
377

382    assert len(error_events) == 1
383

385    assert error_event["code"] == "stream_failed"
386    assert error_event["message"] == "Streaming failed"
387    assert StreamError.model_validate(error_event).code == "stream_failed"

389    assert "upstream model blew up" not in response.text
390
391    assert events[-1]["type"] == "response_complete"
392    assert events[-1]["status"] == "error"
393    assert events[-1]["sources"] == []
