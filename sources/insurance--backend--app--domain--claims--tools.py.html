25def _to_context_result(result: dict[str, Any]) -> dict[str, Any]:
26    """Project one raw retrieval row onto the LLM-facing envelope.
27
28    ``owner_id`` is the caller's own raw user UUID — already implied by the
29    retrieval scoping — and ``_rrf_score`` plus ``distance`` are internal
30    ranking signals; none of them help the model answer, so they are dropped.
31
32    Args:
33        result: Raw retrieval row with ``document``, ``metadata``,
34            ``distance``, and ``_rrf_score`` keys.
35
36    Returns:
37        dict: The row with ``document``, sanitized ``metadata``, and a
38        ``citation`` string in the same shape ``get_claim_status`` returns.
39    """
40    metadata = dict(result.get("metadata") or {})
41    metadata.pop("owner_id", None)
42    metadata.pop("_rrf_score", None)
43    claim_id = metadata.get("claim_id", "")
44    policy_number = metadata.get("policy_number", "")
45    return {
46        "document": result.get("document", ""),
47        "metadata": metadata,
48        "citation": f"Claim {claim_id} in the OmniCare claims system (policy {policy_number}).",
49    }

52async def search_claims(query: str, n_results: int = 5) -> dict[str, Any]:
53    """
54    Search claims using hybrid search (vector + keyword).
55
56    Use this tool when the user asks about their claims, claim status,
57    claim history, or mentions a claim without giving its ID.
58
59    The search combines:
60    - Vector similarity search (pgvector) for semantic matching
61    - PostgreSQL full-text search (tsvector) for keyword matching
62    - Reciprocal Rank Fusion (RRF) to merge results from both methods
63
64    Args:
65        query: Natural language query to search claims.
66            Example: "What is the status of my water damage claim?"
67        n_results: Maximum number of results to return (default: 5).
68
69    Returns:
70        dict: An envelope with the keys:
71            - results: List of matches, each with ``document``, ``metadata``
72              (claim_id, policy_number, claim_type, status, amount,
73              description), and ``citation``.
74            - count: Number of matches in ``results``.
75            - message: Set (with empty ``results``) when the search succeeded
76              but found no matching claim.
77            - error: Set (with empty ``results``) when the lookup itself
78              failed or the caller is not authenticated.
79
80    Examples:
81        >>> # Search for claim status
82        >>> envelope = await search_claims("water damage claim status")
83        >>> if envelope["error"]:
84        ...     print("The lookup failed:", envelope["error"])
85        ... elif envelope["message"]:
86        ...     print("No matching claim:", envelope["message"])
87        ... else:
88        ...     for r in envelope["results"]:
89        ...         print(f"Claim: {r['metadata']['claim_id']}")
90        ...         print(f"Citation: {r['citation']}")
91    """
92    # Resolved at call time so importing this module never
93    # cycles through app.agent (the agent imports this tool
94    # at module level).
95    from app.agent.context import current_user_id  # noqa: PLC0415
96
97    try:
98        user_uuid = current_user_id.get()
99    except LookupError:
100        logger.error("current_user_id not found in context during claims search.")
101        return {
102            "results": [],
103            "count": 0,
104            "message": None,
105            "error": "Unauthorized claim search.",
106        }
107
108    try:
109        results = await retrieve_claims_hybrid(query=query, user_id=user_uuid, n_results=n_results)
110        if not results:
111            return {
112                "results": [],
113                "count": 0,
114                "message": "No relevant claims found matching your search.",
115                "error": None,
116            }
117        return {
118            "results": [_to_context_result(result) for result in results],
119            "count": len(results),
120            "message": None,
121            "error": None,
122        }
123    except Exception as exc:
124        logger.exception("Error during hybrid claims search: %s", exc)
125        return {
126            "results": [],
127            "count": 0,
128            "message": None,
129            "error": "Unable to search claims at this time. Please try again later.",
130        }
