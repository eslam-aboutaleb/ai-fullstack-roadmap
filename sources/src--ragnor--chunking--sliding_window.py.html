1"""Sliding-window chunking.
2
3Splits text into overlapping chunks based on word count: each
4chunk holds ``chunk_size`` words and consecutive chunks overlap
5by ``overlap`` words, so context is preserved across chunk
6boundaries.
7"""
8
9from __future__ import annotations
10
11import uuid
12from typing import Any
13
14from ragnor.types import Chunk
15
16
17def sliding_window_chunk(
18    text: str,
19    chunk_size: int = 600,
20    overlap: int = 100,
21) -> list[str]:
22    """Split text into overlapping chunks based on word count.
23
24    Uses a sliding window approach where each chunk contains ``chunk_size``
25    words and subsequent chunks overlap by ``overlap`` words. This preserves
26    context across chunk boundaries, which improves retrieval quality for
27    queries that span section boundaries.
28
29    Args:
30        text: The input text to split into chunks.
31        chunk_size: Maximum number of words per chunk. Defaults to 600.
32        overlap: Number of words to overlap between consecutive chunks.
33            Defaults to 100. Must satisfy ``0 <= overlap < chunk_size``.
34
35    Returns:
36        list[str]: A list of text chunks. Returns an empty list if the input
37        text is empty or contains no words.
38
39    Raises:
40        ValueError: If ``chunk_size`` is not positive or ``overlap`` is
41            outside ``[0, chunk_size)``.
42    """
43    if chunk_size < 1:
44        raise ValueError(f"chunk_size must be >= 1; got {chunk_size}")
45    if not 0 <= overlap < chunk_size:
46        raise ValueError(
47            f"overlap must satisfy 0 <= overlap < chunk_size ({chunk_size}); got {overlap}"
48        )
49
50    words = text.split()
51    chunks = []
52
53    if not words:
54        return []
55
56    for i in range(0, len(words), chunk_size - overlap):
57        chunk = " ".join(words[i : i + chunk_size])
58        chunks.append(chunk)
59        if i + chunk_size >= len(words):
60            break
61
62    return chunks
63
64
65class SlidingWindowChunker:
66    """Chunker that splits text into overlapping word-count windows."""
67
68    def __init__(self, chunk_size: int = 600, overlap: int = 100) -> None:
69        """Configure the window size and overlap.
70
71        Args:
72            chunk_size: Maximum number of words per chunk.
73            overlap: Words shared between consecutive chunks.
74
75        Raises:
76            ValueError: If ``chunk_size`` is not positive or ``overlap``
77                is outside ``[0, chunk_size)``.
78        """
79        if chunk_size < 1:
80            raise ValueError(f"chunk_size must be >= 1; got {chunk_size}")
81        if not 0 <= overlap < chunk_size:
82            raise ValueError(
83                f"overlap must satisfy 0 <= overlap < chunk_size ({chunk_size}); got {overlap}"
84            )
85        self.chunk_size = chunk_size
86        self.overlap = overlap
87
88    def chunk(self, text: str) -> list[Chunk]:
89        """Split ``text`` into overlapping chunks.
90
91        Args:
92            text: The input text to split.
93
94        Returns:
95            Chunks with generated ids and the sliding-window text.
96        """
97        return [
98            Chunk(id=str(uuid.uuid4()), text=chunk_text)
99            for chunk_text in sliding_window_chunk(text, self.chunk_size, self.overlap)
100        ]
101
102    @property
103    def version(self) -> str:
104        """Version tag for ingestion snapshot comparison."""
105        return "sliding-window-v1"
106
107    @property
108    def config(self) -> dict[str, Any]:
109        """Chunker tunables for ingestion snapshots."""
110        return {"chunk_size": self.chunk_size, "overlap": self.overlap}
