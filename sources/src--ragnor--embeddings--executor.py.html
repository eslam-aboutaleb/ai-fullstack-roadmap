1"""Resilient embedding executor with retry, concurrency, and validation.
2
3:class:`EmbeddingExecutor` wraps an :class:`~ragnor.embeddings.base.EmbeddingFunction`
4and adds production-grade concerns:
5
6* **Bounded concurrency** via :class:`asyncio.Semaphore`.
7* **Bounded input queue** that back-pressures callers when the executor is
8  saturated.
9* **Per-call timeout** via :func:`asyncio.wait_for`.
10* **Exponential back-off with jitter** for transient failures.
11* **Retry-After header / hint** support where the provider exposes it.
12* **Partial-batch error accounting**: a response with the wrong number of
13  vectors raises :class:`EmbeddingResponseError` immediately (never silently
14  misaligns chunks and vectors).
15* **Response validation**: checks count, finite values, and dimension.
16* **Permanent error fast-path**: authentication, dimension, and
17  configuration errors are not retried.
18
19The ``clock`` and ``sleep`` parameters accept injectable callables so tests
20can use a fake clock without patching :mod:`time` globally.
21
22Usage::
23
24    executor = EmbeddingExecutor(
25        fn=my_embedding_function,
26        profile=profile,
27        max_concurrency=4,
28        max_retries=3,
29    )
30    vectors, metrics = await executor.embed(["hello", "world"])
31"""

86_TRANSIENT_PATTERNS = (
87    re.compile(r"rate.?limit", re.IGNORECASE),
88    re.compile(r"timeout", re.IGNORECASE),
89    re.compile(r"connection", re.IGNORECASE),
90    re.compile(r"service.?unavailable", re.IGNORECASE),
91    re.compile(r"internal.?server", re.IGNORECASE),
92    re.compile(r"too many requests", re.IGNORECASE),
93    re.compile(r"5\d\d", re.IGNORECASE),
94)
95
96_PERMANENT_PATTERNS = (
97    re.compile(r"auth", re.IGNORECASE),
98    re.compile(r"invalid.?key", re.IGNORECASE),
99    re.compile(r"permission", re.IGNORECASE),
100    re.compile(r"dimension", re.IGNORECASE),
101    re.compile(r"model.?not.?found", re.IGNORECASE),
102    re.compile(r"not.?found", re.IGNORECASE),
103)
104
105_RETRY_AFTER_PATTERN = re.compile(r"retry[_\s-]*after[:\s]+(\d+\.?\d*)", re.IGNORECASE)

108def _classify_exception(exc: BaseException) -> tuple[bool, float | None]:
109    """Classify *exc* as (is_transient, retry_after_seconds).
110
111    Returns ``(False, None)`` for permanently-failing exceptions.
112    """
113    # asyncio.CancelledError is neither transient nor permanent -- propagate.
114    if isinstance(exc, asyncio.CancelledError):
115        raise exc  # re-raise to honour cancellation
116
117    msg = str(exc)
118
119    # Check permanent patterns first (higher specificity).
120    for pattern in _PERMANENT_PATTERNS:
121        if pattern.search(msg):
122            return False, None
123
124    # Check transient patterns.
125    retry_after: float | None = None
126    match = _RETRY_AFTER_PATTERN.search(msg)
127    if match:
128        import contextlib  # noqa: PLC0415
129
130        with contextlib.suppress(ValueError):
131            retry_after = float(match.group(1))
132
133    for pattern in _TRANSIENT_PATTERNS:
134        if pattern.search(msg):
135            return True, retry_after
136
137    # Unknown exception -- treat as transient (fail-safe for network issues).
138    return True, retry_after

141def _validate_response(
142    texts: list[str],
143    vectors: list[list[float]],
144    profile: EmbeddingProfile,
145) -> None:
146    """Validate that *vectors* is a safe response for *texts*.
147
148    Raises:
149        EmbeddingResponseError: When the response cannot be safely used.
150    """
151    if len(vectors) != len(texts):
152        raise EmbeddingResponseError(
153            f"Provider returned {len(vectors)} vectors for {len(texts)} inputs; "
154            "cannot safely align chunks to vectors."
155        )
156    for i, vec in enumerate(vectors):
157        if len(vec) != profile.dimension:
158            raise EmbeddingResponseError(
159                f"Vector {i} has dimension {len(vec)}; profile expects {profile.dimension}."
160            )
161        for j, val in enumerate(vec):
162            if not math.isfinite(val):
163                raise EmbeddingResponseError(
164                    f"Vector {i}[{j}] is not finite ({val!r}); "
165                    "refusing to write non-finite embeddings."
166                )

169def _estimate_tokens(texts: list[str]) -> int:
170    """Rough token-count estimate (word count * 1.3, rounded up)."""
171    words = sum(len(t.split()) for t in texts)
172    return math.ceil(words * 1.3)

180class EmbeddingExecutor:
181    """Wraps an :class:`EmbeddingFunction` with retry, concurrency, and validation.
182
183    Args:
184        fn: The embedding function to wrap.
185        profile: Profile that describes the expected output geometry.
186        max_batch_size: Maximum texts per provider call.  The executor
187            splits larger inputs into batches automatically.
188        max_input_tokens_estimate: Soft cap on estimated input tokens per
189            batch.  When exceeded the batch is split further.  ``0``
190            disables token-based splitting.
191        max_concurrency: Maximum simultaneous in-flight provider calls.
192        timeout_seconds: Per-call timeout in seconds.  ``0`` means no
193            timeout.
194        base_backoff: Initial back-off duration in seconds.
195        max_backoff: Maximum back-off duration in seconds.
196        jitter: Maximum random jitter added to the back-off.
197        max_retries: Maximum number of retry attempts after the initial
198            attempt.  ``0`` means no retries.
199        clock: Callable returning the current time (injectable for tests).
200        sleep: Async callable that sleeps for *n* seconds (injectable for
201            tests).
202    """
203
204    def __init__(  # noqa: PLR0913, PLR0917
205        self,
206        fn: EmbeddingFunction,
207        profile: EmbeddingProfile,
208        *,
209        max_batch_size: int = 512,
210        max_input_tokens_estimate: int = 0,
211        max_concurrency: int = 4,
212        timeout_seconds: float = 60.0,
213        base_backoff: float = 1.0,
214        max_backoff: float = 60.0,
215        jitter: float = 0.5,
216        max_retries: int = 3,
217        clock: Callable[[], float] | None = None,
218        sleep: Callable[[float], Any] | None = None,
219        event_sink: Any | None = None,
220    ) -> None:
221        if max_batch_size < 1:
222            raise ValueError(f"max_batch_size must be >= 1, got {max_batch_size}")
223        if max_concurrency < 1:
224            raise ValueError(f"max_concurrency must be >= 1, got {max_concurrency}")
225        if max_retries < 0:
226            raise ValueError(f"max_retries must be >= 0, got {max_retries}")
227
228        self._fn = fn
229        self._profile = profile
230        self._max_batch_size = max_batch_size
231        self._max_input_tokens_estimate = max_input_tokens_estimate
232        self._timeout_seconds = timeout_seconds
233        self._base_backoff = base_backoff
234        self._max_backoff = max_backoff
235        self._jitter = jitter
236        self._max_retries = max_retries
237        self._semaphore = asyncio.Semaphore(max_concurrency)
238        self._clock: Callable[[], float] = clock or time.monotonic
239        self._sleep: Callable[[float], Any] = sleep or asyncio.sleep
240        self._event_sink = event_sink if event_sink is not None else NullEventSink()
241

265    async def embed(
266        self,
267        texts: list[str],
268        *,
269        task_mode: TaskMode | None = None,
270    ) -> tuple[list[list[float]], EmbeddingMetrics]:
271        """Embed *texts* with retry, concurrency control, and validation.
272
273        Args:
274            texts: Input texts.  Empty lists return immediately.
275            task_mode: Override for the task mode.  ``None`` means use
276                the profile's default task mode.
277
278        Returns:
279            A ``(vectors, metrics)`` tuple.  ``vectors[i]`` corresponds
280            to ``texts[i]`` without reordering.
281
282        Raises:
283            EmbeddingPermanentError: For authentication, dimension, or
284                configuration failures that must not be retried.
285            EmbeddingResponseError: When the provider returns a response
286                that cannot be safely aligned with the input.
287            EmbeddingTransientError: When all retry attempts are exhausted.
288        """
289        if not texts:
290            return [], EmbeddingMetrics()
291
292        effective_task_mode = task_mode or self._profile.task_mode
293        batches = self._split_into_batches(texts)
294        all_vectors: list[list[float]] = []
295        combined_metrics = EmbeddingMetrics()
296
297        for batch in batches:
298            vectors, batch_metrics = await self._embed_batch_with_retry(batch, effective_task_mode)
299            all_vectors.extend(vectors)
300            combined_metrics = combined_metrics.merge(batch_metrics)
301
302        return all_vectors, combined_metrics

308    def _split_into_batches(self, texts: list[str]) -> list[list[str]]:
309        """Split *texts* into batches respecting size and token limits."""
310        batches: list[list[str]] = []
311        current: list[str] = []
312        current_tokens = 0
313
314        for text in texts:
315            text_tokens = _estimate_tokens([text])
316            token_overflow = (
317                self._max_input_tokens_estimate > 0
318                and current
319                and current_tokens + text_tokens > self._max_input_tokens_estimate
320            )
321            size_overflow = len(current) >= self._max_batch_size
322
323            if size_overflow or token_overflow:
324                batches.append(current)
325                current = []
326                current_tokens = 0
327
328            current.append(text)
329            current_tokens += text_tokens
330
331        if current:
332            batches.append(current)
333
334        return batches

336    async def _embed_batch_with_retry(
337        self,
338        batch: list[str],
339        task_mode: TaskMode,
340    ) -> tuple[list[list[float]], EmbeddingMetrics]:
341        """Embed a single *batch* with exponential back-off retries."""
342        metrics = EmbeddingMetrics()
343        last_exc: BaseException | None = None
344
345        for attempt in range(self._max_retries + 1):
346            if attempt > 0:
347                delay = self._backoff_delay(attempt, last_exc)
348                logger.debug(
349                    "Embedding retry %d/%d after %.2fs (batch size %d)",
350                    attempt,
351                    self._max_retries,
352                    delay,
353                    len(batch),
354                )
355                self._event_sink.emit(
356                    TelemetryEvent(
357                        event_kind=EventKind.RETRY_SCHEDULED,
358                        stage="embedding",
359                        duration_ms=delay * 1000.0,
360                        error=last_exc,
361                        counts={
362                            "attempt": attempt,
363                            "max_retries": self._max_retries,
364                            "batch_size": len(batch),
365                        },
366                    )
367                )
368                await self._sleep(delay)
369                metrics.total_retry_count += 1
370
371            try:
372                vectors = await self._call_provider(batch, task_mode)
373                _validate_response(batch, vectors, self._profile)
374                metrics.embed_calls += 1
375                metrics.cache_misses += len(batch)
376                metrics.total_input_tokens_estimate += _estimate_tokens(batch)
377                self._event_sink.emit(
378                    TelemetryEvent(
379                        event_kind=EventKind.BATCH_EMBEDDED,
380                        stage="embedding",
381                        counts={
382                            "batch_size": len(batch),
383                            "estimated_tokens": _estimate_tokens(batch),
384                            "dimension": self._profile.dimension,
385                        },
386                    )
387                )
388                return vectors, metrics
389
390            except EmbeddingPermanentError:
391                raise
392
393            except asyncio.CancelledError:
394                raise
395
396            except EmbeddingTransientError as exc:
397                last_exc = exc
398                if attempt == self._max_retries:
399                    raise
400
401            except Exception as exc:  # noqa: BLE001
402                is_transient, retry_after = _classify_exception(exc)
403                if not is_transient:
404                    raise EmbeddingPermanentError(str(exc)) from exc
405                last_exc = EmbeddingTransientError(str(exc), retry_after=retry_after)
406                if attempt == self._max_retries:
407                    raise EmbeddingTransientError(
408                        f"Exhausted {self._max_retries} retries: {exc}",
409                        retry_after=retry_after,
410                    ) from exc
411
412        # Should be unreachable.
413        raise EmbeddingTransientError("Retry loop exited unexpectedly.")

415    async def _call_provider(self, batch: list[str], task_mode: TaskMode) -> list[list[float]]:
416        """Call the wrapped embedding function under semaphore and timeout."""
417
418        async def invoke() -> list[list[float]]:
419            parameters = inspect.signature(self._fn).parameters
420            if "task_mode" in parameters:
421                return await self._fn(batch, task_mode=task_mode)
422            if task_mode == TaskMode.DOCUMENT and hasattr(self._fn, "embed_documents"):
423                return await self._fn.embed_documents(batch)
424            if task_mode == TaskMode.QUERY and hasattr(self._fn, "embed_query"):
425                return [await self._fn.embed_query(text) for text in batch]
426            if task_mode != TaskMode.SYMMETRIC:
427                raise EmbeddingPermanentError("Provider does not support the requested task_mode")
428            return await self._fn(batch)
429
430        async with self._semaphore:
431            if self._timeout_seconds > 0:
432                return await asyncio.wait_for(
433                    invoke(),
434                    timeout=self._timeout_seconds,
435                )
436            return await invoke()
437
438    def _backoff_delay(self, attempt: int, last_exc: BaseException | None) -> float:
439        """Compute the back-off delay for *attempt* (1-indexed)."""
440        # Honour Retry-After if the provider supplied one.
441        if isinstance(last_exc, EmbeddingTransientError) and last_exc.retry_after is not None:
442            return last_exc.retry_after
443
444        base = self._base_backoff * (2 ** (attempt - 1))
445        jitter = random.uniform(0, self._jitter)  # noqa: S311
446        return min(base + jitter, self._max_backoff)
