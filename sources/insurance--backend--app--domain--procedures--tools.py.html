1"""
2Procedure playbook tool for the OmniCare AI agent.
3
4This tool loads a domain procedure playbook — the coverage determination
5checklist or the intake triage procedure — out of the procedures index, so the
6agent gets the procedure as grounded text with provenance instead of the whole
7playbook living in the system prompt.
8
9The tool is a plain async function the agent registry can call directly. It uses
10the domain adapter ``app.domain.procedures.retrieval`` for retrieval, which is
11scoped to a single playbook file so a coverage question can never come back
12with triage text attached to it.
13"""

28_PROCEDURE_CHUNK_LIMIT = 16
29"""Chunks pulled per playbook.
30
31Must exceed the chunk count of the largest playbook, or the tool would
32silently return a truncated procedure. A playbook that reaches the limit is
33logged so the corpus split can be revisited.
34"""

36_MAX_NAME_ECHO = 64
37"""Bound on the requested name echoed back in an error.
38
39The name arrives from the model, so it is untrusted text: the echo keeps the
40envelope readable and stops a long name from flooding the response.
41"""

95def _normalise(value: str) -> str:
96    """Fold a requested procedure name into its lookup key."""
97    return re.sub(r"[^a-z0-9]+", "-", value.strip().lower()).strip("-")
98

100def _resolve(name: str) -> _ProcedureSpec | None:
101    """Resolve a requested name to a playbook, or ``None`` when unknown."""
102    key = _normalise(name)
103    canonical = _ALIASES.get(key, key)
104    return _BY_NAME.get(canonical)

107def _sort_key(result: dict[str, Any]) -> tuple[int, int]:
108    """Return the document-order sort key for one chunk.
109
110    Hybrid retrieval ranks by relevance, which would hand the model the
111    playbook's steps out of order. The chunk indexes restore the order the
112    playbook was written in.
113    """
114    metadata = result.get("metadata") or {}
115    chunk_index = metadata.get("chunk_index")
116    sub_chunk_index = metadata.get("sub_chunk_index")
117    return (
118        chunk_index if isinstance(chunk_index, int) else 1 << 30,
119        sub_chunk_index if isinstance(sub_chunk_index, int) else 1 << 30,
120    )

123def _dedupe_sources(results: Iterable[dict[str, Any]]) -> list[dict[str, Any]]:
124    """Build one citation per section, in the order the sections appear."""
125    sources: list[dict[str, Any]] = []
126    seen_sections: set[str] = set()
127    for result in results:
128        metadata = result.get("metadata") or {}
129        section = metadata.get("section") or "Procedure"
130        if section in seen_sections:
131            continue
132        seen_sections.add(section)
133        distance = result.get("distance")
134        sources.append(
135            {
136                "section": section,
137                "source": metadata.get("source") or "procedure playbook",
138                "relevance_score": round(distance, 4) if distance is not None else None,
139            }
140        )
141    return sources

144def _error_envelope(name: str, message: str) -> dict[str, Any]:
145    """Build a structured error envelope that leaks nothing internal."""
146    logger.warning("get_procedure rejected request for %r: %s", name[:_MAX_NAME_ECHO], message)
147    return {
148        "procedure": None,
149        "version": None,
150        "chunk_count": 0,
151        "sources": [],
152        "name": name[:_MAX_NAME_ECHO],
153        "error": message,
154    }

157async def get_procedure(name: str) -> dict[str, Any]:
158    """Load a domain procedure playbook from the procedures index.
159
160    Use this tool when a decision needs a procedure rather than a policy fact:
161    call it before deciding whether a described incident is covered, and before
162    routing a message whose intent is unclear. Then follow the returned
163    ``procedure`` text step by step, and take every policy number, limit, and
164    exclusion from ``query_policy`` rather than from the procedure itself.
165
166    Args:
167        name: Procedure to load. One of ``coverage-review`` or
168            ``claim-triage``.
169
170    Returns:
171        dict: On success, ``procedure`` (the playbook text in section order),
172        ``version`` (the active ingestion version it was served from),
173        ``chunk_count``, ``sources`` (one citation per section), and ``name``.
174        On failure, ``procedure`` is ``None`` and ``error`` explains why.
175    """
176    requested = (name or "").strip()
177    if not requested:
178        return _error_envelope(
179            name or "",
180            "A procedure name is required. Available procedures: "
181            f"{', '.join(sorted(_BY_NAME))}.",
182        )
183
184    spec = _resolve(requested)
185    if spec is None:
186        return _error_envelope(
187            name,
188            f"Unknown procedure '{requested[:_MAX_NAME_ECHO]}'. Available "
189            f"procedures: {', '.join(sorted(_BY_NAME))}.",
190        )
191
192    record = await get_active_version_record(spec.source_file)
193    if record is None:
194        return _error_envelope(
195            name,
196            f"The '{spec.name}' procedure is not in the index yet. Answer from "
197            "the policy documents directly and say the procedure is unavailable.",
198        )
199
200    results = await retrieve_procedures(
201        spec.topic,
202        n_results=_PROCEDURE_CHUNK_LIMIT,
203        source_file=spec.source_file,
204    )
205    if not results:
206        return _error_envelope(
207            name,
208            f"The '{spec.name}' procedure could not be read from the index. "
209            "Answer from the policy documents directly and say the procedure "
210            "is unavailable.",
211        )
212
213    results.sort(key=_sort_key)
214    if len(results) >= _PROCEDURE_CHUNK_LIMIT:
215        # The playbook may have grown past the retrieval window, which would
216        # silently truncate the procedure the model is about to follow.
217        logger.warning(
218            "Procedure '%s' returned %d chunks, the retrieval limit; "
219            "the playbook may be truncated.",
220            spec.name,
221            len(results),
222        )
223
224    return {
225        "procedure": "\n\n".join(result.get("document", "") for result in results),
226        "version": record["version"],
227        "chunk_count": len(results),
228        "sources": _dedupe_sources(results),
229        "name": spec.name,
230        "error": None,
231    }
