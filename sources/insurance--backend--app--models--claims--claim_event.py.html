1"""
2SQLAlchemy ORM model for the ``claim_events`` table.
3
4``claims.status`` keeps only the current state, so every transition destroys the
5value it replaced: nothing can answer "when did this claim last change, and who
6changed it?". This table is the audit trail that survives the overwrite.
7
8The claim lifecycle itself is defined here as well. A transition rule only
9holds if the column it writes, the event that records it, and the endpoint that
10enforces it all agree on the same set of values, so the status enum and the
11legal-transition map live next to the model that persists them rather than
12being re-declared in each layer.
13"""

28class ClaimStatus(str, Enum):
29    """The statuses a claim can hold, in the order a claim moves through them.
30
31    Attributes:
32        SUBMITTED: The claim was filed and is waiting for a handler.
33        UNDER_REVIEW: A claims handler is actively assessing the claim.
34        INFO_REQUESTED: A handler needs more from the claimant before deciding.
35        APPROVED: The claim was accepted. No further transition exists.
36        DENIED: The claim was refused. No further transition exists.
37        WITHDRAWN: The claimant pulled the claim back themselves.
38        CLOSED: The claim left the handler's queue without a decision.
39    """
40
41    SUBMITTED = "Submitted"
42    UNDER_REVIEW = "Under Review"
43    INFO_REQUESTED = "Info Requested"
44    APPROVED = "Approved"
45    DENIED = "Denied"
46    WITHDRAWN = "Withdrawn"
47    CLOSED = "Closed"
48
49
50CLAIM_STATUSES: tuple[ClaimStatus, ...] = tuple(ClaimStatus)
51
52# The statuses a policyholder may set on their own claim. Everything else
53# records a decision only a claims handler can make, so those transitions carry
54# an authority check the API enforces before the state machine is consulted.
55POLICYHOLDER_STATUSES: frozenset[ClaimStatus] = frozenset({ClaimStatus.WITHDRAWN})
56
57# Legal transitions. A status absent from this map has no successor at all,
58# which is how the terminal statuses are expressed.
59ALLOWED_STATUS_TRANSITIONS: dict[ClaimStatus, frozenset[ClaimStatus]] = {
60    ClaimStatus.SUBMITTED: frozenset({ClaimStatus.UNDER_REVIEW, ClaimStatus.WITHDRAWN}),
61    ClaimStatus.UNDER_REVIEW: frozenset(
62        {
63            ClaimStatus.INFO_REQUESTED,
64            ClaimStatus.APPROVED,
65            ClaimStatus.DENIED,
66            ClaimStatus.WITHDRAWN,
67        }
68    ),
69    # A handler who asks for more documents either receives them and returns
70    # the claim to review, or closes it when the claimant never answers.
71    ClaimStatus.INFO_REQUESTED: frozenset({ClaimStatus.UNDER_REVIEW, ClaimStatus.CLOSED}),
72    ClaimStatus.APPROVED: frozenset(),
73    ClaimStatus.DENIED: frozenset(),
74    ClaimStatus.WITHDRAWN: frozenset(),
75    ClaimStatus.CLOSED: frozenset(),
76}

79def parse_claim_status(raw: object) -> ClaimStatus | None:
80    """Interpret a stored or submitted value as a lifecycle status.
81
82    The comparison is case-insensitive and whitespace-tolerant because the
83    column was free text before the lifecycle existed, so a legacy row may
84    hold any casing of a known value.
85
86    Args:
87        raw: The value to interpret, typically the contents of ``claims.status``.
88
89    Returns:
90        ClaimStatus | None: The matching status, or None when the value is not
91        part of the claim lifecycle at all.
92    """
93    if isinstance(raw, ClaimStatus):
94        return raw
95    normalised = str(raw).strip().casefold()
96    for candidate in ClaimStatus:
97        if candidate.value.casefold() == normalised:
98            return candidate
99    return None

102def allowed_next_statuses(current: ClaimStatus | None) -> tuple[str, ...]:
103    """List the display names a claim in ``current`` may move to.
104
105    Args:
106        current: The claim's current status, or None when it is not part of the
107            lifecycle.
108
109    Returns:
110        tuple[str, ...]: The permitted target statuses, sorted so a caller and
111        the API's error message name them in the same order. Empty for a
112        terminal or unrecognized status.
113    """
114    if current is None:
115        return ()
116    return tuple(target.value for target in sorted(ALLOWED_STATUS_TRANSITIONS[current]))

119def is_status_transition_allowed(current: ClaimStatus | None, target: ClaimStatus) -> bool:
120    """Report whether moving from ``current`` to ``target`` is a legal transition.
121
122    Args:
123        current: The claim's current status, or None when it is not part of the
124            lifecycle. None is never a legal starting point, because a claim
125            already outside the lifecycle has no defined path forward.
126        target: The status being requested.
127
128    Returns:
129        bool: True when the transition map permits the move.
130    """
131    return current is not None and target in ALLOWED_STATUS_TRANSITIONS[current]

134class ClaimEvent(Base):
135    """Database model for one transition in a claim's lifecycle.
136
137    Attributes:
138        id: Primary key UUID.
139        claim_uuid: Foreign key to the claim that transitioned. The constraint
140            is real, not just an indexed column: an event that outlived its
141            claim would put a fabricated history in front of a claim nobody can
142            read any more.
143        actor_id: Who performed the transition. Nullable so a system-initiated
144            change can be recorded without attributing it to a person, and
145            released by the database when the actor's account is removed.
146        from_status: Status the claim held before the transition, or None for
147            the event that files the claim.
148        to_status: Status the claim moved to.
149        note: Optional free-text reason recorded alongside the transition.
150        created_at: Timestamp of the transition, generated by the database so
151            that a row written by a worker with a skewed clock still orders
152            correctly against its neighbours.
153    """
154
155    __tablename__ = "claim_events"
156
157    id: Mapped[uuid.UUID] = mapped_column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
158    claim_uuid: Mapped[uuid.UUID] = mapped_column(
159        UUID(as_uuid=True),
160        ForeignKey("claims.id", ondelete="CASCADE"),
161        nullable=False,
162        index=True,
163    )
164    actor_id: Mapped[uuid.UUID | None] = mapped_column(
165        UUID(as_uuid=True),
166        ForeignKey("users.id", ondelete="SET NULL"),
167        nullable=True,
168    )
169    from_status: Mapped[str | None] = mapped_column(String(32), nullable=True)
170    to_status: Mapped[str] = mapped_column(String(32), nullable=False)
171    note: Mapped[str | None] = mapped_column(String, nullable=True)
172    created_at: Mapped[datetime] = mapped_column(
173        DateTime(timezone=True), nullable=False, server_default=func.now()
174    )
175
176    __table_args__ = (
177        # The history read filters on claim_uuid and orders by created_at DESC,
178        # so both columns belong in one index.
179        Index("ix_claim_events_claim_uuid_created_at", claim_uuid, created_at),
180    )
