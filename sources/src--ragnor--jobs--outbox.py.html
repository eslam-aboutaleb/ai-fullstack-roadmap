1"""Generic embedding-job outbox.
2
3The outbox pattern
4decouples row insertion from embedding generation: a
5job row is written in the same transaction as the row
6it refers to, and a background worker claims pending
7jobs, processes them, and records the outcome.
8
9The job table itself is owned by the host
10application: :class:`EmbeddingJobStore` is a
11:class:`~typing.Protocol` implemented against any job
12table, and :class:`JobProcessor` describes the work
13done per job (embedding and upserting — that
14processor lives in the host application).
15
16Outbox invariant
17----------------
18A job row must be committed in the **same transaction**
19as the row it refers to. ``enqueue_job(store, payload,
20session=...)`` therefore only flushes when a caller
21session is supplied; without a session the store opens
22its own transaction and commits.
23"""
24

66def claim_jobs(jobs: list[JobPayload], worker_id: str, now: datetime) -> list[JobPayload]:
67    """Transition claimed jobs to ``processing`` and stamp the lock.
68
69    A failed job is first reset to ``pending`` so its retry bookkeeping
70    starts clean; the lock columns then record which worker claimed it
71    and when, for :func:`reclaim_stale_jobs`. Returns the same job list
72    for call chaining.
73    """
74    for job in jobs:
75        if job.status == "failed":
76            job.status = "pending"
77            job.status_detail = None
78            job.next_retry_at = None
79        job.status = "processing"
80        job.locked_at = now
81        job.locked_by = worker_id
82    return jobs

137async def process_pending_jobs(
138    store: EmbeddingJobStore,
139    processor: JobProcessor,
140    limit: int = 10,
141    worker_id: str | None = None,
142) -> int:
143    """Process pending jobs.
144
145    Claims up to ``limit`` jobs, runs each through ``processor``,
146    and records the outcome. Failed jobs are marked with an error
147    detail and a retry timestamp; jobs that exhaust their attempts
148    move to the terminal ``dead_letter`` state.
149
150    Claimed jobs are stamped with ``locked_at``/``locked_by`` for
151    the whole processing transaction. A worker that dies mid-pass
152    therefore leaves a visible lock, which :func:`reclaim_stale_jobs`
153    later resets.
154
155    Args:
156        store: The job store to claim from and update.
157        processor: The per-job processor.
158        limit: Maximum number of jobs to process in one batch.
159        worker_id: Identity stamped into ``locked_by``.
160
161    Returns:
162        int: Number of jobs successfully processed.
163    """
164    jobs = await store.claim_pending(limit, worker_id or "")
165    jobs = await store.mark_processing(jobs, worker_id or "")
166
167    processed = 0
168    for job in jobs:
169        try:
170            await processor.process(job)
171
172            job.status = "completed"
173            job.completed_at = datetime.now(UTC)
174            await store.mark_completed(job)
175            processed += 1
176            logger.info("Processed job '%s'.", job.job_id)
177        except Exception as exc:
178            job.retry_count += 1
179            if job.retry_count >= job.max_attempts:
180                job.status = "dead_letter"
181                job.status_detail = f"Exhausted {job.max_attempts} attempts: {exc}"
182                job.next_retry_at = None
183                job.locked_at = None
184                job.locked_by = None
185            else:
186                backoff = RETRY_SCHEDULE[min(job.retry_count - 1, len(RETRY_SCHEDULE) - 1)]
187                job.next_retry_at = datetime.now(UTC) + backoff
188                job.status = "failed"
189                job.status_detail = str(exc)
190                job.locked_at = None
191                job.locked_by = None
192            await store.mark_failed(job, exc)
193            logger.error(
194                "Job '%s' failed (retry %d): %s",
195                job.job_id,
196                job.retry_count,
197                exc,
198            )
199
200    return processed

203async def reclaim_stale_jobs(
204    store: EmbeddingJobStore,
205    stale_after_seconds: float | None = None,
206) -> int:
207    """Reset jobs abandoned in ``processing`` back to ``pending``.
208
209    A drainer that dies mid-upsert leaves its rows locked in
210    ``processing`` forever: the claim query only selects ``pending``
211    or retry-due ``failed`` rows, so no live worker can pick them
212    up again. This reclaims rows whose ``locked_at`` is older than
213    ``stale_after_seconds`` — or whose lock timestamp is NULL, which
214    marks rows locked before the column existed — so a replacement
215    worker can process them.
216
217    Args:
218        store: The job store to reclaim within.
219        stale_after_seconds: Lock age in seconds that counts as
220            abandoned. Defaults to the store's own default.
221
222    Returns:
223        int: Number of jobs reclaimed.
224    """
225    return await store.reclaim_stale(stale_after_seconds)

228async def enqueue_job(
229    store: EmbeddingJobStore,
230    payload: dict[str, Any],
231    session: Any = None,
232) -> None:
233    """Add a job to the outbox.
234
235    When ``session`` is supplied the job is only added and flushed;
236    the caller owns the commit, so the job and the row it refers to
237    land in one transaction. When it is omitted the store opens its
238    own session and commits.
239
240    Callers that are writing the referred row must pass their
241    session. See the module docstring for why.
242
243    Args:
244        store: The job store to enqueue into.
245        payload: The document text and metadata to embed.
246        session: Optional caller-owned session. Not committed when
247            provided.
248    """
249    await store.enqueue(payload, session=session)

252class InMemoryEmbeddingJobStore:
253    """In-memory reference implementation of EmbeddingJobStore."""
254
255    def __init__(self, jobs: list[JobPayload] | None = None) -> None:
256        self.jobs: list[JobPayload] = list(jobs or [])
257
258    async def enqueue(self, payload: dict[str, Any], session: Any = None) -> None:
259        """Enqueue a new job payload."""
260        job_id = payload.get("idempotency_key") or f"job-{len(self.jobs) + 1}"
261        if any(job.job_id == job_id for job in self.jobs):
262            return
263        status = payload.get("status", "pending")
264        job = JobPayload(
265            job_id=job_id,
266            payload=payload,
267            status=status,
268            status_detail=payload.get("status_detail"),
269            retry_count=payload.get("retries", 0),
270            max_attempts=payload.get("max_attempts", MAX_ATTEMPTS),
271            created_at=datetime.now(UTC),
272        )
273        self.jobs.append(job)
274
275    async def claim_pending(self, limit: int, worker_id: str) -> list[JobPayload]:
276        """Claim pending jobs up to limit."""
277        now = datetime.now(UTC)
278        claimable = [
279            j
280            for j in self.jobs
281            if j.status == "pending"
282            or (
283                j.status == "failed"
284                and j.next_retry_at is not None
285                and j.next_retry_at <= now
286                and j.retry_count < j.max_attempts
287            )
288        ]
289        return claimable[:limit]
290
291    async def mark_processing(self, jobs: list[JobPayload], worker_id: str) -> list[JobPayload]:
292        """Mark jobs as processing with lock stamped."""
293        return claim_jobs(jobs, worker_id, datetime.now(UTC))
294
295    async def mark_completed(self, job: JobPayload) -> None:
296        """Mark job as completed."""
297        job.status = "completed"
298        job.completed_at = datetime.now(UTC)
299
300    async def mark_failed(self, job: JobPayload, error: Exception) -> None:
301        """Persist the transition already computed by process_pending_jobs."""
302        job.status_detail = job.status_detail or str(error)
303
304    async def reclaim_stale(self, stale_after_seconds: float | None = None) -> int:
305        """Reclaim stale jobs."""
306        reclaimed = 0
307        now = datetime.now(UTC)
308        threshold = timedelta(seconds=stale_after_seconds or 300)
309        for j in self.jobs:
310            if j.status == "processing" and j.locked_at and (now - j.locked_at) > threshold:
311                j.status = "pending"
312                j.locked_at = None
313                j.locked_by = None
314                reclaimed += 1
315        return reclaimed
316
317    async def get_status_counts(self) -> dict[str, int]:
318        """Return counts of jobs by status."""
319        counts: dict[str, int] = {}
320        for j in self.jobs:
321            counts[j.status] = counts.get(j.status, 0) + 1
322        return counts
