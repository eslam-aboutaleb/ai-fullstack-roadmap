52class FileDocumentSource:
53    """Document source reading a single file from disk."""
54
55    def __init__(self, path: str, encoding: str = "utf-8") -> None:
56        """Initialize the source.
57
58        Args:
59            path: Path to the document file.
60            encoding: File encoding. Defaults to UTF-8.
61        """
62        self._path = path
63        self._encoding = encoding
64
65    async def load(self) -> list[RawDocument]:
66        """Load the file as one :class:`RawDocument`.
67
68        A missing file is not an error: ingestion of an
69        absent document is a no-op, so the source yields
70        no documents and logs a warning (the historical
71        behavior for a missing document file).
72
73        The read runs in a worker thread so a large file
74        never blocks the event loop.
75
76        Returns:
77            One document named after the file, or an empty
78            list when the file does not exist.
79        """
80        try:
81            content = await asyncio.to_thread(self._read)
82        except FileNotFoundError:
83            logger.warning("Document not found at '%s'; skipping.", self._path)
84            return []
85        return [RawDocument(source_name=Path(self._path).name, content=content)]
86
87    def _read(self) -> str:
88        """Read the file contents (runs in a worker thread)."""
89        with open(self._path, encoding=self._encoding) as f:
90            return f.read()

93class ListDocumentSource:
94    """Document source over a fixed list of documents.
95
96    Accepts :class:`ragnor.types.Document` objects (mapping
97    ``Document.id`` to the raw document's source name and external
98    id) or pre-built :class:`RawDocument` objects, which pass
99    through unchanged. Useful beyond the builder -- tests and
100    one-shot ingestion of in-memory documents.
101    """
102
103    def __init__(self, documents: Sequence[Document | RawDocument]) -> None:
104        """Initialize the source with a fixed document list.
105
106        Args:
107            documents: Documents to serve; ``Document`` objects are
108                converted to :class:`RawDocument` on load.
109        """
110        self._documents = list(documents)
111
112    async def load(self) -> list[RawDocument]:
113        """Return the fixed documents as :class:`RawDocument` objects.
114
115        Returns:
116            The documents; a ``Document``'s id is recorded as both the
117            source name and the external id.
118        """
119        return [
120            document
121            if isinstance(document, RawDocument)
122            else RawDocument(
123                source_name=document.id,
124                content=document.text,
125                external_id=document.id,
126            )
127            for document in self._documents
128        ]

131class DocumentSourceConnector:
132    """Adapt a legacy source to connector pagination.
133
134    Legacy empty lists are ambiguous: the adapter must trust their historical
135    meaning, except FileDocumentSource whose missing-file behavior is detected
136    separately. The source is loaded once per traversal so pages share a snapshot.
137    """
138
139    def __init__(self, source: DocumentSource) -> None:
140        self.source = source
141        self._documents: list[RawDocument] = []
142        self._scan: LoadResult | None = None
143
144    async def _start(self) -> LoadResult:
145        from ragnor.ingestion.connector_core import (  # noqa: PLC0415
146            LoadResult,
147            SourceStatus,
148            _error_record,
149        )
150
151        try:
152            if isinstance(self.source, FileDocumentSource):
153                # Read directly: legacy load() deliberately hides FileNotFoundError.
154                content = await asyncio.to_thread(self.source._read)
155                loaded = [RawDocument(Path(self.source._path).name, content)]
156            else:
157                loaded = await self.source.load()
158        except FileNotFoundError as exc:
159            return LoadResult(
160                status=SourceStatus.NOT_FOUND, errors=[_error_record(exc)], complete=False
161            )
162        except PermissionError as exc:
163            return LoadResult(
164                status=SourceStatus.INACCESSIBLE, errors=[_error_record(exc)], complete=False
165            )
166        except Exception as exc:
167            return LoadResult(
168                status=SourceStatus.ERROR, errors=[_error_record(exc)], complete=False
169            )
170        return loaded if isinstance(loaded, LoadResult) else LoadResult(loaded)
171
172    async def load_page(self, *, cursor: Any = None, limit: int | None = None) -> LoadResult:
173        from ragnor.ingestion.connector_core import (  # noqa: PLC0415
174            LoadResult,
175            SourceStatus,
176            _error_record,
177        )
178
179        if limit is not None and limit < 1:
180            raise ValueError("limit must be positive")
181        if cursor is None:
182            result = await self._start()
183            if not result.authoritative:
184                return result
185            self._documents = list(result)
186            self._scan = result
187        try:
188            offset = 0 if cursor is None else int(cursor)
189            if offset < 0 or offset > len(self._documents):
190                raise ValueError("Invalid adapter cursor")
191        except (TypeError, ValueError) as exc:
192            return LoadResult(
193                status=SourceStatus.ERROR, errors=[_error_record(exc)], complete=False
194            )
195        end = len(self._documents) if limit is None else offset + limit
196        documents = self._documents[offset:end]
197        has_more = end < len(self._documents)
198        return LoadResult(
199            documents,
200            cursor=cursor,
201            has_more=has_more,
202            next_cursor=str(end) if has_more else None,
203            total_items=len(documents),
204            skipped=self._scan.skipped if self._scan is not None and cursor is None else [],
205            metadata=self._scan.metadata if self._scan is not None else {},
206            checkpoint=self._scan.checkpoint if self._scan is not None else None,
207        )
208
209    async def discover(self, *, cursor: Any = None, limit: int | None = None) -> DiscoveryPage:
210        from ragnor.ingestion.connector_core import DiscoveryPage, SourceItem  # noqa: PLC0415
211
212        page = await self.load_page(cursor=cursor, limit=limit)
213        return DiscoveryPage(
214            [SourceItem(source_name=doc.source_name, external_id=doc.external_id) for doc in page],
215            cursor=cursor,
216            next_cursor=page.next_cursor,
217            has_more=page.has_more,
218            complete=page.complete,
219            status=page.status,
220            errors=page.errors,
221            skipped=page.skipped,
222        )
223
224    async def load(self, *, cursor: Any = None, limit: int | None = None) -> LoadResult:
225        from ragnor.ingestion.connector_core import collect_pages  # noqa: PLC0415
226
227        if cursor is not None:
228            return await self.load_page(cursor=cursor, limit=limit)
229        return await collect_pages(self, page_size=limit)
