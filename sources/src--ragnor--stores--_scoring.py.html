1"""Shared scoring helpers for vector store backends.
2
3The hybrid-search contract ranks two independent stages -- a
4dense vector stage and a sparse keyword stage -- and merges
5them with Reciprocal Rank Fusion. Backends without a native
6full-text search (everything except PostgreSQL) reuse these
7token-overlap and RRF helpers so their result shape and
8ranking stay comparable to the pgvector backend.
9"""

26def l2_distance(left: list[float], right: list[float]) -> float:
27    """Return the L2 distance between two vectors.
28
29    Vectors of different lengths are compared positionally up to
30    the shorter length; stores validate dimensions up front
31    whenever ``embedding_dim`` is configured.
32    """
33    return math.sqrt(sum((a - b) ** 2 for a, b in zip(left, right, strict=False)))

36def token_overlap(query_tokens: set[str], text_tokens: list[str]) -> float:
37    """Return the fraction of query tokens present in ``text_tokens``."""
38    if not query_tokens:
39        return 0.0
40    present = set(text_tokens)
41    return len(query_tokens & present) / len(query_tokens)

44def rrf_merge(  # noqa: PLR0913
45    vector_ranked: Sequence[str],
46    keyword_ranked: Sequence[str],
47    n_results: int,
48) -> list[tuple[str, float]]:
49    """Merge two ranked document-id lists with Reciprocal Rank Fusion.
50
51    Both stages rank independently; a document's score is the sum
52    of ``1 / (60 + rank)`` over the stages that found it (1-based
53    ranks). Ties break by document id so merges are deterministic.
54
55    Args:
56        vector_ranked: Document ids ranked by vector similarity
57            (best first).
58        keyword_ranked: Document ids ranked by keyword score
59            (best first).
60        n_results: Maximum number of merged results to return.
61
62    Returns:
63        At most ``n_results`` ``(doc_id, rrf_score)`` pairs
64        sorted by score descending.
65    """
66    vector_ranks = {doc_id: rank for rank, doc_id in enumerate(vector_ranked, start=1)}
67    keyword_ranks = {doc_id: rank for rank, doc_id in enumerate(keyword_ranked, start=1)}
68
69    def _rrf(doc_id: str) -> float:
70        score = 0.0
71        if doc_id in vector_ranks:
72            score += 1.0 / (60 + vector_ranks[doc_id])
73        if doc_id in keyword_ranks:
74            score += 1.0 / (60 + keyword_ranks[doc_id])
75        return score
76
77    merged = sorted(
78        set(vector_ranks) | set(keyword_ranks),
79        key=lambda doc_id: (-_rrf(doc_id), doc_id),
80    )
81    return [(doc_id, _rrf(doc_id)) for doc_id in merged[:n_results]]

84def fuse_hybrid(  # noqa: PLR0913
85    vector_results: Sequence[dict[str, Any]],
86    keyword_results: Sequence[dict[str, Any]],
87    n_results: int,
88) -> list[dict[str, Any]]:
89    """Merge the two truncated stage results into hybrid results.
90
91    Both stages are already ranked and truncated to
92    ``n_results`` by their capability methods; this function
93    merges the two result lists by document id via
94    :func:`rrf_merge`, preserving ``distance`` from the
95    vector row and ``keyword_score`` from the keyword row.
96    A document found by both stages outranks a single-stage
97    hit; ties break by document id.
98
99    Args:
100        vector_results: The vector stage's result dicts
101            (``id``, ``document``, ``metadata``,
102            ``distance``, ``keyword_score``), best first.
103        keyword_results: The keyword stage's result dicts
104            (same keys), best first.
105        n_results: Maximum number of merged results to
106            return.
107
108    Returns:
109        At most ``n_results`` result dicts sorted by RRF
110        score, each carrying ``id``, ``document``,
111        ``metadata``, ``distance`` (``None`` for
112        keyword-only hits), ``keyword_score`` (``None`` for
113        vector-only hits), and ``_rrf_score``.
114    """
115    vector_ranked = [row["id"] for row in vector_results]
116    keyword_ranked = [row["id"] for row in keyword_results]
117    vector_by_id = {row["id"]: row for row in vector_results}
118    keyword_by_id = {row["id"]: row for row in keyword_results}
119
120    results: list[dict[str, Any]] = []
121    for doc_id, rrf_score in rrf_merge(vector_ranked, keyword_ranked, n_results):
122        vector_row = vector_by_id.get(doc_id)
123        keyword_row = keyword_by_id.get(doc_id)
124        source = vector_row if vector_row is not None else keyword_row
125        results.append(
126            format_search_hit(
127                doc_id=doc_id,
128                document=str(source.get("document", "")),
129                metadata=source.get("metadata") or {},
130                distance=vector_row.get("distance") if vector_row is not None else None,
131                keyword_score=(
132                    keyword_row.get("keyword_score") if keyword_row is not None else None
133                ),
134                rrf_score=rrf_score,
135            )
136        )
137    return results

140def project_metadata(
141    record: dict[str, Any],
142    metadata_fields: Sequence[str] | None,
143) -> dict[str, Any]:
144    """Project ``metadata_fields`` out of ``record``."""
145    if not metadata_fields:
146        return {}
147    return {
148        field: record[field]
149        for field in metadata_fields
150        if field in record and record[field] is not None
151    }

154def format_search_hit(  # noqa: PLR0913
155    doc_id: str,
156    document: str,
157    metadata: dict[str, Any],
158    *,
159    distance: float | None = None,
160    keyword_score: float | None = None,
161    rrf_score: float | None = None,
162) -> dict[str, Any]:
163    """Construct a standard search hit dictionary across store backends."""
164    hit: dict[str, Any] = {
165        "id": doc_id,
166        "document": document,
167        "metadata": metadata,
168        "distance": distance,
169        "keyword_score": keyword_score,
170    }
171    if rrf_score is not None:
172        hit["_rrf_score"] = rrf_score
173    return hit

176def filter_record_map(
177    records: Mapping[str, dict[str, Any]],
178    *,
179    ids: Sequence[str] | None = None,
180    filters: Mapping[str, Any] | None = None,
181    text_field: str = "text",
182    metadata_fields: Sequence[str] | None = None,
183) -> list[dict[str, Any]]:
184    """Filter in-memory records by id and equality filters."""
185    id_set = set(ids) if ids is not None else None
186    results = []
187    for doc_id, record in records.items():
188        if id_set is not None and doc_id not in id_set:
189            continue
190        if filters:
191            match = True
192            for key, value in filters.items():
193                record_val = record.get(key)
194                if isinstance(value, (list, tuple, set, frozenset)):
195                    if record_val not in value:
196                        match = False
197                        break
198                elif record_val != value:
199                    match = False
200                    break
201            if not match:
202                continue
203        results.append(
204            {
205                "id": str(doc_id),
206                "document": record.get(text_field),
207                "metadata": project_metadata(record, metadata_fields),
208            }
209        )
210    return results

213def build_upsert_record(
214    doc: dict[str, Any],
215    id_field: str,
216    text_field: str,
217    extra_fields: Mapping[str, Any] | None = None,
218) -> dict[str, Any]:
219    """Construct a stored record dict from an input document."""
220    doc_id = doc.get("id")
221    record: dict[str, Any] = {
222        id_field: doc_id,
223        text_field: doc[text_field],
224    }
225    for key, value in doc.get("metadata", {}).items():
226        record[key] = value
227    if extra_fields:
228        record.update(extra_fields)
229    return record

232def validate_upsert_documents(  # noqa: PLR0913, PLR0917
233    documents: list[dict[str, Any]],
234    text_field: str,
235    embedding_field: str,
236    id_field: str | None,
237    default_id_field: str,
238    extra_fields: list[str] | None = None,
239    dimension: int | None = None,
240    metric: Any = None,
241) -> str:
242    """Validate identifiers and embeddings for an upsert operation."""
243    from ragnor.validation import (  # noqa: PLC0415
244        validate_embedding,
245        validate_identifier,
246        validate_upsert_identifiers,
247    )
248
249    effective_id = validate_upsert_identifiers(
250        text_field, embedding_field, id_field, default_id_field, extra_fields
251    )
252    if documents:
253        for doc in documents:
254            if dimension is not None:
255                validate_embedding(
256                    doc[embedding_field],
257                    expected_dim=dimension,
258                    label="document embedding",
259                )
260            for key in doc.get("metadata", {}):
261                validate_identifier(key, "metadata key")
262    return effective_id

280def validate_delete_args(ids: Sequence[str] | None, filters: Mapping[str, Any] | None) -> bool:
281    """Validate delete arguments. Return False if ids is empty list (no-op)."""
282    if ids is None and filters is None:
283        raise ValueError("delete requires at least one of ids or filters")
284    if ids is not None:
285        if len(ids) == 0:
286            return False
287        for item in ids:
288            if not isinstance(item, str):
289                raise TypeError(f"All ids must be strings; got {type(item).__name__}")
290    return True

293def score_records_overlap(
294    records: Mapping[str, dict[str, Any]],
295    query: str,
296    text_field: str,
297    filters: Mapping[str, Any] | None,
298    matches_fn: Any,
299) -> dict[str, float]:
300    """Calculate token overlap keyword scores across filtered records."""
301    query_tokens = set(tokenize(query))
302    scores: dict[str, float] = {}
303    for doc_id, record in records.items():
304        if filters and not matches_fn(record, filters):
305            continue
306        score = token_overlap(query_tokens, tokenize(str(record.get(text_field, ""))))
307        if score > 0.0:
308            scores[doc_id] = score
309    return scores
