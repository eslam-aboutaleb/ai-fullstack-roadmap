1"""Claims embedding-job outbox adapter over ragnor.
2
3The claim embedding job is the OmniCare binding of
4ragnor's generic outbox: :class:`SqlAlchemyJobStore`
5implements ragnor's ``EmbeddingJobStore`` protocol
6against the ``EmbeddingJob`` model,
7:func:`enqueue_embedding_job` writes a pending job row
8through :func:`ragnor.jobs.enqueue_job` — in the
9caller's transaction when a session is supplied, so the
10claim row and its job row commit together —
11:class:`ClaimJobProcessor` is the ragnor
12``JobProcessor`` that embeds a claim and upserts its
13entry into the claims vector store (the ``claims``
14binding lives in :mod:`app.domain.claims.retriever`),
15and :func:`process_pending_jobs` /
16:func:`reclaim_stale_jobs` wire the store and processor
17into ragnor's generic job functions.
18
19Invariant for anything that writes an ``embedding_jobs`` row
20------------------------------------------------------------
21A job row must be committed in the **same transaction** as the claim it refers to,
22or written through this outbox by a caller that has already committed that claim.
23
24The reason is that ``embedding_jobs.claim_uuid`` has no foreign key to ``claims``,
25so the database will not stop an orphan job from appearing. An enqueue that commits
26on its own session races the claim transaction: if the claim transaction rolls back
27afterwards, the job survives, points at a claim that does not exist, and the worker
28then upserts a claim index entry for a row nobody can ever read.
29
30Therefore:
31
32- ``enqueue_embedding_job(session=...)`` only flushes. Pass the caller's session.
33- ``enqueue_embedding_job()`` with no session opens its own and commits. Use it only
34  where the claim is already durable.
35- ``submit_claim`` adds the ``EmbeddingJob`` to the caller's session directly.
36"""

91    async def claim_pending(self, limit: int, _worker_id: str) -> list[JobPayload]:
92        """Select claimable jobs with ``FOR UPDATE SKIP LOCKED``.
93
94        ``worker_id`` is part of the ``EmbeddingJobStore`` protocol
95        but is not needed here: the lock columns are stamped by
96        :func:`claim_jobs` after selection.
97        """
98        from sqlalchemy import select  # noqa: PLC0415
99
100        from app.models.claims.embedding_job import EmbeddingJob  # noqa: PLC0415
101
102        now = datetime.now(UTC)
103        async with self._session_factory() as session:
104            result = await session.execute(
105                select(EmbeddingJob)
106                .where(
107                    (EmbeddingJob.status == "pending")
108                    | (
109                        (EmbeddingJob.status == "failed")
110                        & (EmbeddingJob.next_retry_at <= now)
111                        & (EmbeddingJob.retry_count < EmbeddingJob.max_attempts)
112                    )
113                )
114                .order_by(EmbeddingJob.created_at)
115                .limit(limit)
116                .with_for_update(skip_locked=True)
117            )
118            rows = result.scalars().all()
119        return [_payload_from_row(row) for row in rows]

121    async def mark_processing(self, jobs: list[JobPayload], worker_id: str) -> list[JobPayload]:
122        """Transition jobs to ``processing`` and stamp the lock.
123
124        The guarded update (``status IN ('pending', 'failed')``) keeps
125        two concurrent drainers from claiming the same row: only the
126        worker whose update matches still owns the job.
127        """
128        from sqlalchemy import update  # noqa: PLC0415
129
130        from app.models.claims.embedding_job import EmbeddingJob  # noqa: PLC0415
131
132        if not jobs:
133            return []
134
135        now = datetime.now(UTC)
136        claim_jobs(jobs, worker_id, now)
137
138        claimed: list[JobPayload] = []
139        async with self._session_factory() as session:
140            for job in jobs:
141                result = await session.execute(
142                    update(EmbeddingJob)
143                    .where(
144                        EmbeddingJob.id == uuid.UUID(str(job.job_id)),
145                        EmbeddingJob.status.in_(["pending", "failed"]),
146                    )
147                    .values(
148                        status=job.status,
149                        status_detail=job.status_detail,
150                        next_retry_at=job.next_retry_at,
151                        locked_at=job.locked_at,
152                        locked_by=job.locked_by,
153                    )
154                    .returning(EmbeddingJob.id)
155                )
156                if result.scalar() is not None:
157                    claimed.append(job)
158            await session.commit()
159        return claimed

196    async def reclaim_stale(self, stale_after_seconds: float | None = None) -> int:
197        """Reset jobs abandoned in ``processing`` back to ``pending``."""
198        from sqlalchemy import select  # noqa: PLC0415
199
200        from app.models.claims.embedding_job import EmbeddingJob  # noqa: PLC0415
201
202        if stale_after_seconds is None:
203            from app.core.config import settings  # noqa: PLC0415
204
205            stale_after_seconds = settings.job_stale_after_seconds
206
207        cutoff = datetime.now(UTC) - timedelta(seconds=stale_after_seconds)
208        reclaimed = 0
209        async with self._session_factory() as session:
210            result = await session.execute(
211                select(EmbeddingJob)
212                .where(
213                    (EmbeddingJob.status == "processing")
214                    & (EmbeddingJob.locked_at.is_(None) | (EmbeddingJob.locked_at <= cutoff))
215                )
216                .with_for_update(skip_locked=True)
217            )
218            stale_jobs = result.scalars().all()
219            for job in stale_jobs:
220                job.status = "pending"
221                job.status_detail = None
222                job.locked_at = None
223                job.locked_by = None
224                reclaimed += 1
225            if reclaimed:
226                await session.commit()
227
228        if reclaimed:
229            logger.info("Reclaimed %d stale embedding job(s).", reclaimed)
230        return reclaimed

232    async def enqueue(self, payload: dict[str, Any], session: Any = None) -> None:
233        """Write a new pending ``embedding_jobs`` row.
234
235        When ``session`` is supplied the row is only added and flushed;
236        the caller owns the commit, so the job and the claim land in one
237        transaction. Without a session this opens its own session and
238        commits.
239        """
240        from app.models.claims.embedding_job import EmbeddingJob  # noqa: PLC0415
241
242        job = EmbeddingJob(
243            claim_uuid=uuid.UUID(str(payload["claim_uuid"])),
244            claim_id=payload["claim_id"],
245            owner_id=uuid.UUID(str(payload["owner_id"])),
246            claim_type=payload["claim_type"],
247            description=payload["description"],
248            policy_number=payload["policy_number"],
249            claim_status=payload["claim_status"],
250            status="pending",
251            status_detail="pending",
252        )
253        if session is not None:
254            session.add(job)
255            await session.flush()
256        else:
257            async with self._session_factory() as owned_session:
258                owned_session.add(job)
259                await owned_session.commit()

265    async def process(self, job: JobPayload) -> None:
266        """Embed the claim text and upsert it into the claims vector store."""
267        from sqlalchemy import select  # noqa: PLC0415
268
269        from app.core.config import get_settings  # noqa: PLC0415
270        from app.domain.embeddings import EmbeddingFactory, get_vector_store  # noqa: PLC0415
271        from app.models.claims.claim import Claim  # noqa: PLC0415
272        from ragnor.embeddings import get_embedding_dimension  # noqa: PLC0415
273        from ragnor.validation import validate_embedding  # noqa: PLC0415
274
275        payload = job.payload
276        claim_id = payload["claim_id"]
277
278        async with async_session_factory() as session:
279            claim_amount = await session.scalar(
280                select(Claim.amount).where(Claim.claim_id == claim_id)
281            )
282        claim_amount = claim_amount if claim_amount is not None else 0.0
283
284        embed_fn = EmbeddingFactory.get_embedding_function()
285        text_content = f"Claim {claim_id}: {payload['claim_type']} - {payload['description']}"
286        embeddings = await embed_fn([text_content])
287        embedding = embeddings[0]
288        validate_embedding(
289            embedding,
290            expected_dim=get_embedding_dimension(get_settings().embedding_model),
291            label="claim embedding",
292        )
293
294        store = get_vector_store(
295            table_name=CLAIMS_RETRIEVER_SPEC.table_name,
296            id_field=CLAIMS_RETRIEVER_SPEC.id_field,
297        )
298        await store.upsert(
299            documents=[
300                {
301                    "id": payload["claim_uuid"],
302                    "text": text_content,
303                    "embedding": embedding,
304                    "metadata": {
305                        "claim_id": claim_id,
306                        "policy_number": payload["policy_number"],
307                        "claim_type": payload["claim_type"],
308                        "status": payload["claim_status"],
309                        "amount": float(claim_amount),
310                        "description": payload["description"],
311                        "owner_id": payload["owner_id"],
312                    },
313                }
314            ]
315        )

318async def enqueue_embedding_job(  # noqa: PLR0913, PLR0917
319    claim_uuid: uuid.UUID,
320    claim_id: str,
321    owner_id: uuid.UUID,
322    claim_type: str,
323    description: str,
324    policy_number: str,
325    claim_status: str,
326    session: Any = None,
327) -> None:
328    """Add an embedding job for a newly submitted claim.
329
330    Delegates to ragnor's :func:`ragnor.jobs.enqueue_job`
331    with the caller's session: when ``session`` is supplied,
332    the job row is only added and flushed, so the claim and
333    its embedding job are persisted atomically by the
334    caller's commit. When ``session`` is None, ragnor opens
335    and commits its own transaction.
336
337    A failure to enqueue is logged and never raised when this function owns the
338    transaction: the claim it describes is already durable, and failing a
339    committed fact to protect an index would trade one for the other.
340
341    When ``session`` is supplied the situation is the opposite. The job row has
342    only been *flushed*, not committed, so a flush failure has already poisoned
343    the caller's transaction: its next ``commit()`` raises
344    ``PendingRollbackError`` with a traceback that says nothing about the real
345    cause. Re-raising here lets the caller fail on the actual error instead of a
346    misleading one several statements later.
347
348    Args:
349        claim_uuid: Internal UUID of the claim row.
350        claim_id: Human-readable claim identifier.
351        owner_id: UUID of the user who owns the claim.
352        claim_type: Category of the claim.
353        description: Factual description of the incident.
354        policy_number: The policyholder's policy number.
355        claim_status: Current processing status.
356        session: Optional existing database session. When
357            provided, the job row joins the caller's
358            transaction and the caller commits.
359    """
360    store = SqlAlchemyJobStore()
361    payload = {
362        "claim_uuid": str(claim_uuid),
363        "claim_id": claim_id,
364        "owner_id": str(owner_id),
365        "claim_type": claim_type,
366        "description": description,
367        "policy_number": policy_number,
368        "claim_status": claim_status,
369    }
370    try:
371        await enqueue_job(store, payload, session=session)
372        logger.info("Enqueued embedding job for claim '%s'.", claim_id)
373    except Exception as exc:
374        if session is not None:
375            raise
376        logger.exception("Failed to enqueue embedding job for claim '%s': %s", claim_id, exc)

406async def process_pending_jobs(limit: int = 10, worker_id: str | None = None) -> int:
407    """Process pending embedding jobs.
408
409    Fetches up to ``limit`` pending jobs, generates embeddings, and upserts
410    them into the vector store. Failed jobs are marked with an error detail
411    for retry.
412
413    Claimed jobs are stamped with ``locked_at``/``locked_by`` for the whole
414    processing transaction. A worker that dies mid-pass therefore leaves a
415    visible lock, which :func:`reclaim_stale_jobs` later resets.
416
417    Args:
418        limit: Maximum number of jobs to process in one batch.
419        worker_id: Identity stamped into ``locked_by``. Defaults to
420            ``settings.worker_id``.
421
422    Returns:
423        int: Number of jobs successfully processed.
424    """
425    from app.core.config import settings  # noqa: PLC0415
426
427    if worker_id is None:
428        worker_id = settings.worker_id
429
430    store = SqlAlchemyJobStore()
431    processor = ClaimJobProcessor()
432    return await _ragnor_process_pending_jobs(
433        store,
434        processor,
435        limit=limit,
436        worker_id=worker_id,
437    )

440async def reclaim_stale_jobs(stale_after_seconds: float | None = None) -> int:
441    """Reset jobs abandoned in ``processing`` back to ``pending``.
442
443    A drainer that dies mid-upsert leaves its rows locked in
444    ``processing`` forever: the claim query only selects ``pending``
445    or retry-due ``failed`` rows, so no live worker can pick them up
446    again. This reclaims rows whose ``locked_at`` is older than
447    ``stale_after_seconds`` — or whose lock timestamp is NULL, which
448    marks rows locked before the column existed — so a replacement
449    worker can process them.
450
451    Args:
452        stale_after_seconds: Lock age in seconds that counts as
453            abandoned. Defaults to ``settings.job_stale_after_seconds``.
454
455    Returns:
456        int: Number of jobs reclaimed.
457    """
458    if stale_after_seconds is None:
459        from app.core.config import settings  # noqa: PLC0415
460
461        stale_after_seconds = settings.job_stale_after_seconds
462
463    store = SqlAlchemyJobStore()
464    return await _ragnor_reclaim_stale_jobs(
465        store,
466        stale_after_seconds=stale_after_seconds,
467    )
