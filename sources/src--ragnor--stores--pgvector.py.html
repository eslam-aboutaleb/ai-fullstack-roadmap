1"""
2pgvector implementation of the VectorStore interface.
3
4Provides hybrid search (vector + PostgreSQL full-text search RRF), document counting, and upsert
5operations against PostgreSQL tables with pgvector and tsvector columns.
6
7Security model:
8  - All user-supplied **values** are passed as bound parameters via SQLAlchemy's
9    parameterized queries. They never appear in the SQL string, so they cannot
10    inject SQL.
11  - All **SQL identifiers** (table names, column names, filter keys) are
12    validated against a strict allowlist regex (``^[A-Za-z_][A-Za-z0-9_]*$``)
13    before interpolation into raw SQL strings. This prevents SQL injection even
14    when identifiers are derived from configuration.
15  - Raw SQL is used because the hybrid search query requires complex CTEs with
16    window functions, pgvector operators, and PostgreSQL full-text search that are
17    impractical to express in SQLAlchemy ORM/Core while remaining readable.
18
19Configuration (no host imports): the store receives everything through
20constructor arguments — ``session_factory`` (a
21:class:`ragnor.db.session.SessionProvider`), ``embedding_dim``, the host's
22``RagSettings``, and an optional ``embedding_fn`` used to auto-embed queries
23when the caller passes an empty embedding.
24
25Distance metric: the vector stage ranks by the store's metric
26(:mod:`ragnor.metrics`). The operator comes from an internal map — ``l2``
27maps to pgvector's ``<->`` (the default) and ``cosine`` to ``<=>`` — so it
28can never come from user input. pgvector's ``<=>`` is the exact cosine
29distance in ``[0, 2]`` and needs no pre-normalization. Thresholds are
30distances under that metric and are therefore metric-relative: never
31comparable across metrics, so a host that switches metrics must re-tune.
32The matching index opclass (``vector_l2_ops`` vs ``vector_cosine_ops``) is
33host-owned DDL — ragnor never creates indexes.
34"""

68def _keyword_score(row: Any) -> float | None:
69    """Return the PostgreSQL full-text rank from a hybrid search row, if present.
70
71    The column is absent for a vector-only hit, and the projection is not guaranteed
72    across every caller, so membership is checked rather than assumed.
73    """
74    if "keyword_score" not in row:
75        return None
76    value = row["keyword_score"]
77    return None if value is None else float(value)

80@vector_store_registry.register("pgvector")
81class PgVectorStore(VectorStore):
82    """pgvector-based vector store implementation.
83
84    Supports hybrid search combining vector similarity (the pgvector
85    operator selected by the store's distance metric: ``<->`` for
86    the default ``l2`` metric, ``<=>`` for ``cosine``) with
87    PostgreSQL full-text search (tsvector/tsquery) using Reciprocal
88    Rank Fusion. The metric is fixed at store-construction time: the
89    distance operator is resolved once in ``__init__`` from
90    ``self.metric``, so every search uses that metric for the store's
91    lifetime.
92    """

94    def __init__(  # noqa: PLR0913, PLR0917
95        self,
96        table_name: str,
97        id_field: str = "id",
98        embedding_dim: int | None = None,
99        session_factory: SessionProvider | None = None,
100        settings: RagSettings | None = None,
101        embedding_fn: EmbeddingFunction | None = None,
102        tsquery_config: str = "english",
103        sql_factory: SQLFactory | None = None,
104        metric: str | None = None,
105    ):
106        """Initialize the vector store with target table configuration.
107
108        Args:
109            table_name: Name of the database table containing vector data.
110            id_field: Name of the primary key column. Defaults to "id".
111            embedding_dim: Dimensionality of the embedding vectors. Required;
112                pass the dimension configured for the active embedding model.
113            session_factory: Session provider used when a method is called
114                without an explicit session. When omitted and ``settings``
115                carries a ``database_url``, one is created lazily through
116                the SQL factory.
117            settings: Host settings. Used only to resolve a default session
118                factory when ``session_factory`` is omitted, and to read the
119                optional ``rag_distance_metric`` field when ``metric`` is
120                omitted.
121            embedding_fn: Async callable that embeds a list of texts. Used
122                only when ``hybrid_search`` is called with an empty embedding.
123            tsquery_config: PostgreSQL text-search configuration used for
124                ``plainto_tsquery`` (must match the tsvector column's
125                configuration). Defaults to 'english'.
126            sql_factory: SQL factory that runs every statement the
127                store issues; defaults to the SQLAlchemy factory.
128                Inject a custom :class:`ragnor.db.sql_factory.SQLFactory`
129                to route the store's SQL through another stack.
130            metric: Distance metric for the vector stage (``"l2"`` by
131                default, ``"cosine"`` opt-in). Resolution order: this
132                argument, the optional ``rag_distance_metric`` host
133                settings field, then ``"l2"``.
134
135        Raises:
136            ValueError: If ``table_name`` or ``id_field`` contain unsafe
137                characters, if ``embedding_dim`` is omitted, if
138                ``tsquery_config`` is not a lowercase identifier, or if
139                ``metric`` is not a supported distance metric.
140        """
141        validate_identifier(table_name, "table_name")
142        validate_identifier(id_field, "id_field")
143        self.table_name = table_name
144        self.id_field = id_field
145        if embedding_dim is None:
146            raise ValueError(
147                "embedding_dim is required; pass the dimension configured "
148                "for the active embedding model"
149            )
150        self.embedding_dim = embedding_dim
151        if not re.fullmatch(r"[a-z][a-z0-9_]*", tsquery_config):
152            raise ValueError(
153                f"tsquery_config must be a lowercase identifier; got {tsquery_config!r}"
154            )
155        self.tsquery_config = tsquery_config
156        self._settings = settings
157        self._session_factory = session_factory
158        self._resolved_session_factory: SessionProvider | None = session_factory
159        self._embedding_fn = embedding_fn
160        self._sql = sql_factory or SQLAlchemyFactory()
161        self.metric: DistanceMetric = resolve_distance_metric(metric, settings)
162        self._distance_operator = _DISTANCE_OPERATORS[self.metric]

164    def _resolve_session_factory(self) -> SessionProvider:
165        """Return the session factory, creating one from settings if needed.
166
167        The factory is resolved at most once per store instance so the
168        underlying engine is never duplicated.
169
170        Raises:
171            RuntimeError: If neither a session factory nor settings with a
172                ``database_url`` were provided.
173        """
174        if self._resolved_session_factory is None:
175            if self._settings is None:
176                raise RuntimeError(
177                    "PgVectorStore needs a session_factory or settings with "
178                    "a database_url to open its own sessions"
179                )
180            self._resolved_session_factory = self._sql.create_session_factory(
181                self._settings.database_url
182            )
183        return self._resolved_session_factory

185    async def _execute_search_query(  # noqa: PLR0913
186        self,
187        session: Any,
188        stmt: str,
189        params: dict[str, Any],
190        text_field: str,
191        metadata_fields: list[str],
192        *,
193        distance_field: str | None = None,
194        keyword_field: str | None = None,
195    ) -> list[dict[str, Any]]:
196        owns_session = session is None
197        if owns_session:
198            session_factory = self._resolve_session_factory()
199            async with session_factory() as owned_session:
200                rows = await self._sql.fetch_all(owned_session, stmt, params)
201        else:
202            rows = await self._sql.fetch_all(session, stmt, params)
203
204        results = []
205        for row in rows:
206            dist = row[distance_field] if distance_field and distance_field in row else None
207            kw = row[keyword_field] if keyword_field and keyword_field in row else None
208            results.append(
209                format_search_hit(
210                    doc_id=str(row[self.id_field]),
211                    document=row[text_field],
212                    metadata=project_metadata(row, metadata_fields),
213                    distance=dist,
214                    keyword_score=kw,
215                )
216            )
217        return results

226    async def hybrid_search(  # noqa: PLR0913, PLR0915, PLR0917
227        self,
228        query: str,
229        embedding: list[float],
230        n_results: int,
231        threshold: float,
232        text_field: str = "text",
233        metadata_fields: list[str] | None = None,
234        session: Any = None,
235        strict: bool = False,
236        **filters: Any,
237    ) -> list[dict[str, Any]]:
238        """Hybrid search with vector + PostgreSQL full-text search RRF.
239
240        Executes a two-stage retrieval:
241          1. Vector search using the pgvector distance operator for the
242             store's metric (``<->`` L2 by default, ``<=>`` cosine) with
243             a distance threshold.
244          2. PostgreSQL full-text search using tsvector/tsquery.
245
246        Results from both stages are merged using Reciprocal Rank Fusion (RRF),
247        which combines the ranks from each retrieval method to produce a unified
248        ranking. This approach is robust when one retrieval method fails to
249        find relevant documents.
250
251        Args:
252            query: Natural language search query.
253            embedding: Query embedding vector. If empty, the query is embedded
254                automatically using the injected ``embedding_fn``.
255            n_results: Maximum number of results to return.
256            threshold: Maximum distance under the store's metric (L2 or
257                cosine) for the vector search part. Smaller values enforce
258                stricter similarity matching, and thresholds are
259                metric-relative: never comparable across metrics.
260            text_field: Name of the column containing the document text.
261            metadata_fields: List of additional column names to include in the
262                result metadata dict.
263            session: Optional external SQLAlchemy async session. If provided,
264                this session is reused instead of creating a new one. When an
265                external session is supplied, the caller is responsible for
266                committing or rolling back the transaction.
267            strict: When ``True``, a failed search raises
268                :class:`ragnor.types.RetrievalError`. The default ``False``
269                preserves the historical behavior of logging the error and
270                returning an empty list.
271            **filters: Additional equality filters applied to both search CTEs
272                (e.g., ``tenant_id="..."``). Keys must be valid SQL identifiers;
273                values are bound as parameters to prevent SQL injection.
274
275        Returns:
276            list[dict]: A list of result dicts sorted by RRF score, each
277            containing:
278                - ``id`` (str): The document id.
279                - ``document`` (str): The retrieved text.
280                - ``metadata`` (dict): Structured metadata fields.
281                - ``distance`` (float): Distance under the store's metric
282                  (L2 or cosine) from the vector search, or ``None`` for
283                  keyword-only hits.
284                - ``keyword_score`` (float): PostgreSQL full-text rank, or
285                  ``None`` for vector-only hits.
286                - ``_rrf_score`` (float): Combined RRF relevance score.
287
288        Raises:
289            ragnor.types.RetrievalError: If the search fails and
290                ``strict=True``.
291        """
292        validate_identifier(text_field, "text_field")
293        for key in filters:
294            validate_identifier(key, "filter key")
295        metadata_fields = metadata_fields or []
296        for field in metadata_fields:
297            validate_identifier(field, "metadata_field")
298
299        if n_results < 1:
300            raise ValueError(f"n_results must be a positive integer; got {n_results}")
301
302        if not embedding:
303            if self._embedding_fn is None:
304                raise ValueError(
305                    "embedding_fn is required to auto-embed the query; "
306                    "pass an embedding or construct the store with embedding_fn"
307                )
308            embedding_list = await self._embedding_fn([query])
309            embedding = embedding_list[0]
310
311        validate_embedding(embedding, self.embedding_dim, label="query embedding")
312        metadata_fields = metadata_fields or []
313
314        # Build SELECT clause for metadata fields so they are returned in
315        # both the vector and keyword CTEs. This ensures metadata is available
316        # regardless of which search method found the document.
317        vector_select = [
318            self.id_field,
319            text_field,
320        ]
321        vector_select.extend(metadata_fields)
322        vector_select_sql = ", ".join(vector_select)
323
324        keyword_select = [
325            self.id_field,
326            text_field,
327        ]
328        keyword_select.extend(metadata_fields)
329        keyword_select_sql = ", ".join(keyword_select)
330
331        if metadata_fields:
332            # Each COALESCE column is aliased with its field name.
333            # Without the alias PostgreSQL labels every one of them
334            # "coalesce", so the result mapping never contains the
335            # field names and every result would carry empty metadata.
336            meta_coalesce = ", ".join(f"COALESCE(v.{f}, k.{f}) AS {f}" for f in metadata_fields)
337        else:
338            meta_coalesce = ""
339
340        # The outer projection is assembled as a list so that an empty
341        # ``metadata_fields`` contributes no element. Interpolating the metadata
342        # expression as ``f"{meta_coalesce},"`` emits a bare comma when it is empty,
343        # producing ``AS document, , v.distance`` — a syntax error that is then
344        # swallowed by the broad ``except`` below and reported as "no results".
345        outer_select = [
346            f"COALESCE(v.{self.id_field}, k.{self.id_field}) AS id",
347            f"COALESCE(v.{text_field}, k.{text_field}) AS document",
348        ]
349        if metadata_fields:
350            outer_select.append(meta_coalesce)
351        outer_select.extend(
352            [
353                "v.distance AS distance",
354                "k.score AS keyword_score",
355                (
356                    "COALESCE(1.0 / (60 + v.rank), 0.0) "
357                    "+ COALESCE(1.0 / (60 + k.rank), 0.0) AS rrf_score"
358                ),
359            ]
360        )
361        outer_select_sql = ",\n                 ".join(outer_select)
362
363        # Build JOIN condition between the two CTEs. The primary join is on
364        # the document ID field, with optional additional filters applied
365        # to both sides to ensure consistent scoping.
366        join_conditions = [f"v.{self.id_field} = k.{self.id_field}"]
367
368        # Build query parameters and WHERE fragments from validated filters.
369        # All user-supplied values are bound as parameters to prevent SQL
370        # injection; only pre-validated identifiers are interpolated.
371        embedding_str = f"[{','.join(str(x) for x in embedding)}]"
372        params: dict[str, Any] = {
373            "query": query,
374            "embedding": embedding_str,
375            "distance_threshold": threshold,
376            "n_results": n_results,
377            "tsquery_config": self.tsquery_config,
378        }
379        # Filter parameters are namespaced with a ``filter_`` prefix so a
380        # filter key can never shadow the reserved bind parameters above
381        # (``query``, ``embedding``, ``distance_threshold``, ``n_results``).
382        filter_parts: list[str] = []
383        for key, value in filters.items():
384            param_name = f"filter_{key}"
385            filter_parts.append(f"{key} = :{param_name}")
386            params[param_name] = value
387
388        extra_where = " AND ".join(filter_parts) if filter_parts else None
389        if extra_where:
390            join_conditions.append(" AND ".join(f"v.{key} = :filter_{key}" for key in filters))
391
392        join_sql = " AND ".join(join_conditions)
393
394        # The text-search expression is built once and interpolated
395        # so the SQL string stays readable; the config value itself
396        # is a bound parameter (cast to regconfig), never interpolated.
397        tsquery_expr = "plainto_tsquery(cast(:tsquery_config as regconfig), :query)"
398
399        # The distance operator comes from the internal metric map only,
400        # never from user input, so the SQL stays injection-safe. The
401        # expression is interpolated (not bound) because pgvector
402        # operators cannot be parameterized.
403        distance_expr = f"embedding {self._distance_operator} cast(:embedding as vector)"
404
405        stmt = f"""
406            WITH vector_search AS (
407                SELECT {vector_select_sql},
408                       {distance_expr} AS distance,
409                       ROW_NUMBER() OVER (
410                           ORDER BY
411                               {distance_expr},
412                               {self.id_field}
413                       ) AS rank
414                FROM {self.table_name}
415                WHERE {distance_expr} < :distance_threshold
416                {f"AND {extra_where}" if extra_where else ""}
417                LIMIT :n_results
418            ),
419            keyword_search AS (
420                SELECT {keyword_select_sql},
421                       ts_rank(tsvector, {tsquery_expr}) AS score,
422                       ROW_NUMBER() OVER (
423                           ORDER BY
424                               ts_rank(tsvector, {tsquery_expr}) DESC,
425                               {self.id_field}
426                       ) AS rank
427                FROM {self.table_name}
428                WHERE tsvector @@ {tsquery_expr}
429                {f"AND {extra_where}" if extra_where else ""}
430                LIMIT :n_results
431            )
432            SELECT
433                {outer_select_sql}
434            FROM vector_search v
435            FULL OUTER JOIN keyword_search k ON {join_sql}
436            ORDER BY rrf_score DESC, {self.id_field}
437            LIMIT :n_results
438        """
439
440        try:
441            if session is not None:
442                rows = await self._sql.fetch_all(session, stmt, params)
443            else:
444                session_factory = self._resolve_session_factory()
445                async with session_factory() as owned_session:
446                    rows = await self._sql.fetch_all(owned_session, stmt, params)
447
448            retrieved = []
449            for row in rows:
450                metadata = {}
451                for field in metadata_fields:
452                    if field in row and row[field] is not None:
453                        metadata[field] = row[field]
454
455                retrieved.append(
456                    {
457                        "id": str(row["id"]) if row["id"] is not None else None,
458                        "document": row["document"],
459                        "metadata": metadata,
460                        "distance": float(row["distance"]) if row["distance"] is not None else None,
461                        "keyword_score": _keyword_score(row),
462                        "_rrf_score": float(row["rrf_score"]),
463                    }
464                )
465
466            logger.debug(
467                "hybrid_search('%s') on %s: found %d results.",
468                query,
469                self.table_name,
470                len(retrieved),
471            )
472            return retrieved
473        except Exception as exc:
474            logger.error("Hybrid search failed on %s: %s", self.table_name, exc)
475            if strict:
476                raise RetrievalError(f"Hybrid search failed on {self.table_name}: {exc}") from exc
477            return []

479    async def vector_search(  # noqa: PLR0913, PLR0917
480        self,
481        embedding: list[float],
482        n_results: int,
483        threshold: float,
484        text_field: str = "text",
485        metadata_fields: list[str] | None = None,
486        session: Any = None,
487        **filters: Any,
488    ) -> list[dict[str, Any]]:
489        """Standalone dense vector search (the hybrid search's vector stage).
490
491        One round-trip: a single ``SELECT`` ordered by the store's
492        distance expression (``embedding <-> :embedding`` for ``l2``,
493        ``embedding <=> :embedding`` for ``cosine``) with the
494        threshold and equality filters applied as bound
495        parameters.
496
497        Args:
498            embedding: Query embedding vector (required; the
499                method does not auto-embed).
500            n_results: Maximum number of results to return.
501            threshold: Maximum distance under the store's metric
502                (L2 or cosine) for vector search; metric-relative,
503                never comparable across metrics.
504            text_field: Name of the column containing the
505                document text.
506            metadata_fields: Additional column names to
507                include in the result metadata dict.
508            session: Optional external session. When
509                supplied, the caller owns the transaction
510                (commit/rollback). When omitted, the store
511                opens its own session.
512            **filters: Additional filters (e.g., tenant_id).
513
514        Returns:
515            list[dict]: Result dicts ordered by distance
516            ascending (tie-broken by id), each carrying
517            ``id``, ``document``, ``metadata``,
518            ``distance``, and ``keyword_score`` (``None``).
519
520        Raises:
521            ValueError: If identifiers are unsafe,
522                ``n_results`` is not positive, or the
523                embedding has the wrong dimension.
524            Exception: Re-raises database errors.
525        """
526        validate_search_identifiers(text_field, filters, metadata_fields)
527        metadata_fields = metadata_fields or []
528        if n_results < 1:
529            raise ValueError(f"n_results must be a positive integer; got {n_results}")
530
531        validate_embedding(embedding, self.embedding_dim, label="query embedding")
532
533        # The distance operator comes from the internal metric map only,
534        # never from user input, so the SQL stays injection-safe.
535        distance_expr = f"embedding {self._distance_operator} cast(:embedding as vector)"
536
537        select_columns = [self.id_field, text_field, *metadata_fields]
538        select_sql = ", ".join(select_columns)
539
540        filter_parts = []
541        params: dict[str, Any] = {
542            "embedding": f"[{','.join(str(x) for x in embedding)}]",
543            "distance_threshold": threshold,
544            "n_results": n_results,
545        }
546        for key, value in filters.items():
547            param_name = f"filter_{key}"
548            filter_parts.append(f"{key} = :{param_name}")
549            params[param_name] = value
550
551        where_clause = f"WHERE {distance_expr} < :distance_threshold"
552        if filter_parts:
553            where_clause += " AND " + " AND ".join(filter_parts)
554
555        stmt = f"""
556            SELECT {select_sql},
557                   {distance_expr} AS distance
558            FROM {self.table_name}
559            {where_clause}
560            ORDER BY {distance_expr}, {self.id_field}
561            LIMIT :n_results
562        """
563
564        return await self._execute_search_query(
565            session=session,
566            stmt=stmt,
567            params=params,
568            text_field=text_field,
569            metadata_fields=metadata_fields,
570            distance_field="distance",
571        )

573    async def keyword_search(  # noqa: PLR0913, PLR0917
574        self,
575        query: str,
576        n_results: int,
577        text_field: str = "text",
578        metadata_fields: list[str] | None = None,
579        session: Any = None,
580        **filters: Any,
581    ) -> list[dict[str, Any]]:
582        """Standalone keyword search (the hybrid search's keyword stage).
583
584        One round-trip: a single ``SELECT`` using PostgreSQL
585        full-text search (``tsvector @@ plainto_tsquery``)
586        ordered by ``ts_rank`` descending, with equality
587        filters applied as bound parameters.
588
589        Args:
590            query: Natural language search query.
591            n_results: Maximum number of results to return.
592            text_field: Column storing the text matching plainto_tsquery.
593            metadata_fields: Extra metadata column names for the output dict.
594            session: Optional caller transaction session; otherwise a new
595                session is acquired for the duration of this read.
596            **filters: Bound equality filters for row selection.
597
598        Returns:
599            list[dict]: Result dicts ordered by
600            ``ts_rank`` descending (tie-broken by id),
601            each carrying ``id``, ``document``,
602            ``metadata``, ``distance`` (``None``), and
603            ``keyword_score``.
604
605        Raises:
606            ValueError: If identifiers are unsafe or
607                ``n_results`` is not positive.
608            Exception: Re-raises database errors.
609        """
610        validate_search_identifiers(text_field, filters, metadata_fields)
611        metadata_fields = metadata_fields or []
612        if n_results < 1:
613            raise ValueError(f"n_results must be a positive integer; got {n_results}")
614
615        select_columns = [self.id_field, text_field, *metadata_fields]
616        select_sql = ", ".join(select_columns)
617        tsquery_expr = "plainto_tsquery(cast(:tsquery_config as regconfig), :query)"
618
619        filter_parts = []
620        params: dict[str, Any] = {
621            "query": query,
622            "tsquery_config": self.tsquery_config,
623            "n_results": n_results,
624        }
625        for key, value in filters.items():
626            param_name = f"filter_{key}"
627            filter_parts.append(f"{key} = :{param_name}")
628            params[param_name] = value
629
630        where_clause = f"tsvector @@ {tsquery_expr}"
631        if filter_parts:
632            where_clause += " AND " + " AND ".join(filter_parts)
633
634        stmt = f"""
635            SELECT {select_sql},
636                   ts_rank(tsvector, {tsquery_expr}) AS score
637            FROM {self.table_name}
638            WHERE {where_clause}
639            ORDER BY ts_rank(tsvector, {tsquery_expr}) DESC, {self.id_field}
640            LIMIT :n_results
641        """
642
643        return await self._execute_search_query(
644            session=session,
645            stmt=stmt,
646            params=params,
647            text_field=text_field,
648            metadata_fields=metadata_fields,
649            keyword_field="score",
650        )

652    async def get_embeddings(self, ids: list[str], session: Any = None) -> dict[str, list[float]]:
653        if not ids:
654            return {}
655        import json  # noqa: PLC0415
656
657        placeholders = ", ".join(f":id_{i}" for i in range(len(ids)))
658        stmt = (
659            f"SELECT {self.id_field}, embedding FROM {self.table_name} "
660            f"WHERE {self.id_field} IN ({placeholders})"
661        )
662        params = {f"id_{i}": doc_id for i, doc_id in enumerate(ids)}
663        if session is None:
664            async with self._resolve_session_factory()() as owned:
665                rows = await self._sql.fetch_all(owned, stmt, params)
666        else:
667            rows = await self._sql.fetch_all(session, stmt, params)
668        return {
669            str(row[self.id_field]): (
670                json.loads(row["embedding"])
671                if isinstance(row["embedding"], str)
672                else list(row["embedding"])
673            )
674            for row in rows
675        }

677    async def delete(
678        self,
679        ids: list[Any] | None = None,
680        filters: dict[str, Any] | None = None,
681        session: Any = None,
682    ) -> int:
683        """Delete documents selected by ``ids`` and/or ``filters``.
684
685        At least one selector must be supplied; an empty
686        ``ids`` list is rejected. When both are supplied,
687        both restrictions apply (conjunction). The deletion
688        count is reported by the statement itself
689        (``RETURNING`` over the deleted rows), so the
690        returned count is the number of rows the database
691        deleted.
692
693        Args:
694            ids: Document ids to delete.
695            filters: Equality filters (e.g., tenant_id).
696            session: Optional external session. When
697                supplied, the caller owns the transaction
698                (commit/rollback). When omitted, the store
699                opens its own session and commits.
700
701        Returns:
702            int: Number of deleted documents.
703
704        Raises:
705            ValueError: If no selector is supplied, ``ids``
706                is empty, or a filter key is unsafe.
707            Exception: Re-raises database errors.
708        """
709        validate_delete_selectors(ids, filters)
710
711        conditions = []
712        params: dict[str, Any] = {}
713        if ids is not None:
714            conditions.append(f"{self.id_field} = ANY(:ids)")
715            params["ids"] = list(ids)
716        for key, value in (filters or {}).items():
717            param_name = f"filter_{key}"
718            conditions.append(f"{key} = :{param_name}")
719            params[param_name] = value
720
721        stmt = f"""
722            WITH deleted AS (
723                DELETE FROM {self.table_name}
724                WHERE {" AND ".join(conditions)}
725                RETURNING 1
726            )
727            SELECT count(*) FROM deleted
728        """
729
730        owns_session = session is None
731        if owns_session:
732            session_factory = self._resolve_session_factory()
733            async with session_factory() as owned_session:
734                count = await self._sql.fetch_scalar(owned_session, stmt, params)
735                await owned_session.commit()
736        else:
737            count = await self._sql.fetch_scalar(session, stmt, params)
738        return int(count) if count is not None else 0

740    async def get(  # noqa: PLR0913, PLR0917
741        self,
742        ids: list[Any] | None = None,
743        filters: dict[str, Any] | None = None,
744        session: Any = None,
745        text_field: str = "text",
746        metadata_fields: list[str] | None = None,
747    ) -> list[dict[str, Any]]:
748        """Fetch documents selected by ``ids`` and/or ``filters``.
749
750        The selection is a single ``SELECT`` by primary key
751        (``{id_field} = ANY(:ids)``) and/or equality
752        filters. With no selector, every document is
753        returned.
754
755        Args:
756            ids: Document ids to fetch.
757            filters: Equality filters (e.g., tenant_id).
758            session: Optional external session. When
759                supplied, the caller owns the transaction
760                (commit/rollback). When omitted, the store
761                opens its own session.
762            text_field: Name of the column containing the
763                document text.
764            metadata_fields: Additional column names to
765                include in the result metadata dict.
766
767        Returns:
768            list[dict]: ``id``/``document``/``metadata``
769            dicts (score keys are meaningless for direct
770            fetches).
771
772        Raises:
773            ValueError: If any identifier is unsafe.
774            Exception: Re-raises database errors.
775        """
776        validate_search_identifiers(text_field, filters, metadata_fields)
777        metadata_fields = metadata_fields or []
778
779        select_columns = [self.id_field, text_field, *metadata_fields]
780        select_sql = ", ".join(select_columns)
781
782        conditions = []
783        params: dict[str, Any] = {}
784        if ids is not None:
785            conditions.append(f"{self.id_field} = ANY(:ids)")
786            params["ids"] = list(ids)
787        for key, value in (filters or {}).items():
788            param_name = f"filter_{key}"
789            conditions.append(f"{key} = :{param_name}")
790            params[param_name] = value
791
792        where_clause = f"WHERE {' AND '.join(conditions)}" if conditions else ""
793        stmt = f"""
794            SELECT {select_sql}
795            FROM {self.table_name}
796            {where_clause}
797        """
798
799        owns_session = session is None
800        if owns_session:
801            session_factory = self._resolve_session_factory()
802            async with session_factory() as owned_session:
803                rows = await self._sql.fetch_all(owned_session, stmt, params)
804        else:
805            rows = await self._sql.fetch_all(session, stmt, params)
806
807        return [
808            {
809                "id": str(row[self.id_field]),
810                "document": row[text_field],
811                "metadata": project_metadata(row, metadata_fields),
812            }
813            for row in rows
814        ]

816    async def count(
817        self,
818        session: Any = None,
819        strict: bool = False,
820        **filters: Any,
821    ) -> int:
822        """Count documents in the store.
823
824        Args:
825            session: Optional external SQLAlchemy async session. If provided,
826                this session is reused instead of creating a new one. When an
827                external session is supplied, the caller is responsible for
828                committing or rolling back the transaction.
829            strict: When ``True``, a failed count raises
830                :class:`ragnor.types.RetrievalError`. The default ``False``
831                preserves the historical behavior of logging the error and
832                returning ``0``.
833            **filters: Optional equality filters (e.g., ``tenant_id="..."``).
834                Keys must be valid SQL identifiers; values are bound as
835                parameters to prevent SQL injection.
836
837        Returns:
838            int: Number of matching documents, or 0 on error.
839
840        Raises:
841            ragnor.types.RetrievalError: If the count fails and
842                ``strict=True``.
843        """
844        for key in filters:
845            validate_identifier(key, "filter key")
846
847        params: dict[str, Any] = {}
848        filter_parts: list[str] = []
849        for key, value in filters.items():
850            param_name = f"filter_{key}"
851            filter_parts.append(f"{key} = :{param_name}")
852            params[param_name] = value
853
854        where_clause = f"WHERE {' AND '.join(filter_parts)}" if filter_parts else ""
855
856        stmt = f"""
857            SELECT COUNT(*) FROM {self.table_name}
858            {where_clause}
859        """
860
861        try:
862            if session is not None:
863                count = await self._sql.fetch_scalar(session, stmt, params)
864            else:
865                session_factory = self._resolve_session_factory()
866                async with session_factory() as owned_session:
867                    count = await self._sql.fetch_scalar(owned_session, stmt, params)
868            return int(count) if count else 0
869        except Exception as exc:
870            logger.error("Count failed on %s: %s", self.table_name, exc)
871            if strict:
872                raise RetrievalError(f"Count failed on {self.table_name}: {exc}") from exc
873            return 0

875    @staticmethod
876    def _collect_metadata_keys(documents: list[dict[str, Any]]) -> list[str]:
877        """Return every metadata key used by ``documents``, in first-seen order.
878
879        Each key is validated as a safe SQL identifier while collecting, so
880        the caller can interpolate the keys into the INSERT column list.
881        """
882        all_meta_keys: list[str] = []
883        seen_meta_keys: set[str] = set()
884        for doc in documents:
885            for key in doc.get("metadata", {}):
886                validate_identifier(key, "metadata key")
887                if key not in seen_meta_keys:
888                    seen_meta_keys.add(key)
889                    all_meta_keys.append(key)
890        return all_meta_keys

892    @staticmethod
893    def _build_values_clauses(  # noqa: PLR0913, PLR0917
894        documents: list[dict[str, Any]],
895        id_field: str,
896        text_field: str,
897        embedding_field: str,
898        all_meta_keys: list[str],
899        extra_fields: dict[str, Any] | None,
900    ) -> tuple[list[str], dict[str, Any]]:
901        """Build one VALUES row per document and its bind parameters.
902
903        Returns the list of ``(placeholder, ...)`` clauses and the flat
904        ``params`` mapping the clauses refer to. Parameter names are
905        namespaced by row index so rows never collide.
906        """
907        values_clauses: list[str] = []
908        params: dict[str, Any] = {}
909
910        for idx, doc in enumerate(documents):
911            row_placeholders = []
912            for key in [id_field, text_field, embedding_field]:
913                # Column parameters use a ``_col_`` prefix so a metadata
914                # (or extra) key equal to a column name can never shadow
915                # the column's bind value.
916                param_name = f"_col_{key}__{idx}"
917                if key == id_field:
918                    params[param_name] = doc.get("id")
919                elif key == embedding_field:
920                    params[param_name] = f"[{','.join(str(x) for x in doc[key])}]"
921                else:
922                    params[param_name] = doc[key]
923                row_placeholders.append(f":{param_name}")
924            for meta_key in all_meta_keys:
925                param_name = f"{meta_key}__{idx}"
926                params[param_name] = doc.get("metadata", {}).get(meta_key)
927                row_placeholders.append(f":{param_name}")
928            if extra_fields:
929                for extra_key, extra_val in extra_fields.items():
930                    param_name = f"_extra_{extra_key}__{idx}"
931                    params[param_name] = extra_val
932                    row_placeholders.append(f":{param_name}")
933            values_clauses.append(f"({', '.join(row_placeholders)})")
934
935        return values_clauses, params

937    async def upsert(  # noqa: PLR0913, PLR0917
938        self,
939        documents: list[dict[str, Any]],
940        session: Any = None,
941        text_field: str = "text",
942        embedding_field: str = "embedding",
943        id_field: str | None = None,
944        extra_fields: dict[str, Any] | None = None,
945    ) -> None:
946        """Upsert documents into the store.
947
948        Inserts or updates documents using PostgreSQL's ``ON CONFLICT ... DO UPDATE``
949        syntax. This allows the ingestion pipeline to re-run without creating
950        duplicate rows.
951
952        Args:
953            documents: List of document dicts with keys: id, text, metadata,
954                embedding. Metadata keys are included as additional columns.
955            session: Optional external SQLAlchemy async session. If provided,
956                this session is reused instead of creating a new one. The caller
957                is responsible for committing the transaction after this call.
958            text_field: Name of the text column. Defaults to "text".
959            embedding_field: Name of the embedding column. Defaults to "embedding".
960            id_field: Name of the primary key column. Defaults to the store's
961                configured ``id_field``.
962            extra_fields: Additional static fields to include in every INSERT
963                (e.g., ``{"tenant_id": "..."}``).
964
965        Raises:
966            ValueError: If any column name contains unsafe characters.
967            Exception: Re-raises database errors after logging.
968        """
969        id_field = validate_upsert_identifiers(
970            text_field, embedding_field, id_field, self.id_field, extra_fields
971        )
972
973        if not documents:
974            return
975
976        owns_session = session is None
977        try:
978            for doc in documents:
979                validate_embedding(
980                    doc["embedding"],
981                    self.embedding_dim,
982                    label="document embedding",
983                )
984
985            all_meta_keys = self._collect_metadata_keys(documents)
986
987            columns = [id_field, text_field, embedding_field]
988            if extra_fields:
989                columns.extend(extra_fields.keys())
990            columns.extend(all_meta_keys)
991
992            for col in columns:
993                validate_identifier(col, "upsert column")
994
995            non_id_columns = [c for c in columns if c != id_field]
996            update_str = ", ".join(f"{c} = EXCLUDED.{c}" for c in non_id_columns)
997
998            values_clauses, params = self._build_values_clauses(  # noqa: PLR0913, PLR0917
999                documents,
1000                id_field,
1001                text_field,
1002                embedding_field,
1003                all_meta_keys,
1004                extra_fields,
1005            )
1006
1007            values_sql = ",\n    ".join(values_clauses)
1008            columns_str = ", ".join(columns)
1009
1010            stmt = f"""
1011                INSERT INTO {self.table_name} ({columns_str})
1012                VALUES
1013                    {values_sql}
1014                ON CONFLICT ({id_field}) DO UPDATE SET {update_str}
1015            """
1016
1017            if owns_session:
1018                session_factory = self._resolve_session_factory()
1019                async with session_factory() as owned_session:
1020                    await self._sql.execute(owned_session, stmt, params)
1021                    await owned_session.commit()
1022            else:
1023                await self._sql.execute(session, stmt, params)
1024            logger.info("Upserted %d documents to %s.", len(documents), self.table_name)
1025        except Exception as exc:
1026            logger.error("Upsert failed on %s: %s", self.table_name, exc)
1027            raise
