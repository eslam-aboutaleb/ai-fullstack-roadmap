1"""Embedding provider registry.
2
3Providers plug in through :data:`embedding_registry` without
4editing core code. A provider is a factory class registered
5under a name that:
6
71. Accepts a :class:`ragnor.config.RagSettings` object in its
8   constructor, and
92. Exposes a ``create()`` method returning an
10   :class:`ragnor.embeddings.base.EmbeddingFunction`.
11
12Example::
13
14    @embedding_registry.register("my-provider")
15    class MyEmbeddingProvider:
16        def __init__(self, settings: RagSettings) -> None:
17            self._settings = settings
18
19        def create(self) -> EmbeddingFunction:
20            return MyEmbeddingFunction(self._settings.api_key)
21
22:func:`get_embedding_function` resolves the provider name from
23``settings.embedding_provider`` (defaulting to ``"litellm"`` when
24the settings object does not define the key) and instantiates the
25registered provider. No second provider ships with ragnor -- the
26seam is the deliverable.
27"""
28
29from __future__ import annotations
30
31from ragnor.config import RagSettings
32from ragnor.embeddings.base import EmbeddingFunction
33from ragnor.registry import Registry
34
35embedding_registry: Registry[EmbeddingFunction] = Registry()
36
37
38@embedding_registry.register("litellm")
39class LiteLLMEmbeddingProvider:
40    """Factory that builds :class:`LitellmEmbeddingFunction` from settings."""
41
42    def __init__(self, settings: RagSettings) -> None:
43        self._settings = settings
44
45    def create(self) -> EmbeddingFunction:
46        """Return a LiteLLM embedding function configured from settings."""
47        try:
48            from ragnor.embeddings.litellm import (  # noqa: PLC0415
49                LitellmEmbeddingFunction,
50            )
51        except ImportError as exc:
52            raise ImportError(
53                "The LiteLLM integration requires the 'litellm' extra: "
54                "pip install 'ragnor[litellm]'"
55            ) from exc
56        return LitellmEmbeddingFunction(
57            api_key=self._settings.openai_api_key,
58            model_name=self._settings.embedding_model,
59            api_base=self._settings.openai_api_base,
60        )
61
62
63def get_embedding_function(settings: RagSettings) -> EmbeddingFunction:
64    """Return the embedding function selected by ``settings``.
65
66    The provider name is read from ``settings.embedding_provider``.
67    The key is optional: settings objects that do not define it
68    (such as the host application's settings until they adopt the
69    field) fall back to the ``"litellm"`` provider, which preserves
70    the historical behavior.
71
72    Args:
73        settings: Settings exposing ``embedding_model`` and
74            ``openai_api_key`` (and optionally ``embedding_provider``).
75
76    Returns:
77        An embedding function instance ready for use by the
78        retrieval and ingestion pipelines.
79
80    Raises:
81        KeyError: If the provider name is not registered. The
82            message lists the registered provider names.
83    """
84    provider_name = getattr(settings, "embedding_provider", "litellm")
85    provider_cls = embedding_registry.get(provider_name)
86    return provider_cls(settings).create()
