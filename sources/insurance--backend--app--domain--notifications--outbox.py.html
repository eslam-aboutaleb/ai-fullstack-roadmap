1"""Notification outbox over ragnor's generic job store.
2
3Claimant-facing notifications follow the same protocol as the claim embedding
4jobs. :class:`NotificationJobStore` implements ragnor's ``EmbeddingJobStore``
5protocol against the ``notification_jobs`` table,
6:func:`enqueue_notification_job` writes a pending row through
7:func:`ragnor.jobs.enqueue_job` — in the caller's transaction when a session is
8supplied, so the claim change and the notification about it commit together —
9:class:`NotificationJobProcessor` renders the message from a per-job-type template
10and materialises it in the ``notifications`` feed, and
11:func:`process_pending_notification_jobs` /
12:func:`reclaim_stale_notification_jobs` wire the store and processor into ragnor's
13generic job functions.
14
15Invariant for anything that writes a ``notification_jobs`` row
16----------------------------------------------------------------
17A notification job must be committed in the **same transaction** as the claim
18change it describes, or written through this outbox by a caller that has already
19committed that change.
20
21The reason is delivery, not referential integrity: a job row that survives a
22rolled-back status change notifies a policyholder about a transition that never
23happened, and there is no way to take it back once it has been read. The claim
24change is the fact; the notification is a promise about it, so the two land
25together or neither lands.
26
27Therefore:
28
29- ``enqueue_notification_job(session=...)`` only flushes. Pass the caller's session.
30- ``enqueue_notification_job()`` with no session opens its own and commits. Use it
31  only where the claim change is already durable.
32
33Delivery is never inline
34------------------------
35No request thread calls a provider. The API writes the job row; the drain worker
36in :mod:`app.workers.notification_drain` delivers it. That keeps a slow or failing
37provider off the claim lifecycle endpoints and makes the notification retryable.
38"""

93    async def claim_pending(self, limit: int, _worker_id: str) -> list[JobPayload]:
94        """Select claimable jobs with ``FOR UPDATE SKIP LOCKED``.
95
96        ``worker_id`` is part of the ``EmbeddingJobStore`` protocol
97        but is not needed here: the lock columns are stamped by
98        :func:`claim_jobs` after selection.
99        """
100        from sqlalchemy import select  # noqa: PLC0415
101
102        now = datetime.now(UTC)
103        async with self._session_factory() as session:
104            result = await session.execute(
105                select(NotificationJob)
106                .where(
107                    (NotificationJob.status == "pending")
108                    | (
109                        (NotificationJob.status == "failed")
110                        & (NotificationJob.next_retry_at <= now)
111                        & (NotificationJob.retry_count < NotificationJob.max_attempts)
112                    )
113                )
114                .order_by(NotificationJob.created_at)
115                .limit(limit)
116                .with_for_update(skip_locked=True)
117            )
118            rows = result.scalars().all()
119        return [_payload_from_row(row) for row in rows]

121    async def mark_processing(self, jobs: list[JobPayload], worker_id: str) -> list[JobPayload]:
122        """Transition jobs to ``processing`` and stamp the lock.
123
124        The guarded update (``status IN ('pending', 'failed')``) keeps
125        two concurrent drainers from claiming the same row: only the
126        worker whose update matches still owns the job.
127        """
128        from sqlalchemy import update  # noqa: PLC0415
129
130        if not jobs:
131            return []
132
133        now = datetime.now(UTC)
134        claim_jobs(jobs, worker_id, now)
135
136        claimed: list[JobPayload] = []
137        async with self._session_factory() as session:
138            for job in jobs:
139                result = await session.execute(
140                    update(NotificationJob)
141                    .where(
142                        NotificationJob.id == uuid.UUID(str(job.job_id)),
143                        NotificationJob.status.in_(["pending", "failed"]),
144                    )
145                    .values(
146                        status=job.status,
147                        status_detail=job.status_detail,
148                        next_retry_at=job.next_retry_at,
149                        locked_at=job.locked_at,
150                        locked_by=job.locked_by,
151                    )
152                    .returning(NotificationJob.id)
153                )
154                if result.scalar() is not None:
155                    claimed.append(job)
156            await session.commit()
157        return claimed

190    async def reclaim_stale(self, stale_after_seconds: float | None = None) -> int:
191        """Reset jobs abandoned in ``processing`` back to ``pending``."""
192        from sqlalchemy import select  # noqa: PLC0415
193
194        if stale_after_seconds is None:
195            from app.core.config import settings  # noqa: PLC0415
196
197            stale_after_seconds = settings.job_stale_after_seconds
198
199        cutoff = datetime.now(UTC) - timedelta(seconds=stale_after_seconds)
200        reclaimed = 0
201        async with self._session_factory() as session:
202            result = await session.execute(
203                select(NotificationJob)
204                .where(
205                    (NotificationJob.status == "processing")
206                    & (NotificationJob.locked_at.is_(None) | (NotificationJob.locked_at <= cutoff))
207                )
208                .with_for_update(skip_locked=True)
209            )
210            stale_jobs = result.scalars().all()
211            for job in stale_jobs:
212                job.status = "pending"
213                job.status_detail = None
214                job.locked_at = None
215                job.locked_by = None
216                reclaimed += 1
217            if reclaimed:
218                await session.commit()
219
220        if reclaimed:
221            logger.info("Reclaimed %d stale notification job(s).", reclaimed)
222        return reclaimed

224    async def enqueue(self, payload: dict[str, Any], session: Any = None) -> None:
225        """Write a new pending ``notification_jobs`` row.
226
227        When ``session`` is supplied the row is only added and flushed;
228        the caller owns the commit, so the job and the claim change it
229        describes land in one transaction. Without a session this opens
230        its own session and commits.
231
232        ``payload`` carries the addressing (``job_type``, ``user_id``,
233        ``channel``, ``max_attempts``) alongside the job's own ``data``,
234        which is what the processor's templates read back.
235        """
236        job = NotificationJob(
237            user_id=uuid.UUID(str(payload["user_id"])),
238            job_type=str(payload["job_type"]),
239            payload=dict(payload.get("data") or {}),
240            channel=str(payload.get("channel") or CHANNEL_IN_APP),
241            status="pending",
242            status_detail="pending",
243            max_attempts=int(payload.get("max_attempts") or MAX_ATTEMPTS),
244        )
245        if session is not None:
246            session.add(job)
247            await session.flush()
248        else:
249            async with self._session_factory() as owned_session:
250                owned_session.add(job)
251                await owned_session.commit()

257    async def process(self, job: JobPayload) -> None:
258        """Render and deliver one claimed notification job.
259
260        The row is re-read rather than trusting the payload alone, because
261        ``job_type`` and ``channel`` are columns on the outbox and the payload is
262        the business data the templates read. A row that has since been deleted
263        fails the attempt instead of delivering a notification about nothing.
264
265        Args:
266            job: The claimed job.
267
268        Raises:
269            ValueError: The outbox row is gone, or its ``job_type`` is not one the
270                templates can render.
271        """
272        async with async_session_factory() as session:
273            row = await session.get(NotificationJob, uuid.UUID(str(job.job_id)))
274            if row is None:
275                raise ValueError(f"Notification job '{job.job_id}' no longer exists.")
276            job_type = row.job_type
277            channel = row.channel
278            payload = dict(row.payload or {})
279            user_id = row.user_id
280
281        if job_type not in NOTIFICATION_JOB_TYPES:
282            raise ValueError(f"Unknown notification job type '{job_type}'.")
283
284        title, body = render_notification(job_type, payload)
285        await deliver_notification(
286            job_id=uuid.UUID(str(job.job_id)),
287            user_id=user_id,
288            job_type=job_type,
289            channel=channel,
290            payload=payload,
291            title=title,
292            body=body,
293        )

296async def _deliver_in_app(  # noqa: PLR0913
297    *,
298    job_id: uuid.UUID,
299    user_id: uuid.UUID,
300    job_type: str,
301    title: str,
302    body: str,
303    claim_id: str | None,
304) -> None:
305    """Materialise a delivered notification in the owner's feed.
306
307    Written outside the claim transaction, which is the point: a claimant must not
308    wait on a feed write to move a claim, and a feed write must not be able to
309    roll back a status change. The unique key on ``job_id`` makes the write
310    idempotent, so a redelivery after a reclaimed lock refreshes nothing instead
311    of notifying twice.
312    """
313    statement = (
314        insert(Notification)
315        .values(
316            job_id=job_id,
317            user_id=user_id,
318            job_type=job_type,
319            channel=CHANNEL_IN_APP,
320            title=title,
321            body=body,
322            claim_id=claim_id,
323        )
324        .on_conflict_do_nothing(index_elements=[Notification.job_id])
325    )
326
327    async with async_session_factory() as session:
328        await session.execute(statement)
329        await session.commit()

332async def deliver_notification(  # noqa: PLR0913, PLR0917
333    *,
334    job_id: uuid.UUID,
335    user_id: uuid.UUID,
336    job_type: str,
337    channel: str,
338    payload: dict[str, Any],
339    title: str,
340    body: str,
341) -> None:
342    """Send one rendered notification through its channel's provider.
343
344    This is the provider seam. ``in_app`` has a provider; ``email``, ``sms``, and
345    ``webhook`` are reserved names that no provider is registered for, and they
346    raise rather than succeeding. A silent success would mark the job completed and
347    drop the notification on the floor, which is precisely the failure this outbox
348    exists to prevent, so an unimplemented channel has to fail loudly, retry, and
349    dead-letter where an operator can see it.
350
351    Args:
352        job_id: The outbox row delivering this notification.
353        user_id: The recipient's UUID.
354        job_type: Which delivery instruction produced this notification.
355        channel: The delivery target named on the job.
356        payload: The job's business payload, for a provider that needs it.
357        title: The rendered subject line.
358        body: The rendered message text.
359
360    Raises:
361        ValueError: No provider is registered for ``channel``.
362    """
363    if channel != CHANNEL_IN_APP:
364        raise ValueError(
365            f"No notification provider is registered for the '{channel}' channel. "
366            "Register one in app.domain.notifications.outbox.deliver_notification; "
367            f"'{channel}' is reserved."
368        )
369
370    claim_id = payload.get("claim_id")
371    await _deliver_in_app(
372        job_id=job_id,
373        user_id=user_id,
374        job_type=job_type,
375        title=title,
376        body=body,
377        claim_id=str(claim_id) if claim_id else None,
378    )

401def _render_claim_status_changed(payload: dict[str, Any]) -> tuple[str, str]:
402    """Render the notification for a claim that moved to a new status.
403
404    Args:
405        payload: The job payload, carrying ``claim_id``, ``from_status``,
406            ``to_status``, and any ``note`` the transition recorded.
407
408    Returns:
409        tuple[str, str]: The notification's title and body.
410    """
411    claim_id = payload.get("claim_id", "your claim")
412    to_status = payload.get("to_status", "a new status")
413    body = f"Claim {claim_id} is now {to_status}."
414    from_status = payload.get("from_status")
415    if from_status:
416        body = f"Claim {claim_id} moved from {from_status} to {to_status}."
417    note = payload.get("note")
418    if note:
419        body = f"{body} Note: {note}"
420    return f"Claim {claim_id} is now {to_status}", body

447def render_notification(job_type: str, payload: dict[str, Any]) -> tuple[str, str]:
448    """Render a deterministic title and body for one notification job.
449
450    A template table rather than a model call: a notification is a fixed string
451    about a stored fact, the worker must not cost a model invocation per delivery,
452    and the text a claimant reads must not change between two workers running two
453    prompt versions.
454
455    Args:
456        job_type: The delivery instruction, one of ``NOTIFICATION_JOB_TYPES``.
457        payload: The job's business data.
458
459    Returns:
460        tuple[str, str]: The rendered title and body.
461
462    Raises:
463        ValueError: ``job_type`` has no template.
464    """
465    template = _TEMPLATES.get(job_type)
466    if template is None:
467        raise ValueError(f"No notification template is registered for '{job_type}'.")
468    return template(payload)

504async def enqueue_notification_job(  # noqa: PLR0913, PLR0917
505    *,
506    job_type: str,
507    user_id: uuid.UUID,
508    payload: dict[str, Any],
509    channel: str = CHANNEL_IN_APP,
510    max_attempts: int = MAX_ATTEMPTS,
511    session: AsyncSession | None = None,
512) -> None:
513    """Queue a notification for asynchronous delivery.
514
515    Delegates to ragnor's :func:`ragnor.jobs.enqueue_job`
516    with the caller's session: when ``session`` is supplied,
517    the job row is only added and flushed, so the claim change and
518    the notification about it are persisted atomically by the
519    caller's commit. When ``session`` is None, ragnor opens
520    and commits its own transaction.
521
522    A failure to enqueue is logged and never raised when this function owns the
523    transaction: the claim change it describes is already durable, and failing a
524    committed fact to protect a notification would trade one for the other.
525
526    When ``session`` is supplied the situation is the opposite. The job row has
527    only been *flushed*, not committed, so a flush failure has already poisoned
528    the caller's transaction: its next ``commit()`` raises
529    ``PendingRollbackError`` with a traceback that says nothing about the real
530    cause. Re-raising here is what lets the caller fail on the actual error
531    instead of a misleading one several statements later.
532
533    Args:
534        job_type: The delivery instruction, one of ``NOTIFICATION_JOB_TYPES``.
535        user_id: The recipient's UUID.
536        payload: The business data the template renders from.
537        channel: The delivery target. Defaults to the in-app feed.
538        max_attempts: Attempt ceiling before the job is dead-lettered.
539        session: Optional existing database session. When provided, the job row
540            joins the caller's transaction and the caller commits.
541    """
542    store_payload = enqueue_notification_job_payload(
543        job_type=job_type,
544        user_id=user_id,
545        payload=payload,
546        channel=channel,
547        max_attempts=max_attempts,
548    )
549    store = NotificationJobStore()
550    try:
551        await enqueue_job(store, store_payload, session=session)
552        logger.info("Enqueued '%s' notification for user '%s'.", job_type, user_id)
553    except Exception as exc:
554        if session is not None:
555            raise
556        logger.exception("Failed to enqueue '%s' notification: %s", job_type, exc)

559async def process_pending_notification_jobs(
560    limit: int = 10,
561    worker_id: str | None = None,
562) -> int:
563    """Deliver pending notification jobs.
564
565    Fetches up to ``limit`` pending jobs, renders and delivers each, and records
566    the outcome. Failed deliveries are marked with an error detail and a retry
567    timestamp; jobs that exhaust their attempts are dead-lettered.
568
569    Claimed jobs are stamped with ``locked_at``/``locked_by`` for the whole
570    processing transaction. A worker that dies mid-pass therefore leaves a
571    visible lock, which :func:`reclaim_stale_notification_jobs` later resets.
572
573    Args:
574        limit: Maximum number of jobs to deliver in one batch.
575        worker_id: Identity stamped into ``locked_by``. Defaults to
576            ``settings.worker_id``.
577
578    Returns:
579        int: Number of jobs successfully delivered.
580    """
581    from app.core.config import settings  # noqa: PLC0415
582
583    if worker_id is None:
584        worker_id = settings.worker_id
585
586    store = NotificationJobStore()
587    processor = NotificationJobProcessor()
588    return await _ragnor_process_pending_jobs(
589        store,
590        processor,
591        limit=limit,
592        worker_id=worker_id,
593    )

596async def reclaim_stale_notification_jobs(
597    stale_after_seconds: float | None = None,
598) -> int:
599    """Reset notification jobs abandoned in ``processing`` back to ``pending``.
600
601    A drainer that dies mid-delivery leaves its rows locked in ``processing``
602    forever: the claim query only selects ``pending`` or retry-due ``failed`` rows,
603    so no live worker can pick them up again. This reclaims rows whose
604    ``locked_at`` is older than ``stale_after_seconds`` — or whose lock timestamp
605    is NULL — so a replacement worker can deliver them.
606
607    Args:
608        stale_after_seconds: Lock age in seconds that counts as abandoned.
609            Defaults to ``settings.job_stale_after_seconds``.
610
611    Returns:
612        int: Number of jobs reclaimed.
613    """
614    if stale_after_seconds is None:
615        from app.core.config import settings  # noqa: PLC0415
616
617        stale_after_seconds = settings.job_stale_after_seconds
618
619    store = NotificationJobStore()
620    return await _ragnor_reclaim_stale_jobs(
621        store,
622        stale_after_seconds=stale_after_seconds,
623    )
