3The LiteLLM integration ships behind the ``litellm`` extra and
4is imported lazily (via the module ``__getattr__`` below), so
5``import ragnor.embeddings`` works without litellm installed.
6
7New in this release:
8
9* :class:`~ragnor.embeddings.profile.EmbeddingProfile` — provider/model/dimension
10  identity and canonical hash.
11* :class:`~ragnor.embeddings.executor.EmbeddingExecutor` — resilient embedding
12  executor with retry, concurrency control, and response validation.
13* :class:`~ragnor.embeddings.cache.EmbeddingCache` — scoped, profile-keyed
14  in-process cache with LRU eviction and optional TTL.
15* :class:`~ragnor.embeddings.metrics.EmbeddingMetrics` — accumulated call/hit/retry
16  counters.

67def __getattr__(name: str) -> Any:
68    """Lazily import the LiteLLM integration so litellm stays optional."""
69    if name in _LAZY_PROVIDERS:
70        module_name, attribute = _LAZY_PROVIDERS[name]
71        try:
72            module = importlib.import_module(module_name)
73        except ImportError as exc:
74            raise ImportError(
75                "The LiteLLM integration requires the 'litellm' extra: "
76                "pip install 'ragnor[litellm]'"
77            ) from exc
78        return getattr(module, attribute)
79    raise AttributeError(f"module {__name__!r} has no attribute {name!r}")
