1"""Retrieval defense-in-depth with mandatory scope and tombstone enforcement.
2
3Enforces zero-trust deletion privacy: even if physical cleanup in a vector
4database is lagging, asynchronous, or retrying, tombstoned documents and
5inactive chunk generations are filtered out before results are returned.
6"""
7
8from __future__ import annotations
9
10import logging
11from typing import Any
12
13from ragnor.stores.base import VectorStore
14from ragnor.sync.manifest import ManifestStatus, ManifestStore
15from ragnor.sync.state import SyncStateStore
16
17logger = logging.getLogger(__name__)
18
19
20class SyncRetriever:
21    """Retriever wrapper enforcing active generation and tombstone checks.
22
23    Readers only see active generations of permitted, non-tombstoned documents.
24    """
25
26    def __init__(
27        self,
28        vector_store: VectorStore,
29        manifest_store: ManifestStore,
30        state_store: SyncStateStore | None = None,
31    ) -> None:
32        self._vector_store = vector_store
33        self._manifest_store = manifest_store
34        self._state_store = state_store
35
36    async def hybrid_search(  # noqa: PLR0913, PLR0917
37        self,
38        query: str,
39        embedding: list[float],
40        scope: str,
41        n_results: int = 10,
42        threshold: float = 1.0,
43        text_field: str = "text",
44        metadata_fields: list[str] | None = None,
45        session: Any = None,
46        strict: bool = False,
47        **filters: Any,
48    ) -> list[dict[str, Any]]:
49        """Perform hybrid search with mandatory active generation and tombstone filtering."""
50        # Ensure mandatory metadata fields for generation and tombstone validation
51        system_fields = ["doc_id", "scope", "generation", "active", *filters]
52        if metadata_fields is None:
53            effective_fields = system_fields
54        else:
55            effective_fields = sorted(set(metadata_fields) | set(system_fields))
56
57        # Query underlying store filtering by scope and active=True
58        results = await self._vector_store.hybrid_search(
59            query=query,
60            embedding=embedding,
61            n_results=n_results * 2,  # Oversample to account for potential filtering
62            threshold=threshold,
63            text_field=text_field,
64            metadata_fields=effective_fields,
65            session=session,
66            strict=strict,
67            scope=scope,
68            active=True,
69            **filters,
70        )
71
72        # Read one manifest snapshot so a collection commit cannot be observed halfway.
73        records = {record.document_id: record for record in await self._manifest_store.list(scope)}
74        filtered: list[dict[str, Any]] = []
75        for res in results:
76            meta = res.get("metadata") or {}
77            doc_id = meta.get("doc_id")
78            if not doc_id or meta.get("scope") != scope or meta.get("active") is not True:
79                continue
80            rec = records.get(doc_id)
81            if rec is None:
82                continue
83
84            # Strict deletion privacy: tombstoned, failed, or inactive records are dropped
85            if rec.status != ManifestStatus.ACTIVE:
86                continue
87
88            # Ensure generation matches currently active generation
89            chunk_gen = meta.get("generation")
90            if rec.active_generation != chunk_gen:
91                continue
92
93            # Ensure chunk ID is actually in the active chunk list
94            if res.get("id") not in rec.chunk_ids:
95                continue
96
97            # If caller didn't ask for system fields, project only what was requested
98            if metadata_fields is not None:
99                clean_meta = {k: v for k, v in meta.items() if k in metadata_fields}
100                res["metadata"] = clean_meta
101
102            filtered.append(res)
103            if len(filtered) >= n_results:
104                break
105
106        return filtered
