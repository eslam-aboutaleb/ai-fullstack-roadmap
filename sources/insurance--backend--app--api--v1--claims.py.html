102async def prepare_claim_submission(
103    payload: ClaimSubmissionPrepareRequest,
104    incident_date: date | None = Query(
105        default=None,
106        description=(
107            "Date the incident occurred. Optional; a claim filed without one stores "
108            "a null incident date rather than guessing from the filing date."
109        ),
110    ),
111    current_user_id: str = Depends(get_current_user),
112) -> ClaimSubmissionPrepareResponse:
113    """Validate claim data and create a pending submission awaiting confirmation.
114
115    This is the first step of the two-step claim submission flow. The agent
116    collects the required fields, calls this endpoint to validate them, and
117    receives a confirmation_token. The frontend displays a confirmation UI
118    and calls /confirm with the token to complete the submission.
119    """
120    try:
121        user_uuid = uuid.UUID(current_user_id)
122    except ValueError:
123        raise HTTPException(
124            status_code=status.HTTP_401_UNAUTHORIZED,
125            detail="Invalid user",
126        ) from None
127
128    # Validate via Pydantic (already done by FastAPI) and prepare claim data
129    claim_data = payload.model_dump(mode="json")
130    # Serialize Decimal to string for JSONB storage to preserve precision
131    # and avoid float rounding issues.
132    claim_data["amount"] = str(claim_data["amount"])
133    # The pending submission is the only place the incident date can travel between
134    # prepare and confirm, because the confirm request carries just a token.
135    if incident_date is not None:
136        claim_data["incident_date"] = incident_date.isoformat()
137
138    confirmation_token = str(uuid.uuid4())
139    expires_at = datetime.now(UTC) + timedelta(minutes=15)
140
141    try:
142        async with async_session_factory() as session:
143            submission = ClaimSubmission(
144                user_id=user_uuid,
145                confirmation_token=confirmation_token,
146                claim_data=claim_data,
147                status="pending",
148                expires_at=expires_at,
149            )
150            session.add(submission)
151            await session.commit()
152    except SQLAlchemyError as exc:
153        logger.exception("Failed to create pending claim submission: %s", exc)
154        raise HTTPException(
155            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
156            detail="Failed to prepare claim submission. Please try again.",
157        ) from exc
158
159    logger.info(
160        "Prepared claim submission for user '%s' with token '%s'.",
161        current_user_id,
162        confirmation_token,
163    )
164
165    return ClaimSubmissionPrepareResponse(
166        confirmation_token=confirmation_token,
167        expires_at=expires_at,
168        status="pending",
169    )

196async def confirm_claim_submission(
197    payload: ClaimConfirmationRequest,
198    current_user_id: str = Depends(get_current_user),
199) -> ClaimConfirmationResponse:
200    """Confirm and submit a pending claim submission.
201
202    Validates the confirmation token belongs to the authenticated user,
203    checks it is not expired, and then calls the internal submit logic.
204    """
205    try:
206        user_uuid = uuid.UUID(current_user_id)
207    except ValueError:
208        raise HTTPException(
209            status_code=status.HTTP_401_UNAUTHORIZED,
210            detail="Invalid user",
211        ) from None
212
213    async with async_session_factory() as session:
214        submission = (
215            await session.execute(
216                select(ClaimSubmission)
217                .where(
218                    ClaimSubmission.confirmation_token == payload.confirmation_token,
219                    ClaimSubmission.user_id == user_uuid,
220                    ClaimSubmission.status == "pending",
221                )
222                .with_for_update()
223            )
224        ).scalar_one_or_none()
225
226        if submission is None:
227            logger.warning(
228                "Invalid or expired confirmation token '%s' for user '%s'.",
229                payload.confirmation_token,
230                current_user_id,
231            )
232            raise HTTPException(
233                status_code=status.HTTP_400_BAD_REQUEST,
234                detail="Invalid, expired, or already used confirmation token.",
235            )
236
237        if submission.expires_at < datetime.now(UTC):
238            submission.status = "expired"
239            await session.commit()
240            raise HTTPException(
241                status_code=status.HTTP_400_BAD_REQUEST,
242                detail="Confirmation token has expired.",
243            )
244
245        claim_data = submission.claim_data
246        result = await submit_claim_internal(
247            policy_number=claim_data["policy_number"],
248            claim_type=claim_data["claim_type"],
249            amount=float(claim_data["amount"]),
250            description=claim_data["description"],
251            user_uuid=user_uuid,
252            session=session,
253        )
254
255        if not result.get("success"):
256            logger.error(
257                "Claim submission failed for token '%s': %s",
258                payload.confirmation_token,
259                result.get("error", "Unknown error"),
260            )
261            raise HTTPException(
262                status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
263                detail=result.get("error", "Claim submission failed."),
264            )
265
266        await _record_incident_date(
267            session,
268            owner_uuid=user_uuid,
269            claim_id=result.get("confirmation_id"),
270            incident_date=_parse_incident_date(submission.claim_data.get("incident_date")),
271        )
272        await _record_filing_event(
273            session,
274            owner_uuid=user_uuid,
275            claim_id=result.get("confirmation_id"),
276        )
277        # Same transaction as the claim, the filing event, and the embedding job:
278        # the notification describes a claim that only exists once this commit
279        # lands. The session is passed so the job row cannot outlive a rollback and
280        # tell a policyholder their claim was filed when it was not.
281        await enqueue_notification_job(
282            job_type=JOB_TYPE_CLAIM_SUBMITTED,
283            user_id=user_uuid,
284            payload={
285                "claim_id": result.get("confirmation_id"),
286                "claim_type": result.get("claim_type"),
287                "status": result.get("status"),
288            },
289            session=session,
290        )
291
292        submission.status = "completed"
293        await session.commit()
294
295    logger.info(
296        "Claim confirmed for user '%s': %s",
297        current_user_id,
298        result.get("message", ""),
299    )
300
301    return ClaimConfirmationResponse(
302        success=result.get("success", False),
303        claim_id=result.get("confirmation_id"),
304        status=result.get("status"),
305        message=result.get("message", "Claim submission processed."),
306    )

338    """Cancel a pending claim submission the caller has not confirmed yet.
339
340    The submission row is looked up under the same ``FOR UPDATE`` lock the
341    confirm flow takes, so a cancel racing a confirm cannot leave the row half
342    written: one of the two holds the lock and the other sees the row's new
343    status.
344
345    Args:
346        payload: The confirmation token identifying the pending submission.
347        current_user_id: The authenticated user's UUID, injected by
348            ``get_current_user``.
349
350    Returns:
351        ClaimCancelResponse: Success plus a human-readable message.
352
353    Raises:
354        HTTPException: 400 if the token is unknown to this user, expired, or
355            already used; 401 if the authenticated user id is not a valid UUID.
356    """
357    try:
358        user_uuid = uuid.UUID(current_user_id)
359    except ValueError:
360        raise HTTPException(
361            status_code=status.HTTP_401_UNAUTHORIZED,
362            detail="Invalid user",
363        ) from None
364
365    async with async_session_factory() as session:
366        submission = (
367            await session.execute(
368                select(ClaimSubmission)
369                .where(
370                    ClaimSubmission.confirmation_token == payload.confirmation_token,
371                    ClaimSubmission.user_id == user_uuid,
372                    ClaimSubmission.status == "pending",
373                )
374                .with_for_update()
375            )
376        ).scalar_one_or_none()
377
378        # A token belonging to another user is indistinguishable from one that
379        # was never issued, so both report the same refusal.
380        if submission is None:
381            logger.warning(
382                "Cancel requested with an unusable confirmation token '%s' for user '%s'.",
383                payload.confirmation_token,
384                current_user_id,
385            )
386            raise HTTPException(
387                status_code=status.HTTP_400_BAD_REQUEST,
388                detail="Invalid, expired, or already used confirmation token.",
389            )
390
391        if submission.expires_at < datetime.now(UTC):
392            submission.status = "expired"
393            await session.commit()
394            raise HTTPException(
395                status_code=status.HTTP_400_BAD_REQUEST,
396                detail="Confirmation token has expired.",
397            )
398
399        submission.status = "cancelled"
400        await session.commit()
401
402    logger.info(
403        "Cancelled pending claim submission for user '%s'.",
404        current_user_id,
405    )
406
407    return ClaimCancelResponse(
408        success=True,
409        message="Claim submission cancelled. No claim was filed.",
410    )

432async def list_claims(  # noqa: PLR0913
433    claim_status: str | None = Query(
434        default=None,
435        alias="status",
436        description="Filter to claims in this status (e.g. 'Submitted', 'Under Review').",
437    ),
438    claim_type: str | None = Query(
439        default=None,
440        description="Filter to claims of this type (e.g. 'Water Damage').",
441    ),
442    limit: int = Query(
443        default=20,
444        ge=1,
445        le=50,
446        description="Maximum number of claims to return.",
447    ),
448    offset: int = Query(
449        default=0,
450        ge=0,
451        description="Number of newer claims to skip.",
452    ),
453    current_user_id: str = Depends(get_current_user),
454    session: AsyncSession = Depends(get_db),
455) -> ClaimListResponse:
456    """List the authenticated user's claims, newest first.
457
458    The query is scoped to ``owner_id == current_user_id`` before any filter or
459    pagination is applied, so a caller who guesses another user's claim id still
460    gets an empty page rather than that claim's row.
461
462    Args:
463        claim_status: Optional status filter.
464        claim_type: Optional claim-type filter.
465        limit: Page size, defaulting to 20 and capped at 50.
466        offset: Number of newest claims to skip.
467        current_user_id: The authenticated user's UUID, injected by
468            ``get_current_user``.
469        session: Async database session.
470
471    Returns:
472        ClaimListResponse: One page of the caller's claims plus the pagination
473        values that produced it.
474
475    Raises:
476        HTTPException: 401 if the authenticated user id is not a valid UUID.
477    """
478    try:
479        owner_uuid = uuid.UUID(current_user_id)
480    except ValueError:
481        raise HTTPException(
482            status_code=status.HTTP_401_UNAUTHORIZED,
483            detail="Invalid user",
484        ) from None
485
486    stmt = select(Claim).where(Claim.owner_id == owner_uuid)
487    if claim_status is not None:
488        stmt = stmt.where(Claim.status == claim_status)
489    if claim_type is not None:
490        stmt = stmt.where(Claim.claim_type == claim_type)
491    # claim_id breaks ties so a page ordered on a same-microsecond created_at is
492    # still deterministic across repeated requests.
493    stmt = stmt.order_by(desc(Claim.created_at), desc(Claim.claim_id))
494    stmt = stmt.limit(limit).offset(offset)
495
496    claims = (await session.execute(stmt)).scalars().all()
497
498    return ClaimListResponse(
499        claims=[ClaimSummary(**_claim_fields(claim)) for claim in claims],
500        count=len(claims),
501        limit=limit,
502        offset=offset,
503    )

723            ``get_current_user``.
724        session: Async database session.
725
726    Returns:
727        ClaimTransitionResponse: The status the claim moved from and to, plus
728        the statuses it may still move to.
729
730    Raises:
731        HTTPException: 401 for an invalid user id, 404 when the claim is not the
732            caller's, 409 when the lifecycle does not allow a withdrawal from the
733            claim's current status.
734    """
735    try:
736        owner_uuid = uuid.UUID(current_user_id)
737    except ValueError:
738        raise HTTPException(
739            status_code=status.HTTP_401_UNAUTHORIZED,
740            detail="Invalid user",
741        ) from None
742
743    return await _transition_claim(
744        session,
745        owner_uuid=owner_uuid,
746        actor_uuid=owner_uuid,
747        claim_id=claim_id,
748        target=ClaimStatus.WITHDRAWN,
749        note=payload.reason,
750        actor_summary="the policyholder",
751    )

775async def _claim_events(session: AsyncSession, claim: Claim) -> list[dict[str, Any]]:
776    """Read one claim's lifecycle history, newest transition first.
777
778    The query is scoped by the claim row the caller already owns, so a history
779    can never be read for a claim the caller cannot see.
780
781    Args:
782        session: The session reading the history.
783        claim: The claim whose events are read.
784
785    Returns:
786        list[dict[str, Any]]: One serialized event per transition, ordered so
787        the most recent transition comes first. Empty for a claim that has no
788        recorded history.
789    """
790    events = (
791        (
792            await session.execute(
793                select(ClaimEvent)
794                .where(ClaimEvent.claim_uuid == claim.id)
795                .order_by(desc(ClaimEvent.created_at), desc(ClaimEvent.id))
796            )
797        )
798        .scalars()
799        .all()
800    )
801    return [
802        ClaimEventItem.model_validate(event, from_attributes=True).model_dump(mode="json")
803        for event in events
804    ]

807async def _transition_claim(  # noqa: PLR0913
808    session: AsyncSession,
809    *,
810    owner_uuid: uuid.UUID,
811    actor_uuid: uuid.UUID,
812    claim_id: str,
813    target: ClaimStatus,
814    note: str | None,
815    actor_summary: str,
816) -> ClaimTransitionResponse:
817    """Validate and apply one lifecycle transition in the caller's transaction.
818
819    Args:
820        session: The transaction the claim row, the event row, the embedding job,
821            and the notification job are written in. The caller commits, so a
822            rejected transition writes nothing at all.
823        owner_uuid: The user who must own the claim for the write to proceed.
824        actor_uuid: Who is performing the transition, recorded on the event.
825        claim_id: The human-readable claim identifier.
826        target: The status to move the claim to.
827        note: Optional reason to record on the transition.
828        actor_summary: How the actor is described in log messages.
829
830    Returns:
831        ClaimTransitionResponse: The status the claim moved from and to, plus
832        the statuses it may still move to.
833
834    Raises:
835        HTTPException: 404 when the claim is not the caller's, 409 when the
836            lifecycle forbids the transition, 403 when the transition is
837            administrative and the caller has no authority for it.
838    """
839    claim = (
840        await session.execute(
841            select(Claim)
842            .where(Claim.claim_id == claim_id.strip().upper(), Claim.owner_id == owner_uuid)
843            .with_for_update()
844        )
845    ).scalar_one_or_none()
846
847    if claim is None:
848        logger.warning(
849            "Claim '%s' status change requested by non-owner '%s'; reported as not found.",
850            claim_id,
851            owner_uuid,
852        )
853        raise HTTPException(status_code=status.HTTP_404_NOT_FOUND, detail="Claim not found")
854
855    previous_status = claim.status
856    current = parse_claim_status(previous_status)
857    if not is_status_transition_allowed(current, target):
858        allowed = allowed_next_statuses(current)
859        raise HTTPException(
860            status_code=status.HTTP_409_CONFLICT,
861            detail=_illegal_transition_detail(claim.claim_id, previous_status, target, allowed),
862        )
863
864    await _require_transition_authority(actor_uuid, target)
865
866    claim.status = target.value
867    session.add(
868        ClaimEvent(
869            claim_uuid=claim.id,
870            actor_id=actor_uuid,
871            from_status=current.value if current is not None else previous_status,
872            to_status=target.value,
873            note=note,
874        )
875    )
876    await session.flush()
877
878    # The claim's indexed metadata carries the status and amount as they were at
879    # embed time. Without this enqueue the vector store would answer with the
880    # status the claim had when it was filed, however many times it had moved on.
881    # The session is passed, so the job row commits with the status it describes
882    # instead of racing it.
883    await enqueue_embedding_job(
884        claim_uuid=claim.id,
885        claim_id=claim.claim_id,
886        owner_id=claim.owner_id,
887        claim_type=claim.claim_type,
888        description=claim.description,
889        policy_number=claim.policy_number,
890        claim_status=claim.status,
891        session=session,
892    )
893    await _enqueue_transition_notification(
894        session,
895        claim=claim,
896        owner_uuid=owner_uuid,
897        previous_status=previous_status,
898        target=target,
899        note=note,
900    )
901    await session.commit()
902
903    logger.info(
904        "Claim '%s' moved from '%s' to '%s' by %s.",
905        claim.claim_id,
906        previous_status,
907        target.value,
908        actor_summary,
909    )
910
911    allowed = allowed_next_statuses(target)
912    message = f"Claim {claim.claim_id} is now {target.value}."
913    if allowed:
914        message = f"{message} It can move to: {', '.join(allowed)}."
915    return ClaimTransitionResponse(
916        claim_id=claim.claim_id,
917        previous_status=previous_status,
918        status=target.value,
919        note=note,
920        allowed_next_statuses=list(allowed),
921        message=message,
922    )

925async def _require_transition_authority(
926    actor_uuid: uuid.UUID,
927    target: ClaimStatus,
928) -> None:
929    """Refuse an administrative transition the caller has no authority for.
930
931    The authority check is the one place this route needs a role, and no role
932    exists yet. Until one does, a caller who is not recorded in
933    ``ADMINISTRATIVE_TRANSITION_ACTORS`` cannot move anyone's claim through a
934    handler decision at all.
935
936    Args:
937        actor_uuid: The user performing the transition.
938        target: The status being requested.
939
940    Raises:
941        HTTPException: 403 when the target status is administrative and the
942            caller is not one of the interim administrative actors.
943    """
944    if target in POLICYHOLDER_STATUSES:
945        return
946    if str(actor_uuid) in ADMINISTRATIVE_TRANSITION_ACTORS:
947        return
948    raise HTTPException(
949        status_code=status.HTTP_403_FORBIDDEN,
950        detail=(
951            f"Moving a claim to '{target.value}' requires a claims handler. "
952            "A policyholder may only withdraw their own claim."
953        ),
954    )

957def _illegal_transition_detail(
958    claim_id: str,
959    current_status: str,
960    target: ClaimStatus,
961    allowed: tuple[str, ...],
962) -> str:
963    """Build the 409 message that names the statuses a claim may still move to.
964
965    Args:
966        claim_id: The human-readable claim identifier.
967        current_status: The status the claim currently holds.
968        target: The status that was requested.
969        allowed: The statuses the lifecycle permits from the current one.
970
971    Returns:
972        str: A message naming the refused transition and the legal next states.
973    """
974    prefix = f"Cannot move claim '{claim_id}' from '{current_status}' to '{target.value}'. "
975    if allowed:
976        return f"{prefix}Allowed next statuses: {', '.join(allowed)}."
977    return f"{prefix}'{current_status}' has no allowed next status."

980async def _enqueue_transition_notification(  # noqa: PLR0913
981    session: AsyncSession,
982    *,
983    claim: Claim,
984    owner_uuid: uuid.UUID,
985    previous_status: str,
986    target: ClaimStatus,
987    note: str | None,
988) -> None:
989    """Queue the notification that tells the owner their claim moved.
990
991    Runs inside the caller's transaction and never commits: the status change, its
992    event, the re-index job, and this notification become durable together or not
993    at all. Without the shared transaction a notification could describe a
994    transition that was rolled back, and a claimant has no way to un-read it.
995
996    A withdrawal is its own delivery kind rather than a status change, because the
997    claimant performed it themselves: the rendered message must say so instead of
998    reporting a move as something that happened to their claim.
999
1000    Args:
1001        session: The transaction the transition is written in.
1002        claim: The claim that moved, already updated in this session.
1003        owner_uuid: The user the notification belongs to.
1004        previous_status: The status the claim held before the transition.
1005        target: The status the claim moved to.
1006        note: Optional reason the caller recorded on the transition.
1007    """
1008    job_type = (
1009        JOB_TYPE_CLAIM_WITHDRAWN
1010        if target is ClaimStatus.WITHDRAWN
1011        else JOB_TYPE_CLAIM_STATUS_CHANGED
1012    )
1013    await enqueue_notification_job(
1014        job_type=job_type,
1015        user_id=owner_uuid,
1016        payload={
1017            "claim_id": claim.claim_id,
1018            "claim_uuid": str(claim.id),
1019            "user_id": str(owner_uuid),
1020            "from_status": previous_status,
1021            "to_status": target.value,
1022            "note": note,
1023        },
1024        session=session,
1025    )

1028def _parse_incident_date(raw: object) -> date | None:
1029    """Interpret the incident date captured at prepare time.
1030
1031    Args:
1032        raw: The stored value, expected to be an ISO-8601 date string.
1033
1034    Returns:
1035        date | None: The parsed date, or None when no date was captured or the
1036        stored value is unusable. A malformed value is discarded rather than
1037        failing a claim the policyholder already confirmed.
1038    """
1039    if raw is None:
1040        return None
1041    try:
1042        return date.fromisoformat(str(raw))
1043    except ValueError:
1044        logger.warning("Discarding malformed incident date on a pending claim submission.")
1045        return None

1048async def _record_incident_date(
1049    session: AsyncSession,
1050    owner_uuid: uuid.UUID,
1051    claim_id: str | None,
1052    incident_date: date | None,
1053) -> None:
1054    """Write the incident date onto the claim created by this confirmation.
1055
1056    Runs inside the caller's transaction and never commits: the surrounding
1057    confirm flow already commits the claim, the embedding job, and the
1058    submission status change together.
1059
1060    Args:
1061        session: The transaction the claim was written in.
1062        owner_uuid: The confirming user, so only that user's claim is touched.
1063        claim_id: The claim id returned by the submission, or None when it is
1064            absent.
1065        incident_date: The date to store, or None to skip the update entirely.
1066    """
1067    if incident_date is None or not claim_id:
1068        return
1069
1070    claim = (
1071        await session.execute(
1072            select(Claim).where(Claim.claim_id == claim_id, Claim.owner_id == owner_uuid)
1073        )
1074    ).scalar_one_or_none()
1075
1076    if claim is None:
1077        logger.warning("Could not attach incident date: claim '%s' was not found.", claim_id)
1078        return
1079
1080    claim.incident_date = incident_date

1083async def _record_filing_event(
1084    session: AsyncSession,
1085    owner_uuid: uuid.UUID,
1086    claim_id: str | None,
1087) -> None:
1088    """Write the opening entry of the lifecycle history for a filed claim.
1089
1090    Runs inside the caller's transaction and never commits, so the claim, its
1091    embedding job, and the first entry in its history become durable together.
1092    Without this entry a claim's history would start empty and stay empty until
1093    somebody moved it, which would leave "how long has this been pending?"
1094    unanswerable for a claim that has only ever been pending.
1095
1096    Args:
1097        session: The transaction the claim was written in.
1098        owner_uuid: The confirming user, so only that user's claim is touched.
1099        claim_id: The claim id returned by the submission, or None when it is
1100            absent.
1101    """
1102    if not claim_id:
1103        return
1104
1105    claim = (
1106        await session.execute(
1107            select(Claim).where(Claim.claim_id == claim_id, Claim.owner_id == owner_uuid)
1108        )
1109    ).scalar_one_or_none()
1110
1111    if claim is None:
1112        logger.warning("Could not record the filing event: claim '%s' was not found.", claim_id)
1113        return
1114
1115    session.add(
1116        ClaimEvent(
1117            claim_uuid=claim.id,
1118            actor_id=owner_uuid,
1119            from_status=None,
1120            to_status=ClaimStatus.SUBMITTED.value,
1121            note="Claim filed.",
1122        )
1123    )
1124    await session.flush()
