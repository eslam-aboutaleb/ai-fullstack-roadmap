1"""
2Regression tests for the single-transaction guarantee of the claim confirmation flow.
3
4The defect this pins down: ``submit_claim._submit`` called ``session.commit()`` even
5when the caller passed its own session. ``confirm_claim_submission`` had already taken
6``SELECT ... FOR UPDATE`` on the submission row, so that commit released the row lock.
7A second concurrent confirmation could then re-read the same row as ``pending`` and
8insert a second claim for a single confirmation token.
9
10The invariant that matters: one confirmation token yields exactly one claim, and
11``_submit`` never commits a transaction it does not own.
12"""

33class _RecordingSession:
34    """Minimal session double that records flush and commit calls separately."""
35
36    def __init__(self) -> None:
37        self.added: list[object] = []
38        self.commits = 0
39        self.flushes = 0
40
41    def add(self, obj) -> None:
42        self.added.append(obj)
43
44    async def flush(self) -> None:
45        self.flushes += 1
46
47    async def commit(self) -> None:
48        self.commits += 1
49
50    async def rollback(self) -> None:
51        return None

54def test_submit_claim_never_commits_a_caller_supplied_session() -> None:
55    """``submit_claim_internal`` must flush, not commit, on a caller-owned session.
56
57    Committing inside the tool releases the ``FOR UPDATE`` row lock the confirmation
58    endpoint relies on, which is what allowed two claims for a single token.
59    """
60    session = _RecordingSession()
61
62    result = asyncio.run(
63        submit_claim_internal(
64            policy_number=_VALID_CLAIM["policy_number"],
65            claim_type=_VALID_CLAIM["claim_type"],
66            amount=_VALID_CLAIM["amount"],
67            description=_VALID_CLAIM["description"],
68            user_uuid=uuid.uuid4(),
69            session=session,  # type: ignore[arg-type]
70        )
71    )
72
73    assert result["success"] is True, result
74    assert session.commits == 0, "tool committed a caller-owned session"
75    assert session.flushes >= 1, "tool must flush so the caller can commit later"
76    assert len(session.added) == 2, "expected one claim and one embedding job"

79def test_prepare_and_confirm_produce_exactly_one_claim(
80    test_client: TestClient,
81    user_a: SeededUser,
82) -> None:
83    """One token produces one claim and one embedding job, both committed together."""
84    from app.core.database import async_session_factory
85    from app.models.claims.claim import Claim
86    from app.models.claims.embedding_job import EmbeddingJob
87    from sqlalchemy import func, select
88
89    prepared = test_client.post(
90        "/api/v1/claims/prepare",
91        json=_VALID_CLAIM,
92        headers=user_a.headers,
93    )
94    assert prepared.status_code == 200, prepared.text
95    token = prepared.json()["confirmation_token"]
96
97    confirmed = test_client.post(
98        "/api/v1/claims/confirm",
99        json={"confirmation_token": token},
100        headers=user_a.headers,
101    )
102    assert confirmed.status_code == 200, confirmed.text
103    claim_id = confirmed.json()["claim_id"]
104
105    async def _counts() -> tuple[int, int]:
106        async with async_session_factory() as session:
107            claims = await session.scalar(
108                select(func.count()).select_from(Claim).where(Claim.claim_id == claim_id)
109            )
110            jobs = await session.scalar(
111                select(func.count())
112                .select_from(EmbeddingJob)
113                .where(EmbeddingJob.claim_id == claim_id)
114            )
115            return claims, jobs
116
117    claims, jobs = asyncio.run(_counts())
118    assert claims == 1, "confirmation must create exactly one claim"
119    assert jobs == 1, "confirmation must enqueue exactly one embedding job"
