32# Querying through the ORM keeps the embedding-job schema the single source of

35_DEPTH_BY_STATUS = select(EmbeddingJob.status, func.count()).group_by(EmbeddingJob.status)
36_OLDEST_PENDING = select(func.min(EmbeddingJob.created_at)).where(EmbeddingJob.status == "pending")

208async def _ping_database() -> DatabaseMetrics:
209    """Probe the database and measure how long it took.
210
211    Returns:
212        DatabaseMetrics: The probe result, with ``connected`` false when the
213        probe raised or timed out.
214    """
215    started = time.perf_counter()
216
217    async def _probe() -> None:
218        async with engine.connect() as connection:
219            await connection.execute(select(1))
220
221    try:
222        await asyncio.wait_for(_probe(), timeout=_DB_PING_TIMEOUT_SECONDS)
223    except Exception as exc:
224        logger.warning("Detailed health check: database probe failed: %s", exc)
225        return DatabaseMetrics(connected=False)
226    latency_ms = round((time.perf_counter() - started) * 1000, 3)
227    return DatabaseMetrics(connected=True, latency_ms=latency_ms)

230def _age_seconds(timestamp: datetime | None) -> float | None:
231    """Turn a stored timestamp into an age in seconds.
232
233    Args:
234        timestamp: The value read back from PostgreSQL, or ``None``.
235
236    Returns:
237        float | None: Seconds elapsed, or ``None`` when there was no row.
238    """
239    if timestamp is None:
240        return None
241    if timestamp.tzinfo is None:
242        # The column is timezone-aware, so a naive value means the driver handed
243        # back UTC without an offset; treating it as local would skew the age.
244        timestamp = timestamp.replace(tzinfo=UTC)
245    return max((datetime.now(UTC) - timestamp).total_seconds(), 0.0)

248async def _read_outbox() -> OutboxMetrics | None:
249    """Read the embedding outbox depth per status and the oldest pending age.
250
251    Returns:
252        OutboxMetrics | None: The queue depth and backlog age, or ``None`` when
253        the query failed. ``None`` is distinct from an empty depth: a failed read
254        is reported as degraded, an empty queue is not.
255    """
256    try:
257        async with async_session_factory() as session:
258            depth_rows = (await session.execute(_DEPTH_BY_STATUS)).all()
259            oldest_created_at = (await session.execute(_OLDEST_PENDING)).scalar_one_or_none()
260    except Exception as exc:
261        logger.warning("Detailed health check: outbox read failed: %s", exc)
262        return None
263
264    depth = {str(job_status): int(count) for job_status, count in depth_rows}
265    return OutboxMetrics(
266        depth_by_status=depth, oldest_pending_age_seconds=_age_seconds(oldest_created_at)
267    )

298async def detailed_health_check(response: Response) -> DetailedHealthResponse:
299    """Collect operational telemetry for the running service.
300
301    The metrics store is in-process and the database probe is bounded by a
302    timeout, so dependency problems are reported in the body rather than raised;
303    only an unreachable database produces a non-200 status.
304
305    Args:
306        response: FastAPI response object, used to set 503 when the database is
307            unreachable.
308
309    Returns:
310        DetailedHealthResponse: Database probe result, LLM call count and cost
311        over the window, tool latency and outcome counts, and outbox depth with
312        the oldest pending job age.
313    """
314    snapshot = get_metrics().snapshot(DEFAULT_WINDOW_SECONDS)
315    llm = LlmMetrics.model_validate(snapshot["llm"])
316    tools = ToolSummary.model_validate(snapshot["tools"])
317    rate_limits = RateLimitMetrics.model_validate(snapshot["rate_limits"])
318
319    database = await _ping_database()
320    if not database.connected:
321        response.status_code = status.HTTP_503_SERVICE_UNAVAILABLE
322        return DetailedHealthResponse(
323            status="unhealthy",
324            database=database,
325            llm=llm,
326            tools=tools,
327            outbox=OutboxMetrics(),
328            rate_limits=rate_limits,
329            tracing=_tracing_metrics(),
330        )
331
332    outbox = await _read_outbox()
333    return DetailedHealthResponse(
334        # The metrics store and the probe are in-process and cannot fail here, so
335        # only the outbox read can leave this endpoint with a partial answer.
336        status="degraded" if outbox is None else "healthy",
337        database=database,
338        llm=llm,
339        tools=tools,
340        outbox=outbox or OutboxMetrics(),
341        rate_limits=rate_limits,
342        tracing=_tracing_metrics(),
343    )
