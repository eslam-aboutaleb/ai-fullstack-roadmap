2Policy RAG tool for the OmniCare AI agent.
3
4This tool allows the AI agent to search through policy
5documents using hybrid search (vector similarity + PostgreSQL
6full-text search) to answer customer questions about insurance
7policies, coverage, claims process, etc.
8
9The tool is implemented as a plain async function that the
10agent registry can call directly. It uses the domain adapter
11``app.domain.policies.retriever`` for retrieval.
12"""

18from typing import Any
19
20from app.core.config import get_settings
21from app.domain.policies.retriever import retrieve_hybrid

26async def query_policy(query: str) -> dict[str, Any]:
27    """Searches OmniCare insurance policy documents to answer coverage questions.
28
29    Uses hybrid retrieval: pgvector similarity search combined with PostgreSQL
30    full-text search using Reciprocal Rank Fusion.
31
32    Use this tool whenever the user asks about:
33    - What is or is not covered under a policy
34    - Coverage limits or maximum payout amounts
35    - Deductibles or out-of-pocket amounts
36    - Policy exclusions or conditions
37    - Any general policy or insurance question
38
39    The tool returns relevant policy text sections with citations. Always
40    base your answer on the returned ``answer_context`` — do not add details
41    that are not present in the context.
42
43    Args:
44        query (str): The user's policy coverage question or search topic.
45            Example: "water damage from pipe burst"
46
47    Returns:
48        dict with keys:
49            - ``answer_context`` (str): Concatenated relevant policy sections
50              to use as grounding context in your answer.
51            - ``sources`` (list[dict]): Unique source citations, each with
52              ``section`` (str), ``source`` (str filename), and
53              ``relevance_score`` (float, lower is more relevant).
54            - ``chunks_found`` (int): Number of relevant chunks retrieved.
55    """
56    settings = get_settings()
57
58    results = await retrieve_hybrid(
59        query=query,
60        n_results=5,
61        distance_threshold=settings.rag_distance_threshold,
62    )
63
64    if not results:
65        return {
66            "answer_context": (
67                "No relevant policy information found for this query. "
68                "The user may be asking about something not covered in the policy documents, "
69                "or should contact OmniCare support directly."
70            ),
71            "sources": [],
72            "chunks_found": 0,
73        }
74
75    context_parts: list[str] = []
76    sources: list[dict[str, Any]] = []
77    seen_sections: set[str] = set()
78
79    for result in results:
80        doc_text = result["document"]
81        metadata = result["metadata"]
82        distance = result["distance"]
83
84        section = metadata.get("section")
85        if not section:
86            heading_match = re.search(r"^##\s+(.+)$", doc_text, re.MULTILINE)
87            section = heading_match.group(1) if heading_match else "General Policy"
88
89        source_file = metadata.get("source")
90        if not source_file:
91            source_file = "sample_policy.md"
92
93        context_parts.append(f"[{section}]:\n{doc_text}")
94
95        # Deduplicate citations — multiple overlapping sub-chunks from the
96        # same section should not generate duplicate source entries.
97        if section not in seen_sections:
98            source_entry: dict[str, Any] = {
99                "section": section,
100                "source": source_file,
101            }
102            if distance is not None:
103                source_entry["relevance_score"] = round(distance, 4)
104            sources.append(source_entry)
105            seen_sections.add(section)
106
107    return {
108        "answer_context": "\n\n---\n\n".join(context_parts),
109        "sources": sources,
110        "chunks_found": len(results),
111    }
