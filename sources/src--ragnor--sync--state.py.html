18@dataclass
19class FenceToken:
20    """Monotonic fencing token and lease proof for a specific sync scope.
21
22    Attributes:
23        scope: Scoped namespace, collection, or source identifier.
24        owner: Unique worker or coordinator identifier.
25        epoch: Monotonically increasing epoch number.
26        lease_expires_at: Epoch timestamp (seconds) when lease expires.
27    """
28
29    scope: str
30    owner: str
31    epoch: int
32    lease_expires_at: float
33
34    def is_valid(self, now: float | None = None) -> bool:
35        """Check if lease has not yet expired."""
36        if now is None:
37            now = time.time()
38        return self.lease_expires_at > now

93class InMemorySyncStateStore:
94    """Thread-safe and async-safe in-memory SyncStateStore."""
95
96    def __init__(self) -> None:
97        self._leases: dict[str, FenceToken] = {}
98        self._epochs: dict[str, int] = {}
99        self._generations: dict[str, int] = {}
100        self._checkpoints: dict[str, dict[str, Any]] = {}
101        self._lock = asyncio.Lock()
102        self._scope_locks: dict[str, asyncio.Lock] = {}
103        self._guarded: set[tuple[str, int]] = set()
104
105    @asynccontextmanager
106    async def fenced(self, token: FenceToken) -> AsyncIterator[None]:
107        async with self._scope_locks.setdefault(token.scope, asyncio.Lock()):
108            if not await self.validate_token(token):
109                raise ValueError("Expired or superseded fencing token")
110            self._guarded.add((token.scope, token.epoch))
111            try:
112                yield
113            finally:
114                self._guarded.discard((token.scope, token.epoch))
115
116    async def acquire_lease(
117        self, scope: str, owner: str, ttl_seconds: float = 30.0
118    ) -> FenceToken | None:
119        async with self._scope_locks.setdefault(scope, asyncio.Lock()), self._lock:
120            now = time.time()
121            current_lease = self._leases.get(scope)
122            if current_lease is not None and current_lease.is_valid(now):
123                return None
124
125            epoch = self._epochs.get(scope, 0) + 1
126            self._epochs[scope] = epoch
127            token = FenceToken(
128                scope=scope,
129                owner=owner,
130                epoch=epoch,
131                lease_expires_at=now + ttl_seconds,
132            )
133            self._leases[scope] = token
134            if scope not in self._generations:
135                self._generations[scope] = 1
136            return token
137
138    async def renew_lease(self, token: FenceToken, ttl_seconds: float = 30.0) -> FenceToken | None:
139        async with self._lock:
140            now = time.time()
141            current = self._leases.get(token.scope)
142            if (
143                current is None
144                or current.owner != token.owner
145                or current.epoch != token.epoch
146                or not current.is_valid(now)
147            ):
148                return None
149
150            renewed = FenceToken(
151                scope=token.scope,
152                owner=token.owner,
153                epoch=token.epoch,
154                lease_expires_at=now + ttl_seconds,
155            )
156            self._leases[token.scope] = renewed
157            return renewed
158
159    async def release_lease(self, token: FenceToken) -> bool:
160        async with self._lock:
161            current = self._leases.get(token.scope)
162            if (
163                current is not None
164                and current.owner == token.owner
165                and current.epoch == token.epoch
166            ):
167                del self._leases[token.scope]
168                return True
169            return False
170
171    async def validate_token(self, token: FenceToken) -> bool:
172        async with self._lock:
173            now = time.time()
174            current = self._leases.get(token.scope)
175            return (
176                current is not None
177                and current.owner == token.owner
178                and current.epoch == token.epoch
179                and current.is_valid(now)
180            )
181
182    async def get_active_generation(self, scope: str) -> int:
183        async with self._lock:
184            return self._generations.get(scope, 1)
185
186    async def advance_generation(self, scope: str, expected_generation: int) -> int:
187        async with self._lock:
188            current = self._generations.get(scope, 1)
189            if current != expected_generation:
190                return -1
191            self._generations[scope] = current + 1
192            return self._generations[scope]
193
194    async def get_checkpoint(self, scope: str) -> dict[str, Any] | None:
195        async with self._lock:
196            cp = self._checkpoints.get(scope)
197            return dict(cp) if cp is not None else None
198
199    async def save_checkpoint(
200        self, scope: str, checkpoint: dict[str, Any], token: FenceToken
201    ) -> bool:
202        async with self._lock:
203            now = time.time()
204            current = self._leases.get(scope)
205            if (
206                current is None
207                or current.owner != token.owner
208                or current.epoch != token.epoch
209                or (not current.is_valid(now) and (scope, token.epoch) not in self._guarded)
210            ):
211                return False
212            self._checkpoints[scope] = dict(checkpoint)
213            return True
