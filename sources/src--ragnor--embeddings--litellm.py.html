18def _cache_key(model_name: str, text: str) -> str:
19    """Return the content-addressed cache key for a model/text pair."""
20    digest = hashlib.sha256(f"{model_name}\0{text}".encode())
21    return digest.hexdigest()
22

24class LitellmEmbeddingFunction:
25    """Callable embedding function backed by LiteLLM.
26
27    Wraps ``litellm.embedding`` so any LiteLLM-supported
28    provider can serve embeddings.
29    """
30
31    def __init__(  # noqa: PLR0913
32        self,
33        api_key: str,
34        model_name: str,
35        api_base: str | None = None,
36        *,
37        cache_max_entries: int = 0,
38        max_batch_size: int | None = None,
39    ) -> None:
40        """Initialize the embedding function.
41
42        Args:
43            api_key: API key forwarded to the provider.
44            model_name: Embedding model name forwarded to litellm.
45            api_base: Optional provider base URL override.
46            cache_max_entries: Number of vectors kept in the bounded
47                LRU cache. ``0`` disables caching (the default).
48            max_batch_size: Maximum number of inputs sent in a single
49                ``litellm.embedding`` call. ``None`` (the default)
50                sends every input in one call.
51
52        Raises:
53            ValueError: If ``cache_max_entries`` is negative, or
54                ``max_batch_size`` is given and smaller than one.
55        """
56        if cache_max_entries < 0:
57            raise ValueError(f"cache_max_entries must be >= 0, got {cache_max_entries}")
58        if max_batch_size is not None and max_batch_size < 1:
59            raise ValueError(f"max_batch_size must be >= 1, got {max_batch_size}")
60        self.api_key = api_key
61        self.model_name = model_name
62        self.api_base = api_base
63        self.cache_max_entries = cache_max_entries
64        self.max_batch_size = max_batch_size
65        self._cache: OrderedDict[str, list[float]] = OrderedDict()
66
67    def _cache_key(self, text: str) -> str:
68        """Return the content-addressed cache key for ``text``."""
69        return _cache_key(self.model_name, text)
70
71    def _cache_store(self, key: str, vector: list[float]) -> None:
72        """Store a vector, evicting the oldest entries past the bound."""
73        self._cache[key] = vector
74        while len(self._cache) > self.cache_max_entries:
75            self._cache.popitem(last=False)
76
77    def _embed_sync(self, input: list[str]) -> list[list[float]]:
78        """Synchronous embedding call (runs in threadpool)."""
79        kwargs = {"api_key": self.api_key}
80        if self.api_base:
81            kwargs["api_base"] = self.api_base
82        response = litellm.embedding(
83            model=self.model_name,
84            input=input,
85            **kwargs,
86        )
87        return [item["embedding"] for item in response.data]
88
89    async def _embed_batched(self, texts: list[str]) -> list[list[float]]:
90        """Embed texts, splitting into ``max_batch_size`` chunks when set."""
91        if self.max_batch_size is None:
92            return await asyncio.to_thread(self._embed_sync, texts)
93        vectors: list[list[float]] = []
94        for start in range(0, len(texts), self.max_batch_size):
95            chunk = texts[start : start + self.max_batch_size]
96            chunk_vectors = await asyncio.to_thread(self._embed_sync, chunk)
97            vectors.extend(chunk_vectors)
98        return vectors
99
100    async def __call__(self, input: list[str]) -> list[list[float]]:
101        """Embed a list of text strings asynchronously.
102
103        With caching enabled, only cache misses are embedded and the
104        results are returned in the original input order. With caching
105        disabled, the whole input is embedded in a single
106        ``litellm.embedding`` call (or one call per
107        ``max_batch_size`` chunk when batch splitting is enabled).
108
109        Args:
110            input: List of text strings to embed.
111
112        Returns:
113            List of embedding vectors (each a list of floats) in the
114            same order as ``input``. Every returned vector is a fresh
115            copy, so mutating one cannot poison the cache.
116        """
117        if self.cache_max_entries == 0:
118            return await self._embed_batched(input)
119
120        cache_keys = [self._cache_key(text) for text in input]
121        results: dict[int, list[float]] = {}
122        misses: list[int] = []
123        for index, key in enumerate(cache_keys):
124            cached = self._cache.get(key)
125            if cached is None:
126                misses.append(index)
127            else:
128                self._cache.move_to_end(key)
129                results[index] = list(cached)
130
131        if misses:
132            miss_texts = [input[index] for index in misses]
133            miss_vectors = await self._embed_batched(miss_texts)
134            for index, vector in zip(misses, miss_vectors, strict=True):
135                self._cache_store(cache_keys[index], vector)
136                results[index] = list(vector)
137
138        return [results[index] for index in range(len(input))]
139
140    async def embed_query(self, input: str | list[str]) -> list[float]:
141        """Embed a single query string asynchronously.
142
143        Args:
144            input: Query string or list containing one string.
145
146        Returns:
147            A single embedding vector as a list of floats.
148        """
149        if isinstance(input, str):
150            input = [input]
151        result = await self(input)
152        return result[0]
