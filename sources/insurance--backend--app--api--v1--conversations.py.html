41async def _load_owned_conversation(
42    session: AsyncSession,
43    conversation_id: str,
44    current_user_id: str,
45) -> Conversation:
46    """Load a conversation only when it belongs to the authenticated user.
47
48    Both predicates live in the WHERE clause and every failure raises the same
49    404, so a conversation belonging to somebody else is answered exactly like
50    one that was never created: the endpoints cannot be used to learn whether
51    an id exists.
52
53    Args:
54        session: Async database session.
55        conversation_id: The conversation id as it arrived on the request path.
56        current_user_id: The authenticated user's UUID.
57
58    Returns:
59        Conversation: The conversation row owned by the caller.
60
61    Raises:
62        HTTPException: 400 if the conversation id is not a UUID, 404 if no
63            conversation with that id belongs to the caller.
64    """
65    try:
66        conv_uuid = uuid.UUID(conversation_id)
67    except ValueError as err:
68        raise HTTPException(status_code=400, detail="Invalid conversation ID") from err
69
70    conv = (
71        await session.execute(
72            select(Conversation)
73            .where(Conversation.id == conv_uuid)
74            .where(Conversation.user_id == uuid.UUID(current_user_id))
75        )
76    ).scalar_one_or_none()
77
78    if not conv:
79        raise HTTPException(status_code=404, detail="Conversation not found")
80
81    return conv

84@router.get(
85    "/chat/conversations",

89)
90async def list_conversations(
91    q: str | None = Query(
92        default=None,
93        max_length=100,
94        description=(
95            "Case-insensitive substring the conversation title must contain. "
96            "Omit it to list every conversation."
97        ),
98    ),
99    current_user_id: str = Depends(get_current_user),
100    session: AsyncSession = Depends(get_db),
101) -> ConversationListResponse:
102    """List all conversations for the authenticated user.
103
104    Retrieves conversation metadata (id, title, timestamps) ordered by most
105    recently updated. The message history is not included to keep the response
106    lightweight for sidebar list views.
107
108    Args:
109        q: Optional case-insensitive substring the title must contain. When it
110            is omitted or empty the filter is skipped entirely, so the
111            unfiltered response is unchanged.
112        current_user_id: The authenticated user's UUID, injected by
113            ``get_current_user``.
114        session: Async database session.
115
116    Returns:
117        ConversationListResponse: A list of conversation metadata objects
118        including id, title, and timestamps, ordered by most recently updated.
119    """
120    stmt = (
121        select(Conversation)
122        .where(Conversation.user_id == uuid.UUID(current_user_id))
123        .order_by(desc(Conversation.updated_at))
124    )
125    if q:
126        stmt = stmt.where(Conversation.title.ilike(f"%{q}%"))
127    result = await session.execute(stmt)
128    conversations = result.scalars().all()
129
130    return ConversationListResponse(
131        conversations=[
132            {
133                "id": c.id,
134                "title": c.title,
135                "created_at": c.created_at,
136                "updated_at": c.updated_at,
137            }
138            for c in conversations
139        ]
140    )

143@router.get(
144    "/chat/conversations/{conversation_id}",
145    response_model=ConversationDetailResponse,
146    summary="Get Conversation History",
147    description=(
148        "Returns the full message history for a specific conversation, ensuring the user owns it."
149    ),
150)
151async def get_conversation(
152    conversation_id: str,
153    current_user_id: str = Depends(get_current_user),
154    session: AsyncSession = Depends(get_db),
155) -> ConversationDetailResponse:
156    """Retrieve the full history of a single conversation.
157
158    Validates that the conversation belongs to the authenticated user before
159    returning it, preventing horizontal privilege escalation.
160
161    Args:
162        conversation_id: The UUID of the conversation to retrieve.
163
164    Returns:
165        ConversationDetailResponse: The conversation metadata plus the full
166        list of messages.
167
168    Raises:
169        HTTPException: 400 if the conversation ID is malformed, 404 if the
170        conversation does not exist or does not belong to the user.
171    """
172    conv = await _load_owned_conversation(session, conversation_id, current_user_id)
173
174    stmt_messages = (
175        select(ConversationMessage)
176        .where(ConversationMessage.conversation_id == conv.id)
177        .order_by(ConversationMessage.timestamp)
178    )
179    result_messages = await session.execute(stmt_messages)
180    db_messages = result_messages.scalars().all()
181
182    messages = []
183    for db_message in db_messages:
184        message_metadata = db_message.message_metadata or {}
185        messages.append(
186            {
187                "id": str(db_message.id),
188                "role": db_message.role,
189                "content": db_message.content,
190                "timestamp": db_message.timestamp.isoformat(),
191                "sources": message_metadata.get("sources") or [],
192                "tool_calls": message_metadata.get("tool_calls") or [],
193            }
194        )
195
196    return ConversationDetailResponse(
197        id=conv.id,
198        title=conv.title,
199        created_at=conv.created_at,
200        updated_at=conv.updated_at,
201        messages=messages,
202    )

218    session: AsyncSession = Depends(get_db),
219) -> ConversationMetadata:
220    """Rename a conversation owned by the authenticated user.
221
222    The request schema trims the title, so what is stored is what the user
223    sees rather than the string they typed.
224
225    Args:
226        conversation_id: The UUID of the conversation to rename.
227        payload: The new title, validated to be 1-255 characters.
228        current_user_id: The authenticated user's UUID, injected by
229            ``get_current_user``.
230        session: Async database session.
231
232    Returns:
233        ConversationMetadata: The conversation metadata after the rename,
234        carrying the timestamp the update assigned.
235
236    Raises:
237        HTTPException: 400 if the conversation ID is malformed, 404 if the
238            conversation does not exist or does not belong to the user.
239    """
240    conv = await _load_owned_conversation(session, conversation_id, current_user_id)
241
242    conv.title = payload.title
243    await session.commit()
244
245    logger.info(
246        "User '%s' renamed conversation '%s'.",
247        current_user_id,
248        conv.id,
249    )
250
251    return ConversationMetadata(
252        id=conv.id,
253        title=conv.title,
254        created_at=conv.created_at,
255        updated_at=conv.updated_at,

270async def delete_conversation(
271    conversation_id: str,
272    current_user_id: str = Depends(get_current_user),
273    session: AsyncSession = Depends(get_db),
274) -> None:
275    """Delete a conversation and its messages.
276
277    The message rows are deleted explicitly rather than relying on the
278    ``ON DELETE CASCADE`` the model declares, so the behaviour does not depend
279    on the FK definition a given migration happened to apply. Both deletes
280    share the transaction that commits them, so a failure leaves the
281    conversation intact.
282
283    The conversation's agent session is released once the row is gone: the
284    runner would otherwise keep resuming -- and citing -- a transcript the user
285    has already deleted on the next message.
286
287    Args:
288        conversation_id: The UUID of the conversation to delete.
289        current_user_id: The authenticated user's UUID, injected by
290            ``get_current_user``.
291        session: Async database session.
292
293    Raises:
294        HTTPException: 400 if the conversation ID is malformed, 404 if the
295            conversation does not exist or does not belong to the user.
296    """
297    conv = await _load_owned_conversation(session, conversation_id, current_user_id)
298
299    await session.execute(
300        delete(ConversationMessage).where(ConversationMessage.conversation_id == conv.id)
301    )
302    await session.execute(delete(Conversation).where(Conversation.id == conv.id))
303    await session.commit()
304
305    await delete_user_session(current_user_id, conv.session_id)
306
307    logger.info(
308        "User '%s' deleted conversation '%s'.",
309        current_user_id,
310        conv.id,
311    )
