1"""Tool registry for the OmniCare agent.
2
3The registry is the single place the agent's toolset is
4declared: adding a tool is one ``register`` call, with no
5``agent.py`` edit. The agent assembly
6(:func:`app.agent.agent.create_agent`) consumes the registry
7through :meth:`ToolRegistry.build_tool_list`.
8
9``ToolSpec.description`` is documentation-only: it never
10reaches the model, because :meth:`ToolRegistry.build_tool_list`
11returns just the handlers and the model sees each function's
12own docstring instead. The model-facing contract therefore
13lives in the tool docstrings and ``SYSTEM_INSTRUCTION``, not
14here.
15"""
16

30@dataclass
31class ToolSpec:
32    """A single agent tool declaration.
33
34    Attributes:
35        name: Tool name the model invokes.
36        description: What the tool does (for registry
37            consumers and documentation).
38        handler: The async tool function.
39        metadata: Optional annotations (e.g.
40            ``requires_auth``, scopes).
41    """
42
43    name: str
44    description: str
45    handler: Callable[..., Any]
46    metadata: dict[str, Any] = field(default_factory=dict)
47
48
49class ToolRegistry:
50    """Registry of the agent's tools."""
51
52    def __init__(self) -> None:
53        self._tools: dict[str, ToolSpec] = {}
54
55    def register(self, spec: ToolSpec) -> None:
56        """Register a tool (the last registration wins per name)."""
57        self._tools[spec.name] = spec
58
59    def get(self, name: str) -> ToolSpec:
60        """Return the tool registered under ``name``."""
61        try:
62            return self._tools[name]
63        except KeyError:
64            raise KeyError(f"Unknown tool: {name}") from None
65
66    def build_tool_list(self) -> list[Callable[..., Any]]:
67        """Return the registered handlers in registration order."""
68        return [spec.handler for spec in self._tools.values()]
69

71def build_default_registry() -> ToolRegistry:
72    """Register the five OmniCare tools.
73
74    Returns:
75        A registry with ``query_policy``, ``get_claim_status``,

77        ``get_procedure``.
78    """

81        ToolSpec(

84                "Search OmniCare insurance policy documents to answer coverage questions."

86            handler=query_policy,

88        )

90    registry.register(
91        ToolSpec(
92            name="get_claim_status",
93            description=(
94                "Look up the status of an existing claim by its claim ID when the "
95                "user already knows it; a returned found=false means no such "
96                "claim exists and must not be treated as one."
97            ),
98            handler=get_claim_status,
99            metadata={"requires_auth": True},
100        )
101    )
102    registry.register(
103        ToolSpec(
104            name="search_claims",
105            description=("Search the authenticated user's claims history using natural language."),
106            handler=search_claims,
107            metadata={"requires_auth": True},
108        )

111        ToolSpec(
112            name="prepare_claim_submission",
113            description="Validate and prepare a claim submission for confirmation.",
114            handler=prepare_claim_submission,
115            metadata={"requires_auth": True},
116        )
117    )
118    registry.register(
119        ToolSpec(
120            name="get_procedure",
121            description=(
122                "Load a domain procedure playbook (coverage-review or claim-triage) "

126            handler=get_procedure,
