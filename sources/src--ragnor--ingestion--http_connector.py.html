49def _normalize(url: str) -> str:
50    """Reject credential-bearing URLs before DNS or network access."""
51    if not isinstance(url, str) or len(url) > 8192 or any(ord(c) < 33 for c in url):
52        raise ValueError("Invalid HTTP URL")
53    parts = urlsplit(url)
54    if parts.scheme.lower() not in {"http", "https"} or not parts.hostname:
55        raise ValueError("An absolute HTTP(S) URL is required")
56    if parts.username is not None or parts.password is not None or "\\" in url:
57        raise ValueError("URL credentials are forbidden; inject headers instead")
58    if any(_SECRET_KEY.search(key) for key, _ in parse_qsl(parts.query, keep_blank_values=True)):
59        raise ValueError("Credential-bearing query parameters are forbidden")
60    host = parts.hostname.encode("idna").decode("ascii").lower().rstrip(".")
61    try:
62        port = parts.port
63    except ValueError:
64        raise ValueError("Invalid HTTP URL port") from None
65    authority = f"[{host}]" if ":" in host else host
66    if port is not None and port != (443 if parts.scheme.lower() == "https" else 80):
67        authority += f":{port}"
68    path = posixpath.normpath(parts.path or "/")
69    if parts.path.endswith("/") and not path.endswith("/"):
70        path += "/"
71    return urlunsplit((parts.scheme.lower(), authority, path, parts.query, ""))

74def _safe_url(url: str) -> str:
75    parts = urlsplit(url)
76    # This helper also handles untrusted URL strings in callback metadata.
77    host = parts.hostname or "redacted"
78    authority = f"[{host}]" if ":" in host else host
79    try:
80        if parts.port:
81            authority += f":{parts.port}"
82    except ValueError:
83        pass
84    query = urlencode([(key, "[redacted]") for key, _ in parse_qsl(parts.query)])
85    return urlunsplit((parts.scheme, authority, parts.path, query, ""))

88def _safe_metadata(value: Any) -> Any:
89    if isinstance(value, Mapping):
90        return {
91            str(key): "[redacted]" if _SECRET_KEY.search(str(key)) else _safe_metadata(item)
92            for key, item in value.items()
93        }
94    if isinstance(value, (list, tuple)):
95        return [_safe_metadata(item) for item in value]
96    if isinstance(value, str):
97        return re.sub(r"https?://[^\s<>\"']+", lambda m: _safe_url(m.group()), value)
98    return value if value is None or isinstance(value, (int, float, bool)) else "[redacted]"

140class HttpApiConnector:
141    """Paginated API/document loader with fail-closed completeness semantics.
142
143    ``page_fetcher(self, url, cursor)`` may return an awaitable or a
144    ``(list[RawDocument], next_cursor, has_more)`` tuple. Continuation tokens are
145    validated and stored only behind opaque connector-local cursors. Fetchers
146    must not claim exhaustion after an upstream failure. Limits never authorize
147    deletion, including when a limit exactly matches the final page.
148
149    Query values are never persisted or hashed: query-bearing document locators
150    include an opaque per-instance identifier. Reuse a connector instance when
151    stable query-specific identities are needed; checkpoints are instance-local.
152
153    ``transport`` accepts an httpx transport for testing; requests still target a
154    pinned, public IP. DNS, proxy environment variables, and automatic redirects
155    cannot override that pin. Disabling network/redirect protections is rejected.
156    """
157
158    def __init__(self, base_url: str, **options: Any) -> None:
159        if options.get("private_network_protection", True) is not True:
160            raise ValueError("Private-network protections cannot be disabled")
161        if options.get("redirect_check", True) is not True:
162            raise ValueError("Redirect protections cannot be disabled")
163        self._base_url = _normalize(base_url)
164        paths = options.get("paths", options.get("endpoints", [self._base_url]))
165        if isinstance(paths, str):
166            paths = [paths]
167        self._urls = [_normalize(urljoin(self._base_url.rstrip("/") + "/", p)) for p in paths]
168        self.headers = dict(options.get("headers") or {})
169        self.credentials = options.get("credentials", options.get("auth"))
170        self.timeout = float(options.get("timeout", 30.0))
171        if self.timeout <= 0:
172            raise ValueError("timeout must be positive")
173        self.max_pages = _positive(
174            100 if options.get("max_pages") is None else options["max_pages"],
175            "max_pages",

177        self.max_items = _positive(
178            10000 if options.get("max_items") is None else options["max_items"],

182        self.max_response_bytes = _positive(
183            options.get("max_response_bytes", options.get("max_bytes", 10 * 1024 * 1024)),
184            "max_response_bytes",
185        )
186        self.max_requests = _positive(options.get("max_requests", 1000), "max_requests")
187        self.max_redirects = _positive(options.get("max_redirects", 10), "max_redirects")
188        domains = options.get("domains") or options.get("allowed_domains") or []
189        if isinstance(domains, str):
190            domains = [domains]
191        self.domains = [str(d).encode("idna").decode("ascii").lower().rstrip(".") for d in domains]
192        self.allowed_domains = self.domains

254    def _headers(self) -> dict[str, str]:
255        headers = dict(self.headers)
256        credential = self.credentials
257        if isinstance(credential, Mapping):
258            Credential(**credential).apply(headers)
259        elif isinstance(credential, Credential):
260            credential.apply(headers)
261        elif callable(credential):
262            credential(headers)
263        elif credential is not None:
264            raise _Failure("unsupported_credentials", SourceStatus.INACCESSIBLE)
265        # Cookies and arbitrary custom headers may be credentials too.
266        return {str(key): str(value) for key, value in headers.items()}

272    async def _public_ip(self, url: str) -> str:
273        parts = urlsplit(url)
274        host = parts.hostname or ""
275        try:
276            addresses = [ipaddress.ip_address(host)]
277        except ValueError:
278            infos = await asyncio.to_thread(
279                socket.getaddrinfo,
280                host,
281                parts.port or (443 if parts.scheme == "https" else 80),
282                type=socket.SOCK_STREAM,
283            )
284            addresses = [ipaddress.ip_address(info[4][0]) for info in infos]
285        if not addresses or any(not address.is_global for address in addresses):
286            raise _Failure("private_address", SourceStatus.INACCESSIBLE)
287        return str(addresses[0])

289    async def _fetch(self, url: str) -> Any:  # noqa: PLR0915
290        try:
291            import httpx  # noqa: PLC0415
292        except ImportError as exc:
293            raise _Failure("httpx_not_installed", SourceStatus.INACCESSIBLE) from exc
294        current = _normalize(url)
295        headers = self._headers()
296        async with httpx.AsyncClient(
297            timeout=self.timeout,
298            follow_redirects=False,
299            trust_env=False,
300            transport=self._transport,
301        ) as client:
302            for _ in range(self.max_redirects + 1):
303                if not self._domain_allowed(current):
304                    raise _Failure("domain_not_allowed", SourceStatus.INACCESSIBLE)
305                if self._request_count >= self.max_requests:
306                    raise _Failure("max_requests")
307                address = await self._public_ip(current)
308                original = httpx.URL(current)
309                pinned = original.copy_with(host=address)
310                request_headers = {k: v for k, v in headers.items() if k.lower() != "host"}
311                request_headers["Host"] = original.netloc.decode("ascii")
312                request_headers["Accept-Encoding"] = "identity"
313                client.cookies.clear()
314                self._request_count += 1
315                async with client.stream(
316                    "GET",
317                    pinned,
318                    headers=request_headers,
319                    extensions={"sni_hostname": original.host.encode("ascii")},
320                ) as response:
321                    if response.status_code in _REDIRECTS:
322                        location = response.headers.get("location")
323                        if not location:
324                            raise _Failure("redirect_missing_location")
325                        target = _normalize(urljoin(current, location))
326                        old, new = urlsplit(current), urlsplit(target)
327                        if old.scheme == "https" and new.scheme != "https":
328                            raise _Failure("https_downgrade", SourceStatus.INACCESSIBLE)
329                        if headers and (old.scheme, old.netloc) != (new.scheme, new.netloc):
330                            raise _Failure(
331                                "authenticated_cross_origin_redirect", SourceStatus.INACCESSIBLE
332                            )
333                        current = target
334                        continue
335                    code = response.status_code
336                    if code == 404:
337                        raise _Failure("not_found", SourceStatus.NOT_FOUND, status_code=code)
338                    if code in {401, 403}:
339                        raise _Failure("inaccessible", SourceStatus.INACCESSIBLE, status_code=code)
340                    if code == 429:
341                        retry = response.headers.get("retry-after", "")
342                        try:
343                            seconds = max(0.0, float(retry))
344                        except ValueError:
345                            try:
346                                date = parsedate_to_datetime(retry)
347                                seconds = max(0.0, (date - datetime.now(UTC)).total_seconds())
348                            except (TypeError, ValueError, OverflowError):
349                                seconds = None
350                        raise _Failure(
351                            "rate_limited",
352                            SourceStatus.TRANSIENT,
353                            status_code=code,
354                            retry_after=seconds,
355                        )
356                    if code >= 500 or code in {408, 425}:
357                        raise _Failure("http_transient", SourceStatus.TRANSIENT, status_code=code)
358                    if not 200 <= code < 300:
359                        raise _Failure("http_error", SourceStatus.ERROR, status_code=code)
360                    if response.headers.get("content-encoding", "identity").lower() != "identity":
361                        raise _Failure("unsupported_content_encoding")
362                    length = response.headers.get("content-length")
363                    if length is not None and int(length) > self.max_response_bytes:
364                        raise _Failure("response_too_large")
365                    data = bytearray()
366                    async for chunk in response.aiter_bytes(chunk_size=65536):
367                        if len(data) + len(chunk) > self.max_response_bytes:
368                            raise _Failure("response_too_large")
369                        data.extend(chunk)
370                    # Do not retain the pinned request or raw headers in persisted metadata.
371                    return httpx.Response(
372                        code,
373                        content=bytes(data),
374                        headers={"content-type": response.headers.get("content-type", "")},
375                        request=httpx.Request("GET", current),
376                    )
377        raise _Failure("too_many_redirects")

420    async def _read_batch(self, state: dict[str, Any]) -> list[RawDocument]:
421        url = self._urls[state["index"]]
422        if not self._domain_allowed(url):
423            raise _Failure("domain_not_allowed", SourceStatus.INACCESSIBLE)
424        value = await self.fetch_page(url, state["upstream"])
425        if not isinstance(value, tuple) or len(value) != 3:
426            raise _Failure("malformed_page")
427        docs, token, more = value
428        if not isinstance(docs, list) or any(not isinstance(doc, RawDocument) for doc in docs):
429            raise _Failure("malformed_documents")
430        if not isinstance(more, bool):
431            raise _Failure("malformed_has_more")
432        if more:
433            if (
434                not isinstance(token, (str, int))
435                or isinstance(token, bool)
436                or token == ""
437                or token in state["seen"]
438                or token == state["upstream"]
439            ):
440                raise _Failure("invalid_or_repeated_cursor")
441            state["seen"].append(token)
442            state["upstream"] = token
443        else:
444            if token is not None:
445                raise _Failure("cursor_without_continuation")
446            state["index"] += 1
447            state["upstream"], state["seen"] = None, []
448            state["done"] = state["index"] >= len(self._urls)
449        return docs

468    async def load_page(self, *, cursor: Any = None, limit: int | None = None) -> LoadResult:
469        """Read one bounded page; cursors belong only to this connector instance."""
470        size = self.page_size if limit is None else _positive(limit, "limit")
471        self._page_skipped = []
472        documents: list[RawDocument] = []
473        errors: list[Any] = []
474        state = self._initial_state()
475        status = SourceStatus.OK
476        try:
477            if cursor is not None:
478                if not isinstance(cursor, str) or cursor not in self._states:
479                    raise _Failure("invalid_cursor")
480                state = copy.deepcopy(self._states[cursor])
481            self._request_count = state["requests"]
482            if not state["buffer"] and not state["done"]:
483                if state["pages"] >= self.max_pages:
484                    raise _Failure("max_pages")
485                async with asyncio.timeout(self.timeout):
486                    state["buffer"] = await self._read_batch(state)
487                state["pages"] += 1
488            remaining = min(size, self.max_items - state["items"])
489            documents = [self._safe_document(doc) for doc in state["buffer"][:remaining]]
490            state["buffer"] = state["buffer"][remaining:]
491            state["items"] += len(documents)
492            if state.get("terminal_error"):
493                raise _Failure(state["terminal_error"])
494            if state["items"] >= self.max_items:
495                raise _Failure("max_items")
496            if state["pages"] >= self.max_pages:
497                raise _Failure("max_pages")
498        except _Failure as exc:
499            errors.append(exc.error)
500            status = exc.status
501        except Exception as exc:
502            # Exception text often embeds request URLs, header values, or tokens.
503            errors.append({"reason": "request_failed", "error_type": type(exc).__name__})
504            status = SourceStatus.TRANSIENT
505        state["requests"] = self._request_count
506        has_more = bool(state["buffer"] or not state["done"])
507        next_cursor = None
508        if has_more and not errors:
509            next_cursor = uuid4().hex
510            self._states[next_cursor] = state
511        complete = not has_more and not errors
512        if complete and not documents:
513            status = SourceStatus.EMPTY
514        result = LoadResult(
515            documents,
516            status=status,
517            skipped=self._page_skipped,
518            errors=errors,
519            cursor=cursor,
520            next_cursor=next_cursor,
521            has_more=has_more,
522            complete=complete,
523            pages=1,
524            total_items=len(documents),
525        )
526        self._last_result = result
527        return result

597class WebsiteConnector(HttpApiConnector):
598    """Explicit-domain BFS crawler with independent URL/depth/queue/request caps.
599
600    Every encountered in-scope link beyond a cap makes the scan incomplete.
601    Relative links resolve against the final response URL after checked redirects.
602    Off-domain links are outside the configured source scope and are not crawled.
603    """
604
605    def __init__(self, start_urls: Any, **options: Any) -> None:
606        urls = [start_urls] if isinstance(start_urls, str) else list(start_urls)
607        if not urls:
608            raise ValueError("At least one start URL is required")
609        if not (options.get("allowed_domains") or options.get("domains")):
610            raise ValueError("WebsiteConnector requires explicit allowed_domains")
611        super().__init__(urls[0], **options)
612        self._urls = list(dict.fromkeys(_normalize(url) for url in urls))
613        self.max_urls = _positive(options.get("max_urls", 50), "max_urls")
614        self.max_queue = _positive(options.get("max_queue", 100), "max_queue")
615        depth = options.get("max_depth", 2)
616        if isinstance(depth, bool) or not isinstance(depth, int) or depth < 0:
617            raise ValueError("max_depth must be a nonnegative integer")
618        self.max_depth = depth
619        if len(self._urls) > min(self.max_urls, self.max_queue):
620            raise ValueError("Start URLs exceed crawl limits")
621        if any(not self._domain_allowed(url) for url in self._urls):
622            raise ValueError("Start URL is outside allowed_domains")
623
624    @property
625    def id(self) -> str:
626        return "website"
627
628    @property
629    def start_urls(self) -> list[str]:
630        return self.paths
631
632    @property
633    def options(self) -> dict[str, Any]:
634        return {
635            **super().options,
636            "max_urls": self.max_urls,
637            "max_queue": self.max_queue,
638            "max_depth": self.max_depth,
639        }
640
641    def _initial_state(self) -> dict[str, Any]:
642        return {
643            **super()._initial_state(),
644            "queue": deque((url, 0) for url in self._urls),
645            "known": set(self._urls),
646            "visited": set(),
647            "truncated": False,
648        }
649
650    async def _read_batch(self, state: dict[str, Any]) -> list[RawDocument]:
651        if len(state["visited"]) >= self.max_urls:
652            raise _Failure("max_urls")
653        url, depth = state["queue"].popleft()
654        state["visited"].add(url)
655        response = await self._fetch(url)
656        final_url = _normalize(str(response.url))
657        state["known"].add(final_url)
658        docs = list(self._extract(response, depth=depth))
659        parser = _Links(self.max_queue + self.max_urls)
660        if "html" in response.headers.get("content-type", "").lower():
661            parser.feed(response.text)
662        state["truncated"] |= parser.truncated
663        for href in parser.links:
664            try:
665                link = _normalize(urljoin(final_url, href))
666            except ValueError:
667                state["truncated"] = True
668                continue
669            if not self._domain_allowed(link) or link in state["known"]:
670                continue
671            if (
672                depth >= self.max_depth
673                or len(state["known"]) >= self.max_urls
674                or len(state["queue"]) >= self.max_queue
675            ):
676                state["truncated"] = True
677                continue
678            state["known"].add(link)
679            state["queue"].append((link, depth + 1))
680        state["done"] = not state["queue"]
681        if state["done"] and state["truncated"]:
682            state["terminal_error"] = "crawl_limit"
683        return docs
