1"""Embedding metrics: counters and usage accounting.
2
3:class:`EmbeddingMetrics` accumulates observable events from an
4:class:`~ragnor.embeddings.executor.EmbeddingExecutor` run: provider
5calls, cache hits, input size estimates, retries, and any usage
6statistics the provider reports.
7
8All token-count fields are explicitly labeled as *estimates* because
9the true token count is model-specific and rarely returned by providers.
10Callers must not treat these as billing-accurate figures.
11
12Usage::
13
14    metrics = EmbeddingMetrics()
15    metrics.embed_calls += 1
16    metrics.total_input_tokens_estimate += len(text.split())
17
18    combined = metrics.merge(other_metrics)
19"""
20
21from __future__ import annotations
22
23from dataclasses import dataclass, field
24from typing import Any
25
26
27@dataclass
28class EmbeddingMetrics:
29    """Accumulated metrics from one or more embedding operations.
30
31    Attributes:
32        embed_calls: Number of provider embedding calls made (after
33            cache resolution; does not count cache hits).
34        cache_hits: Number of vectors served from the local cache.
35        cache_misses: Number of vectors that required a provider call.
36        total_input_tokens_estimate: Estimated total input tokens across
37            all provider calls.  This is a *rough estimate* (e.g. word
38            count × 1.3) and must not be presented as exact billing usage.
39        total_retry_count: Total number of retry attempts across all calls
40            (not counting the initial attempt).
41        provider_usage: Raw usage dict reported by the provider (e.g.
42            ``{"prompt_tokens": N, "total_tokens": M}``).  Provider-specific
43            and may be empty when the provider does not report usage.
44        is_token_estimate: Always ``True``; signals to callers that
45            ``total_input_tokens_estimate`` is an estimate, not an exact count.
46    """
47
48    embed_calls: int = 0
49    cache_hits: int = 0
50    cache_misses: int = 0
51    total_input_tokens_estimate: int = 0
52    total_retry_count: int = 0
53    provider_usage: dict[str, Any] = field(default_factory=dict)
54    is_token_estimate: bool = True
55
56    def merge(self, other: EmbeddingMetrics) -> EmbeddingMetrics:
57        """Return a new :class:`EmbeddingMetrics` combining ``self`` and ``other``.
58
59        ``provider_usage`` values are summed when both sides carry the same
60        integer key; non-integer values from ``other`` overwrite ``self``.
61
62        Args:
63            other: Metrics to combine with ``self``.
64
65        Returns:
66            A fresh :class:`EmbeddingMetrics` instance with summed counters.
67        """
68        merged_usage: dict[str, Any] = dict(self.provider_usage)
69        for key, value in other.provider_usage.items():
70            both_int = isinstance(merged_usage.get(key), int) and isinstance(value, int)
71            if key in merged_usage and both_int:
72                merged_usage[key] = merged_usage[key] + value
73            else:
74                merged_usage[key] = value
75
76        return EmbeddingMetrics(
77            embed_calls=self.embed_calls + other.embed_calls,
78            cache_hits=self.cache_hits + other.cache_hits,
79            cache_misses=self.cache_misses + other.cache_misses,
80            total_input_tokens_estimate=(
81                self.total_input_tokens_estimate + other.total_input_tokens_estimate
82            ),
83            total_retry_count=self.total_retry_count + other.total_retry_count,
84            provider_usage=merged_usage,
85            is_token_estimate=True,
86        )
87
88    def __repr__(self) -> str:  # pragma: no cover
89        return (
90            f"EmbeddingMetrics(calls={self.embed_calls}, hits={self.cache_hits}, "
91            f"misses={self.cache_misses}, retries={self.total_retry_count})"
92        )
