32class HealthState(StrEnum):
33    """Subsystem or system-wide health classification."""
34
35    READY = "ready"
36    DEGRADED = "degraded"
37    UNAVAILABLE = "unavailable"
38    UNKNOWN = "unknown"
39
40
41# Precedence order for state rollup: higher index = worse state
42_STATE_SEVERITY: dict[HealthState, int] = {
43    HealthState.READY: 0,
44    HealthState.UNKNOWN: 1,
45    HealthState.DEGRADED: 2,
46    HealthState.UNAVAILABLE: 3,
47}

69@dataclass
70class HealthReport:
71    """Rollup health report representing overall system readiness.
72
73    Attributes:
74        state: Worst-case aggregate HealthState across all components.
75        components: Map of component names to ComponentHealth objects.
76        reasons: Aggregated non-ready diagnostic messages.
77        timestamp: Unix epoch timestamp when health was evaluated.
78    """
79
80    state: HealthState = HealthState.READY
81    components: dict[str, ComponentHealth] = field(default_factory=dict)
82    reasons: list[str] = field(default_factory=list)
83    timestamp: float = field(default_factory=time.time)
84
85    def __init__(  # noqa: PLR0913
86        self,
87        state: HealthState | None = None,
88        components: Sequence[ComponentHealth] | Mapping[str, ComponentHealth] | None = None,
89        reasons: list[str] | None = None,
90        timestamp: float | None = None,
91        *,
92        overall: HealthState | None = None,
93    ) -> None:
94        if isinstance(components, (list, tuple)):
95            comp_map = {c.name: c for c in components}
96        elif isinstance(components, (dict, Mapping)):
97            comp_map = dict(components)
98        else:
99            comp_map = {}
100
101        self.components = comp_map
102        self.timestamp = timestamp if timestamp is not None else time.time()
103
104        if overall is not None:
105            self.state = overall
106        elif state is not None:
107            self.state = state
108        elif comp_map:
109            worst_severity = max(_STATE_SEVERITY[c.state] for c in comp_map.values())
110            self.state = next(st for st, sev in _STATE_SEVERITY.items() if sev == worst_severity)
111        else:
112            self.state = HealthState.READY
113
114        if reasons is not None:
115            self.reasons = list(reasons)
116        else:
117            self.reasons = [
118                f"[{c.name}] {c.reason}"
119                for c in comp_map.values()
120                if c.reason and c.state != HealthState.READY
121            ]
122
123    @property
124    def is_healthy(self) -> bool:
125        """Return True if all components are ready."""
126        return self.state == HealthState.READY
127
128    @property
129    def is_ready(self) -> bool:
130        """Return True if all components are healthy and ready."""
131        return self.state == HealthState.READY
132
133    @property
134    def is_degraded(self) -> bool:
135        """Return True if system is operational with warnings or stale components."""
136        return self.state == HealthState.DEGRADED
137
138    @property
139    def is_unavailable(self) -> bool:
140        """Return True if a critical subsystem is offline or incompatible."""
141        return self.state == HealthState.UNAVAILABLE

143    def to_dict(self) -> dict[str, Any]:
144        """Convert report to serializable dictionary."""
145        return {
146            "state": str(self.state),
147            "timestamp": self.timestamp,
148            "reasons": self.reasons,
149            "components": {
150                name: {
151                    "state": str(comp.state),
152                    "reason": comp.reason,
153                    "latency_ms": comp.latency_ms,
154                    "details": comp.details,
155                }
156                for name, comp in self.components.items()
157            },
158        }

203    def check_configuration(self) -> ComponentHealth:
204        """Validate configuration settings and dimensional compatibility."""
205        start = time.perf_counter()
206        if self.settings is None:
207            latency = (time.perf_counter() - start) * 1000
208            return ComponentHealth(
209                name="configuration",
210                state=HealthState.UNKNOWN,
211                reason="No settings configured",
212                latency_ms=latency,
213            )
214
215        model = getattr(self.settings, "embedding_model", None)
216        details: dict[str, Any] = {
217            "embedding_model": model,
218        }
219
220        if model is None:
221            latency = (time.perf_counter() - start) * 1000
222            return ComponentHealth(
223                name="configuration",
224                state=HealthState.DEGRADED,
225                reason="Embedding model not specified in settings",
226                latency_ms=latency,
227                details=details,
228            )
229
230        if model not in _DIMENSIONS:
231            latency = (time.perf_counter() - start) * 1000
232            return ComponentHealth(
233                name="configuration",
234                state=HealthState.DEGRADED,
235                reason=(f"Unknown embedding model {model!r} (not in registered dimensions table)"),
236                latency_ms=latency,
237                details=details,
238            )
239
240        try:
241            expected_dim = get_embedding_dimension(model)
242            details["expected_dimension"] = expected_dim
243        except Exception as exc:  # noqa: BLE001
244            latency = (time.perf_counter() - start) * 1000
245            return ComponentHealth(
246                name="configuration",
247                state=HealthState.DEGRADED,
248                reason=f"Unknown embedding model {model!r}: {exc}",
249                latency_ms=latency,
250                details=details,
251            )
252
253        # Check embedding dimension compatibility with vector store
254        if self.vector_store is not None:
255            store_dim = getattr(self.vector_store, "embedding_dim", None)
256            details["store_dimension"] = store_dim
257
258            if store_dim is not None and expected_dim != store_dim:
259                latency = (time.perf_counter() - start) * 1000
260                return ComponentHealth(
261                    name="configuration",
262                    state=HealthState.UNAVAILABLE,
263                    reason=(
264                        f"Dimension mismatch: store configured for {store_dim} "
265                        f"but model {model!r} produces {expected_dim}"
266                    ),
267                    latency_ms=latency,
268                    details=details,
269                )
270
271        latency = (time.perf_counter() - start) * 1000
272        return ComponentHealth(
273            name="configuration",
274            state=HealthState.READY,
275            reason="Configuration compatible",
276            latency_ms=latency,
277            details=details,
278        )

328    async def check_counts(self, scope: str | None = None) -> ComponentHealth:
329        """Inspect active indexed chunks distinguishing deliberate vs unexpected zero counts."""
330        start = time.perf_counter()
331        target_scope = scope or self.scope
332        docs_count = 0
333        chunk_count = 0
334
335        if self.manifest_store is not None:
336            try:
337                if hasattr(self.manifest_store, "list"):
338                    active_records = await asyncio.wait_for(
339                        self.manifest_store.list(target_scope, status=ManifestStatus.ACTIVE),
340                        timeout=self.timeout_seconds,
341                    )
342                else:
343                    active_records = await asyncio.wait_for(
344                        self.manifest_store.list_active(target_scope),
345                        timeout=self.timeout_seconds,
346                    )
347                docs_count = len(active_records)
348                chunk_count = sum(len(r.chunk_ids) for r in active_records)
349            except Exception as exc:  # noqa: BLE001
350                latency = (time.perf_counter() - start) * 1000
351                return ComponentHealth(
352                    name="active_counts",
353                    state=HealthState.UNAVAILABLE,
354                    reason=f"Failed to query manifest records: {exc}",
355                    latency_ms=latency,
356                )
357        elif self.vector_store is not None:
358            try:
359                chunk_count = await asyncio.wait_for(
360                    self.vector_store.count(strict=True),
361                    timeout=self.timeout_seconds,
362                )
363            except Exception as exc:  # noqa: BLE001
364                latency = (time.perf_counter() - start) * 1000
365                return ComponentHealth(
366                    name="active_counts",
367                    state=HealthState.UNAVAILABLE,
368                    reason=f"Failed to query chunk counts: {exc}",
369                    latency_ms=latency,
370                )
371        else:
372            latency = (time.perf_counter() - start) * 1000
373            return ComponentHealth(
374                name="active_counts",
375                state=HealthState.UNKNOWN,
376                reason="Neither manifest store nor vector store configured",
377                latency_ms=latency,
378            )
379
380        latency = (time.perf_counter() - start) * 1000
381        details = {
382            "active_documents": docs_count,
383            "active_chunks": chunk_count,
384            "allow_empty": self.allow_empty,
385        }
386
387        if chunk_count > 0 or docs_count > 0:
388            return ComponentHealth(
389                name="active_counts",
390                state=HealthState.READY,
391                latency_ms=latency,
392                details=details,
393            )
394
395        # 0 chunks / 0 docs
396        if self.allow_empty:
397            details["deliberate_empty"] = True
398            return ComponentHealth(
399                name="active_counts",
400                state=HealthState.READY,
401                reason="Collection is deliberately empty",
402                latency_ms=latency,
403                details=details,
404            )
405
406        return ComponentHealth(
407            name="active_counts",
408            state=HealthState.DEGRADED,
409            reason="Configured collection unexpectedly indexing zero chunks",
410            latency_ms=latency,
411            details=details,
412        )

414    async def check_sync_status(  # noqa: PLR0911
415        self, scope: str | None = None
416    ) -> ComponentHealth:
417        """Check last successful sync and scan completeness."""
418        start = time.perf_counter()
419        if self.sync_state_store is None:
420            latency = (time.perf_counter() - start) * 1000
421            return ComponentHealth(
422                name="sync_status",
423                state=HealthState.UNKNOWN,
424                reason="No sync state store configured",
425                latency_ms=latency,
426            )
427
428        target_scope = scope or self.scope
429        try:
430            checkpoint = None
431            if hasattr(self.sync_state_store, "get_checkpoint"):
432                checkpoint = await asyncio.wait_for(
433                    self.sync_state_store.get_checkpoint(target_scope),
434                    timeout=self.timeout_seconds,
435                )
436            elif hasattr(self.sync_state_store, "get_state"):
437                state = await asyncio.wait_for(
438                    self.sync_state_store.get_state(target_scope),
439                    timeout=self.timeout_seconds,
440                )
441                if state is not None:
442                    checkpoint = {
443                        "phase": "completed" if state.last_sync_timestamp > 0 else "initial",
444                        "authoritative": state.is_authoritative,
445                        "last_error": state.last_error,
446                    }
447
448            latency = (time.perf_counter() - start) * 1000
449            if checkpoint is None:
450                if self.allow_empty:
451                    return ComponentHealth(
452                        name="sync_status",
453                        state=HealthState.READY,
454                        reason="Collection is deliberately empty; initial sync not yet run",
455                        latency_ms=latency,
456                    )
457                return ComponentHealth(
458                    name="sync_status",
459                    state=HealthState.UNKNOWN,
460                    reason=f"No sync checkpoint found for configured scope {target_scope!r}",
461                    latency_ms=latency,
462                )
463
464            is_authoritative = checkpoint.get("authoritative", True)
465            last_error = checkpoint.get("last_error")
466            phase = checkpoint.get("phase", "")
467
468            details: dict[str, Any] = {
469                "checkpoint": checkpoint,
470                "is_authoritative": is_authoritative,
471            }
472
473            if last_error:
474                return ComponentHealth(
475                    name="sync_status",
476                    state=HealthState.DEGRADED,
477                    reason=f"Last sync recorded failure: {last_error}",
478                    latency_ms=latency,
479                    details=details,
480                )
481
482            if not is_authoritative or phase != "completed":
483                return ComponentHealth(
484                    name="sync_status",
485                    state=HealthState.DEGRADED,
486                    reason="Last sync completed with incomplete or non-authoritative scan coverage",
487                    latency_ms=latency,
488                    details=details,
489                )
490
491            return ComponentHealth(
492                name="sync_status",
493                state=HealthState.READY,
494                reason="Sync status healthy and authoritative",
495                latency_ms=latency,
496                details=details,
497            )
498        except Exception as exc:  # noqa: BLE001
499            latency = (time.perf_counter() - start) * 1000
500            return ComponentHealth(
501                name="sync_status",
502                state=HealthState.UNAVAILABLE,
503                reason=f"Failed to query sync state: {exc}",
504                latency_ms=latency,
505            )

507    async def check_jobs(self, scope: str | None = None) -> ComponentHealth:
508        """Inspect background embedding jobs for pending or dead-letter items."""
509        start = time.perf_counter()
510        if self.job_store is None:
511            latency = (time.perf_counter() - start) * 1000
512            return ComponentHealth(
513                name="job_queue",
514                state=HealthState.READY,
515                reason="Job queue store not configured (in-process execution)",
516                latency_ms=latency,
517            )
518
519        try:
520            dead_letter_count = 0
521            pending_count = 0
522            details: dict[str, Any] = {}
523
524            if hasattr(self.job_store, "get_status_counts"):
525                counts = await asyncio.wait_for(
526                    self.job_store.get_status_counts(), timeout=self.timeout_seconds
527                )
528                details = dict(counts)
529                dead_letter_count = counts.get("dead_letter", 0) + counts.get("failed", 0)
530                pending_count = counts.get("pending", 0)
531            elif hasattr(self.job_store, "jobs"):
532                for j in getattr(self.job_store, "jobs", []):
533                    status = getattr(j, "status", "")
534                    details[status] = details.get(status, 0) + 1
535                    if status in ("dead_letter", "failed"):
536                        dead_letter_count += 1
537                    elif status == "pending":
538                        pending_count += 1
539
540            latency = (time.perf_counter() - start) * 1000
541
542            if dead_letter_count > 0:
543                return ComponentHealth(
544                    name="job_queue",
545                    state=HealthState.DEGRADED,
546                    reason=(
547                        f"Found {dead_letter_count} dead-letter embedding jobs "
548                        "requiring intervention"
549                    ),
550                    latency_ms=latency,
551                    details=details,
552                )
553
554            return ComponentHealth(
555                name="job_queue",
556                state=HealthState.READY,
557                latency_ms=latency,
558                details=details or {"pending_count": pending_count},
559            )
560        except Exception as exc:  # noqa: BLE001
561            latency = (time.perf_counter() - start) * 1000
562            return ComponentHealth(
563                name="job_queue",
564                state=HealthState.DEGRADED,
565                reason=f"Could not inspect job queue: {exc}",
566                latency_ms=latency,
567            )

569    def check_evaluations(
570        self,
571        last_eval_results: Mapping[str, Any] | None = None,
572        thresholds: Mapping[str, float] | None = None,
573    ) -> ComponentHealth:
574        """Inspect recent evaluation or benchmark results."""
575        if last_eval_results is None:
576            return ComponentHealth(
577                name="evaluation",
578                state=HealthState.UNKNOWN,
579                reason="No evaluation results recorded",
580            )
581
582        details = dict(last_eval_results)
583        if last_eval_results.get("passed") is False:
584            return ComponentHealth(
585                name="evaluation",
586                state=HealthState.DEGRADED,
587                reason="Evaluation failed or below threshold",
588                details=details,
589            )
590
591        thresh = thresholds or {"accuracy": 0.5, "recall": 0.5}
592        for metric, min_val in thresh.items():
593            val = last_eval_results.get(metric)
594            if val is not None and float(val) < min_val:
595                return ComponentHealth(
596                    name="evaluation",
597                    state=HealthState.DEGRADED,
598                    reason=(
599                        f"Evaluation metric {metric!r} ({val:.3f}) fell below "
600                        f"threshold ({min_val:.3f})"
601                    ),
602                    details=details,
603                )
604
605        return ComponentHealth(
606            name="evaluation",
607            state=HealthState.READY,
608            reason="Evaluation passed",
609            details=details,
610        )

612    async def check_expensive_consistency(
613        self, scope: str | None = None, limit: int = 100
614    ) -> ComponentHealth:
615        """Perform bounded, explicit consistency verification across manifest and store."""
616        start = time.perf_counter()
617        target_scope = scope or self.scope
618        if self.manifest_store is None:
619            latency = (time.perf_counter() - start) * 1000
620            return ComponentHealth(
621                name="expensive_consistency",
622                state=HealthState.UNKNOWN,
623                reason="Manifest store unavailable for consistency probe",
624                latency_ms=latency,
625            )
626
627        try:
628
629            async def _probe() -> tuple[int, int]:
630                if hasattr(self.manifest_store, "list"):
631                    records = await self.manifest_store.list(
632                        target_scope, status=ManifestStatus.ACTIVE
633                    )
634                else:
635                    records = await self.manifest_store.list_active(target_scope)
636                active_count = len(records[:limit])
637                store_count = 0
638                if self.vector_store is not None:
639                    store_count = await self.vector_store.count(strict=True)
640                return active_count, store_count
641
642            active_docs, store_chunks = await asyncio.wait_for(
643                _probe(), timeout=self.timeout_seconds
644            )
645            latency = (time.perf_counter() - start) * 1000
646            details = {
647                "manifest_active_documents": active_docs,
648                "vector_store_chunks": store_chunks,
649                "bounded_limit": limit,
650            }
651            return ComponentHealth(
652                name="expensive_consistency",
653                state=HealthState.READY,
654                latency_ms=latency,
655                details=details,
656            )
657        except TimeoutError:
658            latency = (time.perf_counter() - start) * 1000
659            return ComponentHealth(
660                name="expensive_consistency",
661                state=HealthState.DEGRADED,
662                reason=f"Consistency check timed out after {self.timeout_seconds}s",
663                latency_ms=latency,
664            )
665        except Exception as exc:  # noqa: BLE001
666            latency = (time.perf_counter() - start) * 1000
667            return ComponentHealth(
668                name="expensive_consistency",
669                state=HealthState.DEGRADED,
670                reason=f"Consistency check failed: {exc}",
671                latency_ms=latency,

694        self,
695        scope: str | None = None,
696        *,
697        include_expensive: bool = False,
698        expensive_limit: int = 100,
699        last_evaluation_results: Mapping[str, Any] | None = None,
700        last_eval_results: Mapping[str, Any] | None = None,
701        eval_thresholds: Mapping[str, float] | None = None,
702    ) -> HealthReport:
703        """Run health checks across components and return an aggregate HealthReport.
704
705        Args:
706            scope: Optional target scope to evaluate.
707            include_expensive: When True, executes bounded, deep consistency checks.
708                Defaults to False to prevent unneeded I/O during routine liveness polling.
709            expensive_limit: Bounded document count limit for expensive probe.
710            last_evaluation_results: Optional recent benchmark or evaluation metrics.
711            last_eval_results: Alias for last_evaluation_results.
712            eval_thresholds: Optional minimum thresholds for evaluation metrics.
713
714        Returns:
715            :class:`HealthReport` with composite state and component breakdowns.
716        """
717        target_scope = scope or self.scope
718        eval_res = last_evaluation_results or last_eval_results
719
720        tasks = [
721            self.check_backend_connectivity(),
722            self.check_counts(scope=target_scope),
723            self.check_sync_status(scope=target_scope),
724            self.check_jobs(scope=target_scope),
725        ]
726        if include_expensive:
727            tasks.append(
728                self.check_expensive_consistency(scope=target_scope, limit=expensive_limit)
729            )
730
731        components_list = [self.check_configuration()]
732        async_results = await asyncio.gather(*tasks)
733        components_list.extend(async_results)
734
735        if eval_res is not None:
736            components_list.append(self.check_evaluations(eval_res, eval_thresholds))
737
738        return HealthReport(components=components_list)
