1"""CI quality gate for the OmniCare RAG evaluation.
2
3The gate is the only thing standing between a change to the retriever, the
4agent, or the evaluation harness and a silent regression in the numbers those
5produce. It runs in CI with no API key, on the isolated test database, and it
6drives the same ``run_evaluation`` entry point and the same
7``app.domain.evaluation_cli`` summary that an operator runs, so the numbers CI
8gates on are the numbers the CLI reports.
9
10Three things are checked here:
11
121. The retrieval machinery still surfaces the sections the dataset expects
13   (``recall@5`` and ``mrr`` floors over the answerable subset).
142. The refusal machinery still detects an answer that invents coverage on a
15   question the corpus cannot answer, and the metric that reports it is still
16   computed -- not vacuously satisfied by an empty subset.
173. The live end-to-end run, which scores the real agent, stays opt-in behind
18   ``OMNICARE_EVAL_LIVE`` so CI never spends model calls or flakiness budget on
19   it by default.
20
21The answer generator used for the refusal check is a deterministic double, not
22the agent: it decides from the retrieved text alone whether the text answers the
23question. That makes the check meaningful -- a double that simply returned the
24gold answer would pass whatever the metric computed -- while keeping the gate
25free of a network call.
26"""

69def _eval_embedding(text: str, dim: int = EMBEDDING_DIM) -> list[float]:
70    """Return a deterministic hashing embedding for ``text``.
71
72    Words, light stems, and character tri- and quad-grams are all hashed into
73    the vector. The character n-grams are what make this double a usable
74    stand-in: word tokens alone leave "coverage" and "covered", or "property"
75    and "Personal Property Protection", at an L2 distance beyond the product's
76    retrieval threshold, and a double that cannot reach half the corpus would
77    measure the double rather than the retriever.
78
79    No network call, and identical input always produces the identical vector.
80    """
81    tokens = re.findall(r"[a-z0-9]+", text.lower())
82    words: list[str] = []
83    for token in tokens:
84        words.append(token)
85        if len(token) > 4 and token.endswith("s"):
86            words.append(token[:-1])
87        if len(token) > 5 and token.endswith("ing"):
88            words.append(token[:-3])
89        if len(token) > 4 and token.endswith("ed"):
90            words.append(token[:-2])
91
92    joined = f" {' '.join(tokens)} "
93    features = words + re.findall(r"[a-z]{3}", joined) + re.findall(r"[a-z]{4}", joined)
94
95    vector = [0.0] * dim
96    for feature in features:
97        digest = hashlib.sha256(feature.encode("utf-8")).digest()
98        index = int.from_bytes(digest[:2], "big") % dim
99        vector[index] += 1.0 if digest[2] % 2 == 0 else -1.0
100
101    norm = math.sqrt(sum(value * value for value in vector))
102    return [value / norm for value in vector] if norm > 0.0 else vector

105class _EmbeddingResponse:
106    """Mimic the LiteLLM embedding response shape (``response.data``)."""

112def _eval_embedding_double(**kwargs: Any) -> _EmbeddingResponse:
113    """Replace the LiteLLM embedding call with the deterministic double."""
114    inputs = kwargs.get("input", [])
115    if isinstance(inputs, str):
116        inputs = [inputs]
117    return _EmbeddingResponse(list(inputs))

124def _substantive_words(text: str) -> set[str]:
125    """Return the stems of ``text`` that could, on their own, make it about a subject.
126
127    Function words are dropped, then so is anything under five characters:
128    "home", "car", and "hit" are the words that accidentally tie an out-of-corpus
129    question to a section heading, and a rule that counts them makes the double
130    answer questions it has no business answering. Each word is reduced to one
131    canonical stem, so "flooded" and "flood" match but a single matching word
132    pair cannot count twice by yielding both "covered" and "cover".
133    """
134    words = set()
135    for token in _CONTENT_WORDS.findall(text.lower()):
136        stem = _stem(token)
137        if stem in _content_stopwords() or len(stem) < 5:
138            continue
139        if any(character.isalpha() for character in stem):
140            words.add(stem)
141    return words

144def _stem(token: str) -> str:
145    """Return ``token`` with a trailing plural or tense suffix removed."""
146    if len(token) > 5 and token.endswith("ing"):
147        return token[:-3]
148    if len(token) > 4 and token.endswith("ed"):
149        return token[:-2]
150    if len(token) > 4 and token.endswith("s"):
151        return token[:-1]
152    return token

162async def grounded_answer_fn(query: str, context: str) -> str:
163    """Answer ``query`` by quoting the retrieved sentences that address it.
164
165    A deterministic model of the assistant's own Uncertainty Ladder: answer from
166    the retrieved text, and decline when the retrieved text does not address the
167    question. The rule -- a sentence supports the answer when it shares at least
168    two substantive words with the question -- is deliberately crude, and it is
169    crude in a way a reader can check: the unanswerable samples are exactly the
170    ones where no sentence qualifies.
171
172    It over-refuses on questions the corpus can only reach through the
173    embedding model's generalisation ("what is the maximum payout for water
174    damage?" has two words in common with the section; "I want to claim for a
175    $3,000 laptop" has none). That is why the false-refusal direction is gated
176    on the live run, where the agent answers them, and only the refusal
177    direction is gated here.
178
179    Args:
180        query: The evaluation question.
181        context: The context the retriever surfaced for it.
182
183    Returns:
184        The sentences that share at least two substantive words with the
185        question, or the assistant's own no-information sentence when there are
186        none.
187    """
188    wanted = _substantive_words(query)
189    supporting = [
190        sentence
191        for sentence in _SENTENCE_SPLIT.split(context or "")
192        if len(_substantive_words(sentence) & wanted) >= 2
193    ]
194    if not supporting:
195        return (
196            "I don't have that information in my policy documents. Please contact our support team."
197        )
198    return " ".join(" ".join(supporting).split())

201async def inventing_answer_fn(query: str, context: str) -> str:
202    """Answer every question the same way, with invented coverage terms.
203
204    This double exists to prove the refusal metric is not a tautology: an answer
205    generator that states a limit and a deductible for every question must score
206    refusal accuracy zero on the unanswerable subset. If it did not, the metric
207    would be measuring nothing and the floor would be decoration.
208    """
209    return (
210        f"Regarding {query.strip()} you are covered up to $10,000 with a $250 "
211        "deductible, and gradual damage is excluded."
212    )
