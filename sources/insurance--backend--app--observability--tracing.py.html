1"""The observability layer's public tracing surface.
2
3Everything outside this package that wants a turn traced imports from this
4module and from nowhere else. ``app.main``, ``app.agent.agent``,
5``app.observability.tool_instrumentation``, and ``app.observability.routes``
6name this module and nothing vendor-specific, and that is what keeps the
7application uncoupled from a tracing vendor: which service the traces go to,
8what it is called, and how its credentials are spelled belong to
9``app.observability.langsmith``, the implementation this module delegates to.
10A change of vendor is therefore a change inside that one module -- this
11module's four names stay.
12
13The API is three verbs and one question:
14
15* :func:`setup_tracing` -- what the application lifespan calls to bring
16  tracing up, reporting whether it came up.
17* :func:`trace_turn` -- one span per agent turn, the root of its trace.
18* :func:`trace_tool` -- one span per tool call, nested under its turn.
19* :func:`tracing_status` -- what the telemetry endpoint reports.
20"""
21

36@dataclass(frozen=True)
37class TracingStatus:
38    """The tracing configuration a deployment is actually running.
39
40    Attributes:
41        enabled: Whether turns are traced right now. The flag and a key are
42            both required, so this is False for a deployment that enabled
43            tracing without configuring a key -- the state that most needs
44            surfacing.
45        backend: The vendor providing tracing. Named even when tracing is
46            off, because the configured-but-inactive case is exactly what an
47            operator reading the telemetry endpoint needs to see.
48        project: The project traces are grouped under.
49        endpoint: The API base URL traces are posted to, with the vendor's
50            cloud default spelled out rather than left blank.
51    """
52
53    enabled: bool
54    backend: str
55    project: str
56    endpoint: str

59def setup_tracing() -> bool:
60    """Bring tracing up, exactly as the application lifespan calls it.
61
62    Ordering is the implementation's to decide and this function's to fix:
63    the environment the tracing library and the model router read has to be
64    exported before either resolves a client or a credential, so
65    configuration comes first and the model-call observer after it.
66
67    Returns:
68        bool: Whether tracing is active afterwards. A disabled or unkeyed
69        deployment reports ``False`` -- and the implementation logs why, so a
70        deployment that meant to trace cannot silently run untraced -- and a
71        failure is contained here rather than raised, because telemetry must
72        never take the application down with it.
73    """
74    try:
75        destination = langsmith.tracing_destination()
76        if not langsmith.configure_langsmith():
77            return False
78        langsmith.register_langsmith_callback()
79        langsmith.install_llm_trace_context()
80        logger.info("Tracing enabled (project '%s').", destination["project"])
81        return True
82    except Exception:
83        logger.warning("Tracing setup skipped", exc_info=True)
84        return False

87def trace_turn(run_name: str) -> Callable[[F], F]:
88    """Return the decorator that traces one agent turn.
89
90    The turn's span is the root of its trace: the model calls the turn makes
91    and the tool calls it triggers nest under it, so one turn of a
92    conversation is one trace the operator can read end to end. Tracing off
93    returns the function untouched.
94
95    Args:
96        run_name: Name the trace is listed under. Callers pass their own so
97            the plain and streaming entry points read apart.
98
99    Returns:
100        Callable[[F], F]: The decorator.
101    """
102    return langsmith.trace_turn(run_name)
103
104
105def trace_tool(tool_name: str) -> Callable[[F], F]:
106    """Return the decorator that traces one tool call.
107
108    The span nests under the turn that invoked the tool, so a trace shows
109    which tool ran, with what arguments, and what it returned. Tracing off
110    returns the handler unchanged.
111
112    Args:
113        tool_name: Tool name the span is recorded under.
114
115    Returns:
116        Callable[[F], F]: The decorator.
117    """
118    return langsmith.trace_tool(tool_name)

121def tracing_status() -> TracingStatus:
122    """Report the tracing configuration this deployment is running.
123
124    Returns:
125        TracingStatus: Whether tracing is live, which backend provides it,
126        and where traces are reported to.

128    destination = langsmith.tracing_destination()
129    return TracingStatus(backend="langsmith", **destination)
