1"""Input validation and security sanitization for tool bridges and adapters.
2
3Prevents arbitrary filesystem paths, credentials/secrets, and unrestricted
4fetch URLs from reaching internal retrieval, storage, or execution engines.
5"""

7from __future__ import annotations

9import re
10from typing import Any
11from urllib.parse import urlparse

13# Pattern matching potential credentials or API keys
14_CREDENTIAL_PATTERNS = [
15    re.compile(r"(?i)\bbearer\s+[a-zA-Z0-9_\-\.]{15,}"),
16    re.compile(r"-----BEGIN [A-Z ]*PRIVATE KEY-----"),
17    re.compile(r"\b(sk-[a-zA-Z0-9]{20,})\b"),
18    re.compile(r"\b(ghp_[a-zA-Z0-9]{20,})\b"),
19    re.compile(r"\b(glpat-[a-zA-Z0-9]{20,})\b"),
20    re.compile(r"\b(xoxb-[a-zA-Z0-9-]{20,})\b"),
21    re.compile(r"(?i)\b(password|passwd|secret_key|api_key)\s*[:=]\s*['\"]?[^\s'\"]{6,}"),
22]
23
24# Patterns detecting path traversal or arbitrary filesystem paths
25_PATH_TRAVERSAL_PATTERNS = [
26    re.compile(r"\.\.[/\\]"),
27    re.compile(r"^[/\\]etc[/\\]"),
28    re.compile(r"^[/\\]var[/\\]"),
29    re.compile(r"^[/\\]usr[/\\]"),
30    re.compile(r"^[/\\]root"),
31    re.compile(r"^[A-Za-z]:[/\\]"),  # Windows drive letter C:\
32    re.compile(r"^~[/\\]"),
33    re.compile(r"(?i)^file://"),
34]
35
36# Prohibited SSRF hostnames and IP ranges
37_PROHIBITED_HOSTS = {
38    "localhost",
39    "127.0.0.1",
40    "0.0.0.0",  # noqa: S104
41    "::1",
42    "169.254.169.254",  # AWS/GCP/Azure link-local metadata service
43    "metadata.google.internal",
44}
45
46_SAFE_IDENTIFIER_PATTERN = re.compile(r"^[a-zA-Z0-9_.-]{1,128}$")
47_SQL_INJECTION_PATTERN = re.compile(r"(?i)(--|;|\bUNION\b|\bDROP\b|\bDELETE\b|\bTRUNCATE\b|/\*)")

50class SecurityValidationError(ValueError):
51    """Raised when tool arguments contain unsafe paths, credentials, or URLs."""
52

54def validate_safe_string(
55    val: str,
56    field_name: str,
57    *,
58    allow_path_traversal: bool = False,
59    max_length: int = 10000,
60) -> None:
61    """Validate that a string contains no null bytes, path traversal, or credentials.
62
63    Args:
64        val: String value to inspect.
65        field_name: Name of the argument or field for error reporting.
66        allow_path_traversal: If True, skips path traversal checks.
67        max_length: Maximum allowed character length.
68
69    Raises:
70        SecurityValidationError: If the value violates security constraints.
71    """
72    if not isinstance(val, str):
73        raise SecurityValidationError(f"{field_name} must be a string, got {type(val).__name__}")
74
75    if len(val) > max_length:
76        raise SecurityValidationError(
77            f"{field_name} exceeds maximum allowed length of {max_length} characters"
78        )
79
80    if "\0" in val:
81        raise SecurityValidationError(f"{field_name} contains prohibited null byte")
82
83    # Check for credentials
84    for pattern in _CREDENTIAL_PATTERNS:
85        if pattern.search(val):
86            raise SecurityValidationError(
87                f"{field_name} contains prohibited credential or secret pattern"
88            )
89
90    # Check for path traversal if not explicitly allowed
91    if not allow_path_traversal:
92        for pattern in _PATH_TRAVERSAL_PATTERNS:
93            if pattern.search(val):
94                raise SecurityValidationError(
95                    f"{field_name} contains prohibited arbitrary filesystem path or traversal"
96                )

99def validate_safe_identifier(val: str, field_name: str) -> None:
100    """Validate that an identifier contains only alphanumeric, dash, dot, or underscore.
101
102    Prevents SQL injection, path traversal, and command injection in collection/document IDs.
103
104    Args:
105        val: Identifier to validate.
106        field_name: Field name for error reporting.
107
108    Raises:
109        SecurityValidationError: If identifier contains unsafe characters.
110    """
111    validate_safe_string(val, field_name, max_length=128)
112    if not _SAFE_IDENTIFIER_PATTERN.match(val) or ".." in val:
113        raise SecurityValidationError(
114            f"{field_name} must be an alphanumeric identifier (1-128 chars, [a-zA-Z0-9_.-]); "
115            f"got {val!r}"
116        )

119def validate_safe_uri(uri: str | None, field_name: str = "uri") -> None:
120    """Validate that a URI uses an approved scheme and does not target internal services.
121
122    Args:
123        uri: The URI string to inspect.
124        field_name: Argument name for error reporting.
125
126    Raises:
127        SecurityValidationError: If scheme is unsafe or target is an internal SSRF host.
128    """
129    if uri is None:
130        return
131
132    validate_safe_string(uri, field_name, max_length=2048, allow_path_traversal=False)
133
134    parsed = urlparse(uri)
135    scheme = parsed.scheme.lower()
136
137    if not scheme:
138        # Bare safe identifier or relative safe ref
139        return
140
141    if scheme not in ("https", "http", "urn", "doc"):
142        raise SecurityValidationError(
143            f"{field_name} specifies prohibited scheme {scheme!r}; "
144            "only https, http, urn, and doc schemes are permitted"
145        )
146
147    if scheme in ("http", "https"):
148        hostname = (parsed.hostname or "").lower()
149        if not hostname:
150            raise SecurityValidationError(f"{field_name} is missing a valid hostname")
151
152        if hostname in _PROHIBITED_HOSTS or hostname.endswith(".internal"):
153            raise SecurityValidationError(
154                f"{field_name} targets prohibited internal or link-local host {hostname!r}"
155            )
156
157        # Check private IP ranges
158        if hostname.startswith("10.") or hostname.startswith("192.168."):
159            raise SecurityValidationError(
160                f"{field_name} targets prohibited private network address {hostname!r}"
161            )
162        if re.match(r"^172\.(1[6-9]|2[0-9]|3[0-1])\.", hostname):
163            raise SecurityValidationError(
164                f"{field_name} targets prohibited private network address {hostname!r}"
165            )

168def validate_safe_filters(filters: dict[str, Any] | None) -> None:
169    """Validate filter dictionary keys and values.
170
171    Args:
172        filters: Filter key-value mapping.
173
174    Raises:
175        SecurityValidationError: If any key or value is unsafe.
176    """
177    if filters is None:
178        return
179
180    if not isinstance(filters, dict):
181        raise SecurityValidationError(f"filters must be a dict, got {type(filters).__name__}")
182
183    for k, v in filters.items():
184        validate_safe_identifier(str(k), f"filter key {k!r}")
185        if isinstance(v, str):
186            validate_safe_string(v, f"filter value for {k!r}", max_length=1000)
187            if _SQL_INJECTION_PATTERN.search(v):
188                raise SecurityValidationError(
189                    f"filter value for {k!r} contains prohibited SQL injection characters"
190                )
191        elif isinstance(v, (int, float, bool)):
192            continue
193        elif isinstance(v, (list, tuple, set)):
194            for item in v:
195                if isinstance(item, str):
196                    validate_safe_string(item, f"filter item in {k!r}", max_length=1000)
197                    if _SQL_INJECTION_PATTERN.search(item):
198                        raise SecurityValidationError(
199                            f"filter list item for {k!r} contains prohibited SQL syntax"
200                        )
