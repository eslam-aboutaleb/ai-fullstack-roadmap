1"""Procedure playbook retrieval adapter over ragnor.
2
3The OmniCare procedure chunk binding — the ``procedure_chunks`` table, its
4``text`` column, and the metadata columns returned in chunk metadata — lives
5here as :data:`PROCEDURE_RETRIEVER_SPEC`. The generic hybrid retrieval
6machinery is ragnor's :class:`ragnor.retrieval.HybridRetriever`.
7
8The procedures index is a *second* index alongside the policy one, not an
9extension of it: the pipeline, the chunk table, and the version table are all
10separate, so re-ingesting a playbook can never disturb the policy index and a
11policy re-ingest can never retire a procedure version.
12"""
13

16import logging
17from typing import Any

19from app.core.config import get_settings
20from app.core.database import async_session_factory
21from ragnor.retrieval import HybridRetriever, RetrieverSpec, as_dicts

23logger = logging.getLogger(__name__)

25PROCEDURE_RETRIEVER_SPEC = RetrieverSpec(
26    table_name="procedure_chunks",
27    id_field="id",
28    text_field="text",
29    metadata_fields=["section", "source", "chunk_index", "sub_chunk_index"],
30)
31"""Binding of the hybrid retriever to the OmniCare procedure chunk table."""

34class ProcedureRetriever:
35    """Hybrid retrieval over the OmniCare procedure chunk table.
36
37    Wraps ragnor's :class:`ragnor.retrieval.HybridRetriever` with the procedure
38    chunk binding. Any retrieval failure is logged and reported as an empty
39    result set rather than raised.
40
41    Args:
42        settings: Application settings. Defaults to ``get_settings()``.
43        session_factory: Session provider the store opens its transactions
44            with. Defaults to the application's ``async_session_factory``.
45    """
46
47    def __init__(self, settings: Any = None, session_factory: Any = None) -> None:
48        self._settings = settings or get_settings()
49        self._session_factory = session_factory or async_session_factory
50        self._retriever = HybridRetriever(
51            spec=PROCEDURE_RETRIEVER_SPEC,
52            settings=self._settings,
53            session_factory=self._session_factory,
54        )
55
56    async def retrieve_hybrid(  # noqa: PLR0913
57        self,
58        query: str,
59        n_results: int = 5,
60        distance_threshold: float | None = None,
61        source_file: str | None = None,
62    ) -> list[dict[str, Any]]:
63        """Run hybrid (vector + PostgreSQL full-text) search over procedure chunks.
64
65        Args:
66            query: Natural language query, or the playbook's subject seed.
67            n_results: Maximum number of results to return.
68            distance_threshold: Maximum L2 distance for the vector search
69                part. Defaults to ``settings.rag_distance_threshold``.
70            source_file: Restrict the search to one playbook file. The
71                filter is pushed into both search stages, so chunks from a
72                different playbook are never returned.
73
74        Returns:
75            List of result dicts with ``document``, ``metadata``,
76            ``distance``, and ``_rrf_score`` keys.
77        """
78        if distance_threshold is None:
79            distance_threshold = self._settings.rag_distance_threshold
80        results = await self._retriever.retrieve(
81            query,
82            n_results=n_results,
83            distance_threshold=distance_threshold,
84            **({"source": source_file} if source_file else {}),
85        )
86        return as_dicts(results)
87

89async def retrieve_procedures(  # noqa: PLR0913
90    query: str,
91    n_results: int = 5,
92    distance_threshold: float | None = None,
93    source_file: str | None = None,
94) -> list[dict[str, Any]]:
95    """Run hybrid search over the OmniCare procedure chunk table.
96
97    Module-level entry point. Any retrieval failure is logged and reported as
98    an empty result set rather than raised.
99
100    Args:
101        query: Natural language query, or the playbook's subject seed.
102        n_results: Maximum number of results to return.
103        distance_threshold: Maximum L2 distance for the vector search part.
104            Defaults to ``settings.rag_distance_threshold``.
105        source_file: Restrict the search to one playbook file.
106
107    Returns:
108        List of result dicts, or an empty list on failure.
109    """
110    try:
111        return await ProcedureRetriever().retrieve_hybrid(
112            query,
113            n_results=n_results,
114            distance_threshold=distance_threshold,
115            source_file=source_file,
116        )
117    except Exception as exc:
118        logger.error("Procedure retrieval failed: %s", exc)
119        return []
